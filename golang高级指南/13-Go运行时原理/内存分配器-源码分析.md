## 1. 内存分配器概述

### 1.1 TCMalloc设计

Go的内存分配器基于Google的TCMalloc（Thread-Caching Malloc）设计。

**核心思想**：
- 🔥 **多级缓存**：减少锁竞争
- 🔥 **大小分级**：不同大小的对象使用不同的分配策略
- 🔥 **线程缓存**：每个P有独立的缓存
- 🔥 **中心缓存**：全局共享的缓存

### 1.2 内存分配层级

```go
package main

import "fmt"

// 🔥 Go内存分配的三个层级

// 1. mcache（线程缓存）
// - 每个P拥有一个mcache
// - 无锁分配，性能最高
// - 缓存小对象（<32KB）

// 2. mcentral（中心缓存）
// - 全局共享
// - 需要加锁
// - 管理特定大小的span

// 3. mheap（堆）
// - 全局唯一
// - 管理所有的span
// - 从操作系统申请内存

func main() {
    fmt.Println("内存分配层级:")
    fmt.Println("mcache -> mcentral -> mheap -> OS")
}
```

### 1.3 对象大小分类

```go
package main

import "fmt"

func main() {
    // 🔥 Go将对象分为三类
    
    // 1. 微对象（Tiny）：< 16B
    // - 多个微对象可以组合到一个16B的块中
    // - 减少内存碎片
    
    // 2. 小对象（Small）：16B - 32KB
    // - 使用mcache分配
    // - 按大小分为67个等级
    
    // 3. 大对象（Large）：> 32KB
    // - 直接从mheap分配
    // - 不经过mcache和mcentral
    
    fmt.Println("对象大小分类:")
    fmt.Println("微对象: < 16B")
    fmt.Println("小对象: 16B - 32KB")
    fmt.Println("大对象: > 32KB")
}
```

## 2. 核心数据结构

### 2.1 mspan结构

```go
package main

// 🔥 mspan是内存管理的基本单位
// 源码位置：runtime/mheap.go

type mspan struct {
    // 链表指针
    next *mspan     // 下一个span
    prev *mspan     // 上一个span
    
    // span的基本信息
    startAddr uintptr   // 起始地址
    npages    uintptr   // 页数（1页=8KB）
    
    // 对象分配信息
    spanclass   spanClass  // span类别
    elemsize    uintptr    // 元素大小
    nelems      uint16     // 元素个数
    allocCount  uint16     // 已分配对象数
    
    // 位图信息
    allocBits  *gcBits    // 分配位图
    gcmarkBits *gcBits    // GC标记位图
}

// 🔥 spanClass定义
type spanClass uint8

const (
    numSpanClasses = 67 * 2  // 67个大小等级 * 2（是否需要扫描）
)

func main() {
    // mspan示例
    // 假设分配16B的对象
    // - npages = 1（8KB）
    // - elemsize = 16B
    // - nelems = 512（8KB / 16B）
}
```

### 2.2 mcache结构

```go
package main

// 🔥 mcache是每个P的本地缓存
// 源码位置：runtime/mcache.go

type mcache struct {
    // 微对象分配器
    tiny       uintptr  // 当前tiny块的起始地址
    tinyoffset uintptr  // tiny块的偏移量
    local_tinyallocs uintptr  // tiny对象分配计数
    
    // 小对象分配器
    // 67个大小等级 * 2（scan/noscan）= 134个span
    alloc [numSpanClasses]*mspan
    
    // 统计信息
    local_scan  uintptr  // 扫描的字节数
    local_alloc uintptr  // 分配的字节数
}

func main() {
    // mcache示例
    // 每个P有一个mcache
    // 分配小对象时，直接从mcache的alloc数组中获取对应的span
}
```

### 2.3 mcentral结构

```go
package main

// 🔥 mcentral管理特定大小的span
// 源码位置：runtime/mcentral.go

type mcentral struct {
    spanclass spanClass  // span类别
    
    // 部分空闲的span列表
    partial [2]spanSet
    
    // 完全空闲的span列表
    full [2]spanSet
}

type spanSet struct {
    spineLock mutex
    spine     unsafe.Pointer  // *[N]*span
    spineLen  uintptr
    spineCap  uintptr
}

func main() {
    // mcentral示例
    // 当mcache的span用完时，从mcentral获取新的span
    // 当mcache的span完全空闲时，归还给mcentral
}
```

### 2.4 mheap结构

```go
package main

// 🔥 mheap是全局的堆管理器
// 源码位置：runtime/mheap.go

type mheap struct {
    lock      mutex
    pages     pageAlloc  // 页分配器
    
    // 所有的mcentral
    central [numSpanClasses]struct {
        mcentral mcentral
        pad      [cpu.CacheLinePadSize - unsafe.Sizeof(mcentral{})%cpu.CacheLinePadSize]byte
    }
    
    // span分配器
    spanalloc fixalloc  // span分配器
    cachealloc fixalloc // mcache分配器
    
    // 大对象分配
    largealloc  uint64  // 大对象分配字节数
    nlargealloc uint64  // 大对象分配次数
}

func main() {
    // mheap示例
    // 全局唯一的堆管理器
    // 管理所有的内存分配
}
```

## 3. 内存分配流程

### 3.1 小对象分配流程

```go
package main

import (
    "fmt"
    "runtime"
    "unsafe"
)

func main() {
    // 🔥 小对象分配流程（16B - 32KB）
    
    // 步骤1：计算对象大小等级
    size := 32  // 假设分配32B
    sizeclass := getSizeClass(size)
    
    // 步骤2：从mcache获取对应的span
    // span := mcache.alloc[sizeclass]
    
    // 步骤3：从span中分配对象
    // 如果span有空闲对象，直接分配
    // 如果span已满，执行步骤4
    
    // 步骤4：从mcentral获取新的span
    // 如果mcentral有空闲span，返回给mcache
    // 如果mcentral没有空闲span，执行步骤5
    
    // 步骤5：从mheap分配新的span
    // mheap从操作系统申请内存
    
    fmt.Printf("分配32B对象，大小等级: %d\n", sizeclass)
}

func getSizeClass(size int) int {
    // 简化的大小等级计算
    // 实际实现更复杂
    if size <= 8 {
        return 0
    } else if size <= 16 {
        return 1
    } else if size <= 32 {
        return 2
    }
    return 3
}
```

## 4. 内存回收机制

### 4.1 span回收

```go
package main

import "fmt"

func main() {
    // 🔥 span回收流程
    
    // 场景1：span完全空闲
    // - 从mcache归还到mcentral
    // - 加入mcentral的full列表
    
    // 场景2：span部分空闲
    // - 保留在mcache中
    // - 继续用于分配
    
    // 场景3：mcentral的span过多
    // - 归还给mheap
    // - mheap可能归还给操作系统
    
    fmt.Println("span回收策略:")
    fmt.Println("1. 完全空闲 -> mcentral")
    fmt.Println("2. 部分空闲 -> 保留在mcache")
    fmt.Println("3. mcentral过多 -> mheap")
}
```

### 4.2 内存归还操作系统

```go
package main

import (
    "fmt"
    "runtime"
    "runtime/debug"
)

func main() {
    // 🔥 内存归还操作系统
```

## 5. 内存对齐

### 5.1 对齐规则

```go
package main

import (
    "fmt"
    "unsafe"
)

func main() {
    // 🔥 Go的内存对齐规则
    
    // 规则1：基本类型按自身大小对齐
    var i8 int8
    var i16 int16
    var i32 int32
    var i64 int64
    
    fmt.Printf("int8对齐: %d\n", unsafe.Alignof(i8))    // 1
    fmt.Printf("int16对齐: %d\n", unsafe.Alignof(i16))  // 2
    fmt.Printf("int32对齐: %d\n", unsafe.Alignof(i32))  // 4
    fmt.Printf("int64对齐: %d\n", unsafe.Alignof(i64))  // 8
    
    // 规则2：结构体按最大字段对齐
    type S1 struct {
        a int8
        b int64
        c int8
    }
    
    fmt.Printf("S1大小: %d\n", unsafe.Sizeof(S1{}))      // 24（不是10）
    fmt.Printf("S1对齐: %d\n", unsafe.Alignof(S1{}))    // 8
    
    // 规则3：优化字段顺序可以减少内存占用
    type S2 struct {
        b int64
        a int8
        c int8
    }
    
    fmt.Printf("S2大小: %d\n", unsafe.Sizeof(S2{}))      // 16（优化后）
}
```

### 5.2 内存对齐优化

```go
    "fmt"

    fmt.Printf("GoodStruct大小: %d\n", unsafe.Sizeof(GoodStruct{})) // 16

    fmt.Println("1. 将大字段放在前面")
    fmt.Println("2. 将小字段放在后面")
    fmt.Println("3. 相同大小的字段放在一起")
}
```

## 6. 性能优化

### 6.1 减少内存分配

```go
package main

import (
    "fmt"
    "sync"
)

// ❌ 频繁分配内存
func badAlloc() {
    for i := 0; i < 1000000; i++ {
        data := make([]byte, 1024)
        _ = data
    }
}

// ✅ 使用对象池
var bufferPool = sync.Pool{
    New: func() interface{} {
        return make([]byte, 1024)
    },
}

func goodAlloc() {
    for i := 0; i < 1000000; i++ {
        data := bufferPool.Get().([]byte)
        // 使用data
        bufferPool.Put(data)
    }
}

func main() {
    fmt.Println("内存分配优化:")
    fmt.Println("1. 使用sync.Pool复用对象")
    fmt.Println("2. 预分配切片容量")
    fmt.Println("3. 避免逃逸到堆")
}
```

### 6.2 预分配容量

```go
package main

import "fmt"

// ❌ 未预分配容量
func badSlice() []int {
    s := make([]int, 0)
    for i := 0; i < 1000; i++ {
        s = append(s, i)  // 多次扩容
    }
    return s
}

// ✅ 预分配容量
func goodSlice() []int {
    s := make([]int, 0, 1000)  // 预分配容量
    for i := 0; i < 1000; i++ {
        s = append(s, i)  // 无需扩容
    }
    return s
}

func main() {
    fmt.Println("切片优化:")
    fmt.Println("- 预分配容量避免多次扩容")
    fmt.Println("- 减少内存分配次数")
    fmt.Println("- 提高性能")
}
```
