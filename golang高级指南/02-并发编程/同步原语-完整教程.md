## 同步原语概述

Go 提供了多种同步原语来协调 goroutine 之间的执行，主要包含在 `sync` 和 `sync/atomic` 包中。

### 核心原语

1. **Mutex**：互斥锁
2. **RWMutex**：读写锁
3. **WaitGroup**：等待组
4. **Once**：单次执行
5. **Cond**：条件变量
6. **Atomic**：原子操作

## Mutex 互斥锁

Mutex 用于保护共享资源，确保同一时间只有一个 goroutine 可以访问。

### 基本用法

```go
package main

import (
    "fmt"
    "sync"
)

type Counter struct {
    mu    sync.Mutex
    value int
}

func (c *Counter) Inc() {
    c.mu.Lock()
    defer c.mu.Unlock()
    c.value++
}

func (c *Counter) Value() int {
    c.mu.Lock()
    defer c.mu.Unlock()
    return c.value
}

func main() {
    var counter Counter
    var wg sync.WaitGroup
    
    // 启动 1000 个 goroutine
    for i := 0; i < 1000; i++ {
        wg.Add(1)
        go func() {
            defer wg.Done()
            counter.Inc()
        }()
    }
    
    wg.Wait()
    fmt.Println("Counter:", counter.Value())  // 输出：1000
}
```

### 常见错误

```go
// ❌ 错误：复制包含 Mutex 的结构体
type BadCounter struct {
    mu    sync.Mutex
    value int
}

func (c BadCounter) Inc() {  // 值接收者会复制 Mutex
    c.mu.Lock()
    defer c.mu.Unlock()
    c.value++
}

// ✅ 正确：使用指针接收者
func (c *BadCounter) IncCorrect() {
    c.mu.Lock()
    defer c.mu.Unlock()
    c.value++
}
```

### 死锁示例

```go
// ❌ 死锁：重复加锁
func deadlock() {
    var mu sync.Mutex
    mu.Lock()
    mu.Lock()  // 死锁！
    mu.Unlock()
}

// ✅ 避免死锁：使用 defer
func noDeadlock() {
    var mu sync.Mutex
    mu.Lock()
    defer mu.Unlock()
    // 业务逻辑
}
```

## RWMutex 读写锁

RWMutex 允许多个读操作同时进行，但写操作是互斥的。

### 基本用法

```go
package main

import (
    "fmt"
    "sync"
    "time"
)

type Cache struct {
    mu   sync.RWMutex
    data map[string]string
}

func NewCache() *Cache {
    return &Cache{
        data: make(map[string]string),
    }
}

// 读操作：使用读锁
func (c *Cache) Get(key string) (string, bool) {
    c.mu.RLock()
    defer c.mu.RUnlock()
    
    value, ok := c.data[key]
    return value, ok
}

// 写操作：使用写锁
func (c *Cache) Set(key, value string) {
    c.mu.Lock()
    defer c.mu.Unlock()
    
    c.data[key] = value
}

func main() {
    cache := NewCache()
    var wg sync.WaitGroup
    
    // 写入数据
    cache.Set("key1", "value1")
    
    // 启动 100 个读 goroutine
    for i := 0; i < 100; i++ {
        wg.Add(1)
        go func(id int) {
            defer wg.Done()
            value, _ := cache.Get("key1")
            fmt.Printf("Reader %d: %s\n", id, value)
        }(i)
    }
    
    // 启动 10 个写 goroutine
    for i := 0; i < 10; i++ {
        wg.Add(1)
        go func(id int) {
            defer wg.Done()
            cache.Set(fmt.Sprintf("key%d", id), fmt.Sprintf("value%d", id))
        }(i)
    }
    
    wg.Wait()
}
```

### 性能对比

```go
package main

import (
    "sync"
    "testing"
)

type MutexCache struct {
    mu   sync.Mutex
    data map[string]string
}

type RWMutexCache struct {
    mu   sync.RWMutex
    data map[string]string
}

func BenchmarkMutexRead(b *testing.B) {
    cache := &MutexCache{data: make(map[string]string)}
    cache.data["key"] = "value"
    
    b.RunParallel(func(pb *testing.PB) {
        for pb.Next() {
            cache.mu.Lock()
            _ = cache.data["key"]
            cache.mu.Unlock()
        }
    })
}

func BenchmarkRWMutexRead(b *testing.B) {
    cache := &RWMutexCache{data: make(map[string]string)}
    cache.data["key"] = "value"
    
    b.RunParallel(func(pb *testing.PB) {
        for pb.Next() {
            cache.mu.RLock()
            _ = cache.data["key"]
            cache.mu.RUnlock()
        }
    })
}
```

结果：RWMutex 在读多写少的场景下性能更好。

## WaitGroup 等待组

WaitGroup 用于等待一组 goroutine 完成。

### 基本用法

```go
package main

import (
    "fmt"
    "sync"
    "time"
)

func worker(id int, wg *sync.WaitGroup) {
    defer wg.Done()
    
    fmt.Printf("Worker %d starting\n", id)
    time.Sleep(time.Second)
    fmt.Printf("Worker %d done\n", id)
}

func main() {
    var wg sync.WaitGroup
    
    for i := 1; i <= 5; i++ {
        wg.Add(1)
        go worker(i, &wg)
    }
    
    wg.Wait()
    fmt.Println("All workers done")
}
```

### 常见错误

```go
// ❌ 错误：在 goroutine 内部 Add
func bad() {
    var wg sync.WaitGroup
    for i := 0; i < 5; i++ {
        go func() {
            wg.Add(1)  // 可能在 Wait 之后执行
            defer wg.Done()
            // 工作
        }()
    }
    wg.Wait()
}

// ✅ 正确：在启动 goroutine 前 Add
func good() {
    var wg sync.WaitGroup
    for i := 0; i < 5; i++ {
        wg.Add(1)
        go func() {
            defer wg.Done()
            // 工作
        }()
    }
    wg.Wait()
}
```

### 高级用法：带超时的 Wait

```go
package main

import (
    "fmt"
    "sync"
    "time"
)

func waitWithTimeout(wg *sync.WaitGroup, timeout time.Duration) bool {
    done := make(chan struct{})
    
    go func() {
        wg.Wait()
        close(done)
    }()
    
    select {
    case <-done:
        return true
    case <-time.After(timeout):
        return false
    }
}

func main() {
    var wg sync.WaitGroup
    
    wg.Add(1)
    go func() {
        defer wg.Done()
        time.Sleep(2 * time.Second)
    }()
    
    if waitWithTimeout(&wg, 1*time.Second) {
        fmt.Println("All done")
    } else {
        fmt.Println("Timeout!")
    }
}
```

## Once 单次执行

Once 确保某个操作只执行一次，常用于单例模式。

### 基本用法

```go
package main

import (
    "fmt"
    "sync"
)

var (
    instance *Singleton
    once     sync.Once
)

type Singleton struct {
    data string
}

func GetInstance() *Singleton {
    once.Do(func() {
        fmt.Println("Creating singleton instance")
        instance = &Singleton{data: "singleton"}
    })
    return instance
}

func main() {
    var wg sync.WaitGroup
    
    for i := 0; i < 10; i++ {
        wg.Add(1)
        go func() {
            defer wg.Done()
            s := GetInstance()
            fmt.Println(s.data)
        }()
    }
    
    wg.Wait()
    // "Creating singleton instance" 只会打印一次
}
```

### 延迟初始化

```go
package main

import (
    "fmt"
    "sync"
)

type Config struct {
    once   sync.Once
    data   map[string]string
    loaded bool
}

func (c *Config) Load() {
    c.once.Do(func() {
        fmt.Println("Loading config...")
        c.data = map[string]string{
            "host": "localhost",
            "port": "8080",
        }
        c.loaded = true
    })
}

func (c *Config) Get(key string) string {
    c.Load()  // 确保已加载
    return c.data[key]
}

func main() {
    config := &Config{}
    
    // 多次调用，只加载一次
    fmt.Println(config.Get("host"))
    fmt.Println(config.Get("port"))
}
```

## Cond 条件变量

Cond 用于在 goroutine 之间进行信号通知。

### 基本用法

```go
package main

import (
    "fmt"
    "sync"
    "time"
)

func main() {
    var mu sync.Mutex
    cond := sync.NewCond(&mu)
    ready := false
    
    // 等待者
    go func() {
        mu.Lock()
        for !ready {
            fmt.Println("Waiting...")
            cond.Wait()  // 释放锁并等待信号
        }
        fmt.Println("Ready!")
        mu.Unlock()
    }()
    
    // 通知者
    time.Sleep(2 * time.Second)
    mu.Lock()
    ready = true
    cond.Signal()  // 发送信号
    mu.Unlock()
    
    time.Sleep(time.Second)
}
```

### 生产者-消费者模式

```go
package main

import (
    "fmt"
    "sync"
    "time"
)

type Queue struct {
    mu    sync.Mutex
    cond  *sync.Cond
    items []int
}

func NewQueue() *Queue {
    q := &Queue{
        items: make([]int, 0),
    }
    q.cond = sync.NewCond(&q.mu)
    return q
}

func (q *Queue) Enqueue(item int) {
    q.mu.Lock()
    defer q.mu.Unlock()
    
    q.items = append(q.items, item)
    q.cond.Signal()  // 通知消费者
}

func (q *Queue) Dequeue() int {
    q.mu.Lock()
    defer q.mu.Unlock()
    
    for len(q.items) == 0 {
        q.cond.Wait()  // 等待生产者
    }
    
    item := q.items[0]
    q.items = q.items[1:]
    return item
}

func main() {
    queue := NewQueue()
    
    // 消费者
    go func() {
        for i := 0; i < 5; i++ {
            item := queue.Dequeue()
            fmt.Printf("Consumed: %d\n", item)
        }
    }()
    
    // 生产者
    for i := 0; i < 5; i++ {
        time.Sleep(time.Second)
        queue.Enqueue(i)
        fmt.Printf("Produced: %d\n", i)
    }
    
    time.Sleep(2 * time.Second)
}
```

## 原子操作

原子操作提供了无锁的并发安全操作。

### 基本类型

```go
package main

import (
    "fmt"
    "sync"
    "sync/atomic"
)

func main() {
    var counter int64
    var wg sync.WaitGroup
    
    for i := 0; i < 1000; i++ {
        wg.Add(1)
        go func() {
            defer wg.Done()
            atomic.AddInt64(&counter, 1)
        }()
    }
    
    wg.Wait()
    fmt.Println("Counter:", atomic.LoadInt64(&counter))
}
```

### 常用原子操作

```go
package main

import (
    "fmt"
    "sync/atomic"
)

func main() {
    var value int64 = 10
    
    // Add：增加
    atomic.AddInt64(&value, 5)
    fmt.Println("After Add:", value)  // 15
    
    // Load：读取
    loaded := atomic.LoadInt64(&value)
    fmt.Println("Loaded:", loaded)  // 15
    
    // Store：存储
    atomic.StoreInt64(&value, 20)
    fmt.Println("After Store:", value)  // 20
    
    // Swap：交换
    old := atomic.SwapInt64(&value, 30)
    fmt.Println("Old:", old, "New:", value)  // Old: 20 New: 30
    
    // CompareAndSwap：比较并交换
    swapped := atomic.CompareAndSwapInt64(&value, 30, 40)
    fmt.Println("Swapped:", swapped, "Value:", value)  // true, 40
}
```

### atomic.Value

```go
package main

import (
    "fmt"
    "sync/atomic"
)

type Config struct {
    Host string
    Port int
}

func main() {
    var config atomic.Value
    
    // 存储配置
    config.Store(&Config{
        Host: "localhost",
        Port: 8080,
    })
    
    // 读取配置
    c := config.Load().(*Config)
    fmt.Printf("Config: %s:%d\n", c.Host, c.Port)
    
    // 更新配置
    config.Store(&Config{
        Host: "0.0.0.0",
        Port: 9090,
    })
    
    c = config.Load().(*Config)
    fmt.Printf("New Config: %s:%d\n", c.Host, c.Port)
}
```

## 实战案例

### 案例 1：线程安全的缓存

```go
package main

import (
    "sync"
    "time"
)

type CacheItem struct {
    value      interface{}
    expiration time.Time
}

type SafeCache struct {
    mu    sync.RWMutex
    items map[string]*CacheItem
}

func NewSafeCache() *SafeCache {
    cache := &SafeCache{
        items: make(map[string]*CacheItem),
    }
    
    // 启动清理 goroutine
    go cache.cleanup()
    
    return cache
}

func (c *SafeCache) Set(key string, value interface{}, ttl time.Duration) {
    c.mu.Lock()
    defer c.mu.Unlock()
    
    c.items[key] = &CacheItem{
        value:      value,
        expiration: time.Now().Add(ttl),
    }
}

func (c *SafeCache) Get(key string) (interface{}, bool) {
    c.mu.RLock()
    defer c.mu.RUnlock()
    
    item, ok := c.items[key]
    if !ok {
        return nil, false
    }
    
    if time.Now().After(item.expiration) {
        return nil, false
    }
    
    return item.value, true
}

func (c *SafeCache) cleanup() {
    ticker := time.NewTicker(time.Minute)
    defer ticker.Stop()
    
    for range ticker.C {
        c.mu.Lock()
        now := time.Now()
        for key, item := range c.items {
            if now.After(item.expiration) {
                delete(c.items, key)
            }
        }
        c.mu.Unlock()
    }
}
```

### 案例 2：并发安全的计数器

```go
package main

import (
    "fmt"
    "sync"
    "sync/atomic"
)

type Metrics struct {
    requests int64
    errors   int64
    mu       sync.RWMutex
    latency  []int64
}

func (m *Metrics) IncRequests() {
    atomic.AddInt64(&m.requests, 1)
}

func (m *Metrics) IncErrors() {
    atomic.AddInt64(&m.errors, 1)
}

func (m *Metrics) AddLatency(latency int64) {
    m.mu.Lock()
    defer m.mu.Unlock()
    m.latency = append(m.latency, latency)
}

func (m *Metrics) Stats() (requests, errors int64, avgLatency float64) {
    requests = atomic.LoadInt64(&m.requests)
    errors = atomic.LoadInt64(&m.errors)
    
    m.mu.RLock()
    defer m.mu.RUnlock()
    
    if len(m.latency) > 0 {
        var sum int64
        for _, l := range m.latency {
            sum += l
        }
        avgLatency = float64(sum) / float64(len(m.latency))
    }
    
    return
}

func main() {
    metrics := &Metrics{}
    var wg sync.WaitGroup
    
    // 模拟请求
    for i := 0; i < 1000; i++ {
        wg.Add(1)
        go func() {
            defer wg.Done()
            metrics.IncRequests()
            metrics.AddLatency(100)
        }()
    }
    
    wg.Wait()
    
    requests, errors, avgLatency := metrics.Stats()
    fmt.Printf("Requests: %d, Errors: %d, Avg Latency: %.2f\n", 
        requests, errors, avgLatency)
}
```

## 最佳实践

### 1. 选择合适的同步原语

- **简单计数**：使用 atomic
- **保护共享资源**：使用 Mutex
- **读多写少**：使用 RWMutex
- **等待完成**：使用 WaitGroup
- **单次初始化**：使用 Once
- **条件等待**：使用 Cond

### 2. 避免常见错误

```go
// ❌ 复制 Mutex
type Bad struct {
    mu sync.Mutex
}
func (b Bad) Method() {}  // 值接收者

// ✅ 使用指针
func (b *Bad) MethodCorrect() {}

// ❌ 忘记解锁
func forget() {
    var mu sync.Mutex
    mu.Lock()
    // 忘记 Unlock
}

// ✅ 使用 defer
func remember() {
    var mu sync.Mutex
    mu.Lock()
    defer mu.Unlock()
}
```

### 3. 性能优化

```go
// 优先使用 atomic
var counter int64
atomic.AddInt64(&counter, 1)  // 比 Mutex 快

// 读多写少使用 RWMutex
var cache sync.Map  // 或使用 sync.Map
```
