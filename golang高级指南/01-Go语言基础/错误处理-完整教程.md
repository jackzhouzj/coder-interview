## 1. 错误处理基础

### 1.1 error接口

```go
package main

import (
    "errors"
    "fmt"
)

func main() {
    // 🔥 error是一个接口
    // type error interface {
    //     Error() string
    // }
    
    // 🔥 创建错误
    err1 := errors.New("发生了一个错误")
    fmt.Println(err1)
    
    // 🔥 格式化错误
    err2 := fmt.Errorf("用户 %s 不存在", "张三")
    fmt.Println(err2)
}
```

### 1.2 基本错误处理

```go
package main

import (
    "errors"
    "fmt"
)

func divide(a, b float64) (float64, error) {
    if b == 0 {
        return 0, errors.New("除数不能为0")
    }
    return a / b, nil
}

func main() {
    // 🔥 处理错误
    result, err := divide(10, 0)
    if err != nil {
        fmt.Println("错误:", err)
        return
    }
    fmt.Println("结果:", result)
    
    // 🔥 正常情况
    result, err = divide(10, 2)
    if err != nil {
        fmt.Println("错误:", err)
        return
    }
    fmt.Println("结果:", result)
}
```

## 2. 自定义错误

### 2.1 实现error接口

```go
package main

import "fmt"

// 🔥 自定义错误类型
type ValidationError struct {
    Field   string
    Message string
}

func (e ValidationError) Error() string {
    return fmt.Sprintf("验证失败 [%s]: %s", e.Field, e.Message)
}

func validateAge(age int) error {
    if age < 0 {
        return ValidationError{
            Field:   "age",
            Message: "年龄不能为负数",
        }
    }
    if age > 150 {
        return ValidationError{
            Field:   "age",
            Message: "年龄不能超过150",
        }
    }
    return nil
}

func main() {
    err := validateAge(-1)
    if err != nil {
        fmt.Println(err)
    }
    
    err = validateAge(200)
    if err != nil {
        fmt.Println(err)
    }
}
```

### 2.2 带状态码的错误

```go
package main

import "fmt"

// 🔥 带状态码的错误
type APIError struct {
    Code    int
    Message string
}

func (e APIError) Error() string {
    return fmt.Sprintf("API错误 [%d]: %s", e.Code, e.Message)
}

func callAPI(endpoint string) error {
    // 模拟API调用
    if endpoint == "" {
        return APIError{
            Code:    400,
            Message: "端点不能为空",
        }
    }
    return nil
}

func main() {
    err := callAPI("")
    if err != nil {
        if apiErr, ok := err.(APIError); ok {
            fmt.Printf("状态码: %d, 消息: %s\n", apiErr.Code, apiErr.Message)
        }
    }
}
```

## 3. 错误包装

### 3.1 使用fmt.Errorf包装

```go
package main

import (
    "errors"
    "fmt"
)

func readConfig() error {
    return errors.New("配置文件不存在")
}

func initialize() error {
    err := readConfig()
    if err != nil {
        // 🔥 使用%w包装错误
        return fmt.Errorf("初始化失败: %w", err)
    }
    return nil
}

func main() {
    err := initialize()
    if err != nil {
        fmt.Println(err)
        
        // 🔥 解包错误
        if errors.Is(err, errors.New("配置文件不存在")) {
            fmt.Println("配置文件问题")
        }
    }
}
```

### 3.2 多层错误包装

```go
package main

import (
    "errors"
    "fmt"
)

func dbQuery() error {
    return errors.New("连接超时")
}

func getUser(id int) error {
    err := dbQuery()
    if err != nil {
        return fmt.Errorf("查询用户失败 [id=%d]: %w", id, err)
    }
    return nil
}

func handleRequest() error {
    err := getUser(123)
    if err != nil {
        return fmt.Errorf("处理请求失败: %w", err)
    }
    return nil
}

func main() {
    err := handleRequest()
    if err != nil {
        fmt.Println("错误:", err)
        // 输出: 错误: 处理请求失败: 查询用户失败 [id=123]: 连接超时
    }
}
```

## 4. 错误检查

### 4.1 errors.Is

```go
package main

import (
    "errors"
    "fmt"
)

var (
    ErrNotFound     = errors.New("未找到")
    ErrUnauthorized = errors.New("未授权")
    ErrInternal     = errors.New("内部错误")
)

func getUser(id int) error {
    if id == 0 {
        return fmt.Errorf("用户查询失败: %w", ErrNotFound)
    }
    return nil
}

func main() {
    err := getUser(0)
    
    // 🔥 使用errors.Is检查错误
    if errors.Is(err, ErrNotFound) {
        fmt.Println("用户不存在")
    }
    
    if errors.Is(err, ErrUnauthorized) {
        fmt.Println("未授权")
    }
}
```

### 4.2 errors.As

```go
package main

import (
    "errors"
    "fmt"
)

type ValidationError struct {
    Field   string
    Message string
}

func (e ValidationError) Error() string {
    return fmt.Sprintf("%s: %s", e.Field, e.Message)
}

func validateUser(name string) error {
    if name == "" {
        return ValidationError{
            Field:   "name",
            Message: "名称不能为空",
        }
    }
    return nil
}

func main() {
    err := validateUser("")
    
    // 🔥 使用errors.As提取错误类型
    var validationErr ValidationError
    if errors.As(err, &validationErr) {
        fmt.Printf("验证错误 - 字段: %s, 消息: %s\n",
            validationErr.Field, validationErr.Message)
    }
}
```

## 5. panic和recover

### 5.1 panic基础

```go
package main

import "fmt"

func main() {
    defer fmt.Println("defer 1")
    defer fmt.Println("defer 2")
    
    // 🔥 panic会终止程序
    panic("发生了严重错误")
    
    fmt.Println("这行不会执行")
}
```

### 5.2 recover捕获panic

```go
package main

import "fmt"

func safeDivide(a, b int) (result int, err error) {
    // 🔥 使用defer和recover捕获panic
    defer func() {
        if r := recover(); r != nil {
            err = fmt.Errorf("panic: %v", r)
        }
    }()
    
    result = a / b  // 可能panic
    return
}

func main() {
    result, err := safeDivide(10, 0)
    if err != nil {
        fmt.Println("错误:", err)
    } else {
        fmt.Println("结果:", result)
    }
    
    result, err = safeDivide(10, 2)
    if err != nil {
        fmt.Println("错误:", err)
    } else {
        fmt.Println("结果:", result)
    }
}
```

### 5.3 何时使用panic

```go
package main

import "fmt"

// ✅ 适合使用panic的场景：
// 1. 程序初始化失败
func init() {
    // 如果配置文件不存在，程序无法运行
    // panic("配置文件不存在")
}

// 2. 不可恢复的错误
func mustConnect() {
    // 如果无法连接数据库，程序无法继续
    // panic("无法连接数据库")
}

// ❌ 不应该使用panic的场景：
// 1. 正常的错误处理
func divide(a, b int) (int, error) {
    if b == 0 {
        // ❌ 不要panic
        // panic("除数不能为0")
        
        // ✅ 返回错误
        return 0, fmt.Errorf("除数不能为0")
    }
    return a / b, nil
}

// 2. 可以恢复的错误
func processFile(filename string) error {
    // ❌ 不要panic
    // if filename == "" {
    //     panic("文件名为空")
    // }
    
    // ✅ 返回错误
    if filename == "" {
        return fmt.Errorf("文件名为空")
    }
    return nil
}
```

## 6. 错误处理模式

### 6.1 哨兵错误

```go
package main

import (
    "errors"
    "fmt"
)

// 🔥 定义哨兵错误
var (
    ErrNotFound     = errors.New("未找到")
    ErrUnauthorized = errors.New("未授权")
    ErrInvalidInput = errors.New("无效输入")
)

func getUser(id int) error {
    if id <= 0 {
        return ErrInvalidInput
    }
    if id == 999 {
        return ErrNotFound
    }
    return nil
}

func main() {
    err := getUser(999)
    
    // 🔥 使用哨兵错误
    if errors.Is(err, ErrNotFound) {
        fmt.Println("用户不存在")
    }
}
```

### 6.2 错误类型

```go
package main

import "fmt"

// 🔥 定义错误类型
type NotFoundError struct {
    Resource string
    ID       int
}

func (e NotFoundError) Error() string {
    return fmt.Sprintf("%s [id=%d] 未找到", e.Resource, e.ID)
}

type ValidationError struct {
    Field   string
    Message string
}

func (e ValidationError) Error() string {
    return fmt.Sprintf("验证失败 [%s]: %s", e.Field, e.Message)
}

func getUser(id int) error {
    if id <= 0 {
        return ValidationError{
            Field:   "id",
            Message: "ID必须大于0",
        }
    }
    if id == 999 {
        return NotFoundError{
            Resource: "User",
            ID:       id,
        }
    }
    return nil
}

func main() {
    err := getUser(999)
    
    // 🔥 类型断言
    switch e := err.(type) {
    case NotFoundError:
        fmt.Printf("资源不存在: %s, ID: %d\n", e.Resource, e.ID)
    case ValidationError:
        fmt.Printf("验证失败: %s - %s\n", e.Field, e.Message)
    default:
        fmt.Println("未知错误:", err)
    }
}
```

### 6.3 不透明错误

```go
package main

import (
    "errors"
    "fmt"
)

// 🔥 不暴露错误的内部结构
type userService struct{}

func (s *userService) GetUser(id int) error {
    // 内部错误不暴露给调用者
    if id <= 0 {
        return errors.New("无效的用户ID")
    }
    return nil
}

func main() {
    service := &userService{}
    err := service.GetUser(0)
    if err != nil {
        // 调用者只知道发生了错误，不知道具体原因
        fmt.Println("获取用户失败:", err)
    }
}
```

## 7. 第三方错误库

### 7.1 pkg/errors

```go
package main

import (
    "fmt"
    
    "github.com/pkg/errors"
)

func readConfig() error {
    return errors.New("配置文件不存在")
}

func initialize() error {
    err := readConfig()
    if err != nil {
        // 🔥 添加堆栈信息
        return errors.Wrap(err, "初始化失败")
    }
    return nil
}

func main() {
    err := initialize()
    if err != nil {
        // 🔥 打印堆栈信息
        fmt.Printf("%+v\n", err)
    }
}
```

## 8. 最佳实践

### 8.1 错误处理原则

```go
// ✅ 原则1：总是处理错误
result, err := someFunction()
if err != nil {
    // 处理错误
    return err
}

// ✅ 原则2：不要忽略错误
// ❌ 错误
// someFunction()

// ✅ 正确
_, err := someFunction()
if err != nil {
    log.Println(err)
}

// ✅ 原则3：错误只处理一次
// ❌ 错误：既记录又返回
if err != nil {
    log.Println(err)
    return err  // 调用者可能再次记录
}

// ✅ 正确：要么记录，要么返回
if err != nil {
    return fmt.Errorf("操作失败: %w", err)
}

// ✅ 原则4：添加上下文信息
if err != nil {
    return fmt.Errorf("处理用户 %d 失败: %w", userID, err)
}
```

### 8.2 错误信息规范

```go
// ✅ 错误信息应该：
// 1. 小写开头（除非是专有名词）
errors.New("配置文件不存在")

// 2. 不要以标点符号结尾
errors.New("无法连接数据库")

// 3. 包含足够的上下文
fmt.Errorf("用户 %d 不存在", userID)

// 4. 使用动词短语
errors.New("无法打开文件")

// ❌ 不好的错误信息
errors.New("Error!")
errors.New("Something went wrong.")
errors.New("失败")
```

### 8.3 错误处理模式

```go
package main

import (
    "errors"
    "fmt"
)

// ✅ 模式1：提前返回
func process(id int) error {
    if id <= 0 {
        return errors.New("无效的ID")
    }
    
    user, err := getUser(id)
    if err != nil {
        return fmt.Errorf("获取用户失败: %w", err)
    }
    
    if err := validateUser(user); err != nil {
        return fmt.Errorf("验证用户失败: %w", err)
    }
    
    return nil
}

// ✅ 模式2：错误变量
func processWithErrorVar() (err error) {
    defer func() {
        if err != nil {
            err = fmt.Errorf("处理失败: %w", err)
        }
    }()
    
    // 处理逻辑
    return nil
}

// ✅ 模式3：错误聚合
type MultiError struct {
    Errors []error
}

func (m MultiError) Error() string {
    return fmt.Sprintf("发生 %d 个错误", len(m.Errors))
}

func processMultiple(ids []int) error {
    var errs []error
    
    for _, id := range ids {
        if err := process(id); err != nil {
            errs = append(errs, err)
        }
    }
    
    if len(errs) > 0 {
        return MultiError{Errors: errs}
    }
    
    return nil
}

func getUser(id int) (interface{}, error) {
    return nil, nil
}

func validateUser(user interface{}) error {
    return nil
}
```
