## 1. 函数基础

### 1.1 函数定义

```go
package main

import "fmt"

// 🔥 基本函数定义
func sayHello() {
    fmt.Println("Hello, World!")
}

// 🔥 带参数的函数
func greet(name string) {
    fmt.Printf("Hello, %s!\n", name)
}

// 🔥 带返回值的函数
func add(a, b int) int {
    return a + b
}

// 🔥 多个参数相同类型可以简写
func multiply(a, b, c int) int {
    return a * b * c
}

func main() {
    sayHello()
    greet("张三")
    result := add(1, 2)
    fmt.Println("1 + 2 =", result)
    fmt.Println("2 * 3 * 4 =", multiply(2, 3, 4))
}
```

### 1.2 函数作为值

```go
package main

import "fmt"

func add(a, b int) int {
    return a + b
}

func subtract(a, b int) int {
    return a - b
}

func main() {
    // 🔥 函数赋值给变量
    var f func(int, int) int
    f = add
    fmt.Println("f(1, 2) =", f(1, 2))
    
    f = subtract
    fmt.Println("f(5, 3) =", f(5, 3))
    
    // 🔥 函数作为参数
    calculate := func(a, b int, op func(int, int) int) int {
        return op(a, b)
    }
    
    result1 := calculate(10, 5, add)
    result2 := calculate(10, 5, subtract)
    fmt.Println("10 + 5 =", result1)
    fmt.Println("10 - 5 =", result2)
}
```

## 2. 函数参数

### 2.1 值传递和引用传递

```go
package main

import "fmt"

// 🔥 值传递：传递副本
func modifyValue(x int) {
    x = 100
    fmt.Println("函数内:", x)
}

// 🔥 指针传递：传递地址
func modifyPointer(x *int) {
    *x = 100
    fmt.Println("函数内:", *x)
}

// 🔥 切片传递：传递引用
func modifySlice(s []int) {
    s[0] = 100
}

// 🔥 Map传递：传递引用
func modifyMap(m map[string]int) {
    m["key"] = 100
}

func main() {
    // 值传递
    a := 10
    modifyValue(a)
    fmt.Println("函数外:", a)  // 10（未改变）
    
    // 指针传递
    b := 10
    modifyPointer(&b)
    fmt.Println("函数外:", b)  // 100（已改变）
    
    // 切片传递
    slice := []int{1, 2, 3}
    modifySlice(slice)
    fmt.Println("切片:", slice)  // [100 2 3]（已改变）
    
    // Map传递
    m := make(map[string]int)
    modifyMap(m)
    fmt.Println("Map:", m)  // map[key:100]（已改变）
}
```

### 2.2 可变参数

```go
package main

import "fmt"

// 🔥 可变参数函数
func sum(nums ...int) int {
    total := 0
    for _, num := range nums {
        total += num
    }
    return total
}

// 🔥 可变参数必须是最后一个参数
func printf(format string, args ...interface{}) {
    fmt.Printf(format, args...)
}

func main() {
    // 调用可变参数函数
    fmt.Println(sum(1, 2, 3))
    fmt.Println(sum(1, 2, 3, 4, 5))
    
    // 传递切片
    nums := []int{1, 2, 3, 4, 5}
    fmt.Println(sum(nums...))  // 使用...展开切片
    
    // 使用可变参数
    printf("Name: %s, Age: %d\n", "张三", 25)
}
```

## 3. 函数返回值

### 3.1 多返回值

```go
package main

import (
    "fmt"
    "errors"
)

// 🔥 多返回值
func divide(a, b float64) (float64, error) {
    if b == 0 {
        return 0, errors.New("除数不能为0")
    }
    return a / b, nil
}

// 🔥 命名返回值
func swap(a, b int) (x, y int) {
    x = b
    y = a
    return  // 裸返回
}

// 🔥 忽略返回值
func getValues() (int, int, int) {
    return 1, 2, 3
}

func main() {
    // 接收多个返回值
    result, err := divide(10, 2)
    if err != nil {
        fmt.Println("错误:", err)
    } else {
        fmt.Println("结果:", result)
    }
    
    // 使用命名返回值
    x, y := swap(1, 2)
    fmt.Printf("swap(1, 2) = %d, %d\n", x, y)
    
    // 忽略部分返回值
    a, _, c := getValues()
    fmt.Printf("a=%d, c=%d\n", a, c)
}
```

### 3.2 返回函数

```go
package main

import "fmt"

// 🔥 返回函数
func makeAdder(x int) func(int) int {
    return func(y int) int {
        return x + y
    }
}

// 🔥 返回多个函数
func makeOperators() (func(int, int) int, func(int, int) int) {
    add := func(a, b int) int { return a + b }
    sub := func(a, b int) int { return a - b }
    return add, sub
}

func main() {
    // 使用返回的函数
    add5 := makeAdder(5)
    fmt.Println(add5(3))  // 8
    fmt.Println(add5(10)) // 15
    
    // 使用返回的多个函数
    add, sub := makeOperators()
    fmt.Println(add(10, 5))  // 15
    fmt.Println(sub(10, 5))  // 5
}
```

## 4. 匿名函数和闭包

### 4.1 匿名函数

```go
package main

import "fmt"

func main() {
    // 🔥 定义并立即调用
    func() {
        fmt.Println("匿名函数")
    }()
    
    // 🔥 带参数的匿名函数
    func(name string) {
        fmt.Println("Hello,", name)
    }("张三")
    
    // 🔥 赋值给变量
    add := func(a, b int) int {
        return a + b
    }
    fmt.Println(add(1, 2))
    
    // 🔥 作为参数传递
    calculate := func(a, b int, op func(int, int) int) int {
        return op(a, b)
    }
    
    result := calculate(10, 5, func(a, b int) int {
        return a * b
    })
    fmt.Println("10 * 5 =", result)
}
```

### 4.2 闭包

```go
package main

import "fmt"

// 🔥 闭包：函数引用外部变量
func counter() func() int {
    count := 0
    return func() int {
        count++
        return count
    }
}

// 🔥 闭包陷阱
func createFunctions() []func() {
    var funcs []func()
    
    // ❌ 错误：所有函数共享同一个i
    for i := 0; i < 5; i++ {
        funcs = append(funcs, func() {
            fmt.Println(i)  // 都会打印5
        })
    }
    
    return funcs
}

// ✅ 正确：每个函数有自己的i
func createFunctionsCorrect() []func() {
    var funcs []func()
    
    for i := 0; i < 5; i++ {
        i := i  // 创建新变量
        funcs = append(funcs, func() {
            fmt.Println(i)
        })
    }
    
    return funcs
}

func main() {
    // 使用闭包
    c1 := counter()
    fmt.Println(c1())  // 1
    fmt.Println(c1())  // 2
    fmt.Println(c1())  // 3
    
    c2 := counter()
    fmt.Println(c2())  // 1（新的计数器）
    
    // 闭包陷阱
    funcs := createFunctions()
    for _, f := range funcs {
        f()  // 都打印5
    }
    
    // 正确的闭包
    funcs2 := createFunctionsCorrect()
    for _, f := range funcs2 {
        f()  // 打印0-4
    }
}
```

### 4.3 闭包应用

```go
package main

import "fmt"

// 🔥 斐波那契数列生成器
func fibonacci() func() int {
    a, b := 0, 1
    return func() int {
        a, b = b, a+b
        return a
    }
}

// 🔥 缓存函数结果
func memoize(f func(int) int) func(int) int {
    cache := make(map[int]int)
    return func(n int) int {
        if v, ok := cache[n]; ok {
            fmt.Println("从缓存获取:", n)
            return v
        }
        result := f(n)
        cache[n] = result
        return result
    }
}

func main() {
    // 斐波那契数列
    fib := fibonacci()
    for i := 0; i < 10; i++ {
        fmt.Print(fib(), " ")
    }
    fmt.Println()
    
    // 缓存函数
    expensive := func(n int) int {
        fmt.Println("计算:", n)
        return n * n
    }
    
    cached := memoize(expensive)
    fmt.Println(cached(5))  // 计算: 5
    fmt.Println(cached(5))  // 从缓存获取: 5
    fmt.Println(cached(10)) // 计算: 10
}
```

## 5. 方法

### 5.1 方法定义

```go
package main

import "fmt"

type Rectangle struct {
    Width  float64
    Height float64
}

// 🔥 值接收者方法
func (r Rectangle) Area() float64 {
    return r.Width * r.Height
}

// 🔥 指针接收者方法
func (r *Rectangle) Scale(factor float64) {
    r.Width *= factor
    r.Height *= factor
}

func main() {
    rect := Rectangle{Width: 10, Height: 5}
    
    // 调用值接收者方法
    area := rect.Area()
    fmt.Println("面积:", area)
    
    // 调用指针接收者方法
    rect.Scale(2)
    fmt.Println("缩放后:", rect)
    fmt.Println("新面积:", rect.Area())
}
```

### 5.2 值接收者 vs 指针接收者

```go
package main

import "fmt"

type Counter struct {
    Count int
}

// 🔥 值接收者：不会修改原对象
func (c Counter) IncrementValue() {
    c.Count++
}

// 🔥 指针接收者：会修改原对象
func (c *Counter) IncrementPointer() {
    c.Count++
}

func main() {
    counter := Counter{Count: 0}
    
    // 值接收者
    counter.IncrementValue()
    fmt.Println("值接收者后:", counter.Count)  // 0（未改变）
    
    // 指针接收者
    counter.IncrementPointer()
    fmt.Println("指针接收者后:", counter.Count)  // 1（已改变）
}
```

### 5.3 方法集

```go
package main

import "fmt"

type Person struct {
    Name string
}

// 值接收者方法
func (p Person) GetName() string {
    return p.Name
}

// 指针接收者方法
func (p *Person) SetName(name string) {
    p.Name = name
}

func main() {
    // 🔥 值类型可以调用值接收者和指针接收者方法
    p1 := Person{Name: "张三"}
    fmt.Println(p1.GetName())
    p1.SetName("李四")  // Go自动转换为(&p1).SetName("李四")
    fmt.Println(p1.GetName())
    
    // 🔥 指针类型可以调用值接收者和指针接收者方法
    p2 := &Person{Name: "王五"}
    fmt.Println(p2.GetName())  // Go自动转换为(*p2).GetName()
    p2.SetName("赵六")
    fmt.Println(p2.GetName())
}
```

### 5.4 方法链

```go
package main

import "fmt"

type Builder struct {
    str string
}

// 🔥 返回指针实现方法链
func (b *Builder) Append(s string) *Builder {
    b.str += s
    return b
}

func (b *Builder) AppendLine(s string) *Builder {
    b.str += s + "\n"
    return b
}

func (b *Builder) String() string {
    return b.str
}

func main() {
    builder := &Builder{}
    result := builder.
        Append("Hello").
        Append(" ").
        Append("World").
        AppendLine("!").
        Append("Go is awesome").
        String()
    
    fmt.Println(result)
}
```

## 6. defer语句

### 6.1 defer基础

```go
package main

import "fmt"

func main() {
    // 🔥 defer延迟执行
    defer fmt.Println("1")
    defer fmt.Println("2")
    defer fmt.Println("3")
    
    fmt.Println("开始")
    // 输出顺序：开始 3 2 1（后进先出）
}
```

### 6.2 defer应用场景

```go
package main

import (
    "fmt"
    "os"
)

// 🔥 资源清理
func readFile(filename string) error {
    file, err := os.Open(filename)
    if err != nil {
        return err
    }
    defer file.Close()  // 确保文件被关闭
    
    // 读取文件...
    return nil
}

// 🔥 解锁
func processWithLock(mu *sync.Mutex) {
    mu.Lock()
    defer mu.Unlock()  // 确保解锁
    
    // 处理逻辑...
}

// 🔥 记录函数执行时间
func measureTime(name string) func() {
    start := time.Now()
    return func() {
        fmt.Printf("%s 执行时间: %v\n", name, time.Since(start))
    }
}

func slowFunction() {
    defer measureTime("slowFunction")()
    time.Sleep(time.Second)
}
```

### 6.3 defer陷阱

```go
package main

import "fmt"

func main() {
    // ⚠️ 陷阱1：defer参数立即求值
    x := 1
    defer fmt.Println("defer:", x)  // 打印1，不是2
    x = 2
    fmt.Println("main:", x)
    
    // ⚠️ 陷阱2：defer和闭包
    y := 1
    defer func() {
        fmt.Println("defer closure:", y)  // 打印2
    }()
    y = 2
    fmt.Println("main:", y)
    
    // ⚠️ 陷阱3：defer和返回值
    fmt.Println("result:", deferReturn())  // 2
}

func deferReturn() (result int) {
    defer func() {
        result++  // 修改返回值
    }()
    return 1
}
```

## 7. panic和recover

### 7.1 panic

```go
package main

import "fmt"

func main() {
    // 🔥 panic会终止程序
    defer fmt.Println("defer 1")
    defer fmt.Println("defer 2")
    
    panic("发生错误")
    
    fmt.Println("这行不会执行")
}
```

### 7.2 recover

```go
package main

import "fmt"

func safeDivide(a, b int) (result int, err error) {
    // 🔥 使用recover捕获panic
    defer func() {
        if r := recover(); r != nil {
            err = fmt.Errorf("panic: %v", r)
        }
    }()
    
    if b == 0 {
        panic("除数不能为0")
    }
    
    result = a / b
    return
}

func main() {
    result, err := safeDivide(10, 0)
    if err != nil {
        fmt.Println("错误:", err)
    } else {
        fmt.Println("结果:", result)
    }
    
    result, err = safeDivide(10, 2)
    if err != nil {
        fmt.Println("错误:", err)
    } else {
        fmt.Println("结果:", result)
    }
}
```

### 7.3 panic和recover最佳实践

```go
package main

import (
    "fmt"
    "log"
)

// 🔥 在Goroutine中使用recover
func safeGoroutine(f func()) {
    go func() {
        defer func() {
            if r := recover(); r != nil {
                log.Printf("Goroutine panic: %v", r)
            }
        }()
        f()
    }()
}

// 🔥 Web服务器中间件
func recoverMiddleware(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        defer func() {
            if err := recover(); err != nil {
                log.Printf("panic: %v", err)
                http.Error(w, "Internal Server Error", 500)
            }
        }()
        next.ServeHTTP(w, r)
    })
}

func main() {
    safeGoroutine(func() {
        panic("Goroutine panic")
    })
    
    time.Sleep(time.Second)
    fmt.Println("程序继续运行")
}
```

## 8. 最佳实践

### 8.1 函数设计原则

```go
// ✅ 单一职责
func calculateTotal(items []Item) float64 {
    // 只负责计算总价
}

// ✅ 函数名清晰
func getUserByID(id int) (*User, error) {
    // 函数名表明了功能
}

// ✅ 参数不要太多（建议不超过3个）
// ❌ 不好
func createUser(name, email, phone, address, city, country string, age int) {}

// ✅ 好：使用结构体
type UserInfo struct {
    Name    string
    Email   string
    Phone   string
    Address string
    City    string
    Country string
    Age     int
}

func createUser(info UserInfo) {}
```

### 8.2 错误处理

```go
// ✅ 返回错误而不是panic
func divide(a, b float64) (float64, error) {
    if b == 0 {
        return 0, errors.New("除数不能为0")
    }
    return a / b, nil
}

// ✅ 使用命名返回值简化错误处理
func readConfig() (config Config, err error) {
    defer func() {
        if err != nil {
            log.Printf("读取配置失败: %v", err)
        }
    }()
    
    // 读取配置...
    return
}
```

### 8.3 方法接收者选择

```go
// ✅ 使用指针接收者的情况：
// 1. 需要修改接收者
// 2. 接收者是大结构体
// 3. 接收者包含sync.Mutex等不可复制的字段

type LargeStruct struct {
    // 很多字段...
}

func (l *LargeStruct) Method() {
    // 使用指针避免复制
}

// ✅ 使用值接收者的情况：
// 1. 接收者是小结构体
// 2. 接收者是基本类型
// 3. 不需要修改接收者

type Point struct {
    X, Y int
}

func (p Point) Distance() float64 {
    return math.Sqrt(float64(p.X*p.X + p.Y*p.Y))
}
```
