## 1. 接口基础

### 1.1 什么是接口

接口是一组方法签名的集合，定义了对象的行为。

```go
package main

import "fmt"

// 🔥 定义接口
type Shape interface {
    Area() float64
    Perimeter() float64
}

// 🔥 实现接口
type Rectangle struct {
    Width  float64
    Height float64
}

func (r Rectangle) Area() float64 {
    return r.Width * r.Height
}

func (r Rectangle) Perimeter() float64 {
    return 2 * (r.Width + r.Height)
}

type Circle struct {
    Radius float64
}

func (c Circle) Area() float64 {
    return 3.14 * c.Radius * c.Radius
}

func (c Circle) Perimeter() float64 {
    return 2 * 3.14 * c.Radius
}

func main() {
    var s Shape
    
    s = Rectangle{Width: 10, Height: 5}
    fmt.Printf("矩形面积: %.2f\n", s.Area())
    
    s = Circle{Radius: 5}
    fmt.Printf("圆形面积: %.2f\n", s.Area())
}
```

### 1.2 接口的隐式实现

```go
package main

import "fmt"

// 🔥 Go的接口是隐式实现的
type Writer interface {
    Write([]byte) (int, error)
}

type FileWriter struct {
    filename string
}

// 🔥 只要实现了接口的所有方法，就实现了该接口
func (f FileWriter) Write(data []byte) (int, error) {
    fmt.Printf("写入文件 %s: %s\n", f.filename, string(data))
    return len(data), nil
}

func main() {
    var w Writer
    w = FileWriter{filename: "test.txt"}
    w.Write([]byte("Hello, World!"))
}
```

## 2. 接口实现

### 2.1 值接收者和指针接收者

```go
package main

import "fmt"

type Counter interface {
    Increment()
    Value() int
}

// 🔥 值接收者实现
type ValueCounter struct {
    count int
}

func (c ValueCounter) Increment() {
    c.count++  // 不会修改原值
}

func (c ValueCounter) Value() int {
    return c.count
}

// 🔥 指针接收者实现
type PointerCounter struct {
    count int
}

func (c *PointerCounter) Increment() {
    c.count++  // 会修改原值
}

func (c *PointerCounter) Value() int {
    return c.count
}

func main() {
    // 值接收者
    var c1 Counter = ValueCounter{}
    c1.Increment()
    fmt.Println("ValueCounter:", c1.Value())  // 0
    
    // 指针接收者
    var c2 Counter = &PointerCounter{}
    c2.Increment()
    fmt.Println("PointerCounter:", c2.Value())  // 1
}
```

### 2.2 接口嵌套

```go
package main

import "fmt"

// 🔥 基础接口
type Reader interface {
    Read(p []byte) (n int, err error)
}

type Writer interface {
    Write(p []byte) (n int, err error)
}

// 🔥 接口嵌套
type ReadWriter interface {
    Reader
    Writer
}

type File struct {
    name string
}

func (f File) Read(p []byte) (n int, err error) {
    fmt.Println("Reading from", f.name)
    return 0, nil
}

func (f File) Write(p []byte) (n int, err error) {
    fmt.Println("Writing to", f.name)
    return len(p), nil
}

func main() {
    var rw ReadWriter = File{name: "test.txt"}
    rw.Read(nil)
    rw.Write([]byte("data"))
}
```

## 3. 空接口

### 3.1 空接口基础

```go
package main

import "fmt"

func main() {
    // 🔥 空接口可以接收任何类型
    var i interface{}
    
    i = 42
    fmt.Printf("Type: %T, Value: %v\n", i, i)
    
    i = "hello"
    fmt.Printf("Type: %T, Value: %v\n", i, i)
    
    i = []int{1, 2, 3}
    fmt.Printf("Type: %T, Value: %v\n", i, i)
    
    i = struct{ Name string }{"张三"}
    fmt.Printf("Type: %T, Value: %v\n", i, i)
}
```

### 3.2 空接口切片

```go
package main

import "fmt"

func printAll(values ...interface{}) {
    for i, v := range values {
        fmt.Printf("[%d] Type: %T, Value: %v\n", i, v, v)
    }
}

func main() {
    // 🔥 可变参数接收任意类型
    printAll(1, "hello", 3.14, true, []int{1, 2, 3})
}
```

## 4. 类型断言

### 4.1 基本类型断言

```go
package main

import "fmt"

func main() {
    var i interface{} = "hello"
    
    // 🔥 类型断言
    s := i.(string)
    fmt.Println(s)  // hello
    
    // 🔥 安全的类型断言
    s, ok := i.(string)
    if ok {
        fmt.Println("是字符串:", s)
    }
    
    // ⚠️ 错误的类型断言会panic
    // n := i.(int)  // panic!
    
    // ✅ 正确做法
    n, ok := i.(int)
    if ok {
        fmt.Println("是整数:", n)
    } else {
        fmt.Println("不是整数")
    }
}
```

### 4.2 类型switch

```go
package main

import "fmt"

func describe(i interface{}) {
    // 🔥 类型switch
    switch v := i.(type) {
    case int:
        fmt.Printf("整数: %d\n", v)
    case string:
        fmt.Printf("字符串: %s\n", v)
    case bool:
        fmt.Printf("布尔值: %t\n", v)
    case []int:
        fmt.Printf("整数切片: %v\n", v)
    case map[string]int:
        fmt.Printf("Map: %v\n", v)
    case nil:
        fmt.Println("nil值")
    default:
        fmt.Printf("未知类型: %T\n", v)
    }
}

func main() {
    describe(42)
    describe("hello")
    describe(true)
    describe([]int{1, 2, 3})
    describe(map[string]int{"a": 1})
    describe(nil)
    describe(3.14)
}
```

## 5. 接口组合

### 5.1 标准库接口

```go
package main

import (
    "fmt"
    "io"
    "strings"
)

func main() {
    // 🔥 io.Reader接口
    r := strings.NewReader("Hello, World!")
    buf := make([]byte, 8)
    
    for {
        n, err := r.Read(buf)
        if err == io.EOF {
            break
        }
        fmt.Printf("读取 %d 字节: %s\n", n, buf[:n])
    }
}
```

### 5.2 自定义接口组合

```go
package main

import "fmt"

// 🔥 基础接口
type Saver interface {
    Save() error
}

type Loader interface {
    Load() error
}

type Validator interface {
    Validate() error
}

// 🔥 组合接口
type Repository interface {
    Saver
    Loader
    Validator
}

type User struct {
    ID   int
    Name string
}

func (u User) Save() error {
    fmt.Println("保存用户:", u.Name)
    return nil
}

func (u User) Load() error {
    fmt.Println("加载用户:", u.Name)
    return nil
}

func (u User) Validate() error {
    if u.Name == "" {
        return fmt.Errorf("用户名不能为空")
    }
    return nil
}

func main() {
    var repo Repository = User{ID: 1, Name: "张三"}
    repo.Validate()
    repo.Save()
    repo.Load()
}
```

## 6. 反射基础

### 6.1 反射的Type和Value

```go
package main

import (
    "fmt"
    "reflect"
)

func main() {
    var x float64 = 3.14
    
    // 🔥 获取Type
    t := reflect.TypeOf(x)
    fmt.Println("Type:", t)
    fmt.Println("Kind:", t.Kind())
    
    // 🔥 获取Value
    v := reflect.ValueOf(x)
    fmt.Println("Value:", v)
    fmt.Println("Type:", v.Type())
    fmt.Println("Kind:", v.Kind())
    fmt.Println("Float:", v.Float())
}
```

### 6.2 反射修改值

```go
package main

import (
    "fmt"
    "reflect"
)

func main() {
    var x float64 = 3.14
    
    // ❌ 不能直接修改
    // v := reflect.ValueOf(x)
    // v.SetFloat(7.1)  // panic!
    
    // ✅ 使用指针
    v := reflect.ValueOf(&x)
    fmt.Println("CanSet:", v.CanSet())  // false
    
    v = v.Elem()
    fmt.Println("CanSet:", v.CanSet())  // true
    
    v.SetFloat(7.1)
    fmt.Println("x =", x)  // 7.1
}
```

### 6.3 反射结构体

```go
package main

import (
    "fmt"
    "reflect"
)

type User struct {
    ID   int    `json:"id"`
    Name string `json:"name"`
    Age  int    `json:"age"`
}

func main() {
    user := User{ID: 1, Name: "张三", Age: 25}
    
    // 🔥 获取类型信息
    t := reflect.TypeOf(user)
    fmt.Println("Type:", t.Name())
    fmt.Println("Kind:", t.Kind())
    fmt.Println("NumField:", t.NumField())
    
    // 🔥 遍历字段
    for i := 0; i < t.NumField(); i++ {
        field := t.Field(i)
        fmt.Printf("字段%d: %s, 类型: %s, Tag: %s\n",
            i, field.Name, field.Type, field.Tag.Get("json"))
    }
    
    // 🔥 获取值信息
    v := reflect.ValueOf(user)
    for i := 0; i < v.NumField(); i++ {
        field := v.Field(i)
        fmt.Printf("%s = %v\n", t.Field(i).Name, field.Interface())
    }
}
```

## 7. 反射应用

### 7.1 通用打印函数

```go
package main

import (
    "fmt"
    "reflect"
)

func Print(x interface{}) {
    v := reflect.ValueOf(x)
    t := v.Type()
    
    fmt.Printf("Type: %s\n", t)
    
    switch v.Kind() {
    case reflect.Invalid:
        fmt.Println("Invalid")
    case reflect.Int, reflect.Int8, reflect.Int16, reflect.Int32, reflect.Int64:
        fmt.Printf("Int: %d\n", v.Int())
    case reflect.Uint, reflect.Uint8, reflect.Uint16, reflect.Uint32, reflect.Uint64:
        fmt.Printf("Uint: %d\n", v.Uint())
    case reflect.Float32, reflect.Float64:
        fmt.Printf("Float: %f\n", v.Float())
    case reflect.String:
        fmt.Printf("String: %s\n", v.String())
    case reflect.Ptr:
        fmt.Println("Pointer to", v.Elem().Type())
        Print(v.Elem().Interface())
    case reflect.Struct:
        fmt.Println("Struct:")
        for i := 0; i < v.NumField(); i++ {
            fmt.Printf("  %s: %v\n", t.Field(i).Name, v.Field(i).Interface())
        }
    case reflect.Slice, reflect.Array:
        fmt.Printf("Slice/Array of %s:\n", v.Type().Elem())
        for i := 0; i < v.Len(); i++ {
            fmt.Printf("  [%d]: %v\n", i, v.Index(i).Interface())
        }
    case reflect.Map:
        fmt.Println("Map:")
        for _, key := range v.MapKeys() {
            fmt.Printf("  %v: %v\n", key.Interface(), v.MapIndex(key).Interface())
        }
    default:
        fmt.Printf("Unknown kind: %s\n", v.Kind())
    }
}

func main() {
    Print(42)
    Print("hello")
    Print([]int{1, 2, 3})
    Print(map[string]int{"a": 1, "b": 2})
    
    type Person struct {
        Name string
        Age  int
    }
    Print(Person{Name: "张三", Age: 25})
}
```

### 7.2 结构体标签解析

```go
package main

import (
    "fmt"
    "reflect"
)

type User struct {
    ID       int    `json:"id" db:"user_id" validate:"required"`
    Username string `json:"username" db:"username" validate:"required,min=3"`
    Email    string `json:"email" db:"email" validate:"required,email"`
    Age      int    `json:"age" db:"age" validate:"gte=0,lte=150"`
}

func ParseTags(obj interface{}) {
    t := reflect.TypeOf(obj)
    
    if t.Kind() == reflect.Ptr {
        t = t.Elem()
    }
    
    if t.Kind() != reflect.Struct {
        fmt.Println("不是结构体")
        return
    }
    
    fmt.Printf("结构体: %s\n", t.Name())
    
    for i := 0; i < t.NumField(); i++ {
        field := t.Field(i)
        fmt.Printf("\n字段: %s\n", field.Name)
        fmt.Printf("  json: %s\n", field.Tag.Get("json"))
        fmt.Printf("  db: %s\n", field.Tag.Get("db"))
        fmt.Printf("  validate: %s\n", field.Tag.Get("validate"))
    }
}

func main() {
    user := User{}
    ParseTags(user)
}
```

### 7.3 动态调用方法

```go
package main

import (
    "fmt"
    "reflect"
)

type Calculator struct{}

func (c Calculator) Add(a, b int) int {
    return a + b
}

func (c Calculator) Multiply(a, b int) int {
    return a * b
}

func CallMethod(obj interface{}, methodName string, args ...interface{}) []interface{} {
    v := reflect.ValueOf(obj)
    method := v.MethodByName(methodName)
    
    if !method.IsValid() {
        fmt.Printf("方法 %s 不存在\n", methodName)
        return nil
    }
    
    // 🔥 准备参数
    in := make([]reflect.Value, len(args))
    for i, arg := range args {
        in[i] = reflect.ValueOf(arg)
    }
    
    // 🔥 调用方法
    results := method.Call(in)
    
    // 🔥 转换结果
    out := make([]interface{}, len(results))
    for i, result := range results {
        out[i] = result.Interface()
    }
    
    return out
}

func main() {
    calc := Calculator{}
    
    result := CallMethod(calc, "Add", 10, 20)
    fmt.Println("Add:", result[0])
    
    result = CallMethod(calc, "Multiply", 10, 20)
    fmt.Println("Multiply:", result[0])
}
```

## 8. 最佳实践

### 8.1 接口设计原则

```go
// ✅ 接口应该小而精
type Reader interface {
    Read(p []byte) (n int, err error)
}

// ✅ 接口应该描述行为，不是数据
type Saver interface {
    Save() error
}

// ✅ 接受接口，返回具体类型
func ProcessData(r io.Reader) *Result {
    // ...
    return &Result{}
}

// ❌ 不要返回接口
// func ProcessData(r io.Reader) io.Reader {
//     // ...
// }

// ✅ 使用标准库接口
// io.Reader, io.Writer, fmt.Stringer等
```

### 8.2 反射使用建议

```go
// ⚠️ 反射的缺点：
// 1. 性能开销大
// 2. 编译时无法检查类型
// 3. 代码可读性差

// ✅ 何时使用反射：
// 1. 编写通用库（如JSON序列化）
// 2. 处理未知类型
// 3. 实现ORM框架

// ❌ 避免过度使用反射
// 能用接口就用接口，不要用反射

// ✅ 使用类型断言代替反射
func process(v interface{}) {
    // ❌ 使用反射
    // val := reflect.ValueOf(v)
    // if val.Kind() == reflect.String {
    //     s := val.String()
    // }
    
    // ✅ 使用类型断言
    if s, ok := v.(string); ok {
        fmt.Println(s)
    }
}
```

### 8.3 性能对比

```go
package main

import (
    "fmt"
    "reflect"
    "testing"
)

type User struct {
    Name string
    Age  int
}

// 直接访问
func DirectAccess(u User) string {
    return u.Name
}

// 反射访问
func ReflectAccess(u User) string {
    v := reflect.ValueOf(u)
    return v.FieldByName("Name").String()
}

func BenchmarkDirectAccess(b *testing.B) {
    u := User{Name: "张三", Age: 25}
    for i := 0; i < b.N; i++ {
        DirectAccess(u)
    }
}

func BenchmarkReflectAccess(b *testing.B) {
    u := User{Name: "张三", Age: 25}
    for i := 0; i < b.N; i++ {
        ReflectAccess(u)
    }
}

// 结果：直接访问比反射快100倍以上
```
