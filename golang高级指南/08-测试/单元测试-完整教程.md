## 1. 测试基础

### 1.1 第一个测试

```go
// math.go
package math

func Add(a, b int) int {
    return a + b
}

func Subtract(a, b int) int {
    return a - b
}
```

```go
// math_test.go
package math

import "testing"

// 🔥 测试函数必须以Test开头
func TestAdd(t *testing.T) {
    result := Add(2, 3)
    expected := 5
    
    if result != expected {
        t.Errorf("Add(2, 3) = %d; want %d", result, expected)
    }
}

func TestSubtract(t *testing.T) {
    result := Subtract(5, 3)
    expected := 2
    
    if result != expected {
        t.Errorf("Subtract(5, 3) = %d; want %d", result, expected)
    }
}
```

### 1.2 运行测试

```bash
# 🔥 运行所有测试
go test

# 🔥 运行指定测试
go test -run TestAdd

# 🔥 显示详细信息
go test -v

# 🔥 运行指定包的测试
go test ./...

# 🔥 运行测试并显示覆盖率
go test -cover
```

### 1.3 测试辅助函数

```go
package math

import "testing"

// 🔥 使用t.Helper()标记辅助函数
func assertEqual(t *testing.T, got, want int) {
    t.Helper()
    if got != want {
        t.Errorf("got %d; want %d", got, want)
    }
}

func TestAddWithHelper(t *testing.T) {
    result := Add(2, 3)
    assertEqual(t, result, 5)
}
```

## 2. 表格驱动测试

### 2.1 基本表格驱动测试

```go
package math

import "testing"

func TestAddTable(t *testing.T) {
    // 🔥 定义测试用例表格
    tests := []struct {
        name     string
        a, b     int
        expected int
    }{
        {"正数相加", 2, 3, 5},
        {"负数相加", -2, -3, -5},
        {"正负相加", 2, -3, -1},
        {"零相加", 0, 0, 0},
    }
    
    // 🔥 遍历测试用例
    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            result := Add(tt.a, tt.b)
            if result != tt.expected {
                t.Errorf("Add(%d, %d) = %d; want %d",
                    tt.a, tt.b, result, tt.expected)
            }
        })
    }
}
```

### 2.2 复杂表格驱动测试

```go
package user

import "testing"

type User struct {
    Name  string
    Email string
    Age   int
}

func ValidateUser(u User) error {
    if u.Name == "" {
        return errors.New("名称不能为空")
    }
    if u.Age < 0 || u.Age > 150 {
        return errors.New("年龄无效")
    }
    return nil
}

func TestValidateUser(t *testing.T) {
    tests := []struct {
        name    string
        user    User
        wantErr bool
        errMsg  string
    }{
        {
            name:    "有效用户",
            user:    User{Name: "张三", Email: "test@example.com", Age: 25},
            wantErr: false,
        },
        {
            name:    "名称为空",
            user:    User{Name: "", Email: "test@example.com", Age: 25},
            wantErr: true,
            errMsg:  "名称不能为空",
        },
        {
            name:    "年龄无效",
            user:    User{Name: "张三", Email: "test@example.com", Age: -1},
            wantErr: true,
            errMsg:  "年龄无效",
        },
    }
    
    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            err := ValidateUser(tt.user)
            
            if tt.wantErr {
                if err == nil {
                    t.Error("期望错误，但没有错误")
                } else if err.Error() != tt.errMsg {
                    t.Errorf("错误消息 = %v; want %v", err.Error(), tt.errMsg)
                }
            } else {
                if err != nil {
                    t.Errorf("不期望错误，但得到: %v", err)
                }
            }
        })
    }
}
```

### 3.2 提高覆盖率

```go
// calculator.go
package calculator

func Divide(a, b float64) (float64, error) {
    if b == 0 {
        return 0, errors.New("除数不能为0")
    }
    return a / b, nil
}
```

```go
// calculator_test.go
package calculator

import "testing"

func TestDivide(t *testing.T) {
    tests := []struct {
        name    string
        a, b    float64
        want    float64
        wantErr bool
    }{
        {"正常除法", 10, 2, 5, false},
        {"除以零", 10, 0, 0, true},
        {"负数除法", -10, 2, -5, false},
        {"小数除法", 10, 3, 3.333333333333333, false},
    }
    
    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            got, err := Divide(tt.a, tt.b)
            
            if tt.wantErr {
                if err == nil {
                    t.Error("期望错误，但没有错误")
                }
            } else {
                if err != nil {
                    t.Errorf("不期望错误，但得到: %v", err)
                }
                if got != tt.want {
                    t.Errorf("Divide(%v, %v) = %v; want %v",
                        tt.a, tt.b, got, tt.want)
                }
            }
        })
    }
}
```

## 4. 子测试

### 4.1 使用t.Run

```go
package math

import "testing"

func TestMath(t *testing.T) {
    // 🔥 使用t.Run创建子测试
    t.Run("Add", func(t *testing.T) {
        result := Add(2, 3)
        if result != 5 {
            t.Errorf("Add(2, 3) = %d; want 5", result)
        }
    })
    
    t.Run("Subtract", func(t *testing.T) {
        result := Subtract(5, 3)
        if result != 2 {
            t.Errorf("Subtract(5, 3) = %d; want 2", result)
        }
    })
}
```

### 4.2 并行测试

```go
package math

import "testing"

func TestParallel(t *testing.T) {
    tests := []struct {
        name string
        a, b int
        want int
    }{
        {"test1", 1, 2, 3},
        {"test2", 2, 3, 5},
        {"test3", 3, 4, 7},
    }
    
    for _, tt := range tests {
        tt := tt  // 捕获循环变量
        t.Run(tt.name, func(t *testing.T) {
            // 🔥 并行运行测试
            t.Parallel()
            
            result := Add(tt.a, tt.b)
            if result != tt.want {
                t.Errorf("Add(%d, %d) = %d; want %d",
                    tt.a, tt.b, result, tt.want)
            }
        })
    }
}
```

## 5. Mock测试

### 5.1 接口Mock

```go
// user.go
package user

type UserRepository interface {
    GetUser(id int) (*User, error)
    SaveUser(user *User) error
}

type UserService struct {
    repo UserRepository
}

func NewUserService(repo UserRepository) *UserService {
    return &UserService{repo: repo}
}

func (s *UserService) GetUserName(id int) (string, error) {
    user, err := s.repo.GetUser(id)
    if err != nil {
        return "", err
    }
    return user.Name, nil
}
```

```go
// user_test.go
package user

import (
    "errors"
    "testing"
)

// 🔥 Mock实现
type MockUserRepository struct {
    users map[int]*User
}

func (m *MockUserRepository) GetUser(id int) (*User, error) {
    user, ok := m.users[id]
    if !ok {
        return nil, errors.New("用户不存在")
    }
    return user, nil
}

func (m *MockUserRepository) SaveUser(user *User) error {
    m.users[user.ID] = user
    return nil
}

func TestUserService_GetUserName(t *testing.T) {
    // 🔥 创建Mock
    mockRepo := &MockUserRepository{
        users: map[int]*User{
            1: {ID: 1, Name: "张三"},
        },
    }
    
    service := NewUserService(mockRepo)
    
    // 测试存在的用户
    name, err := service.GetUserName(1)
    if err != nil {
        t.Errorf("不期望错误: %v", err)
    }
    if name != "张三" {
        t.Errorf("name = %s; want 张三", name)
    }
    
    // 测试不存在的用户
    _, err = service.GetUserName(999)
    if err == nil {
        t.Error("期望错误，但没有错误")
    }
}
```

### 5.2 使用testify/mock

```go
package user

import (
    "testing"
    
    "github.com/stretchr/testify/mock"
)

// 🔥 使用testify创建Mock
type MockUserRepository struct {
    mock.Mock
}

func (m *MockUserRepository) GetUser(id int) (*User, error) {
    args := m.Called(id)
    if args.Get(0) == nil {
        return nil, args.Error(1)
    }
    return args.Get(0).(*User), args.Error(1)
}

func (m *MockUserRepository) SaveUser(user *User) error {
    args := m.Called(user)
    return args.Error(0)
}

func TestUserServiceWithTestify(t *testing.T) {
    mockRepo := new(MockUserRepository)
    
    // 🔥 设置期望
    mockRepo.On("GetUser", 1).Return(&User{ID: 1, Name: "张三"}, nil)
    
    service := NewUserService(mockRepo)
    name, err := service.GetUserName(1)
    
    if err != nil {
        t.Errorf("不期望错误: %v", err)
    }
    if name != "张三" {
        t.Errorf("name = %s; want 张三", name)
    }
    
    // 🔥 验证Mock调用
    mockRepo.AssertExpectations(t)
}
```

### 6.1 基本基准测试

```go
package math

import "testing"

// 🔥 基准测试函数以Benchmark开头
func BenchmarkAdd(b *testing.B) {
    for i := 0; i < b.N; i++ {
        Add(2, 3)
    }
}

func BenchmarkSubtract(b *testing.B) {
    for i := 0; i < b.N; i++ {
        Subtract(5, 3)
    }
}
```

```bash
# 🔥 运行基准测试
go test -bench=.

# 🔥 运行指定基准测试
go test -bench=BenchmarkAdd

# 🔥 显示内存分配
go test -bench=. -benchmem

# 🔥 指定运行时间
go test -bench=. -benchtime=10s
```

### 6.2 复杂基准测试

```go
package strings

import (
    "strings"
    "testing"
)

func BenchmarkStringConcat(b *testing.B) {
    b.Run("Plus", func(b *testing.B) {
        for i := 0; i < b.N; i++ {
            _ = "hello" + " " + "world"
        }
    })
    
    b.Run("Sprintf", func(b *testing.B) {
        for i := 0; i < b.N; i++ {
            _ = fmt.Sprintf("%s %s", "hello", "world")
        }
    })
    
    b.Run("Builder", func(b *testing.B) {
        for i := 0; i < b.N; i++ {
            var builder strings.Builder
            builder.WriteString("hello")
            builder.WriteString(" ")
            builder.WriteString("world")
            _ = builder.String()
        }
    })
}
```

## 8. 最佳实践

### 8.1 测试组织

```
project/
├── user/
│   ├── user.go
│   ├── user_test.go
│   └── user_integration_test.go
├── order/
│   ├── order.go
│   └── order_test.go
└── testdata/
    ├── users.json
    └── orders.json
```

### 8.2 测试命名规范

```go
// ✅ 好的测试名称
func TestUserService_GetUser_Success(t *testing.T) {}
func TestUserService_GetUser_NotFound(t *testing.T) {}
func TestUserService_GetUser_InvalidID(t *testing.T) {}

// ❌ 不好的测试名称
func TestGetUser(t *testing.T) {}
func TestGetUser2(t *testing.T) {}
func TestGetUserError(t *testing.T) {}
```

### 8.3 测试原则

```go
// ✅ 原则1：测试应该独立
// 每个测试不应该依赖其他测试

// ✅ 原则2：测试应该可重复
// 多次运行应该得到相同结果

// ✅ 原则3：测试应该快速
// 避免耗时操作，使用Mock

// ✅ 原则4：测试应该清晰
// 使用表格驱动测试，清晰的命名

// ✅ 原则5：测试应该覆盖边界情况
tests := []struct {
    name string
    input int
    want int
}{
    {"零", 0, 0},
    {"正数", 5, 5},
    {"负数", -5, -5},
    {"最大值", math.MaxInt, math.MaxInt},
    {"最小值", math.MinInt, math.MinInt},
}
```

### 8.4 Setup和Teardown

```go
package user

import (
    "testing"
)

func TestMain(m *testing.M) {
    // 🔥 全局Setup
    setup()
    
    // 运行测试
    code := m.Run()
    
    // 🔥 全局Teardown
    teardown()
    
    os.Exit(code)
}

func setup() {
    // 初始化测试环境
    fmt.Println("Setup")
}

func teardown() {
    // 清理测试环境
    fmt.Println("Teardown")
}

func TestWithSetup(t *testing.T) {
    // 🔥 测试级别的Setup
    t.Cleanup(func() {
        // 测试结束后清理
        fmt.Println("Test cleanup")
    })
    
    // 测试逻辑
}
```
