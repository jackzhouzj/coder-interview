## 内存泄漏概述

Go 虽然有 GC，但仍可能发生内存泄漏。内存泄漏是指程序持有不再使用的内存引用，导致 GC 无法回收。

### 内存泄漏的表现

1. **内存持续增长**：程序运行时间越长，内存占用越高
2. **GC 频繁触发**：GC 无法有效回收内存
3. **OOM 错误**：最终导致内存溢出
4. **性能下降**：频繁 GC 影响性能

### Go 中的内存泄漏类型

1. **Goroutine 泄漏**：goroutine 无法退出
2. **Channel 泄漏**：channel 阻塞导致 goroutine 泄漏
3. **Timer/Ticker 泄漏**：未正确关闭
4. **全局变量泄漏**：无限增长的全局容器
5. **闭包引用泄漏**：闭包持有大对象引用
6. **第三方库泄漏**：库内部的资源未释放

## 常见泄漏场景

### 1. Goroutine 泄漏

#### 场景 1：Channel 阻塞

```go
// ❌ 泄漏：goroutine 永久阻塞
func leak() {
    ch := make(chan int)
    go func() {
        val := <-ch  // 永远等待，goroutine 泄漏
        fmt.Println(val)
    }()
    // ch 没有发送数据
}

// ✅ 修复：使用 context 或 close channel
func fixed() {
    ch := make(chan int)
    ctx, cancel := context.WithTimeout(context.Background(), time.Second)
    defer cancel()
    
    go func() {
        select {
        case val := <-ch:
            fmt.Println(val)
        case <-ctx.Done():
            return
        }
    }()
}
```

#### 场景 2：无限循环

```go
// ❌ 泄漏：goroutine 无法退出
func leak() {
    go func() {
        for {
            // 没有退出条件
            time.Sleep(time.Second)
        }
    }()
}

// ✅ 修复：添加退出机制
func fixed() {
    ctx, cancel := context.WithCancel(context.Background())
    defer cancel()
    
    go func() {
        for {
            select {
            case <-ctx.Done():
                return
            default:
                time.Sleep(time.Second)
            }
        }
    }()
}
```

#### 场景 3：WaitGroup 误用

```go
// ❌ 泄漏：WaitGroup 永远等待
func leak() {
    var wg sync.WaitGroup
    wg.Add(1)
    go func() {
        // 忘记调用 Done()
        time.Sleep(time.Second)
    }()
    wg.Wait()  // 永远等待
}

// ✅ 修复：确保调用 Done
func fixed() {
    var wg sync.WaitGroup
    wg.Add(1)
    go func() {
        defer wg.Done()
        time.Sleep(time.Second)
    }()
    wg.Wait()
}
```

### 2. Timer/Ticker 泄漏

```go
// ❌ 泄漏：Timer 未停止
func leak() {
    for i := 0; i < 1000; i++ {
        time.AfterFunc(time.Hour, func() {
            // 回调函数
        })
    }
    // 创建了 1000 个 Timer，占用内存
}

// ✅ 修复：及时停止 Timer
func fixed() {
    timers := make([]*time.Timer, 0)
    for i := 0; i < 1000; i++ {
        timer := time.AfterFunc(time.Hour, func() {
            // 回调函数
        })
        timers = append(timers, timer)
    }
    
    // 清理
    for _, timer := range timers {
        timer.Stop()
    }
}

// ❌ 泄漏：Ticker 未停止
func tickerLeak() {
    ticker := time.NewTicker(time.Second)
    go func() {
        for range ticker.C {
            // 处理
        }
    }()
    // 忘记 ticker.Stop()
}

// ✅ 修复：停止 Ticker
func tickerFixed() {
    ticker := time.NewTicker(time.Second)
    defer ticker.Stop()
    
    go func() {
        for range ticker.C {
            // 处理
        }
    }()
}
```

### 3. 全局变量泄漏

```go
// ❌ 泄漏：全局 map 无限增长
var cache = make(map[string][]byte)

func leak(key string, data []byte) {
    cache[key] = data  // 永远不删除
}

// ✅ 修复：使用 LRU 缓存
type LRUCache struct {
    capacity int
    cache    map[string]*list.Element
    list     *list.List
    mu       sync.Mutex
}

func (c *LRUCache) Set(key string, value []byte) {
    c.mu.Lock()
    defer c.mu.Unlock()
    
    if c.list.Len() >= c.capacity {
        // 删除最旧的元素
        oldest := c.list.Back()
        if oldest != nil {
            c.list.Remove(oldest)
            delete(c.cache, oldest.Value.(string))
        }
    }
    
    elem := c.list.PushFront(key)
    c.cache[key] = elem
}
```

### 4. 闭包引用泄漏

```go
// ❌ 泄漏：闭包持有大对象
func leak() {
    data := make([]byte, 10*1024*1024)  // 10MB
    
    go func() {
        // 闭包引用了 data，即使只用了一小部分
        fmt.Println(data[0])
        time.Sleep(time.Hour)
    }()
}

// ✅ 修复：只传递需要的数据
func fixed() {
    data := make([]byte, 10*1024*1024)
    firstByte := data[0]
    
    go func(b byte) {
        fmt.Println(b)
        time.Sleep(time.Hour)
    }(firstByte)
}
```

### 5. HTTP 连接泄漏

```go
// ❌ 泄漏：未关闭响应体
func leak() {
    resp, err := http.Get("https://example.com")
    if err != nil {
        return
    }
    // 忘记关闭 Body
}

// ✅ 修复：确保关闭 Body
func fixed() {
    resp, err := http.Get("https://example.com")
    if err != nil {
        return
    }
    defer resp.Body.Close()
    
    // 读取响应
    io.Copy(io.Discard, resp.Body)
}
```

## 实战案例

### 案例 1：HTTP 服务内存泄漏

#### 问题代码

```go
package main

import (
    "fmt"
    "net/http"
    "time"
)

var clients = make(map[string]*http.Client)

func handler(w http.ResponseWriter, r *http.Request) {
    clientID := r.Header.Get("Client-ID")
    
    // 为每个客户端创建新的 HTTP Client
    client := &http.Client{
        Timeout: 10 * time.Second,
    }
    clients[clientID] = client  // 泄漏：永远不删除
    
    // 使用 client 发送请求
    resp, err := client.Get("https://api.example.com/data")
    if err != nil {
        http.Error(w, err.Error(), http.StatusInternalServerError)
        return
    }
    // 泄漏：忘记关闭 Body
    
    fmt.Fprintf(w, "OK")
}

func main() {
    http.HandleFunc("/", handler)
    http.ListenAndServe(":8080", nil)
}
```

#### 排查过程

```bash
# 1. 监控内存
curl http://localhost:6060/debug/pprof/heap > heap1.prof
# 压测
ab -n 10000 -c 100 http://localhost:8080/
# 再次采集
curl http://localhost:6060/debug/pprof/heap > heap2.prof

# 2. 对比分析
go tool pprof -base heap1.prof heap2.prof
(pprof) top
# 发现 map 占用大量内存

# 3. 检查 goroutine
curl http://localhost:6060/debug/pprof/goroutine > goroutine.prof
go tool pprof goroutine.prof
(pprof) top
# 发现大量 goroutine 阻塞在 HTTP 读取
```

#### 修复代码

```go
package main

import (
    "fmt"
    "io"
    "net/http"
    "sync"
    "time"
)

type ClientPool struct {
    mu      sync.RWMutex
    clients map[string]*clientEntry
    maxSize int
}

type clientEntry struct {
    client    *http.Client
    lastUsed  time.Time
}

func NewClientPool(maxSize int) *ClientPool {
    pool := &ClientPool{
        clients: make(map[string]*clientEntry),
        maxSize: maxSize,
    }
    
    // 启动清理 goroutine
    go pool.cleanup()
    
    return pool
}

func (p *ClientPool) Get(clientID string) *http.Client {
    p.mu.Lock()
    defer p.mu.Unlock()
    
    if entry, ok := p.clients[clientID]; ok {
        entry.lastUsed = time.Now()
        return entry.client
    }
    
    // 检查容量
    if len(p.clients) >= p.maxSize {
        p.evictOldest()
    }
    
    client := &http.Client{
        Timeout: 10 * time.Second,
    }
    
    p.clients[clientID] = &clientEntry{
        client:   client,
        lastUsed: time.Now(),
    }
    
    return client
}

func (p *ClientPool) evictOldest() {
    var oldestID string
    var oldestTime time.Time
    
    for id, entry := range p.clients {
        if oldestTime.IsZero() || entry.lastUsed.Before(oldestTime) {
            oldestID = id
            oldestTime = entry.lastUsed
        }
    }
    
    delete(p.clients, oldestID)
}

func (p *ClientPool) cleanup() {
    ticker := time.NewTicker(5 * time.Minute)
    defer ticker.Stop()
    
    for range ticker.C {
        p.mu.Lock()
        now := time.Now()
        for id, entry := range p.clients {
            if now.Sub(entry.lastUsed) > 10*time.Minute {
                delete(p.clients, id)
            }
        }
        p.mu.Unlock()
    }
}

var clientPool = NewClientPool(1000)

func handler(w http.ResponseWriter, r *http.Request) {
    clientID := r.Header.Get("Client-ID")
    client := clientPool.Get(clientID)
    
    resp, err := client.Get("https://api.example.com/data")
    if err != nil {
        http.Error(w, err.Error(), http.StatusInternalServerError)
        return
    }
    defer resp.Body.Close()
    
    // 确保读取完 Body
    io.Copy(io.Discard, resp.Body)
    
    fmt.Fprintf(w, "OK")
}

func main() {
    http.HandleFunc("/", handler)
    http.ListenAndServe(":8080", nil)
}
```

### 案例 2：Goroutine 泄漏

#### 问题代码

```go
package main

import (
    "fmt"
    "time"
)

func processData(data []int) {
    results := make(chan int)
    
    for _, v := range data {
        go func(val int) {
            time.Sleep(time.Second)
            results <- val * 2  // 如果没有接收者，goroutine 永久阻塞
        }(v)
    }
    
    // 只读取一部分结果
    for i := 0; i < 5; i++ {
        fmt.Println(<-results)
    }
    // 剩余的 goroutine 泄漏
}

func main() {
    data := make([]int, 100)
    for i := 0; i < 100; i++ {
        data[i] = i
    }
    
    processData(data)
    
    time.Sleep(10 * time.Second)
    fmt.Printf("Goroutines: %d\n", runtime.NumGoroutine())
}
```

#### 修复代码

```go
package main

import (
    "context"
    "fmt"
    "runtime"
    "sync"
    "time"
)

func processDataFixed(ctx context.Context, data []int) {
    results := make(chan int, len(data))  // 使用缓冲 channel
    var wg sync.WaitGroup
    
    for _, v := range data {
        wg.Add(1)
        go func(val int) {
            defer wg.Done()
            
            select {
            case <-ctx.Done():
                return
            case <-time.After(time.Second):
                select {
                case results <- val * 2:
                case <-ctx.Done():
                    return
                }
            }
        }(v)
    }
    
    // 等待所有 goroutine 完成
    go func() {
        wg.Wait()
        close(results)
    }()
    
    // 读取结果
    count := 0
    for result := range results {
        fmt.Println(result)
        count++
        if count >= 5 {
            break
        }
    }
}

func main() {
    data := make([]int, 100)
    for i := 0; i < 100; i++ {
        data[i] = i
    }
    
    ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
    defer cancel()
    
    processDataFixed(ctx, data)
    
    time.Sleep(2 * time.Second)
    fmt.Printf("Goroutines: %d\n", runtime.NumGoroutine())
}
```

### 案例 3：缓存泄漏

#### 问题代码

```go
package main

import (
    "sync"
    "time"
)

type Cache struct {
    mu   sync.RWMutex
    data map[string][]byte
}

func (c *Cache) Set(key string, value []byte) {
    c.mu.Lock()
    defer c.mu.Unlock()
    c.data[key] = value  // 永远不删除
}

func (c *Cache) Get(key string) ([]byte, bool) {
    c.mu.RLock()
    defer c.mu.RUnlock()
    val, ok := c.data[key]
    return val, ok
}

func main() {
    cache := &Cache{data: make(map[string][]byte)}
    
    // 不断写入数据
    for i := 0; ; i++ {
        key := fmt.Sprintf("key_%d", i)
        value := make([]byte, 1024*1024)  // 1MB
        cache.Set(key, value)
        time.Sleep(10 * time.Millisecond)
    }
}
```

#### 修复代码

```go
package main

import (
    "container/list"
    "fmt"
    "sync"
    "time"
)

type CacheItem struct {
    key       string
    value     []byte
    expiresAt time.Time
}

type LRUCache struct {
    mu       sync.RWMutex
    capacity int
    items    map[string]*list.Element
    lru      *list.List
}

func NewLRUCache(capacity int) *LRUCache {
    cache := &LRUCache{
        capacity: capacity,
        items:    make(map[string]*list.Element),
        lru:      list.New(),
    }
    
    // 启动过期清理
    go cache.cleanup()
    
    return cache
}

func (c *LRUCache) Set(key string, value []byte, ttl time.Duration) {
    c.mu.Lock()
    defer c.mu.Unlock()
    
    // 检查是否已存在
    if elem, ok := c.items[key]; ok {
        c.lru.MoveToFront(elem)
        item := elem.Value.(*CacheItem)
        item.value = value
        item.expiresAt = time.Now().Add(ttl)
        return
    }
    
    // 检查容量
    if c.lru.Len() >= c.capacity {
        c.evict()
    }
    
    item := &CacheItem{
        key:       key,
        value:     value,
        expiresAt: time.Now().Add(ttl),
    }
    
    elem := c.lru.PushFront(item)
    c.items[key] = elem
}

func (c *LRUCache) Get(key string) ([]byte, bool) {
    c.mu.Lock()
    defer c.mu.Unlock()
    
    elem, ok := c.items[key]
    if !ok {
        return nil, false
    }
    
    item := elem.Value.(*CacheItem)
    
    // 检查是否过期
    if time.Now().After(item.expiresAt) {
        c.lru.Remove(elem)
        delete(c.items, key)
        return nil, false
    }
    
    c.lru.MoveToFront(elem)
    return item.value, true
}

func (c *LRUCache) evict() {
    elem := c.lru.Back()
    if elem != nil {
        c.lru.Remove(elem)
        item := elem.Value.(*CacheItem)
        delete(c.items, item.key)
    }
}

func (c *LRUCache) cleanup() {
    ticker := time.NewTicker(time.Minute)
    defer ticker.Stop()
    
    for range ticker.C {
        c.mu.Lock()
        now := time.Now()
        
        var next *list.Element
        for elem := c.lru.Front(); elem != nil; elem = next {
            next = elem.Next()
            item := elem.Value.(*CacheItem)
            
            if now.After(item.expiresAt) {
                c.lru.Remove(elem)
                delete(c.items, item.key)
            }
        }
        
        c.mu.Unlock()
    }
}

func main() {
    cache := NewLRUCache(1000)  // 限制 1000 个元素
    
    for i := 0; ; i++ {
        key := fmt.Sprintf("key_%d", i)
        value := make([]byte, 1024*1024)
        cache.Set(key, value, 5*time.Minute)
        time.Sleep(10 * time.Millisecond)
    }
}
```

```go
    "github.com/prometheus/client_golang/prometheus"
    "github.com/prometheus/client_golang/prometheus/promhttp"
)

var (
    goroutineGauge = prometheus.NewGauge(prometheus.GaugeOpts{
        Name: "goroutines_count",
        Help: "Number of goroutines",
    })
    
    memoryGauge = prometheus.NewGauge(prometheus.GaugeOpts{
        Name: "memory_usage_bytes",
        Help: "Memory usage in bytes",
    })
)

func init() {
    prometheus.MustRegister(goroutineGauge)
    prometheus.MustRegister(memoryGauge)
    
    go func() {
        ticker := time.NewTicker(10 * time.Second)
        defer ticker.Stop()
        
        for range ticker.C {
            goroutineGauge.Set(float64(runtime.NumGoroutine()))
            
            var m runtime.MemStats
            runtime.ReadMemStats(&m)
            memoryGauge.Set(float64(m.Alloc))
        }
    }()
}
```
