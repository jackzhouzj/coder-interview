## 消息队列基础

### 什么是消息队列

消息队列（Message Queue）是一种异步通信机制，用于在分布式系统中传递消息。

### 核心概念

```python
"""
消息队列核心组件
"""

# 1. 生产者（Producer）：发送消息
# 2. 消费者（Consumer）：接收并处理消息
# 3. 队列（Queue）：存储消息的容器
# 4. 交换机（Exchange）：路由消息到队列
# 5. 绑定（Binding）：交换机和队列的关系
```

### 使用场景

```python
"""
消息队列典型应用场景
"""

# 1. 异步处理
# 场景：用户注册后发送邮件、短信
def register_user(user_data):
    # 同步：保存用户信息
    user = save_user(user_data)
    
    # 异步：发送通知
    send_email_task.delay(user.email)
    send_sms_task.delay(user.phone)
    
    return user

# 2. 流量削峰
# 场景：秒杀活动，将请求放入队列慢慢处理
def handle_seckill_order(order_data):
    # 将订单放入队列
    queue.put(order_data)
    return {"status": "queued"}

# 3. 系统解耦
# 场景：订单系统与库存系统解耦
def create_order(order_data):
    # 创建订单
    order = save_order(order_data)
    
    # 发送消息通知库存系统
    publish_message('order.created', order)
    
    return order

# 4. 日志收集
# 场景：分布式日志聚合
def log_event(event_data):
    send_to_queue('logs', event_data)

# 5. 数据同步
# 场景：主从数据库同步
def sync_data(data):
    publish_message('data.sync', data)
```

## RabbitMQ

### 安装配置

```bash
# Docker 安装 RabbitMQ
docker run -d --name rabbitmq \
  -p 5672:5672 \
  -p 15672:15672 \
  -e RABBITMQ_DEFAULT_USER=admin \
  -e RABBITMQ_DEFAULT_PASS=admin123 \
  rabbitmq:3-management

# Python 客户端安装
pip install pika
```

### 基础使用

```python
"""
RabbitMQ 基础示例
"""
import pika

# 1. 建立连接
def get_connection():
    credentials = pika.PlainCredentials('admin', 'admin123')
    parameters = pika.ConnectionParameters(
        host='localhost',
        port=5672,
        credentials=credentials
    )
    return pika.BlockingConnection(parameters)

# 2. 生产者：发送消息
def send_message(queue_name, message):
    connection = get_connection()
    channel = connection.channel()
    
    # 声明队列
    channel.queue_declare(queue=queue_name, durable=True)
    
    # 发送消息
    channel.basic_publish(
        exchange='',
        routing_key=queue_name,
        body=message,
        properties=pika.BasicProperties(
            delivery_mode=2,  # 持久化消息
        )
    )
    
    print(f"发送消息: {message}")
    connection.close()

# 3. 消费者：接收消息
def consume_messages(queue_name):
    connection = get_connection()
    channel = connection.channel()
    
    channel.queue_declare(queue=queue_name, durable=True)
    
    def callback(ch, method, properties, body):
        print(f"接收消息: {body.decode()}")
        # 处理消息
        process_message(body)
        # 确认消息
        ch.basic_ack(delivery_tag=method.delivery_tag)
    
    channel.basic_qos(prefetch_count=1)
    channel.basic_consume(queue=queue_name, on_message_callback=callback)
    
    print('等待消息...')
    channel.start_consuming()

# 使用示例
if __name__ == '__main__':
    # 发送消息
    send_message('task_queue', 'Hello RabbitMQ!')
    
    # 接收消息
    # consume_messages('task_queue')
```

### 交换机类型

```python
"""
RabbitMQ 四种交换机类型
"""

# 1. Direct Exchange（直连交换机）
def direct_exchange_demo():
    """路由键完全匹配"""
    connection = get_connection()
    channel = connection.channel()
    
    # 声明交换机
    channel.exchange_declare(exchange='direct_logs', exchange_type='direct')
    
    # 发送不同级别的日志
    channel.basic_publish(
        exchange='direct_logs',
        routing_key='error',
        body='Error log message'
    )
    
    channel.basic_publish(
        exchange='direct_logs',
        routing_key='info',
        body='Info log message'
    )
    
    connection.close()

# 2. Fanout Exchange（扇出交换机）
def fanout_exchange_demo():
    """广播到所有绑定的队列"""
    connection = get_connection()
    channel = connection.channel()
    
    channel.exchange_declare(exchange='logs', exchange_type='fanout')
    
    # 发送消息到所有订阅者
    channel.basic_publish(
        exchange='logs',
        routing_key='',
        body='Broadcast message'
    )
    
    connection.close()

# 3. Topic Exchange（主题交换机）
def topic_exchange_demo():
    """支持通配符匹配"""
    connection = get_connection()
    channel = connection.channel()
    
    channel.exchange_declare(exchange='topic_logs', exchange_type='topic')
    
    # 发送消息
    routing_keys = [
        'user.create',
        'user.update',
        'order.create'
    ]
    
    for key in routing_keys:
        channel.basic_publish(
            exchange='topic_logs',
            routing_key=key,
            body=f'Message for {key}'
        )
    
    connection.close()

# 消费者订阅模式
# user.* 匹配 user.create, user.update
# *.create 匹配 user.create, order.create
```

## Kafka

### 安装配置

```bash
# Docker 安装 Kafka
docker-compose up -d

# Python 客户端安装
pip install kafka-python
```

### 基础使用

```python
"""
Kafka 生产者和消费者
"""
from kafka import KafkaProducer, KafkaConsumer
import json

# 生产者
producer = KafkaProducer(
    bootstrap_servers=['localhost:9092'],
    value_serializer=lambda v: json.dumps(v).encode('utf-8')
)

# 发送消息
producer.send('user-events', {'user_id': 123, 'action': 'login'})
producer.flush()

# 消费者
consumer = KafkaConsumer(
    'user-events',
    bootstrap_servers=['localhost:9092'],
    value_deserializer=lambda m: json.loads(m.decode('utf-8')),
    group_id='user-service'
)

for message in consumer:
    print(f"Received: {message.value}")
```

## Celery

### 安装配置

```bash
pip install celery redis
```

### 基础使用

```python
"""
Celery 异步任务
"""
from celery import Celery

app = Celery('tasks', broker='redis://localhost:6379/0')

@app.task
def send_email(email, subject, content):
    # 发送邮件逻辑
    print(f"Sending email to {email}")
    return True

# 调用任务
result = send_email.delay('user@example.com', 'Hello', 'Content')
```
