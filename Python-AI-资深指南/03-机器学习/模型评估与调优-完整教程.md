## 1. 模型评估指标

### 1.1 分类问题评估指标

```python
"""
分类问题评估指标
@author erik.zhou
"""
from sklearn.metrics import (
    accuracy_score, precision_score, recall_score, f1_score,
    confusion_matrix, classification_report, roc_auc_score, roc_curve
)
from sklearn.datasets import load_breast_cancer
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
import numpy as np

# 加载数据
data = load_breast_cancer()
X_train, X_test, y_train, y_test = train_test_split(
    data.data, data.target, test_size=0.2, random_state=42
)

# 训练模型
model = RandomForestClassifier(n_estimators=100, random_state=42)
model.fit(X_train, y_train)
y_pred = model.predict(X_test)
y_pred_proba = model.predict_proba(X_test)[:, 1]

# 1. 准确率
accuracy = accuracy_score(y_test, y_pred)
print(f"准确率 (Accuracy): {accuracy:.4f}")

# 2. 精确率
precision = precision_score(y_test, y_pred)
print(f"精确率 (Precision): {precision:.4f}")

# 3. 召回率
recall = recall_score(y_test, y_pred)
print(f"召回率 (Recall): {recall:.4f}")

# 4. F1分数
f1 = f1_score(y_test, y_pred)
print(f"F1分数: {f1:.4f}")

# 5. 混淆矩阵
cm = confusion_matrix(y_test, y_pred)
print(f"\n混淆矩阵:\n{cm}")

# 6. 分类报告
report = classification_report(y_test, y_pred, target_names=data.target_names)
print(f"\n分类报告:\n{report}")

# 7. ROC-AUC
roc_auc = roc_auc_score(y_test, y_pred_proba)
print(f"ROC-AUC: {roc_auc:.4f}")
```

## 2. 交叉验证

### 2.1 K折交叉验证

```python
"""
K折交叉验证
@author erik.zhou
"""
from sklearn.model_selection import (
    cross_val_score, cross_validate, KFold, StratifiedKFold
)
from sklearn.ensemble import RandomForestClassifier

# 加载数据
data = load_breast_cancer()
X, y = data.data, data.target

# 1. 简单K折交叉验证
model = RandomForestClassifier(n_estimators=100, random_state=42)
scores = cross_val_score(model, X, y, cv=5, scoring='accuracy')

print(f"5折交叉验证准确率: {scores}")
print(f"平均准确率: {scores.mean():.4f} (+/- {scores.std() * 2:.4f})")

# 2. 分层K折交叉验证（保持类别比例）
skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
scores_stratified = cross_val_score(model, X, y, cv=skf, scoring='accuracy')

print(f"\n分层5折交叉验证准确率: {scores_stratified}")
print(f"平均准确率: {scores_stratified.mean():.4f} (+/- {scores_stratified.std() * 2:.4f})")

# 3. 多指标交叉验证
scoring = ['accuracy', 'precision', 'recall', 'f1', 'roc_auc']
scores_multi = cross_validate(model, X, y, cv=5, scoring=scoring)

print("\n多指标交叉验证结果:")
for metric in scoring:
    score_key = f'test_{metric}'
    print(f"{metric}: {scores_multi[score_key].mean():.4f} (+/- {scores_multi[score_key].std() * 2:.4f})")
```

### 3.4 学习曲线分析

```python
"""
学习曲线分析
@author erik.zhou
"""
from sklearn.model_selection import learning_curve
import matplotlib.pyplot as plt

def plot_learning_curve(estimator, X, y, cv=5):
    """
    绘制学习曲线
    
    Args:
        estimator: 模型
        X: 特征数据
        y: 目标数据
        cv: 交叉验证折数
    """
    train_sizes, train_scores, test_scores = learning_curve(
        estimator, X, y, cv=cv, n_jobs=-1,
        train_sizes=np.linspace(0.1, 1.0, 10),
        scoring='accuracy'
    )
    
    train_mean = np.mean(train_scores, axis=1)
    train_std = np.std(train_scores, axis=1)
    test_mean = np.mean(test_scores, axis=1)
    test_std = np.std(test_scores, axis=1)
    
    plt.figure(figsize=(10, 6))
    plt.plot(train_sizes, train_mean, label='训练集分数', color='blue', marker='o')
    plt.fill_between(train_sizes, train_mean - train_std, train_mean + train_std, alpha=0.15, color='blue')
    plt.plot(train_sizes, test_mean, label='验证集分数', color='red', marker='s')
    plt.fill_between(train_sizes, test_mean - test_std, test_mean + test_std, alpha=0.15, color='red')
    
    plt.xlabel('训练样本数')
    plt.ylabel('准确率')
    plt.title('学习曲线')
    plt.legend(loc='best')
    plt.grid(True)
    plt.show()
    
    return train_sizes, train_scores, test_scores

# 使用示例
model = RandomForestClassifier(n_estimators=100, random_state=42)
plot_learning_curve(model, X, y, cv=5)
```

### 4.3 混合法（Blending）

```python
"""
混合法模型融合
@author erik.zhou
"""
from sklearn.model_selection import train_test_split

# 将训练集进一步划分
X_train_blend, X_val_blend, y_train_blend, y_val_blend = train_test_split(
    X_train, y_train, test_size=0.2, random_state=42
)

# 训练基础模型
models = [
    RandomForestClassifier(n_estimators=100, random_state=42),
    LogisticRegression(random_state=42, max_iter=1000),
    SVC(probability=True, random_state=42)
]

# 在训练集上训练基础模型
for model in models:
    model.fit(X_train_blend, y_train_blend)

# 在验证集上生成预测
val_predictions = np.column_stack([
    model.predict_proba(X_val_blend)[:, 1] for model in models
])

# 训练元模型
meta_model = LogisticRegression(random_state=42)
meta_model.fit(val_predictions, y_val_blend)

# 在测试集上预测
test_predictions = np.column_stack([
    model.predict_proba(X_test)[:, 1] for model in models
])
final_predictions = meta_model.predict(test_predictions)

# 评估
from sklearn.metrics import accuracy_score
print(f"混合模型准确率: {accuracy_score(y_test, final_predictions):.4f}")
```

### 5.1 诊断过拟合和欠拟合

```python
"""
诊断过拟合和欠拟合
@author erik.zhou
"""
from sklearn.tree import DecisionTreeClassifier
from sklearn.model_selection import validation_curve

def diagnose_fitting(estimator, X, y):
    """
    诊断模型的拟合情况
    
    Args:
        estimator: 模型
        X: 特征数据
        y: 目标数据
    """
    # 训练模型
    estimator.fit(X_train, y_train)
    
    # 计算训练集和测试集分数
    train_score = estimator.score(X_train, y_train)
    test_score = estimator.score(X_test, y_test)
    
    print(f"训练集分数: {train_score:.4f}")
    print(f"测试集分数: {test_score:.4f}")
    print(f"分数差异: {train_score - test_score:.4f}")
    
    # 诊断
    if train_score > 0.95 and test_score < 0.85:
        print("诊断: 过拟合")
    elif train_score < 0.85 and test_score < 0.85:
        print("诊断: 欠拟合")
    else:
        print("诊断: 拟合良好")

# 示例1：过拟合（决策树深度过大）
model_overfit = DecisionTreeClassifier(max_depth=None, random_state=42)
print("深度不限的决策树:")
diagnose_fitting(model_overfit, X, y)

# 示例2：欠拟合（决策树深度过小）
model_underfit = DecisionTreeClassifier(max_depth=2, random_state=42)
print("\n深度为2的决策树:")
diagnose_fitting(model_underfit, X, y)

# 示例3：拟合良好
model_good = DecisionTreeClassifier(max_depth=5, random_state=42)
print("\n深度为5的决策树:")
diagnose_fitting(model_good, X, y)
```

### 5.4 验证曲线

```python
"""
验证曲线分析
@author erik.zhou
"""
from sklearn.model_selection import validation_curve

def plot_validation_curve(estimator, X, y, param_name, param_range, cv=5):
    """
    绘制验证曲线
    
    Args:
        estimator: 模型
        X: 特征数据
        y: 目标数据
        param_name: 参数名称
        param_range: 参数范围
        cv: 交叉验证折数
    """
    train_scores, test_scores = validation_curve(
        estimator, X, y,
        param_name=param_name,
        param_range=param_range,
        cv=cv,
        scoring='accuracy',
        n_jobs=-1
    )
    
    train_mean = np.mean(train_scores, axis=1)
    train_std = np.std(train_scores, axis=1)
    test_mean = np.mean(test_scores, axis=1)
    test_std = np.std(test_scores, axis=1)
    
    plt.figure(figsize=(10, 6))
    plt.plot(param_range, train_mean, label='训练集分数', color='blue', marker='o')
    plt.fill_between(param_range, train_mean - train_std, train_mean + train_std, alpha=0.15, color='blue')
    plt.plot(param_range, test_mean, label='验证集分数', color='red', marker='s')
    plt.fill_between(param_range, test_mean - test_std, test_mean + test_std, alpha=0.15, color='red')
    
    plt.xlabel(param_name)
    plt.ylabel('准确率')
    plt.title(f'验证曲线 - {param_name}')
    plt.legend(loc='best')
    plt.grid(True)
    plt.show()

# 使用示例：分析决策树深度
param_range = range(1, 21)
plot_validation_curve(
    DecisionTreeClassifier(random_state=42),
    X, y,
    param_name='max_depth',
    param_range=param_range,
    cv=5
)
```

### 6.1 完整的模型评估与调优流程

```python
"""
完整的模型评估与调优实战案例
@author erik.zhou
"""
from sklearn.model_selection import train_test_split, GridSearchCV
from sklearn.preprocessing import StandardScaler
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report, confusion_matrix
import pandas as pd

class ModelEvaluationPipeline:
    """模型评估与调优完整流程"""
    
    def __init__(self, model, param_grid):
        self.model = model
        self.param_grid = param_grid
        self.best_model = None
        self.scaler = StandardScaler()
    
    def preprocess(self, X_train, X_test):
        """数据预处理"""
        X_train_scaled = self.scaler.fit_transform(X_train)
        X_test_scaled = self.scaler.transform(X_test)
        return X_train_scaled, X_test_scaled
    
    def tune_hyperparameters(self, X_train, y_train, cv=5):
        """超参数调优"""
        grid_search = GridSearchCV(
            estimator=self.model,
            param_grid=self.param_grid,
            cv=cv,
            scoring='accuracy',
            n_jobs=-1,
            verbose=1
        )
        
        grid_search.fit(X_train, y_train)
        self.best_model = grid_search.best_estimator_
        
        print(f"最佳参数: {grid_search.best_params_}")
        print(f"最佳交叉验证分数: {grid_search.best_score_:.4f}")
        
        return grid_search
    
    def evaluate(self, X_test, y_test):
        """模型评估"""
        if self.best_model is None:
            raise ValueError("请先进行超参数调优")
        
        y_pred = self.best_model.predict(X_test)
        
        # 分类报告
        print("\n分类报告:")
        print(classification_report(y_test, y_pred))
        
        # 混淆矩阵
        print("\n混淆矩阵:")
        print(confusion_matrix(y_test, y_pred))
        
        # 测试集分数
        test_score = self.best_model.score(X_test, y_test)
        print(f"\n测试集准确率: {test_score:.4f}")
        
        return y_pred
    
    def cross_validate(self, X, y, cv=5):
        """交叉验证"""
        if self.best_model is None:
            raise ValueError("请先进行超参数调优")
        
        scores = cross_val_score(
            self.best_model, X, y, cv=cv, scoring='accuracy'
        )
        
        print(f"\n{cv}折交叉验证分数: {scores}")
        print(f"平均分数: {scores.mean():.4f} (+/- {scores.std() * 2:.4f})")
        
        return scores

# 使用示例
data = load_breast_cancer()
X_train, X_test, y_train, y_test = train_test_split(
    data.data, data.target, test_size=0.2, random_state=42
)

# 定义模型和参数网格

param_grid = {

    'max_depth': [5, 10, 15],
    'min_samples_split': [2, 5, 10]
}
```

## 7. 最佳实践

### 7.1 模型评估原则

1. **使用多个评估指标**：不要只依赖单一指标
2. **交叉验证**：避免数据划分的偶然性
3. **独立测试集**：保留独立的测试集进行最终评估
4. **考虑业务目标**：选择与业务目标一致的评估指标
5. **关注泛化能力**：重点关注模型在未见数据上的表现
