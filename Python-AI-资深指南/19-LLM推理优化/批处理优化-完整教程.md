## 1. 动态批处理

### 1.1 批处理基础

#### 基本批处理实现
```python
"""
批处理：将多个请求合并处理
- 提高GPU利用率
- 减少推理延迟
- 优化吞吐量

@author erik.zhou
"""

import torch
from typing import List, Dict
from transformers import AutoModelForCausalLM, AutoTokenizer
from collections import deque
import time

class SimpleBatchProcessor:
    """简单批处理器"""
    
    def __init__(
        self,
        model_name: str,
        batch_size: int = 8,
        max_wait_time: float = 0.1
    ):
        self.model = AutoModelForCausalLM.from_pretrained(model_name)
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.batch_size = batch_size
        self.max_wait_time = max_wait_time
        
        self.request_queue = deque()
        self.model.eval()
    
    def add_request(self, prompt: str, max_tokens: int = 50):
        """添加请求到队列"""
        request = {
            'prompt': prompt,
            'max_tokens': max_tokens,
            'timestamp': time.time()
        }
        self.request_queue.append(request)
        return len(self.request_queue)
    
    def process_batch(self) -> List[str]:
        """处理一批请求"""
        if not self.request_queue:
            return []
        
        # 收集批次
        batch = []
        while len(batch) < self.batch_size and self.request_queue:
            batch.append(self.request_queue.popleft())
        
        # 准备输入
        prompts = [req['prompt'] for req in batch]
        max_tokens = max(req['max_tokens'] for req in batch)
        
        # 批量编码
        inputs = self.tokenizer(
            prompts,
            return_tensors="pt",
            padding=True,
            truncation=True
        )
        
        # 批量推理
        with torch.no_grad():
            outputs = self.model.generate(
                **inputs,
                max_new_tokens=max_tokens,
                pad_token_id=self.tokenizer.eos_token_id
            )
        
        # 解码结果
        results = self.tokenizer.batch_decode(outputs, skip_special_tokens=True)
        
        return results
    
    def run(self):
        """运行批处理循环"""
        while True:
            if self.request_queue:
                # 检查是否达到批次大小或等待超时
                oldest_request = self.request_queue[0]
                wait_time = time.time() - oldest_request['timestamp']
                
                if len(self.request_queue) >= self.batch_size or wait_time >= self.max_wait_time:
                    results = self.process_batch()
                    print(f"处理了 {len(results)} 个请求")
            
            time.sleep(0.01)

# 使用示例
processor = SimpleBatchProcessor("gpt2", batch_size=4, max_wait_time=0.1)
processor.add_request("Hello, how are you?")
processor.add_request("What is AI?")
results = processor.process_batch()
```


### 1.2 动态批处理

#### 连续批处理实现
```python
"""
连续批处理：动态调整批次大小
- 根据请求到达率调整
- 最小化延迟
- 最大化吞吐量

@author erik.zhou
"""

import asyncio
from dataclasses import dataclass
from typing import Optional
import uuid

@dataclass
class Request:
    """请求数据类"""
    id: str
    prompt: str
    max_tokens: int
    timestamp: float
    future: Optional[asyncio.Future] = None

class DynamicBatchProcessor:
    """动态批处理器"""
    
    def __init__(
        self,
        model_name: str,
        min_batch_size: int = 1,
        max_batch_size: int = 32,
        max_wait_time: float = 0.05
    ):
        self.model = AutoModelForCausalLM.from_pretrained(model_name)
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        
        self.min_batch_size = min_batch_size
        self.max_batch_size = max_batch_size
        self.max_wait_time = max_wait_time
        
        self.request_queue = asyncio.Queue()
        self.running = False
    
    async def add_request(self, prompt: str, max_tokens: int = 50) -> str:
        """异步添加请求"""
        request = Request(
            id=str(uuid.uuid4()),
            prompt=prompt,
            max_tokens=max_tokens,
            timestamp=time.time(),
            future=asyncio.Future()
        )
        
        await self.request_queue.put(request)
        result = await request.future
        return result
    
    async def collect_batch(self) -> List[Request]:
        """收集一批请求"""
        batch = []
        deadline = time.time() + self.max_wait_time
        
        while len(batch) < self.max_batch_size:
            timeout = max(0, deadline - time.time())
            
            try:
                request = await asyncio.wait_for(
                    self.request_queue.get(),
                    timeout=timeout
                )
                batch.append(request)
                
                # 达到最小批次大小后，检查是否还有更多请求
                if len(batch) >= self.min_batch_size:
                    if self.request_queue.empty():
                        break
            except asyncio.TimeoutError:
                break
        
        return batch
    
    async def process_batch(self, batch: List[Request]):
        """处理批次"""
        if not batch:
            return
        
        # 准备输入
        prompts = [req.prompt for req in batch]
        max_tokens = max(req.max_tokens for req in batch)
        
        # 批量编码
        inputs = self.tokenizer(
            prompts,
            return_tensors="pt",
            padding=True,
            truncation=True
        )
        
        # 批量推理
        with torch.no_grad():
            outputs = self.model.generate(
                **inputs,
                max_new_tokens=max_tokens,
                pad_token_id=self.tokenizer.eos_token_id
            )
        
        # 解码并返回结果
        results = self.tokenizer.batch_decode(outputs, skip_special_tokens=True)
        
        for request, result in zip(batch, results):
            request.future.set_result(result)
    
    async def run(self):
        """运行批处理循环"""
        self.running = True
        
        while self.running:
            batch = await self.collect_batch()
            if batch:
                await self.process_batch(batch)
                print(f"处理了批次大小: {len(batch)}")
            else:
                await asyncio.sleep(0.01)
    
    def stop(self):
        """停止批处理"""
        self.running = False

# 使用示例
async def main():
    processor = DynamicBatchProcessor("gpt2", min_batch_size=2, max_batch_size=8)
    
    # 启动处理器
    processor_task = asyncio.create_task(processor.run())
    
    # 发送请求
    tasks = [
        processor.add_request("Hello, how are you?"),
        processor.add_request("What is AI?"),
        processor.add_request("Explain machine learning")
    ]
    
    results = await asyncio.gather(*tasks)
    print("结果:", results)
    
    processor.stop()
    await processor_task

# asyncio.run(main())
```

## 2. 请求调度

### 2.1 优先级调度

#### 优先级队列实现
```python
"""
优先级调度：根据优先级处理请求
- 高优先级请求优先处理
- 防止饥饿
- 公平调度

@author erik.zhou
"""

import heapq
from enum import IntEnum

class Priority(IntEnum):
    """请求优先级"""
    LOW = 3
    NORMAL = 2
    HIGH = 1
    URGENT = 0

@dataclass
class PriorityRequest:
    """带优先级的请求"""
    priority: Priority
    timestamp: float
    id: str
    prompt: str
    max_tokens: int
    future: Optional[asyncio.Future] = None
    
    def __lt__(self, other):
        """优先级比较"""
        if self.priority != other.priority:
            return self.priority < other.priority
        return self.timestamp < other.timestamp


class PriorityScheduler:
    """优先级调度器"""
    
    def __init__(self, model_name: str, batch_size: int = 8):
        self.model = AutoModelForCausalLM.from_pretrained(model_name)
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.batch_size = batch_size
        
        self.request_heap = []
        self.lock = asyncio.Lock()
    
    async def add_request(
        self,
        prompt: str,
        max_tokens: int = 50,
        priority: Priority = Priority.NORMAL
    ) -> str:
        """添加带优先级的请求"""
        request = PriorityRequest(
            priority=priority,
            timestamp=time.time(),
            id=str(uuid.uuid4()),
            prompt=prompt,
            max_tokens=max_tokens,
            future=asyncio.Future()
        )
        
        async with self.lock:
            heapq.heappush(self.request_heap, request)
        
        result = await request.future
        return result
    
    async def get_batch(self) -> List[PriorityRequest]:
        """获取一批请求"""
        batch = []
        
        async with self.lock:
            while len(batch) < self.batch_size and self.request_heap:
                request = heapq.heappop(self.request_heap)
                batch.append(request)
        
        return batch
    
    async def process_batch(self, batch: List[PriorityRequest]):
        """处理批次"""
        if not batch:
            return
        
        prompts = [req.prompt for req in batch]
        max_tokens = max(req.max_tokens for req in batch)
        
        inputs = self.tokenizer(
            prompts,
            return_tensors="pt",
            padding=True,
            truncation=True
        )
        
        with torch.no_grad():
            outputs = self.model.generate(
                **inputs,
                max_new_tokens=max_tokens,
                pad_token_id=self.tokenizer.eos_token_id
            )
        
        results = self.tokenizer.batch_decode(outputs, skip_special_tokens=True)
        
        for request, result in zip(batch, results):
            request.future.set_result(result)

# 使用示例
async def priority_example():
    scheduler = PriorityScheduler("gpt2", batch_size=4)
    
    # 发送不同优先级的请求
    tasks = [
        scheduler.add_request("Low priority", priority=Priority.LOW),
        scheduler.add_request("High priority", priority=Priority.HIGH),
        scheduler.add_request("Normal priority", priority=Priority.NORMAL),
        scheduler.add_request("Urgent!", priority=Priority.URGENT)
    ]
    
    # 处理批次
    batch = await scheduler.get_batch()
    await scheduler.process_batch(batch)
    
    results = await asyncio.gather(*tasks)
    print("结果:", results)

# asyncio.run(priority_example())
```

### 2.2 公平调度

#### 轮询调度实现
```python
"""
公平调度：确保所有用户公平获得资源
- 轮询调度
- 防止单个用户占用资源
- 保证服务质量

@author erik.zhou
"""

from collections import defaultdict, deque

class FairScheduler:
    """公平调度器"""
    
    def __init__(self, model_name: str, batch_size: int = 8):
        self.model = AutoModelForCausalLM.from_pretrained(model_name)
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.batch_size = batch_size
        
        # 每个用户的请求队列
        self.user_queues = defaultdict(deque)
        self.user_order = deque()
        self.lock = asyncio.Lock()
    
    async def add_request(
        self,
        user_id: str,
        prompt: str,
        max_tokens: int = 50
    ) -> str:
        """添加用户请求"""
        request = Request(
            id=str(uuid.uuid4()),
            prompt=prompt,
            max_tokens=max_tokens,
            timestamp=time.time(),
            future=asyncio.Future()
        )
        
        async with self.lock:
            if user_id not in self.user_queues or not self.user_queues[user_id]:
                self.user_order.append(user_id)
            
            self.user_queues[user_id].append(request)
        
        result = await request.future
        return result
    
    async def get_batch(self) -> List[Request]:
        """公平获取批次"""
        batch = []
        
        async with self.lock:
            # 轮询每个用户的队列
            users_checked = 0
            while len(batch) < self.batch_size and users_checked < len(self.user_order):
                if not self.user_order:
                    break
                
                user_id = self.user_order.popleft()
                
                if user_id in self.user_queues and self.user_queues[user_id]:
                    request = self.user_queues[user_id].popleft()
                    batch.append(request)
                    
                    # 如果用户还有请求，重新加入轮询
                    if self.user_queues[user_id]:
                        self.user_order.append(user_id)
                
                users_checked += 1
        
        return batch

# 使用示例
async def fair_scheduling_example():
    scheduler = FairScheduler("gpt2", batch_size=4)
    
    # 多个用户发送请求
    tasks = [
        scheduler.add_request("user1", "Request 1 from user1"),
        scheduler.add_request("user1", "Request 2 from user1"),
        scheduler.add_request("user2", "Request 1 from user2"),
        scheduler.add_request("user3", "Request 1 from user3")
    ]
    
    # 处理批次
    batch = await scheduler.get_batch()
    print(f"批次包含 {len(batch)} 个请求，来自不同用户")

# asyncio.run(fair_scheduling_example())
```

## 3. 内存管理

### 3.1 KV缓存管理

#### PagedAttention实现
```python
"""
KV缓存管理：优化注意力机制的内存使用
- 分页管理KV缓存
- 减少内存碎片
- 支持更大批次

@author erik.zhou
"""

import torch
import torch.nn as nn

class PagedKVCache:
    """分页KV缓存"""
    
    def __init__(
        self,
        num_layers: int,
        num_heads: int,
        head_dim: int,
        page_size: int = 16,
        max_pages: int = 1000
    ):
        self.num_layers = num_layers
        self.num_heads = num_heads
        self.head_dim = head_dim
        self.page_size = page_size
        self.max_pages = max_pages
        
        # 初始化页表
        self.k_cache = torch.zeros(
            max_pages, num_layers, num_heads, page_size, head_dim
        )
        self.v_cache = torch.zeros(
            max_pages, num_layers, num_heads, page_size, head_dim
        )
        
        # 空闲页列表
        self.free_pages = list(range(max_pages))
        # 序列到页的映射
        self.sequence_pages = {}
    
    def allocate_pages(self, sequence_id: str, num_pages: int) -> List[int]:
        """为序列分配页"""
        if len(self.free_pages) < num_pages:
            raise RuntimeError("内存不足，无法分配页")
        
        pages = [self.free_pages.pop() for _ in range(num_pages)]
        self.sequence_pages[sequence_id] = pages
        return pages
    
    def free_sequence(self, sequence_id: str):
        """释放序列的页"""
        if sequence_id in self.sequence_pages:
            pages = self.sequence_pages.pop(sequence_id)
            self.free_pages.extend(pages)
    
    def write_kv(
        self,
        sequence_id: str,
        layer_idx: int,
        position: int,
        k: torch.Tensor,
        v: torch.Tensor
    ):
        """写入KV缓存"""
        pages = self.sequence_pages[sequence_id]
        page_idx = position // self.page_size
        offset = position % self.page_size
        
        page_id = pages[page_idx]
        self.k_cache[page_id, layer_idx, :, offset] = k
        self.v_cache[page_id, layer_idx, :, offset] = v
    
    def read_kv(
        self,
        sequence_id: str,
        layer_idx: int,
        length: int
    ) -> tuple:
        """读取KV缓存"""
        pages = self.sequence_pages[sequence_id]
        num_pages = (length + self.page_size - 1) // self.page_size
        
        k_list = []
        v_list = []
        
        for i in range(num_pages):
            page_id = pages[i]
            start = i * self.page_size
            end = min(start + self.page_size, length)
            actual_len = end - start
            
            k_list.append(self.k_cache[page_id, layer_idx, :, :actual_len])
            v_list.append(self.v_cache[page_id, layer_idx, :, :actual_len])
        
        k = torch.cat(k_list, dim=-2)
        v = torch.cat(v_list, dim=-2)
        
        return k, v

# 使用示例
cache = PagedKVCache(
    num_layers=12,
    num_heads=12,
    head_dim=64,
    page_size=16,
    max_pages=1000
)

# 分配页
sequence_id = "seq_001"
pages = cache.allocate_pages(sequence_id, num_pages=4)
print(f"为序列 {sequence_id} 分配了页: {pages}")

# 释放页
cache.free_sequence(sequence_id)
print(f"释放了序列 {sequence_id} 的页")
```

### 3.2 内存池管理

#### 内存池实现
```python
"""
内存池管理：预分配和复用内存
- 减少内存分配开销
- 避免内存碎片
- 提高内存利用率

@author erik.zhou
"""

class MemoryPool:
    """内存池"""
    
    def __init__(
        self,
        max_batch_size: int,
        max_seq_length: int,
        hidden_size: int,
        device: str = "cuda"
    ):
        self.max_batch_size = max_batch_size
        self.max_seq_length = max_seq_length
        self.hidden_size = hidden_size
        self.device = device
        
        # 预分配内存池
        self.input_pool = torch.zeros(
            max_batch_size, max_seq_length, hidden_size,
            device=device
        )
        self.output_pool = torch.zeros(
            max_batch_size, max_seq_length, hidden_size,
            device=device
        )
        
        # 使用标记
        self.in_use = [False] * max_batch_size
    
    def allocate(self, batch_size: int, seq_length: int):
        """分配内存"""
        if batch_size > self.max_batch_size:
            raise ValueError(f"批次大小 {batch_size} 超过最大值 {self.max_batch_size}")
        
        # 查找可用槽位
        start_idx = None
        for i in range(self.max_batch_size - batch_size + 1):
            if all(not self.in_use[j] for j in range(i, i + batch_size)):
                start_idx = i
                break
        
        if start_idx is None:
            raise RuntimeError("内存池已满")
        
        # 标记为使用中
        for i in range(start_idx, start_idx + batch_size):
            self.in_use[i] = True
        
        # 返回内存视图
        input_buffer = self.input_pool[start_idx:start_idx+batch_size, :seq_length]
        output_buffer = self.output_pool[start_idx:start_idx+batch_size, :seq_length]
        
        return start_idx, input_buffer, output_buffer
    
    def free(self, start_idx: int, batch_size: int):
        """释放内存"""
        for i in range(start_idx, start_idx + batch_size):
            self.in_use[i] = False
    
    def get_usage(self) -> float:
        """获取内存使用率"""
        used = sum(self.in_use)
        return used / self.max_batch_size

# 使用示例
pool = MemoryPool(
    max_batch_size=32,
    max_seq_length=512,
    hidden_size=768,
    device="cuda" if torch.cuda.is_available() else "cpu"
)

# 分配内存
start_idx, input_buf, output_buf = pool.allocate(batch_size=4, seq_length=128)
print(f"分配了内存，起始索引: {start_idx}")
print(f"内存使用率: {pool.get_usage():.2%}")

# 释放内存
pool.free(start_idx, batch_size=4)
print(f"释放后内存使用率: {pool.get_usage():.2%}")
```

## 4. 性能监控

### 4.1 指标收集

#### 性能监控器
```python
"""
性能监控：收集和分析性能指标
- 延迟监控
- 吞吐量监控
- 资源使用监控

@author erik.zhou
"""

from dataclasses import dataclass, field
from typing import List
import statistics

@dataclass
class PerformanceMetrics:
    """性能指标"""
    latencies: List[float] = field(default_factory=list)
    batch_sizes: List[int] = field(default_factory=list)
    queue_lengths: List[int] = field(default_factory=list)
    throughputs: List[float] = field(default_factory=list)
    
    def add_request(
        self,
        latency: float,
        batch_size: int,
        queue_length: int,
        throughput: float
    ):
        """添加请求指标"""
        self.latencies.append(latency)
        self.batch_sizes.append(batch_size)
        self.queue_lengths.append(queue_length)
        self.throughputs.append(throughput)
    
    def get_summary(self) -> Dict:
        """获取统计摘要"""
        if not self.latencies:
            return {}
        
        return {
            'avg_latency': statistics.mean(self.latencies),
            'p50_latency': statistics.median(self.latencies),
            'p95_latency': self._percentile(self.latencies, 0.95),
            'p99_latency': self._percentile(self.latencies, 0.99),
            'avg_batch_size': statistics.mean(self.batch_sizes),
            'avg_queue_length': statistics.mean(self.queue_lengths),
            'avg_throughput': statistics.mean(self.throughputs),
            'total_requests': len(self.latencies)
        }
    
    def _percentile(self, data: List[float], percentile: float) -> float:
        """计算百分位数"""
        sorted_data = sorted(data)
        index = int(len(sorted_data) * percentile)
        return sorted_data[min(index, len(sorted_data) - 1)]


class PerformanceMonitor:
    """性能监控器"""
    
    def __init__(self):
        self.metrics = PerformanceMetrics()
        self.start_time = time.time()
    
    def record_request(
        self,
        request_start: float,
        batch_size: int,
        queue_length: int,
        tokens_generated: int
    ):
        """记录请求"""
        latency = time.time() - request_start
        throughput = tokens_generated / latency if latency > 0 else 0
        
        self.metrics.add_request(
            latency=latency,
            batch_size=batch_size,
            queue_length=queue_length,
            throughput=throughput
        )
    
    def print_summary(self):
        """打印性能摘要"""
        summary = self.metrics.get_summary()
        
        print("\n=== 性能摘要 ===")
        print(f"总请求数: {summary.get('total_requests', 0)}")
        print(f"平均延迟: {summary.get('avg_latency', 0):.3f}s")
```

### 4.2 实时监控

#### 实时监控仪表板
```python
"""
实时监控：实时显示性能指标
- 实时延迟
- 实时吞吐量
- 资源使用情况

@author erik.zhou
"""

import threading
from collections import deque

class RealtimeMonitor:
    """实时监控器"""
    
    def __init__(self, window_size: int = 100):
        self.window_size = window_size
        
        # 滑动窗口
        self.recent_latencies = deque(maxlen=window_size)
        self.recent_throughputs = deque(maxlen=window_size)
        self.recent_batch_sizes = deque(maxlen=window_size)
        
        self.lock = threading.Lock()
        self.monitoring = False
    
    def record(self, latency: float, throughput: float, batch_size: int):
        """记录指标"""
        with self.lock:
            self.recent_latencies.append(latency)
            self.recent_throughputs.append(throughput)
            self.recent_batch_sizes.append(batch_size)
    
    def get_current_stats(self) -> Dict:
        """获取当前统计"""
        with self.lock:
            if not self.recent_latencies:
                return {}
            
            return {
                'current_latency': self.recent_latencies[-1],
                'avg_latency': statistics.mean(self.recent_latencies),
                'current_throughput': self.recent_throughputs[-1],
                'avg_throughput': statistics.mean(self.recent_throughputs),
                'current_batch_size': self.recent_batch_sizes[-1],
                'avg_batch_size': statistics.mean(self.recent_batch_sizes)
            }
    
    def start_monitoring(self, interval: float = 1.0):
        """启动监控线程"""
        self.monitoring = True
        
        def monitor_loop():
            while self.monitoring:
                stats = self.get_current_stats()
                if stats:
                    print(f"\r实时监控 - "
                          f"延迟: {stats['avg_latency']:.3f}s | "
                          f"吞吐量: {stats['avg_throughput']:.1f} tokens/s | "
                          f"批次: {stats['avg_batch_size']:.1f}",
                          end='', flush=True)
                time.sleep(interval)
        
        self.monitor_thread = threading.Thread(target=monitor_loop, daemon=True)
        self.monitor_thread.start()
    
    def stop_monitoring(self):
        """停止监控"""
        self.monitoring = False
        if hasattr(self, 'monitor_thread'):
            self.monitor_thread.join()

# 使用示例
realtime_monitor = RealtimeMonitor(window_size=50)
realtime_monitor.start_monitoring(interval=0.5)

        latency=0.1 + i * 0.01,

realtime_monitor.stop_monitoring()
```

### 5.1 完整批处理系统

#### 生产级批处理服务
```python
"""
生产级批处理系统：整合所有优化技术
- 动态批处理
- 优先级调度
- 内存管理
- 性能监控

@author erik.zhou
"""

class ProductionBatchService:
    """生产级批处理服务"""
    
    def __init__(
        self,
        model_name: str,
        max_batch_size: int = 32,
        max_wait_time: float = 0.05,
        enable_monitoring: bool = True
    ):
        # 初始化模型
        self.model = AutoModelForCausalLM.from_pretrained(model_name)
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model.eval()
        
        # 配置参数
        self.max_batch_size = max_batch_size
        self.max_wait_time = max_wait_time
        
        # 请求队列
        self.request_heap = []
        self.lock = asyncio.Lock()
        
        # 性能监控
        self.enable_monitoring = enable_monitoring
        if enable_monitoring:
            self.monitor = PerformanceMonitor()
            self.realtime_monitor = RealtimeMonitor()
            self.realtime_monitor.start_monitoring()
        
        # 内存池
        self.memory_pool = MemoryPool(
            max_batch_size=max_batch_size,
            max_seq_length=512,
            hidden_size=768
        )
        
        self.running = False
    
    async def add_request(
        self,
        prompt: str,
        max_tokens: int = 50,
        priority: Priority = Priority.NORMAL
    ) -> str:
        """添加请求"""
        request_start = time.time()
        
        request = PriorityRequest(
            priority=priority,
            timestamp=time.time(),
            id=str(uuid.uuid4()),
            prompt=prompt,
            max_tokens=max_tokens,
            future=asyncio.Future()
        )
        
        async with self.lock:
            heapq.heappush(self.request_heap, request)
            queue_length = len(self.request_heap)
        
        result = await request.future
        
        # 记录性能指标
        if self.enable_monitoring:
            self.monitor.record_request(
                request_start=request_start,
                batch_size=1,
                queue_length=queue_length,
                tokens_generated=max_tokens
            )
        
        return result
    
    async def process_loop(self):
        """处理循环"""
        self.running = True
        
        while self.running:
            batch = await self.collect_batch()
            
            if batch:
                await self.process_batch(batch)
            else:
                await asyncio.sleep(0.01)
    
    async def collect_batch(self) -> List[PriorityRequest]:
        """收集批次"""
        batch = []
        deadline = time.time() + self.max_wait_time
        
        async with self.lock:
            while len(batch) < self.max_batch_size and self.request_heap:
                if time.time() >= deadline and len(batch) > 0:
                    break
                
                request = heapq.heappop(self.request_heap)
                batch.append(request)
        
        return batch
    
    async def process_batch(self, batch: List[PriorityRequest]):
        """处理批次"""
        batch_start = time.time()
        
        prompts = [req.prompt for req in batch]
        max_tokens = max(req.max_tokens for req in batch)
        
        # 使用内存池
        try:
            start_idx, input_buf, output_buf = self.memory_pool.allocate(
                batch_size=len(batch),
                seq_length=max_tokens
            )
        except RuntimeError:
            # 内存池已满，回退到普通分配
            pass
        
        # 批量编码
        inputs = self.tokenizer(
            prompts,
            return_tensors="pt",
            padding=True,
            truncation=True
        )
        
        # 批量推理
        with torch.no_grad():
            outputs = self.model.generate(
                **inputs,
                max_new_tokens=max_tokens,
                pad_token_id=self.tokenizer.eos_token_id
            )
        
        # 解码结果
        results = self.tokenizer.batch_decode(outputs, skip_special_tokens=True)
        
        # 返回结果
        for request, result in zip(batch, results):
            request.future.set_result(result)
        
        # 释放内存
        try:
            self.memory_pool.free(start_idx, len(batch))
        except:
            pass
        
        # 记录实时监控
        if self.enable_monitoring:
            batch_latency = time.time() - batch_start
            throughput = max_tokens * len(batch) / batch_latency
            self.realtime_monitor.record(
                latency=batch_latency,
                throughput=throughput,
                batch_size=len(batch)
            )
    
    def start(self):
        """启动服务"""
        return asyncio.create_task(self.process_loop())
    
    def stop(self):
        """停止服务"""
        self.running = False
        if self.enable_monitoring:
            self.realtime_monitor.stop_monitoring()
            self.monitor.print_summary()

# 使用示例
async def production_example():
    service = ProductionBatchService(
        model_name="gpt2",
        max_batch_size=8,
        max_wait_time=0.05,
        enable_monitoring=True
    )
    
    # 启动服务
    service_task = service.start()
    
    # 发送请求
    tasks = [
        service.add_request("Hello, how are you?", priority=Priority.HIGH),
        service.add_request("What is AI?", priority=Priority.NORMAL),
        service.add_request("Explain deep learning", priority=Priority.LOW)
    ]
    
    results = await asyncio.gather(*tasks)
    print("结果:", results)
    
    # 停止服务
    service.stop()
    await service_task
```

### 5.2 负载均衡

#### 多实例负载均衡
```python
"""
负载均衡：在多个模型实例间分配请求
- 轮询策略
- 最少连接策略
- 加权轮询

@author erik.zhou
"""

from enum import Enum

class LoadBalanceStrategy(Enum):
    """负载均衡策略"""
    ROUND_ROBIN = "round_robin"
    LEAST_CONNECTIONS = "least_connections"
    WEIGHTED_ROUND_ROBIN = "weighted_round_robin"

class LoadBalancer:
    """负载均衡器"""
    
    def __init__(
        self,
        num_instances: int = 4,
        strategy: LoadBalanceStrategy = LoadBalanceStrategy.ROUND_ROBIN
    ):
        self.num_instances = num_instances
        self.strategy = strategy
        
        # 实例状态
        self.instance_connections = [0] * num_instances
        self.instance_weights = [1] * num_instances
        self.current_index = 0
    
    def select_instance(self) -> int:
        """选择实例"""
        if self.strategy == LoadBalanceStrategy.ROUND_ROBIN:
            return self._round_robin()
        elif self.strategy == LoadBalanceStrategy.LEAST_CONNECTIONS:
            return self._least_connections()
        elif self.strategy == LoadBalanceStrategy.WEIGHTED_ROUND_ROBIN:
            return self._weighted_round_robin()
        else:
            return 0
    
    def _round_robin(self) -> int:
        """轮询策略"""
        instance = self.current_index
        self.current_index = (self.current_index + 1) % self.num_instances
        return instance
    
    def _least_connections(self) -> int:
        """最少连接策略"""
        return self.instance_connections.index(min(self.instance_connections))
    
    def _weighted_round_robin(self) -> int:
        """加权轮询策略"""
        # 简化实现：根据权重选择
        total_weight = sum(self.instance_weights)
        target = (self.current_index % total_weight)
        
        cumulative = 0
        for i, weight in enumerate(self.instance_weights):
            cumulative += weight
            if target < cumulative:
                self.current_index += 1
                return i
        
        return 0
    
    def acquire(self, instance_id: int):
        """获取实例"""
        self.instance_connections[instance_id] += 1
    
    def release(self, instance_id: int):
        """释放实例"""
        self.instance_connections[instance_id] -= 1

# 使用示例
balancer = LoadBalancer(
    num_instances=4,
    strategy=LoadBalanceStrategy.LEAST_CONNECTIONS
)

# 模拟请求分配
for i in range(10):
    instance = balancer.select_instance()
    balancer.acquire(instance)
    print(f"请求 {i} 分配到实例 {instance}")
    
    # 模拟处理完成
    if i % 3 == 0:
        balancer.release(instance)
```
