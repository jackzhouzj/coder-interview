## 1. 模型量化

### 1.1 量化基础

#### 量化原理
```python
"""
量化：将高精度浮点数转换为低精度表示
- FP32 -> FP16/BF16：减少一半内存
- FP32 -> INT8：减少75%内存
- FP32 -> INT4：减少87.5%内存

@author erik.zhou
"""

import torch
import torch.nn as nn
from transformers import AutoModelForCausalLM, AutoTokenizer

class QuantizationDemo:
    """量化演示类"""
    
    def __init__(self, model_name: str):
        self.model_name = model_name
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
    
    def load_fp32_model(self):
        """加载FP32模型"""
        model = AutoModelForCausalLM.from_pretrained(
            self.model_name,
            torch_dtype=torch.float32
        )
        return model
    
    def load_fp16_model(self):
        """加载FP16模型"""
        model = AutoModelForCausalLM.from_pretrained(
            self.model_name,
            torch_dtype=torch.float16
        )
        return model
    
    def load_int8_model(self):
        """加载INT8量化模型"""
        from transformers import BitsAndBytesConfig
        
        quantization_config = BitsAndBytesConfig(
            load_in_8bit=True,
            llm_int8_threshold=6.0
        )
        
        model = AutoModelForCausalLM.from_pretrained(
            self.model_name,
            quantization_config=quantization_config,
            device_map="auto"
        )
        return model
    
    def compare_model_sizes(self):
        """比较不同精度模型的大小"""
        import os
        
        # FP32模型
        fp32_model = self.load_fp32_model()
        fp32_size = sum(p.numel() * p.element_size() for p in fp32_model.parameters())
        
        # FP16模型
        fp16_model = self.load_fp16_model()
        fp16_size = sum(p.numel() * p.element_size() for p in fp16_model.parameters())
        
        print(f"FP32模型大小: {fp32_size / 1024**3:.2f} GB")
        print(f"FP16模型大小: {fp16_size / 1024**3:.2f} GB")
        print(f"压缩比: {fp32_size / fp16_size:.2f}x")

# 使用示例
demo = QuantizationDemo("gpt2")
demo.compare_model_sizes()
```

```python
"""
GPTQ：基于梯度的后训练量化
- 支持INT4/INT8量化
- 保持模型精度
- 显著减少内存占用

@author erik.zhou

from transformers import AutoModelForCausalLM, GPTQConfig

class GPTQQuantizer:
    """GPTQ量化器"""
    
    def __init__(self, model_name: str):

    def quantize_model(self, bits: int = 4):
        """量化模型"""

            dataset="c4",
            tokenizer=self.model_name
        )
        
        # 加载并量化模型
        model = AutoModelForCausalLM.from_pretrained(
            self.model_name,
            quantization_config=gptq_config,
            device_map="auto"

        return model
    
    def save_quantized_model(self, model, save_path: str):
        """保存量化模型"""
        model.save_pretrained(save_path)
        print(f"量化模型已保存到: {save_path}")
    
    def benchmark_inference(self, model, prompt: str):
        """基准测试"""
        import time
        from transformers import AutoTokenizer
        
        tokenizer = AutoTokenizer.from_pretrained(self.model_name)
        inputs = tokenizer(prompt, return_tensors="pt").to(model.device)
        
        # 预热

        # 测试

        outputs = model.generate(**inputs, max_new_tokens=100)
        end_time = time.time()
        
        tokens_per_second = 100 / (end_time - start_time)
        print(f"推理速度: {tokens_per_second:.2f} tokens/s")
        
        return tokenizer.decode(outputs[0])
```

## 2. 知识蒸馏

### 2.1 蒸馏基础

#### 标准蒸馏
```python
"""
知识蒸馏：将大模型知识迁移到小模型
- Teacher模型：大型预训练模型
- Student模型：小型目标模型
- 蒸馏损失：KL散度 + 交叉熵

@author erik.zhou
"""

import torch
import torch.nn as nn
import torch.nn.functional as F
from transformers import AutoModelForCausalLM, AutoTokenizer

class KnowledgeDistillation:
    """知识蒸馏训练器"""
    
    def __init__(
        self,
        teacher_model_name: str,
        student_model_name: str,
        temperature: float = 2.0,
        alpha: float = 0.5
    ):
        self.teacher = AutoModelForCausalLM.from_pretrained(teacher_model_name)
        self.student = AutoModelForCausalLM.from_pretrained(student_model_name)
        self.tokenizer = AutoTokenizer.from_pretrained(teacher_model_name)
        
        self.temperature = temperature
        self.alpha = alpha
        
        # 冻结教师模型
        for param in self.teacher.parameters():
            param.requires_grad = False
        self.teacher.eval()
    
    def distillation_loss(
        self,
        student_logits: torch.Tensor,
        teacher_logits: torch.Tensor,
        labels: torch.Tensor
    ):
        """计算蒸馏损失"""
        # 软标签损失（KL散度）
        soft_loss = F.kl_div(
            F.log_softmax(student_logits / self.temperature, dim=-1),
            F.softmax(teacher_logits / self.temperature, dim=-1),
            reduction='batchmean'
        ) * (self.temperature ** 2)
        
        # 硬标签损失（交叉熵）
        hard_loss = F.cross_entropy(
            student_logits.view(-1, student_logits.size(-1)),
            labels.view(-1)
        )
        
        # 组合损失
        loss = self.alpha * soft_loss + (1 - self.alpha) * hard_loss
        return loss
    
    def train_step(self, batch):
        """训练步骤"""
        input_ids = batch['input_ids']
        attention_mask = batch['attention_mask']
        labels = batch['labels']
        
        # 教师模型前向传播
        with torch.no_grad():
            teacher_outputs = self.teacher(
                input_ids=input_ids,
                attention_mask=attention_mask
            )
            teacher_logits = teacher_outputs.logits
        
        # 学生模型前向传播
        student_outputs = self.student(
            input_ids=input_ids,
            attention_mask=attention_mask
        )
        student_logits = student_outputs.logits
        
        # 计算损失
        loss = self.distillation_loss(student_logits, teacher_logits, labels)
        
        return loss

# 使用示例
distiller = KnowledgeDistillation(
    teacher_model_name="gpt2-large",
    student_model_name="gpt2",
    temperature=2.0,
    alpha=0.5
)
```

## 3. 剪枝技术

### 3.1 结构化剪枝

#### 注意力头剪枝
```python
"""
结构化剪枝：移除整个结构单元
- 注意力头剪枝
- 层剪枝
- 神经元剪枝

@author erik.zhou
"""

import torch
import torch.nn as nn
from transformers import AutoModelForCausalLM

class AttentionHeadPruner:
    """注意力头剪枝器"""
    
    def __init__(self, model_name: str):
        self.model = AutoModelForCausalLM.from_pretrained(model_name)
        self.head_importance = {}
    
    def compute_head_importance(self, dataloader):
        """计算注意力头重要性"""
        self.model.eval()
        
        for batch in dataloader:
            with torch.no_grad():
                outputs = self.model(**batch, output_attentions=True)
                attentions = outputs.attentions
                
                # 计算每个头的平均注意力分数
                for layer_idx, layer_attention in enumerate(attentions):
                    if layer_idx not in self.head_importance:
                        self.head_importance[layer_idx] = []
                    
                    # layer_attention shape: [batch, num_heads, seq_len, seq_len]
                    head_scores = layer_attention.mean(dim=[0, 2, 3])
                    self.head_importance[layer_idx].append(head_scores)
        
        # 平均所有批次的重要性
        for layer_idx in self.head_importance:
            self.head_importance[layer_idx] = torch.stack(
                self.head_importance[layer_idx]
            ).mean(dim=0)
        
        return self.head_importance
    
    def prune_heads(self, prune_ratio: float = 0.2):
        """剪枝注意力头"""
        heads_to_prune = {}
        
        for layer_idx, importance in self.head_importance.items():
            num_heads = len(importance)
            num_prune = int(num_heads * prune_ratio)
            
            # 选择重要性最低的头进行剪枝
            _, indices = torch.topk(importance, num_prune, largest=False)
            heads_to_prune[layer_idx] = indices.tolist()
        
        # 执行剪枝
        self.model.prune_heads(heads_to_prune)
        
        print(f"剪枝完成，移除了 {prune_ratio*100}% 的注意力头")
        return self.model

# 使用示例
```

### 3.2 非结构化剪枝

#### 权重剪枝
```python
"""
非结构化剪枝：移除单个权重
- 幅度剪枝
- 梯度剪枝
- 稀疏训练

@author erik.zhou
"""

import torch.nn.utils.prune as prune

class WeightPruner:
    """权重剪枝器"""
    
    def __init__(self, model):
        self.model = model
    
    def magnitude_pruning(self, amount: float = 0.3):
        """幅度剪枝：移除绝对值最小的权重"""
        for name, module in self.model.named_modules():
            if isinstance(module, nn.Linear):
                prune.l1_unstructured(module, name='weight', amount=amount)
                # 永久移除剪枝掩码
                prune.remove(module, 'weight')
        
        print(f"幅度剪枝完成，移除了 {amount*100}% 的权重")
        return self.model
    
    def structured_pruning(self, amount: float = 0.3):
        """结构化剪枝：移除整个神经元"""
        for name, module in self.model.named_modules():
            if isinstance(module, nn.Linear):
                prune.ln_structured(
                    module,
                    name='weight',
                    amount=amount,
                    n=2,
                    dim=0
                )
                prune.remove(module, 'weight')
        
        print(f"结构化剪枝完成，移除了 {amount*100}% 的神经元")
        return self.model
    
    def global_pruning(self, amount: float = 0.3):
        """全局剪枝：在所有层中统一剪枝"""
        parameters_to_prune = []
        
        for name, module in self.model.named_modules():
            if isinstance(module, nn.Linear):
                parameters_to_prune.append((module, 'weight'))
        
        prune.global_unstructured(
            parameters_to_prune,
            pruning_method=prune.L1Unstructured,
            amount=amount
        )
        
        # 移除剪枝掩码
        for module, _ in parameters_to_prune:
            prune.remove(module, 'weight')
        
        print(f"全局剪枝完成，移除了 {amount*100}% 的权重")
        return self.model

# 使用示例
from transformers import AutoModelForCausalLM

model = AutoModelForCausalLM.from_pretrained("gpt2")
pruner = WeightPruner(model)
pruned_model = pruner.magnitude_pruning(amount=0.3)
```

## 4. 算子融合

### 4.1 基础算子融合

#### 融合优化
```python
"""
算子融合：合并多个操作减少内存访问
- LayerNorm + Linear融合
- Attention融合
- FFN融合

@author erik.zhou
"""

import torch
import torch.nn as nn

class FusedLinearLayerNorm(nn.Module):
    """融合的Linear + LayerNorm层"""
    
    def __init__(self, in_features: int, out_features: int, eps: float = 1e-5):
        super().__init__()
        self.linear = nn.Linear(in_features, out_features)
        self.layer_norm = nn.LayerNorm(out_features, eps=eps)
    
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """融合前向传播"""
        # 原始实现：两次内存访问
        # x = self.linear(x)
        # x = self.layer_norm(x)
        
        # 融合实现：一次内存访问
        x = self.linear(x)
        return self.layer_norm(x)


class FusedAttention(nn.Module):
    """融合的注意力机制"""
    
    def __init__(self, hidden_size: int, num_heads: int):
        super().__init__()
        self.hidden_size = hidden_size
        self.num_heads = num_heads
        self.head_dim = hidden_size // num_heads
        
        # 融合QKV投影
        self.qkv = nn.Linear(hidden_size, 3 * hidden_size)
        self.out_proj = nn.Linear(hidden_size, hidden_size)
    
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """融合注意力计算"""
        batch_size, seq_len, _ = x.shape
        
        # 一次计算QKV
        qkv = self.qkv(x)
        qkv = qkv.reshape(batch_size, seq_len, 3, self.num_heads, self.head_dim)
        qkv = qkv.permute(2, 0, 3, 1, 4)
        q, k, v = qkv[0], qkv[1], qkv[2]
        
        # 融合的注意力计算
        attn = torch.matmul(q, k.transpose(-2, -1)) / (self.head_dim ** 0.5)
        attn = torch.softmax(attn, dim=-1)
        out = torch.matmul(attn, v)
        
        # 重组输出
        out = out.transpose(1, 2).reshape(batch_size, seq_len, self.hidden_size)
        out = self.out_proj(out)
        
        return out

# 使用示例
fused_ln = FusedLinearLayerNorm(768, 768)
fused_attn = FusedAttention(768, 12)
```

### 4.2 Flash Attention

#### Flash Attention实现
```python
"""
Flash Attention：内存高效的注意力机制
- 分块计算
- 减少HBM访问
- 支持长序列

@author erik.zhou
"""

try:
    from flash_attn import flash_attn_func
    FLASH_ATTN_AVAILABLE = True
except ImportError:
    FLASH_ATTN_AVAILABLE = False

class FlashAttentionWrapper(nn.Module):
    """Flash Attention包装器"""
    
    def __init__(self, hidden_size: int, num_heads: int):
        super().__init__()
        self.hidden_size = hidden_size
        self.num_heads = num_heads
        self.head_dim = hidden_size // num_heads
        
        self.qkv = nn.Linear(hidden_size, 3 * hidden_size)
        self.out_proj = nn.Linear(hidden_size, hidden_size)
    
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """使用Flash Attention的前向传播"""
        batch_size, seq_len, _ = x.shape
        
        # 计算QKV
        qkv = self.qkv(x)
        qkv = qkv.reshape(batch_size, seq_len, 3, self.num_heads, self.head_dim)
        q, k, v = qkv[:, :, 0], qkv[:, :, 1], qkv[:, :, 2]
        
        if FLASH_ATTN_AVAILABLE:
            # 使用Flash Attention
            out = flash_attn_func(q, k, v, causal=False)
        else:
            # 回退到标准注意力
            q = q.transpose(1, 2)
            k = k.transpose(1, 2)
            v = v.transpose(1, 2)
            
            attn = torch.matmul(q, k.transpose(-2, -1)) / (self.head_dim ** 0.5)
            attn = torch.softmax(attn, dim=-1)
            out = torch.matmul(attn, v)
            out = out.transpose(1, 2)
        
        # 重组输出
        out = out.reshape(batch_size, seq_len, self.hidden_size)
        out = self.out_proj(out)
        
        return out

# 使用示例
flash_attn = FlashAttentionWrapper(768, 12)
x = torch.randn(2, 512, 768)
output = flash_attn(x)
```

## 5. 综合优化实战

### 5.1 完整优化流程

#### 端到端优化
```python
"""
综合优化：结合多种技术
- 量化 + 剪枝
- 蒸馏 + 算子融合
- 完整优化流程

@author erik.zhou
"""

from transformers import AutoModelForCausalLM, AutoTokenizer
import time

class ComprehensiveOptimizer:
    """综合优化器"""
    
    def __init__(self, model_name: str):
        self.model_name = model_name
        self.model = AutoModelForCausalLM.from_pretrained(model_name)
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
    
    def optimize_pipeline(self):
        """完整优化流程"""
        print("开始优化流程...")
        
        # 1. 剪枝
        print("\n步骤1: 权重剪枝")
        pruner = WeightPruner(self.model)
        self.model = pruner.magnitude_pruning(amount=0.3)
        
        # 2. 量化
        print("\n步骤2: INT8量化")
        from transformers import BitsAndBytesConfig
        
        quantization_config = BitsAndBytesConfig(
            load_in_8bit=True
        )
        
        # 重新加载量化模型
        self.model = AutoModelForCausalLM.from_pretrained(
            self.model_name,
            quantization_config=quantization_config,
            device_map="auto"
        )
        
        print("\n优化完成！")
        return self.model
    
    def benchmark(self, prompt: str, max_tokens: int = 100):
        """性能基准测试"""
        inputs = self.tokenizer(prompt, return_tensors="pt").to(self.model.device)
        
        # 预热
        _ = self.model.generate(**inputs, max_new_tokens=10)
        
        # 测试
        start_time = time.time()
        outputs = self.model.generate(**inputs, max_new_tokens=max_tokens)
        end_time = time.time()
        
        latency = end_time - start_time
        throughput = max_tokens / latency
        
        print(f"\n性能指标:")
        print(f"延迟: {latency:.2f}s")
        print(f"吞吐量: {throughput:.2f} tokens/s")
        
        result = self.tokenizer.decode(outputs[0])
        return result, latency, throughput

# 使用示例
optimizer = ComprehensiveOptimizer("gpt2")
optimized_model = optimizer.optimize_pipeline()
result, latency, throughput = optimizer.benchmark("Hello, how are you?")
```

```python
class PerformanceComparator:
    """性能对比器"""
    
    def __init__(self, original_model_name: str):
        self.original_model_name = original_model_name
    
    def compare_models(self, optimized_model):
        """对比原始模型和优化模型"""
        # 加载原始模型
        original_model = AutoModelForCausalLM.from_pretrained(
            self.original_model_name
        )
        
        # 计算模型大小
        original_size = self._get_model_size(original_model)
        optimized_size = self._get_model_size(optimized_model)
        
        print("\n模型对比:")
        print(f"原始模型大小: {original_size:.2f} MB")
        print(f"优化模型大小: {optimized_size:.2f} MB")
        print(f"压缩比: {original_size / optimized_size:.2f}x")
        
        return {
            'original_size': original_size,
            'optimized_size': optimized_size,
            'compression_ratio': original_size / optimized_size
        }
    
    def _get_model_size(self, model):
        """计算模型大小（MB）"""
        param_size = sum(
            p.numel() * p.element_size() for p in model.parameters()
        )
        buffer_size = sum(
            b.numel() * b.element_size() for b in model.buffers()
        )
        size_mb = (param_size + buffer_size) / 1024 / 1024
        return size_mb
```
