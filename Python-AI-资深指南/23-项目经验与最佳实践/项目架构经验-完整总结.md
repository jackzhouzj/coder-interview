```python
from fastapi import FastAPI, Depends
from langchain.chat_models import ChatOpenAI
from langchain.vectorstores import Chroma
from langchain.embeddings import OpenAIEmbeddings

app = FastAPI(title="AI QA System")

# 依赖注入
def get_llm():
    """获取 LLM 实例"""
    return ChatOpenAI(
        model="gpt-3.5-turbo",
        temperature=0.7,
        max_tokens=1000
    )

def get_vector_store():
    """获取向量数据库实例"""
    return Chroma(
        persist_directory="./chroma_db",
        embedding_function=OpenAIEmbeddings()
    )

@app.post("/api/v1/chat")
async def chat(
    question: str,
    llm: ChatOpenAI = Depends(get_llm),
    vector_store: Chroma = Depends(get_vector_store)
):
    """问答接口"""
    # 1. 向量检索相关文档
    docs = vector_store.similarity_search(question, k=3)
    
    # 2. 构建上下文
    context = "\n".join([doc.page_content for doc in docs])
    
    # 3. 调用 LLM
    prompt = f"基于以下上下文回答问题：\n{context}\n\n问题：{question}"
    response = llm.predict(prompt)
    
    return {"answer": response}
```

```python
connections.connect(host="localhost", port="19530")

class VectorService:
    """向量检索服务"""
    
    def __init__(self):
        self.collection = Collection("documents")
        # 创建索引
        self.collection.create_index(
            field_name="embedding",
            index_params={
                "metric_type": "L2",
                "index_type": "IVF_FLAT",
                "params": {"nlist": 1024}
            }
        )
        self.collection.load()

        """向量检索"""

            "metric_type": "L2",
            "params": {"nprobe": 10}

            data=[query_embedding],
            anns_field="embedding",
            param=search_params,
            limit=top_k,
            output_fields=["content", "metadata"]
        )
        
        return results[0]
```

**优化 3：多级缓存策略**

```python
import redis
from functools import wraps
import pickle

class CacheService:
    """多级缓存服务"""
    
    def __init__(self):
        # L1: 本地缓存（进程内）
        self.local_cache = {}
        self.local_cache_size = 100
        
        # L2: Redis 缓存（分布式）
        self.redis_client = redis.Redis(
            host="localhost",
            port=6379,
            decode_responses=False
        )
    
    def get(self, key: str):
        """获取缓存"""
        # 先查本地缓存
        if key in self.local_cache:
            return self.local_cache[key]
        
        # 再查 Redis
        value = self.redis_client.get(key)
        if value:
            value = pickle.loads(value)
            # 回写本地缓存
            self._set_local(key, value)
            return value
        
        return None
    
    def set(self, key: str, value, ttl: int = 3600):
        """设置缓存"""
        # 写入 Redis
        self.redis_client.setex(
            key,
            ttl,
            pickle.dumps(value)
        )
        # 写入本地缓存
        self._set_local(key, value)
    
    def _set_local(self, key: str, value):
        """设置本地缓存（LRU）"""
        if len(self.local_cache) >= self.local_cache_size:
            # 删除最旧的
            self.local_cache.pop(next(iter(self.local_cache)))
        self.local_cache[key] = value

# 缓存装饰器
def cache_result(ttl: int = 3600):
    """缓存装饰器"""
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            # 生成缓存 key
            cache_key = f"{func.__name__}:{str(args)}:{str(kwargs)}"
            
            # 尝试从缓存获取
            cached = cache_service.get(cache_key)
            if cached is not None:
                return cached
            
            # 执行函数
            result = await func(*args, **kwargs)
            
            # 写入缓存
            cache_service.set(cache_key, result, ttl)
            
            return result
        return wrapper
    return decorator

# 使用示例
@cache_result(ttl=1800)
async def search_documents(query: str, top_k: int = 3):
    """检索文档（带缓存）"""
    return vector_service.search(query, top_k)

# ✅ 效果：
# - 缓存命中率：70%
# - 平均响应时间：50ms → 5ms
# - 减少 LLM API 调用 60%
```

```python
import grpc

import vector_service_pb2
import vector_service_pb2_grpc

class VectorServicer(vector_service_pb2_grpc.VectorServiceServicer):
    """向量服务实现"""
    
    def Search(self, request, context):
        """检索接口"""
```

```python
# 1. 永远要有 Plan B
class LLMService:
    """LLM 服务（带降级）"""
    
    def __init__(self):
        self.primary_llm = ChatOpenAI(model="gpt-4")
        self.fallback_llm = ChatOpenAI(model="gpt-3.5-turbo")
        self.cache = CacheService()
    
    async def chat(self, prompt: str):
        """对话（带降级和缓存）"""
        # 1. 先查缓存
        cached = self.cache.get(prompt)
        if cached:
            return cached
        
        try:
            # 2. 尝试主模型
            response = await self.primary_llm.apredict(prompt)
        except Exception as e:
            logger.warning(f"Primary LLM failed: {e}")
            try:
                # 3. 降级到备用模型
                response = await self.fallback_llm.apredict(prompt)

                logger.error(f"Fallback LLM failed: {e}")
```

## 📝 架构设计原则

### 1. KISS 原则（Keep It Simple, Stupid）

```python
# ❌ 过度设计
class AbstractFactoryBuilder:
    """抽象工厂建造者（过度设计）"""
    pass

# ✅ 简单直接
def create_user(name: str, email: str) -> User:
    """创建用户"""
    return User(name=name, email=email)
```

### 2. YAGNI 原则（You Aren't Gonna Need It）

```python
# ❌ 提前实现未来可能需要的功能
class User:
    def __init__(self):
        self.cache = {}  # 可能以后需要缓存
        self.queue = []  # 可能以后需要队列
        self.pool = []   # 可能以后需要连接池

# ✅ 只实现当前需要的功能
class User:
    def __init__(self, name: str, email: str):
        self.name = name
        self.email = email
```

### 3. 单一职责原则

```python
# ❌ 一个类做太多事
class UserService:
    def create_user(self): pass
    def send_email(self): pass
    def log_activity(self): pass
    def generate_report(self): pass

# ✅ 职责分离
class UserService:
    def create_user(self): pass

class EmailService:
    def send_email(self): pass

class LogService:
    def log_activity(self): pass

class ReportService:
    def generate_report(self): pass
```
