## 🎯 案例一：接口响应慢优化

### 问题描述

**现象**：用户列表接口响应时间从 200ms 增加到 5000ms

**影响**：用户体验极差，投诉增多

### 问题排查

#### 1. 添加性能监控

```python
import time
import logging
from functools import wraps

logger = logging.getLogger(__name__)

def performance_monitor(func):
    """性能监控装饰器"""
    @wraps(func)
    async def wrapper(*args, **kwargs):
        start_time = time.time()
        
        # 记录开始
        logger.info(f"[PERF] {func.__name__} started")
        
        try:
            result = await func(*args, **kwargs)
            return result
        finally:
            # 记录耗时
            duration = time.time() - start_time
            logger.info(f"[PERF] {func.__name__} took {duration:.3f}s")
            
            # 慢查询告警
            if duration > 1.0:
                logger.warning(
                    f"[SLOW] {func.__name__} took {duration:.3f}s",
                    extra={
                        "function": func.__name__,
                        "duration": duration,
                        "args": str(args)[:100]
                    }
                )
    
    return wrapper

# 使用示例
@app.get("/api/users")
@performance_monitor
async def get_users(page: int = 1, size: int = 20):
    """获取用户列表"""
    return await user_service.get_users(page, size)
```

#### 2. 分段计时定位瓶颈

```python
import time

class PerformanceTracker:
    """性能追踪器"""
    
    def __init__(self):
        self.checkpoints = []
        self.start_time = time.time()
    
    def checkpoint(self, name: str):
        """记录检查点"""
        current_time = time.time()
        duration = current_time - self.start_time
        self.checkpoints.append({
            "name": name,
            "time": current_time,
            "duration": duration
        })
    
    def report(self):
        """生成报告"""
        total = time.time() - self.start_time
        
        print(f"\n性能分析报告 (总耗时: {total:.3f}s)")
        print("-" * 60)
        
        prev_time = self.start_time
        for cp in self.checkpoints:
            step_duration = cp["time"] - prev_time
            percentage = (step_duration / total) * 100
            print(f"{cp['name']:30s} {step_duration:.3f}s ({percentage:.1f}%)")
            prev_time = cp["time"]

# 使用示例
async def get_users(page: int, size: int):
    """获取用户列表（带性能追踪）"""
    tracker = PerformanceTracker()
    
    # 1. 查询用户
    users = await db.query(User).offset(page * size).limit(size).all()
    tracker.checkpoint("查询用户")
    
    # 2. 查询用户角色

    roles = await db.query(Role).filter(Role.user_id.in_(user_ids)).all()
    tracker.checkpoint("查询角色")
    
    # 3. 查询用户权限
    permissions = await db.query(Permission).filter(
        Permission.user_id.in_(user_ids)
    ).all()
    tracker.checkpoint("查询权限")
    
    # 4. 组装数据
    result = assemble_user_data(users, roles, permissions)
    tracker.checkpoint("组装数据")
    
    # 生成报告
    tracker.report()
    
    return result

# 输出示例：

# ------------------------------------------------------------
# 查询用户                        0.120s (2.5%)
# 查询角色                        2.340s (48.2%)  ← 瓶颈！
```

### 问题定位

**发现**：查询角色和权限时产生了 N+1 查询问题


### 优化方案

#### 方案一：批量查询（推荐）

```python
# ✅ 优化后：批量查询
async def get_users_optimized(page: int, size: int):
    """获取用户列表（批量查询）"""
    # 1. 查询用户
    users = await db.query(User).offset(page * size).limit(size).all()
    user_ids = [u.id for u in users]
    
    # 2. 批量查询角色（1次查询）
    roles = await db.query(Role).filter(Role.user_id.in_(user_ids)).all()
    roles_map = {}
    for role in roles:
        if role.user_id not in roles_map:
            roles_map[role.user_id] = []
        roles_map[role.user_id].append(role)
    
    # 3. 批量查询权限（1次查询）
    permissions = await db.query(Permission).filter(
        Permission.user_id.in_(user_ids)
    ).all()
    permissions_map = {}
    for perm in permissions:
        if perm.user_id not in permissions_map:
            permissions_map[perm.user_id] = []
        permissions_map[perm.user_id].append(perm)
    
    # 4. 组装数据
    result = []
    for user in users:
        result.append({
            "user": user,
            "roles": roles_map.get(user.id, []),
            "permissions": permissions_map.get(user.id, [])
        })
    
    return result
    # 总查询次数：1 + 1 + 1 = 3 次！
```

#### 方案三：添加缓存

```python
import redis
import json
from functools import wraps

redis_client = redis.Redis(host="localhost", port=6379, decode_responses=True)

def cache_result(ttl: int = 300):
    """缓存装饰器"""
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            # 生成缓存 key
            cache_key = f"{func.__name__}:{args}:{kwargs}"
            
            # 尝试从缓存获取
            cached = redis_client.get(cache_key)
            if cached:
                logger.info(f"Cache hit: {cache_key}")
                return json.loads(cached)
            
            # 执行函数
            result = await func(*args, **kwargs)
            
            # 写入缓存
            redis_client.setex(
                cache_key,
                ttl,
                json.dumps(result, default=str)
            )
            
            return result
        return wrapper
    return decorator

# 使用缓存
@cache_result(ttl=300)
async def get_users_cached(page: int, size: int):
    """获取用户列表（带缓存）"""
    return await get_users_optimized(page, size)
```

```python
import tracemalloc
import linecache

def display_top_memory(snapshot, key_type='lineno', limit=10):
    """显示内存占用最多的代码"""
    snapshot = snapshot.filter_traces((
        tracemalloc.Filter(False, "<frozen importlib._bootstrap>"),
        tracemalloc.Filter(False, "<unknown>"),
    ))
    
    top_stats = snapshot.statistics(key_type)
    
    print(f"Top {limit} lines:")
    for index, stat in enumerate(top_stats[:limit], 1):
        frame = stat.traceback[0]
        print(f"#{index}: {frame.filename}:{frame.lineno}: {stat.size / 1024 / 1024:.1f} MB")
        line = linecache.getline(frame.filename, frame.lineno).strip()
        if line:
```

```python
# ❌ 问题代码：缓存无限增长
class CacheService:
    """缓存服务（有内存泄漏）"""
    
    def __init__(self):
        self.cache = {}  # 永远不清理
    
    def get(self, key: str):
        return self.cache.get(key)
    
    def set(self, key: str, value):
        self.cache[key] = value  # 一直增长

# ✅ 优化：使用 LRU 缓存
from functools import lru_cache
from collections import OrderedDict

class LRUCache:
    """LRU 缓存"""
    
    def __init__(self, capacity: int = 1000):
        self.cache = OrderedDict()
        self.capacity = capacity
    
    def get(self, key: str):
        """获取缓存"""
        if key not in self.cache:
            return None
        
        # 移到末尾（最近使用）
        self.cache.move_to_end(key)
        return self.cache[key]
    
    def set(self, key: str, value):
        """设置缓存"""
        if key in self.cache:
            # 更新并移到末尾
            self.cache.move_to_end(key)
        
        self.cache[key] = value
        
        # 超过容量，删除最旧的
        if len(self.cache) > self.capacity:
            self.cache.popitem(last=False)

# ✅ 更好：使用 cachetools
from cachetools import LRUCache, TTLCache

# LRU 缓存（最多 1000 个）
lru_cache = LRUCache(maxsize=1000)

# TTL 缓存（5分钟过期）
ttl_cache = TTLCache(maxsize=1000, ttl=300)
```

#### 问题 3：循环引用导致内存无法释放

```python
# ❌ 问题代码：循环引用
class Node:
    """节点（有循环引用）"""
    
    def __init__(self, value):
        self.value = value
        self.parent = None
        self.children = []
    
    def add_child(self, child):
        """添加子节点"""
        child.parent = self  # 循环引用
        self.children.append(child)

# 创建节点树
root = Node("root")
child1 = Node("child1")
child2 = Node("child2")

root.add_child(child1)
root.add_child(child2)

# 即使删除 root，由于循环引用，内存无法释放
del root  # 内存仍然被占用

# ✅ 优化：使用弱引用
import weakref

class Node:
    """节点（使用弱引用）"""
    
    def __init__(self, value):
        self.value = value
        self.parent = None  # 使用弱引用
        self.children = []
    
    def add_child(self, child):
        """添加子节点"""
        child.parent = weakref.ref(self)  # 弱引用
        self.children.append(child)
    
    def get_parent(self):
        """获取父节点"""
        if self.parent is None:
            return None
        return self.parent()  # 解引用

# 现在删除 root 后，内存可以正常释放
root = Node("root")
child1 = Node("child1")
root.add_child(child1)
del root  # 内存正常释放
```

### 关键经验

```python
# ✅ 数据库优化最佳实践

# 1. 为常用查询字段添加索引
# 查询条件、JOIN 字段、ORDER BY 字段都要加索引

# 2. 避免 SELECT *
# ❌ 错误
SELECT * FROM users WHERE id = 1

# ✅ 正确
SELECT id, name, email FROM users WHERE id = 1

# 3. 使用 LIMIT 限制返回行数
SELECT * FROM orders ORDER BY created_at DESC LIMIT 100

# 4. 避免在索引字段上使用函数
# ❌ 错误：索引失效
WHERE DATE(created_at) = '2024-01-01'

# ✅ 正确：使用索引
WHERE created_at >= '2024-01-01' AND created_at < '2024-01-02'

# 5. 批量操作代替循环
# ❌ 错误：循环插入
for user in users:
    db.execute("INSERT INTO users VALUES (...)")

# ✅ 正确：批量插入
db.execute("INSERT INTO users VALUES (...), (...), (...)")
```
