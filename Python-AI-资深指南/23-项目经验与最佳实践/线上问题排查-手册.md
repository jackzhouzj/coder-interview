```python
from contextlib import contextmanager
from sqlalchemy.orm import Session

@contextmanager
def get_db_session():
```

#### 故障现象

```
现象：应用内存从 2GB 增长到 8GB，最终 OOM 被 kill
影响：服务中断 15 分钟
```

```python
@profile
```

**第二步：定位内存泄漏点**

```python
# 问题代码 ❌
class DataProcessor:
    def __init__(self):
        self.cache = {}  # 全局缓存，永不清理
    
    def process(self, data_id: str):
        if data_id not in self.cache:
            # 加载大量数据到内存
            self.cache[data_id] = self._load_data(data_id)
        return self.cache[data_id]
    
    def _load_data(self, data_id: str):
        # 加载 10MB 数据
        return [i for i in range(1000000)]

# 正确实现 ✅
from functools import lru_cache
from datetime import datetime, timedelta

class DataProcessor:
    def __init__(self, max_cache_size=100, cache_ttl=3600):
        self.cache = {}
        self.cache_time = {}
        self.max_cache_size = max_cache_size
        self.cache_ttl = cache_ttl
    
    def process(self, data_id: str):
        # 清理过期缓存
        self._cleanup_expired_cache()
        
        # 检查缓存大小
        if len(self.cache) >= self.max_cache_size:
            self._evict_oldest()
        
        if data_id not in self.cache:
            self.cache[data_id] = self._load_data(data_id)
            self.cache_time[data_id] = datetime.now()
        
        return self.cache[data_id]
    
    def _cleanup_expired_cache(self):
        """清理过期缓存"""
        now = datetime.now()
        expired_keys = [
            key for key, cache_time in self.cache_time.items()
            if (now - cache_time).seconds > self.cache_ttl
        ]
        for key in expired_keys:
            del self.cache[key]
            del self.cache_time[key]
    
    def _evict_oldest(self):
        """淘汰最旧的缓存"""
        if not self.cache_time:
            return
        oldest_key = min(self.cache_time, key=self.cache_time.get)
        del self.cache[oldest_key]
        del self.cache_time[oldest_key]
```

**第三步：使用生成器优化内存**
```python
# 问题代码 ❌ - 一次性加载所有数据
def process_all_users():
    users = User.query.all()  # 加载 100 万用户到内存

# 正确实现 ✅ - 使用生成器分批处理
def process_all_users_generator():
    """使用生成器分批处理"""
    batch_size = 1000
    offset = 0
    
    while True:
        users = User.query.limit(batch_size).offset(offset).all()
        if not users:
            break
        
        for user in users:
            yield process_user(user)
        
        offset += batch_size

# 使用方式

    save_result(result)
```

#### 故障总结

**根本原因**：
- 全局缓存无限增长，未设置过期时间
- 一次性加载大量数据到内存
- 未使用生成器处理大数据集

**解决方案**：
- 实现 LRU 缓存淘汰策略
- 设置缓存过期时间（TTL）
- 使用生成器分批处理数据

**数据对比**：
- 修复前：内存峰值 8GB，OOM 崩溃
- 修复后：内存稳定在 2GB，无 OOM

### 案例 3：Redis 连接池耗尽 🔴

#### 故障现象
```
时间：2024-05-10 10:00
现象：Redis 连接数达到上限，新请求无法获取连接
错误：redis.exceptions.ConnectionError: Too many connections
影响：所有依赖 Redis 的功能不可用
```

#### 排查过程

**第一步：检查 Redis 连接数**
```bash
# 查看当前连接数
redis-cli info clients
# connected_clients:1000
# maxclients:1000  # 已达上限

# 查看连接来源
redis-cli client list | awk '{print $2}' | cut -d: -f1 | sort | uniq -c | sort -rn
```

**第二步：定位连接泄漏**

```python
# 问题代码 ❌ - 每次创建新连接
import redis

def get_cache(key: str):
    # 每次都创建新连接，未复用
    r = redis.Redis(host='localhost', port=6379)
    return r.get(key)
    # 连接未关闭，导致泄漏

# 正确实现 ✅ - 使用连接池
import redis
from redis import ConnectionPool

# 全局连接池
redis_pool = ConnectionPool(
    host='localhost',
    port=6379,
    max_connections=50,  # 最大连接数
    decode_responses=True,
    socket_keepalive=True,
    socket_connect_timeout=5,
    retry_on_timeout=True
)

def get_redis_client():
    """获取 Redis 客户端（复用连接池）"""
    return redis.Redis(connection_pool=redis_pool)

def get_cache(key: str):
    """正确使用 Redis 连接"""
    client = get_redis_client()
    try:
        return client.get(key)
    except redis.RedisError as e:
        logger.error(f"Redis 错误: {e}")
        return None
```

**第三步：实现连接池监控**
```python
import logging
from contextlib import contextmanager

logger = logging.getLogger(__name__)

@contextmanager
def redis_connection():
    """Redis 连接上下文管理器"""
    client = get_redis_client()
    try:
        # 记录连接池状态
        pool_info = redis_pool.connection_kwargs
        logger.debug(f"Redis 连接池状态: {pool_info}")
        
        yield client
    except redis.RedisError as e:
        logger.error(f"Redis 操作失败: {e}")
        raise
    finally:
        # 连接自动归还到连接池
        pass

# 使用方式
with redis_connection() as client:
    client.set("key", "value")
    value = client.get("key")
```

#### 故障总结

**根本原因**：
- 每次请求创建新 Redis 连接
- 连接未正确关闭，导致泄漏
- Redis maxclients 配置过小

**解决方案**：
- 使用全局连接池复用连接
- 增加 Redis maxclients 到 5000
- 添加连接池监控告警

**数据对比**：
- 修复前：连接数 1000/1000，频繁报错
- 修复后：连接数稳定在 50 以内

## 应急处理流程

### 1. 故障响应流程

```
故障发现（监控告警）
    ↓
快速确认（5分钟内）
    ↓
影响评估（确定优先级）
    ↓
临时止血（恢复服务）
    ↓
根本修复（解决问题）
    ↓
复盘总结（避免重复）
```

### 2. 故障等级定义

| 等级 | 影响范围 | 响应时间 | 处理策略 |
|------|---------|---------|---------|
| P0 | 核心功能不可用，影响所有用户 | 5分钟 | 立即回滚/切换备用 |
| P1 | 重要功能异常，影响部分用户 | 15分钟 | 紧急修复上线 |
| P2 | 次要功能异常，影响少量用户 | 1小时 | 计划修复 |
| P3 | 体验问题，不影响功能 | 1天 | 正常迭代 |

```python
# 使用 Prometheus 监控
from prometheus_client import Counter, Histogram, Gauge
import time

# 请求计数器
request_count = Counter(
    'http_requests_total',
    'Total HTTP requests',
    ['method', 'endpoint', 'status']
)

# 响应时间直方图
request_duration = Histogram(
    'http_request_duration_seconds',
    'HTTP request duration',
    ['method', 'endpoint']
)

# 当前活跃连接数
active_connections = Gauge(
    'active_connections',
    'Number of active connections'
)

# 使用示例
def monitor_request(func):
    """请求监控装饰器"""
    def wrapper(*args, **kwargs):
        start_time = time.time()
        
        try:
            result = func(*args, **kwargs)
            status = 200
            return result
        except Exception as e:
            status = 500
            raise
        finally:
            duration = time.time() - start_time
            request_count.labels(
                method='GET',
                endpoint='/api/users',
                status=status
            ).inc()
            request_duration.labels(
                method='GET',
                endpoint='/api/users'
            ).observe(duration)
    
    return wrapper
```

```python
import logging
import json
from datetime import datetime

class JSONFormatter(logging.Formatter):
    """JSON 格式日志"""
    def format(self, record):
        log_data = {
            "timestamp": datetime.utcnow().isoformat(),
            "level": record.levelname,
            "logger": record.name,
            "message": record.getMessage(),
            "module": record.module,
            "function": record.funcName,
            "line": record.lineno
        }
        
        # 添加额外字段
        if hasattr(record, 'user_id'):
            log_data['user_id'] = record.user_id
        if hasattr(record, 'request_id'):
            log_data['request_id'] = record.request_id
        
        # 添加异常信息
        if record.exc_info:
            log_data['exception'] = self.formatException(record.exc_info)
        
        return json.dumps(log_data)

# 配置日志
logger = logging.getLogger(__name__)
handler = logging.StreamHandler()
handler.setFormatter(JSONFormatter())
logger.addHandler(handler)
logger.setLevel(logging.INFO)
```

```python
    BatchSpanProcessor(jaeger_exporter)
)

tracer = trace.get_tracer(__name__)

# 使用追踪
def process_request(request_id: str):
    with tracer.start_as_current_span("process_request") as span:
        span.set_attribute("request_id", request_id)
        
        # 业务逻辑
        with tracer.start_as_current_span("database_query"):
            result = query_database()
        
        with tracer.start_as_current_span("cache_update"):
            update_cache(result)
        
        return result
```
