## 数据加密概述

### 加密类型对比

| 类型 | 特点 | 用途 | 代表算法 |
|------|------|------|---------|
| 哈希 | 单向、不可逆 | 密码存储 | bcrypt、SHA-256 |
| 对称加密 | 加解密同一密钥 | 数据加密 | AES |
| 非对称加密 | 公钥加密、私钥解密 | 数字签名、密钥交换 | RSA、ECC |

```python
import bcrypt

def hash_password(password: str) -> str:
    """使用 bcrypt 哈希密码"""
    # 生成盐值并哈希
    salt = bcrypt.gensalt(rounds=12)  # rounds 越大越安全，但越慢
    hashed = bcrypt.hashpw(password.encode('utf-8'), salt)
    return hashed.decode('utf-8')

def verify_password(password: str, hashed: str) -> bool:
    """验证密码"""
    return bcrypt.checkpw(
        password.encode('utf-8'),
        hashed.encode('utf-8')
    )

# 使用示例
password = "MySecurePassword123!"

# 注册时：哈希密码
hashed_password = hash_password(password)
print(f"哈希后: {hashed_password}")

# 登录时：验证密码
is_valid = verify_password(password, hashed_password)
print(f"验证结果: {is_valid}")

# 错误密码
is_valid = verify_password("WrongPassword", hashed_password)
print(f"错误密码: {is_valid}")
```

### PBKDF2（标准库）

```python
import hashlib
import os
from base64 import b64encode, b64decode

def hash_password_pbkdf2(password: str, salt: bytes = None) -> str:
    """使用 PBKDF2 哈希密码"""
    if salt is None:
        salt = os.urandom(32)
    
    # PBKDF2-HMAC-SHA256
    key = hashlib.pbkdf2_hmac(
        'sha256',
        password.encode('utf-8'),
        salt,
        iterations=100000  # 迭代次数
    )
    
    # 存储格式：salt$hash
    storage = b64encode(salt).decode('utf-8') + '$' + b64encode(key).decode('utf-8')
    return storage

def verify_password_pbkdf2(password: str, stored: str) -> bool:
    """验证密码"""
    salt_b64, key_b64 = stored.split('$')
    salt = b64decode(salt_b64)
    stored_key = b64decode(key_b64)
    
    # 使用相同盐值重新计算
    key = hashlib.pbkdf2_hmac(
        'sha256',
        password.encode('utf-8'),
        salt,
        iterations=100000
    )
    
    return key == stored_key

# 使用示例
password = "MySecurePassword123!"
hashed = hash_password_pbkdf2(password)
print(f"PBKDF2 哈希: {hashed}")

is_valid = verify_password_pbkdf2(password, hashed)
print(f"验证结果: {is_valid}")
```

```python
from cryptography.fernet import Fernet
from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.kdf.pbkdf2 import PBKDF2
from cryptography.hazmat.backends import default_backend
import base64
import os

class AESCipher:
    """AES 加密工具类"""
    
    def __init__(self, password: str):
        """初始化（从密码派生密钥）"""
        self.password = password.encode()
    
    def _get_key(self, salt: bytes) -> bytes:
        """从密码派生密钥"""
        kdf = PBKDF2(
            algorithm=hashes.SHA256(),
            length=32,
            salt=salt,
            iterations=100000,
            backend=default_backend()
        )
        key = base64.urlsafe_b64encode(kdf.derive(self.password))
        return key
    
    def encrypt(self, plaintext: str) -> str:
        """加密"""
        # 生成随机盐值
        salt = os.urandom(16)
        
        # 派生密钥
        key = self._get_key(salt)
        
        # 加密
        f = Fernet(key)
        ciphertext = f.encrypt(plaintext.encode('utf-8'))
        
        # 返回：salt$ciphertext
        return base64.b64encode(salt).decode() + '$' + ciphertext.decode()
    
    def decrypt(self, encrypted: str) -> str:
        """解密"""
        # 分离盐值和密文
        salt_b64, ciphertext = encrypted.split('$')
        salt = base64.b64decode(salt_b64)
        
        # 派生密钥
        key = self._get_key(salt)
        
        # 解密
        f = Fernet(key)
        plaintext = f.decrypt(ciphertext.encode())
        
        return plaintext.decode('utf-8')

# 使用示例
cipher = AESCipher(password="MySecretKey123!")

# 加密
plaintext = "这是需要加密的敏感数据"
encrypted = cipher.encrypt(plaintext)
print(f"加密后: {encrypted}")

# 解密
decrypted = cipher.decrypt(encrypted)
print(f"解密后: {decrypted}")
```

### AES-GCM 模式

```python
from cryptography.hazmat.primitives.ciphers.aead import AESGCM
import os

class AESGCMCipher:
    """AES-GCM 加密（带认证）"""
    
    def __init__(self, key: bytes = None):
        """初始化"""
        if key is None:
            key = AESGCM.generate_key(bit_length=256)
        self.aesgcm = AESGCM(key)
        self.key = key
    
    def encrypt(self, plaintext: str, associated_data: str = None) -> tuple:
        """加密
        
        Args:
            plaintext: 明文
            associated_data: 关联数据（不加密但参与认证）
        
        Returns:
            (nonce, ciphertext)
        """
        nonce = os.urandom(12)  # 96-bit nonce
        
        ad = associated_data.encode() if associated_data else None
        
        ciphertext = self.aesgcm.encrypt(
            nonce,
            plaintext.encode(),
            ad
        )
        
        return nonce, ciphertext
    
    def decrypt(self, nonce: bytes, ciphertext: bytes, associated_data: str = None) -> str:
        """解密"""
        ad = associated_data.encode() if associated_data else None
        
        plaintext = self.aesgcm.decrypt(
            nonce,
            ciphertext,
            ad
        )
        
        return plaintext.decode()

# 使用示例
cipher = AESGCMCipher()

# 加密
plaintext = "敏感数据"
associated_data = "user_id=123"  # 关联数据
nonce, ciphertext = cipher.encrypt(plaintext, associated_data)

print(f"Nonce: {nonce.hex()}")
print(f"密文: {ciphertext.hex()}")

# 解密
decrypted = cipher.decrypt(nonce, ciphertext, associated_data)
print(f"解密: {decrypted}")
```

## 非对称加密

### RSA 加密

```python
from cryptography.hazmat.primitives.asymmetric import rsa, padding
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.hazmat.backends import default_backend

class RSACipher:
    """RSA 加密工具类"""
    
    def __init__(self):
        """生成密钥对"""
        self.private_key = rsa.generate_private_key(
            public_exponent=65537,
            key_size=2048,
            backend=default_backend()
        )
        self.public_key = self.private_key.public_key()
    
    def get_public_key_pem(self) -> str:
        """获取公钥（PEM 格式）"""
        pem = self.public_key.public_bytes(
            encoding=serialization.Encoding.PEM,
            format=serialization.PublicFormat.SubjectPublicKeyInfo
        )
        return pem.decode()
    
    def get_private_key_pem(self, password: str = None) -> str:
        """获取私钥（PEM 格式）"""
        encryption = serialization.NoEncryption()
        if password:
            encryption = serialization.BestAvailableEncryption(
                password.encode()
            )
        
        pem = self.private_key.private_bytes(
            encoding=serialization.Encoding.PEM,
            format=serialization.PrivateFormat.PKCS8,
            encryption_algorithm=encryption
        )
        return pem.decode()
    
    def encrypt(self, plaintext: str) -> bytes:
        """使用公钥加密"""
        ciphertext = self.public_key.encrypt(
            plaintext.encode(),
            padding.OAEP(
                mgf=padding.MGF1(algorithm=hashes.SHA256()),
                algorithm=hashes.SHA256(),
                label=None
            )
        )
        return ciphertext
    
    def decrypt(self, ciphertext: bytes) -> str:
        """使用私钥解密"""
        plaintext = self.private_key.decrypt(
            ciphertext,
            padding.OAEP(
                mgf=padding.MGF1(algorithm=hashes.SHA256()),
                algorithm=hashes.SHA256(),
                label=None
            )
        )
        return plaintext.decode()
    
    def sign(self, message: str) -> bytes:
        """使用私钥签名"""
        signature = self.private_key.sign(
            message.encode(),
            padding.PSS(
                mgf=padding.MGF1(hashes.SHA256()),
                salt_length=padding.PSS.MAX_LENGTH
            ),
            hashes.SHA256()
        )
        return signature
    
    def verify(self, message: str, signature: bytes) -> bool:
        """使用公钥验证签名"""
        try:
            self.public_key.verify(
                signature,
                message.encode(),
                padding.PSS(
                    mgf=padding.MGF1(hashes.SHA256()),
                    salt_length=padding.PSS.MAX_LENGTH
                ),
                hashes.SHA256()
            )
            return True
        except Exception:
            return False

# 使用示例
rsa_cipher = RSACipher()

# 1. 加密和解密
plaintext = "敏感数据"
ciphertext = rsa_cipher.encrypt(plaintext)
print(f"加密: {ciphertext.hex()}")

decrypted = rsa_cipher.decrypt(ciphertext)
print(f"解密: {decrypted}")

# 2. 签名和验证
message = "重要消息"
signature = rsa_cipher.sign(message)
print(f"签名: {signature.hex()}")

is_valid = rsa_cipher.verify(message, signature)
print(f"验证: {is_valid}")

# 3. 导出密钥
public_key_pem = rsa_cipher.get_public_key_pem()
print(f"公钥:\n{public_key_pem}")

private_key_pem = rsa_cipher.get_private_key_pem(password="KeyPassword123")
print(f"私钥:\n{private_key_pem}")
```

### 混合加密（RSA + AES）

```python
from cryptography.fernet import Fernet
import base64

class HybridCipher:
    """混合加密：RSA 加密 AES 密钥，AES 加密数据"""
    
    def __init__(self):
        self.rsa_cipher = RSACipher()
    
    def encrypt(self, plaintext: str) -> dict:
        """加密大数据"""
        # 1. 生成随机 AES 密钥
        aes_key = Fernet.generate_key()
        
        # 2. 使用 AES 加密数据
        fernet = Fernet(aes_key)
        encrypted_data = fernet.encrypt(plaintext.encode())
        
        # 3. 使用 RSA 加密 AES 密钥
        encrypted_key = self.rsa_cipher.encrypt(aes_key.decode())
        
        return {
            'encrypted_key': base64.b64encode(encrypted_key).decode(),
            'encrypted_data': encrypted_data.decode()
        }
    
    def decrypt(self, encrypted_key: str, encrypted_data: str) -> str:
        """解密"""
        # 1. 使用 RSA 解密 AES 密钥
        encrypted_key_bytes = base64.b64decode(encrypted_key)
        aes_key = self.rsa_cipher.decrypt(encrypted_key_bytes)
        
        # 2. 使用 AES 解密数据
        fernet = Fernet(aes_key.encode())
        plaintext = fernet.decrypt(encrypted_data.encode())
        
        return plaintext.decode()

# 使用示例
hybrid = HybridCipher()

# 加密大文件或大数据
large_data = "这是一段很长的数据..." * 1000
result = hybrid.encrypt(large_data)

print(f"加密的密钥: {result['encrypted_key'][:50]}...")
print(f"加密的数据: {result['encrypted_data'][:50]}...")

# 解密
decrypted = hybrid.decrypt(result['encrypted_key'], result['encrypted_data'])
print(f"解密成功: {decrypted[:50]}...")
```

```python
def mask_email(email: str) -> str:
    """邮箱脱敏：u***r@example.com"""
    if not email or '@' not in email:
        return email
    
    local, domain = email.split('@', 1)
    
    if len(local) <= 2:
        masked_local = local[0] + '*'
    else:
        masked_local = local[0] + '*' * (len(local) - 2) + local[-1]
    
    return f"{masked_local}@{domain}"
```

### 通用脱敏工具

```python
from typing import Callable
import re

class DataMasker:
    """数据脱敏工具类"""
    
    @staticmethod
    def mask_string(text: str, start: int, end: int, mask_char: str = '*') -> str:
        """通用字符串脱敏"""
        if not text or len(text) < start + end:
            return text
        
        visible_start = text[:start]
        visible_end = text[-end:] if end > 0 else ''
        masked_middle = mask_char * (len(text) - start - end)
        
        return visible_start + masked_middle + visible_end
    
    @staticmethod
    def mask_by_pattern(text: str, pattern: str, replacement: str) -> str:
        """按正则表达式脱敏"""
        return re.sub(pattern, replacement, text)
    
    @staticmethod
    def mask_json_fields(data: dict, fields: list, masker: Callable) -> dict:
        """脱敏 JSON 中的指定字段"""
        result = data.copy()
        
        for field in fields:
            if field in result:
                result[field] = masker(result[field])
        
        return result

# 使用示例
masker = DataMasker()

# 1. 通用脱敏
print(masker.mask_string("13812345678", 3, 4))  # 138****5678

# 2. 正则脱敏
text = "我的手机号是13812345678，邮箱是user@example.com"
masked = masker.mask_by_pattern(text, r'\d{11}', '***********')
print(masked)  # 我的手机号是***********，邮箱是user@example.com

# 3. JSON 脱敏
user_data = {
    'name': '张三',
    'phone': '13812345678',
    'email': 'user@example.com',
    'id_card': '110101199001011234'
}

masked_data = masker.mask_json_fields(
    user_data,
    ['phone', 'email', 'id_card'],
    lambda x: masker.mask_string(x, 3, 4)
)

print(masked_data)
```

### 密钥管理

```python
from cryptography.fernet import Fernet
import json
import os

class SecretManager:
    """密钥管理器"""
    
    def __init__(self, master_key: str):
        """初始化"""
        self.cipher = Fernet(master_key.encode())
        self.secrets_file = 'secrets.enc'
    
    def set_secret(self, key: str, value: str):
        """设置密钥"""
        secrets = self._load_secrets()
        secrets[key] = value
        self._save_secrets(secrets)
    
    def get_secret(self, key: str) -> str:
        """获取密钥"""
        secrets = self._load_secrets()
        return secrets.get(key)
    
    def delete_secret(self, key: str):
        """删除密钥"""
        secrets = self._load_secrets()
        if key in secrets:
            del secrets[key]
            self._save_secrets(secrets)
    
    def _load_secrets(self) -> dict:
        """加载密钥"""
        if not os.path.exists(self.secrets_file):
            return {}
        
        with open(self.secrets_file, 'rb') as f:
            encrypted = f.read()
        
        if not encrypted:
            return {}
        
        decrypted = self.cipher.decrypt(encrypted)
        return json.loads(decrypted)
    
    def _save_secrets(self, secrets: dict):
        """保存密钥"""
        data = json.dumps(secrets).encode()
        encrypted = self.cipher.encrypt(data)
        
        with open(self.secrets_file, 'wb') as f:
            f.write(encrypted)

# 使用示例
master_key = Fernet.generate_key().decode()
manager = SecretManager(master_key)

# 存储密钥
manager.set_secret('database_password', 'MySecurePassword123')
manager.set_secret('api_key', 'sk-1234567890abcdef')

# 读取密钥
db_password = manager.get_secret('database_password')
print(f"数据库密码: {db_password}")

# 删除密钥
manager.delete_secret('api_key')
```

```python
import boto3
import json
from botocore.exceptions import ClientError

class AWSSecretsManager:
    """AWS Secrets Manager 集成"""
    
    def __init__(self, region_name: str = 'us-east-1'):
        self.client = boto3.client(
            'secretsmanager',
            region_name=region_name
        )
    
    def get_secret(self, secret_name: str) -> dict:
        """获取密钥"""
        try:
            response = self.client.get_secret_value(SecretId=secret_name)
            
            if 'SecretString' in response:
                return json.loads(response['SecretString'])
            else:
                return response['SecretBinary']
        
        except ClientError as e:
            raise Exception(f"获取密钥失败: {e}")
    
    def create_secret(self, secret_name: str, secret_value: dict):
        """创建密钥"""
        try:
            self.client.create_secret(
                Name=secret_name,
```

### 案例 1：用户数据加密存储

```python
from sqlalchemy import Column, Integer, String, create_engine
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker
from cryptography.fernet import Fernet
import bcrypt

Base = declarative_base()

class EncryptedUser(Base):
    """加密用户模型"""
    __tablename__ = 'users'
    
    id = Column(Integer, primary_key=True)
    username = Column(String(50), unique=True, nullable=False)
    password_hash = Column(String(255), nullable=False)  # bcrypt 哈希
    email_encrypted = Column(String(255), nullable=False)  # AES 加密
    phone_encrypted = Column(String(255), nullable=False)  # AES 加密

class UserService:
    """用户服务"""
    
    def __init__(self, encryption_key: bytes):
        self.cipher = Fernet(encryption_key)
    
    def create_user(self, username: str, password: str, email: str, phone: str) -> EncryptedUser:
        """创建用户"""
        user = EncryptedUser(
            username=username,
            password_hash=self._hash_password(password),
            email_encrypted=self._encrypt_field(email),
            phone_encrypted=self._encrypt_field(phone)
        )
        return user
    
    def verify_password(self, user: EncryptedUser, password: str) -> bool:
        """验证密码"""
        return bcrypt.checkpw(
            password.encode(),
            user.password_hash.encode()
        )
    
    def get_email(self, user: EncryptedUser) -> str:
        """获取邮箱（解密）"""
        return self._decrypt_field(user.email_encrypted)
    
    def get_phone(self, user: EncryptedUser) -> str:
        """获取手机号（解密）"""
        return self._decrypt_field(user.phone_encrypted)
    
    def _hash_password(self, password: str) -> str:
        """哈希密码"""
        salt = bcrypt.gensalt()
        hashed = bcrypt.hashpw(password.encode(), salt)
        return hashed.decode()
    
    def _encrypt_field(self, value: str) -> str:
        """加密字段"""
        encrypted = self.cipher.encrypt(value.encode())
        return encrypted.decode()
    
    def _decrypt_field(self, encrypted: str) -> str:
        """解密字段"""
        decrypted = self.cipher.decrypt(encrypted.encode())
        return decrypted.decode()

# 使用示例
```

### 案例 2：API 响应数据脱敏

```python
from flask import Flask, jsonify
from functools import wraps

app = Flask(__name__)

def mask_sensitive_data(data):
    """脱敏敏感数据"""
    if isinstance(data, dict):
        result = {}
        for key, value in data.items():
            if key in ['phone', 'mobile']:
                result[key] = mask_phone(str(value))
            elif key in ['email']:
                result[key] = mask_email(str(value))
            elif key in ['id_card', 'idCard']:
                result[key] = mask_id_card(str(value))
            elif key in ['bank_card', 'bankCard']:
                result[key] = mask_bank_card(str(value))
            elif isinstance(value, (dict, list)):
                result[key] = mask_sensitive_data(value)
            else:
                result[key] = value
        return result
    
    elif isinstance(data, list):
        return [mask_sensitive_data(item) for item in data]
    
    return data

def mask_response(f):
    """响应脱敏装饰器"""
    @wraps(f)
    def decorated(*args, **kwargs):
        response = f(*args, **kwargs)
        
        if isinstance(response, tuple):
            data, status_code = response
            masked_data = mask_sensitive_data(data)
            return jsonify(masked_data), status_code
        else:
            masked_data = mask_sensitive_data(response)
            return jsonify(masked_data)
    
    return decorated

@app.route('/api/user/<int:user_id>')
@mask_response
def get_user(user_id):
    """获取用户信息（自动脱敏）"""
    user = {
        'id': user_id,
        'name': '张三',
        'phone': '13812345678',
        'email': 'zhangsan@example.com',
        'id_card': '110101199001011234',
        'bank_card': '6222021234567891234'
    }
    return user

# 辅助函数
def mask_phone(phone: str) -> str:
    if len(phone) < 11:
        return phone
    return phone[:3] + '****' + phone[-4:]

def mask_email(email: str) -> str:
    if '@' not in email:
        return email
    local, domain = email.split('@', 1)
    if len(local) <= 2:
        return local[0] + '*@' + domain
    return local[0] + '*' * (len(local) - 2) + local[-1] + '@' + domain

def mask_id_card(id_card: str) -> str:
    if len(id_card) < 18:
        return id_card
    return id_card[:3] + '*' * 11 + id_card[-4:]

def mask_bank_card(card: str) -> str:
    card = card.replace(' ', '')
    if len(card) < 16:
        return card
    return card[:4] + ' **** **** ' + card[-4:]
```

## 总结

### 加密最佳实践

1. **密码存储**：使用 bcrypt 或 argon2，不要用 MD5/SHA1
2. **数据加密**：使用 AES-256，密钥长度至少 256 位
3. **密钥管理**：使用环境变量或密钥管理服务
4. **传输加密**：强制使用 HTTPS
5. **数据脱敏**：日志和 API 响应中脱敏敏感信息

### 关键要点

1. 不要自己实现加密算法
2. 使用经过验证的加密库
3. 密钥管理至关重要
4. 定期更新加密算法
5. 加密不等于安全

### 常见错误

- ❌ 使用 MD5/SHA1 存储密码
- ❌ 硬编码密钥在代码中
- ❌ 使用弱密钥（如 "123456"）
- ❌ 不使用盐值
- ❌ 在日志中打印敏感信息

### 学习路径

1. 第 1 周：掌握密码哈希（bcrypt/argon2）
2. 第 2 周：学习对称加密（AES）
3. 第 3 周：理解非对称加密（RSA）
4. 第 4 周：实践数据脱敏
5. 第 5 周：在实际项目中应用

**记住：加密是保护数据的最后一道防线！** 🔐
