## 认证授权概述

### 认证 vs 授权

| 概念 | 英文 | 含义 | 示例 |
|------|------|------|------|
| 认证 | Authentication | 验证用户身份 | 登录验证用户名密码 |
| 授权 | Authorization | 验证用户权限 | 检查用户是否有访问资源的权限 |

### 常见认证方式

```python
"""
1. Session-Cookie 认证
   - 服务端存储 Session
   - 客户端存储 Session ID
   - 适合传统 Web 应用

2. Token 认证（JWT）
   - 无状态
   - 适合分布式系统
   - 适合移动端和 SPA

3. OAuth2
   - 第三方授权
   - 适合开放平台

4. API Key
   - 简单的 API 认证
   - 适合服务间调用
"""
```

## JWT 令牌认证

### JWT 基础

```python
import jwt
import datetime
from typing import Dict, Optional

SECRET_KEY = "your-secret-key-keep-it-safe"
ALGORITHM = "HS256"

def create_jwt_token(user_id: int, username: str) -> str:
    """创建 JWT Token"""
    payload = {
        'user_id': user_id,
        'username': username,
        'exp': datetime.datetime.utcnow() + datetime.timedelta(hours=24),
        'iat': datetime.datetime.utcnow()
    }
    
    token = jwt.encode(payload, SECRET_KEY, algorithm=ALGORITHM)
    return token

def verify_jwt_token(token: str) -> Optional[Dict]:
    """验证 JWT Token"""
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        return payload
    except jwt.ExpiredSignatureError:
        print("Token 已过期")
        return None
    except jwt.InvalidTokenError:
        print("无效的 Token")
        return None

# 使用示例
token = create_jwt_token(user_id=123, username="john")
print(f"Token: {token}")

payload = verify_jwt_token(token)
if payload:
    print(f"用户 ID: {payload['user_id']}")
    print(f"用户名: {payload['username']}")
```

### Refresh Token 机制

```python
import secrets
from datetime import datetime, timedelta
from typing import Tuple

class TokenManager:
    """Token 管理器"""
    
    def __init__(self, secret_key: str):
        self.secret_key = secret_key
        self.refresh_tokens = {}  # 生产环境应使用 Redis
    
    def create_tokens(self, user_id: int) -> Tuple[str, str]:
        """创建 Access Token 和 Refresh Token"""
        # Access Token（短期有效，15 分钟）
        access_token = jwt.encode({
            'user_id': user_id,
            'type': 'access',
            'exp': datetime.utcnow() + timedelta(minutes=15)
        }, self.secret_key, algorithm="HS256")
        
        # Refresh Token（长期有效，7 天）
        refresh_token = secrets.token_urlsafe(32)
        self.refresh_tokens[refresh_token] = {
            'user_id': user_id,
            'exp': datetime.utcnow() + timedelta(days=7)
        }
        
        return access_token, refresh_token
    
    def refresh_access_token(self, refresh_token: str) -> str:
        """使用 Refresh Token 刷新 Access Token"""
        token_data = self.refresh_tokens.get(refresh_token)
        
        if not token_data:
            raise ValueError("无效的 Refresh Token")
        
        if token_data['exp'] < datetime.utcnow():
            del self.refresh_tokens[refresh_token]
            raise ValueError("Refresh Token 已过期")
        
        # 生成新的 Access Token
        access_token = jwt.encode({
            'user_id': token_data['user_id'],
            'type': 'access',
            'exp': datetime.utcnow() + timedelta(minutes=15)
        }, self.secret_key, algorithm="HS256")
        
        return access_token
    
    def revoke_refresh_token(self, refresh_token: str):
        """撤销 Refresh Token"""
        if refresh_token in self.refresh_tokens:
            del self.refresh_tokens[refresh_token]

# 使用示例
token_manager = TokenManager('your-secret-key')

# 登录时创建 Token
access_token, refresh_token = token_manager.create_tokens(user_id=123)

# Access Token 过期后，使用 Refresh Token 刷新
new_access_token = token_manager.refresh_access_token(refresh_token)

# 登出时撤销 Refresh Token
token_manager.revoke_refresh_token(refresh_token)
```

## OAuth2 授权

### OAuth2 流程

```python
"""
OAuth2 授权码流程：

1. 用户点击"使用 GitHub 登录"
2. 重定向到 GitHub 授权页面
3. 用户同意授权
4. GitHub 重定向回应用，带上授权码
5. 应用使用授权码换取 Access Token
6. 使用 Access Token 访问用户信息
"""
```

### 实现 OAuth2 客户端

```python
from flask import Flask, request, redirect, session, url_for
import requests
from urllib.parse import urlencode

app = Flask(__name__)
app.secret_key = 'your-secret-key'

# GitHub OAuth 配置
GITHUB_CLIENT_ID = 'your-client-id'
GITHUB_CLIENT_SECRET = 'your-client-secret'
GITHUB_AUTHORIZE_URL = 'https://github.com/login/oauth/authorize'
GITHUB_TOKEN_URL = 'https://github.com/login/oauth/access_token'
GITHUB_API_URL = 'https://api.github.com/user'

@app.route('/login/github')
def github_login():
    """重定向到 GitHub 授权页面"""
    params = {
        'client_id': GITHUB_CLIENT_ID,
        'redirect_uri': url_for('github_callback', _external=True),
        'scope': 'user:email',
        'state': 'random-state-string'  # 防止 CSRF
    }
    
    auth_url = f"{GITHUB_AUTHORIZE_URL}?{urlencode(params)}"
    return redirect(auth_url)

@app.route('/callback/github')
def github_callback():
    """GitHub 回调处理"""
    # 1. 获取授权码
    code = request.args.get('code')
    state = request.args.get('state')
    
    if not code:
        return "授权失败", 400
    
    # 2. 验证 state（防止 CSRF）
    if state != 'random-state-string':
        return "State 验证失败", 400
    
    # 3. 使用授权码换取 Access Token
    token_data = {
        'client_id': GITHUB_CLIENT_ID,
        'client_secret': GITHUB_CLIENT_SECRET,
        'code': code,
        'redirect_uri': url_for('github_callback', _external=True)
    }
    
    token_response = requests.post(
        GITHUB_TOKEN_URL,
        data=token_data,
        headers={'Accept': 'application/json'}
    )
    
    token_json = token_response.json()
    access_token = token_json.get('access_token')
    
    if not access_token:
        return "获取 Token 失败", 400
    
    # 4. 使用 Access Token 获取用户信息
    user_response = requests.get(
        GITHUB_API_URL,
        headers={'Authorization': f'Bearer {access_token}'}
    )
    
    user_data = user_response.json()
    
    # 5. 保存用户信息到 Session
    session['user'] = {
        'id': user_data['id'],
        'username': user_data['login'],
        'email': user_data.get('email'),
        'avatar': user_data['avatar_url']
    }
    
    return redirect('/dashboard')

@app.route('/dashboard')
def dashboard():
    """用户仪表板"""
    if 'user' not in session:
        return redirect('/login/github')
    
    return f"欢迎，{session['user']['username']}！"

@app.route('/logout')
def logout():
    """登出"""
    session.pop('user', None)
    return redirect('/')
```

## RBAC 角色权限控制

### RBAC 模型

```python
from enum import Enum
from typing import Set, List
from dataclasses import dataclass

class Permission(Enum):
    """权限枚举"""
    READ_USER = "read:user"
    WRITE_USER = "write:user"
    DELETE_USER = "delete:user"
    READ_POST = "read:post"
    WRITE_POST = "write:post"
    DELETE_POST = "delete:post"
    ADMIN = "admin"

@dataclass
class Role:
    """角色"""
    name: str
    permissions: Set[Permission]

# 定义角色
ROLES = {
    'guest': Role(
        name='guest',
        permissions={Permission.READ_POST}
    ),
    'user': Role(
        name='user',
        permissions={
            Permission.READ_USER,
            Permission.READ_POST,
            Permission.WRITE_POST
        }
    ),
    'moderator': Role(
        name='moderator',
        permissions={
            Permission.READ_USER,
            Permission.READ_POST,
            Permission.WRITE_POST,
            Permission.DELETE_POST
        }
    ),
    'admin': Role(
        name='admin',
        permissions={
            Permission.READ_USER,
            Permission.WRITE_USER,
            Permission.DELETE_USER,
            Permission.READ_POST,
            Permission.WRITE_POST,
            Permission.DELETE_POST,
            Permission.ADMIN
        }
    )
}

class User:
    """用户"""
    def __init__(self, user_id: int, username: str, roles: List[str]):
        self.id = user_id
        self.username = username
        self.roles = roles
    
    def has_permission(self, permission: Permission) -> bool:
        """检查用户是否有某个权限"""
        for role_name in self.roles:
            role = ROLES.get(role_name)
            if role and permission in role.permissions:
                return True
        return False
    
    def has_role(self, role_name: str) -> bool:
        """检查用户是否有某个角色"""
        return role_name in self.roles

# 使用示例
user = User(user_id=1, username='john', roles=['user'])
print(user.has_permission(Permission.WRITE_POST))  # True
print(user.has_permission(Permission.DELETE_USER))  # False

admin = User(user_id=2, username='admin', roles=['admin'])
print(admin.has_permission(Permission.DELETE_USER))  # True
```

### Flask RBAC 实现

```python
from flask import Flask, g, jsonify, abort
from functools import wraps

app = Flask(__name__)

def require_permission(permission: Permission):
    """权限检查装饰器"""
    def decorator(f):
        @wraps(f)
        def decorated_function(*args, **kwargs):
            if not hasattr(g, 'current_user'):
                abort(401, "未登录")
            
            if not g.current_user.has_permission(permission):
                abort(403, "权限不足")
            
            return f(*args, **kwargs)
        return decorated_function
    return decorator

def require_role(role_name: str):
    """角色检查装饰器"""
    def decorator(f):
        @wraps(f)
        def decorated_function(*args, **kwargs):
            if not hasattr(g, 'current_user'):
                abort(401, "未登录")
            
            if not g.current_user.has_role(role_name):
                abort(403, "需要 {} 角色".format(role_name))
            
            return f(*args, **kwargs)
        return decorated_function
    return decorator

@app.route('/api/posts', methods=['POST'])
@require_permission(Permission.WRITE_POST)
def create_post():
    """创建文章（需要写权限）"""
    return jsonify({"message": "文章创建成功"})

@app.route('/api/users/<int:user_id>', methods=['DELETE'])
@require_permission(Permission.DELETE_USER)
def delete_user(user_id):
    """删除用户（需要删除用户权限）"""
    return jsonify({"message": f"用户 {user_id} 已删除"})

@app.route('/api/admin/settings', methods=['GET'])
@require_role('admin')
def admin_settings():
    """管理员设置（需要管理员角色）"""
    return jsonify({"settings": {}})
```

### 数据库 RBAC 模型

```python
from sqlalchemy import Column, Integer, String, Table, ForeignKey
from sqlalchemy.orm import relationship
from sqlalchemy.ext.declarative import declarative_base

Base = declarative_base()

# 用户-角色关联表
user_roles = Table(
    'user_roles',
    Base.metadata,
    Column('user_id', Integer, ForeignKey('users.id')),
    Column('role_id', Integer, ForeignKey('roles.id'))
)

# 角色-权限关联表
role_permissions = Table(
    'role_permissions',
    Base.metadata,
    Column('role_id', Integer, ForeignKey('roles.id')),
    Column('permission_id', Integer, ForeignKey('permissions.id'))
)

class User(Base):
    """用户模型"""
    __tablename__ = 'users'
    
    id = Column(Integer, primary_key=True)
    username = Column(String(50), unique=True, nullable=False)
    email = Column(String(100), unique=True, nullable=False)
    
    # 多对多关系
    roles = relationship('Role', secondary=user_roles, back_populates='users')
    
    def has_permission(self, permission_name: str) -> bool:
        """检查用户是否有某个权限"""
        for role in self.roles:
            for permission in role.permissions:
                if permission.name == permission_name:
                    return True
        return False

class Role(Base):
    """角色模型"""
    __tablename__ = 'roles'
    
    id = Column(Integer, primary_key=True)
    name = Column(String(50), unique=True, nullable=False)
    description = Column(String(200))
    
    # 多对多关系
    users = relationship('User', secondary=user_roles, back_populates='roles')
    permissions = relationship(
        'Permission',
        secondary=role_permissions,
        back_populates='roles'
    )

class Permission(Base):
    """权限模型"""
    __tablename__ = 'permissions'
    
    id = Column(Integer, primary_key=True)
    name = Column(String(50), unique=True, nullable=False)
    description = Column(String(200))
    
    # 多对多关系
    roles = relationship('Role', secondary=role_permissions, back_populates='permissions')
```

### TOTP 实现

```python
import pyotp
import qrcode
from io import BytesIO
import base64

class TwoFactorAuth:
    """双因素认证"""
    
    @staticmethod
    def generate_secret() -> str:
        """生成密钥"""
        return pyotp.random_base32()
    
    @staticmethod
    def get_totp_uri(secret: str, username: str, issuer: str = "MyApp") -> str:
        """生成 TOTP URI"""
        return pyotp.totp.TOTP(secret).provisioning_uri(
            name=username,
            issuer_name=issuer
        )
    
    @staticmethod
    def generate_qr_code(uri: str) -> str:
        """生成二维码（Base64）"""
        qr = qrcode.QRCode(version=1, box_size=10, border=5)
        qr.add_data(uri)
        qr.make(fit=True)
        
        img = qr.make_image(fill_color="black", back_color="white")
        
        # 转换为 Base64
        buffer = BytesIO()
        img.save(buffer, format='PNG')
        img_str = base64.b64encode(buffer.getvalue()).decode()
        
        return f"data:image/png;base64,{img_str}"
    
    @staticmethod
    def verify_token(secret: str, token: str) -> bool:
        """验证 TOTP Token"""
        totp = pyotp.TOTP(secret)
        return totp.verify(token, valid_window=1)

# 使用示例
tfa = TwoFactorAuth()

# 1. 用户启用 2FA 时，生成密钥
secret = tfa.generate_secret()
print(f"密钥: {secret}")

# 2. 生成二维码供用户扫描
uri = tfa.get_totp_uri(secret, username="john@example.com")
qr_code = tfa.generate_qr_code(uri)

# 3. 用户登录时验证 Token
token = input("请输入 6 位验证码: ")
is_valid = tfa.verify_token(secret, token)
print(f"验证结果: {is_valid}")
```

from flask import Flask, request, jsonify, g
from werkzeug.security import generate_password_hash, check_password_hash
from functools import wraps
import jwt
import datetime
from typing import Optional

app = Flask(__name__)
app.config['SECRET_KEY'] = 'your-secret-key'

# 模拟数据库
users_db = {}
tokens_blacklist = set()

def token_required(f):
    """认证装饰器"""
    @wraps(f)
    def decorated(*args, **kwargs):
        token = request.headers.get('Authorization')
        
        if not token:
            return jsonify({'error': '缺少 Token'}), 401
        
        try:
            if token.startswith('Bearer '):
                token = token[7:]
            
            # 检查黑名单
            if token in tokens_blacklist:
                return jsonify({'error': 'Token 已失效'}), 401
            
            # 验证 Token
            data = jwt.decode(
                token,
                app.config['SECRET_KEY'],
                algorithms=["HS256"]
            )
            
            g.current_user_id = data['user_id']
            g.token = token
            
        except jwt.ExpiredSignatureError:
            return jsonify({'error': 'Token 已过期'}), 401
        except jwt.InvalidTokenError:
            return jsonify({'error': '无效的 Token'}), 401
        
        return f(*args, **kwargs)
    
    return decorated

@app.route('/register', methods=['POST'])
def register():
    """用户注册"""
    data = request.get_json()
    
    username = data.get('username')
    password = data.get('password')
    email = data.get('email')
    
    # 验证输入
    if not username or not password or not email:
        return jsonify({'error': '缺少必填字段'}), 400
    
    # 检查用户是否存在
    if username in users_db:
        return jsonify({'error': '用户名已存在'}), 400
    
    # 创建用户
    user_id = len(users_db) + 1
    users_db[username] = {
        'id': user_id,
        'username': username,
        'password': generate_password_hash(password),
        'email': email,
        'roles': ['user']
    }
    
    return jsonify({
        'message': '注册成功',
        'user_id': user_id
    }), 201

@app.route('/login', methods=['POST'])
def login():
    """用户登录"""
    data = request.get_json()
    
    username = data.get('username')
    password = data.get('password')
    
    # 验证用户
    user = users_db.get(username)
    if not user or not check_password_hash(user['password'], password):
        return jsonify({'error': '用户名或密码错误'}), 401
    
    # 生成 Token
    token = jwt.encode({
        'user_id': user['id'],
        'username': user['username'],
        'exp': datetime.datetime.utcnow() + datetime.timedelta(hours=24)
    }, app.config['SECRET_KEY'], algorithm="HS256")
    
    return jsonify({
        'token': token,
        'user': {
            'id': user['id'],
            'username': user['username'],
            'email': user['email']
        }
    })

@token_required
def logout():

    tokens_blacklist.add(g.token)
    
    return jsonify({'message': '登出成功'})

@token_required
def get_profile():
    """获取用户信息"""
    user = next(
        (u for u in users_db.values() if u['id'] == g.current_user_id),
        None
    )
    
    if not user:
        return jsonify({'error': '用户不存在'}), 404
    
    return jsonify({
        'id': user['id'],
        'username': user['username'],
        'email': user['email'],
        'roles': user['roles']
    })

    """修改密码"""
    data = request.get_json()
    
    old_password = data.get('old_password')
    new_password = data.get('new_password')

        return jsonify({'error': '旧密码错误'}), 401
    
    # 更新密码
    user['password'] = generate_password_hash(new_password)
    
    # 将当前 Token 加入黑名单

    return jsonify({'message': '密码修改成功，请重新登录'})

## 总结

### 认证授权最佳实践

1. **使用强密码策略**：最少 12 位，包含大小写字母、数字、特殊字符
2. **密码哈希存储**：使用 bcrypt 或 argon2
3. **Token 过期时间**：Access Token 短期（15 分钟），Refresh Token 长期（7 天）
4. **实施 RBAC**：基于角色的权限控制
5. **启用 2FA**：提高账户安全性
6. **安全的 Session**：HttpOnly、Secure、SameSite
7. **Token 黑名单**：支持登出和撤销
8. **审计日志**：记录所有认证授权操作

### 关键要点

1. 认证验证身份，授权验证权限
2. JWT 适合无状态分布式系统
3. OAuth2 适合第三方授权
4. RBAC 简化权限管理
5. 2FA 提高安全性
