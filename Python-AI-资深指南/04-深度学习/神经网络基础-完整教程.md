## 1. 感知机

### 1.1 单层感知机

```python
"""
单层感知机实现
@author erik.zhou
"""
import numpy as np

class Perceptron:
    """单层感知机"""
    
    def __init__(self, input_size, learning_rate=0.01):
        """
        初始化感知机
        
        Args:
            input_size: 输入特征维度
            learning_rate: 学习率
        """
        self.weights = np.zeros(input_size)
        self.bias = 0
        self.learning_rate = learning_rate
    
    def predict(self, x):
        """预测"""
        return 1 if np.dot(x, self.weights) + self.bias > 0 else 0
    
    def train(self, X, y, epochs=100):
        """训练感知机"""
        for epoch in range(epochs):
            errors = 0
            for xi, yi in zip(X, y):
                prediction = self.predict(xi)
                error = yi - prediction
                
                if error != 0:
                    self.weights += self.learning_rate * error * xi
                    self.bias += self.learning_rate * error
                    errors += 1
            
            if errors == 0:
                print(f"收敛于第 {epoch + 1} 轮")
                break
        
        return self

# 使用示例：AND逻辑门
X = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
y = np.array([0, 0, 0, 1])

perceptron = Perceptron(input_size=2)
perceptron.train(X, y, epochs=100)

print("AND逻辑门测试:")
for xi, yi in zip(X, y):
    pred = perceptron.predict(xi)
    print(f"输入: {xi}, 预测: {pred}, 真实: {yi}")
```

## 2. 多层感知机

### 2.1 MLP架构

```python
"""
多层感知机（MLP）实现
@author erik.zhou
"""
import numpy as np

class MLP:
    """多层感知机"""
    
    def __init__(self, layer_sizes):
        """
        初始化MLP
        
        Args:
            layer_sizes: 各层神经元数量，如[2, 4, 1]表示输入2个，隐藏层4个，输出1个
        """
        self.layer_sizes = layer_sizes
        self.num_layers = len(layer_sizes)
        
        # 初始化权重和偏置
        self.weights = []
        self.biases = []
        
        for i in range(self.num_layers - 1):
            w = np.random.randn(layer_sizes[i], layer_sizes[i+1]) * 0.01
            b = np.zeros((1, layer_sizes[i+1]))
            self.weights.append(w)
            self.biases.append(b)
    
    def sigmoid(self, z):
        """Sigmoid激活函数"""
        return 1 / (1 + np.exp(-np.clip(z, -500, 500)))
    
    def sigmoid_derivative(self, z):
        """Sigmoid导数"""
        s = self.sigmoid(z)
        return s * (1 - s)
    
    def forward(self, X):
        """前向传播"""
        self.activations = [X]
        self.z_values = []
        
        for i in range(self.num_layers - 1):
            z = np.dot(self.activations[-1], self.weights[i]) + self.biases[i]
            self.z_values.append(z)
            a = self.sigmoid(z)
            self.activations.append(a)
        
        return self.activations[-1]
    
    def backward(self, X, y, learning_rate=0.01):
        """反向传播"""
        m = X.shape[0]
        
        # 计算输出层误差
        delta = self.activations[-1] - y
        
        # 反向传播误差
        deltas = [delta]
        for i in range(self.num_layers - 2, 0, -1):
            delta = np.dot(delta, self.weights[i].T) * self.sigmoid_derivative(self.z_values[i-1])
            deltas.insert(0, delta)
        
        # 更新权重和偏置
        for i in range(self.num_layers - 1):
            self.weights[i] -= learning_rate * np.dot(self.activations[i].T, deltas[i]) / m
            self.biases[i] -= learning_rate * np.sum(deltas[i], axis=0, keepdims=True) / m
    
    def train(self, X, y, epochs=1000, learning_rate=0.1):
        """训练MLP"""
        losses = []
        
        for epoch in range(epochs):
            # 前向传播
            output = self.forward(X)
            
            # 计算损失
            loss = np.mean((output - y) ** 2)
            losses.append(loss)
            
            # 反向传播
            self.backward(X, y, learning_rate)
            
            if epoch % 100 == 0:
                print(f"Epoch {epoch}, Loss: {loss:.4f}")
        
        return losses

# 使用示例：解决XOR问题
X = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
y = np.array([[0], [1], [1], [0]])

mlp = MLP([2, 4, 1])
losses = mlp.train(X, y, epochs=5000, learning_rate=0.5)

print("\nXOR问题测试:")
for xi, yi in zip(X, y):
    pred = mlp.forward(xi.reshape(1, -1))
    print(f"输入: {xi}, 预测: {pred[0][0]:.4f}, 真实: {yi[0]}")
```

```python
"""
常用激活函数实现
@author erik.zhou
"""
import numpy as np
import matplotlib.pyplot as plt

class ActivationFunctions:
    """激活函数集合"""
    
    @staticmethod
    def sigmoid(x):
        """Sigmoid函数"""
        return 1 / (1 + np.exp(-np.clip(x, -500, 500)))
    
    @staticmethod
    def sigmoid_derivative(x):
        """Sigmoid导数"""
        s = ActivationFunctions.sigmoid(x)
        return s * (1 - s)
    
    @staticmethod
    def tanh(x):
        """Tanh函数"""
        return np.tanh(x)
    
    @staticmethod
    def tanh_derivative(x):
        """Tanh导数"""
        return 1 - np.tanh(x) ** 2
    
    @staticmethod
    def relu(x):
        """ReLU函数"""
        return np.maximum(0, x)
    
    @staticmethod
    def relu_derivative(x):
        """ReLU导数"""
        return (x > 0).astype(float)
    
    @staticmethod
    def leaky_relu(x, alpha=0.01):
        """Leaky ReLU函数"""
        return np.where(x > 0, x, alpha * x)
    
    @staticmethod
    def leaky_relu_derivative(x, alpha=0.01):
        """Leaky ReLU导数"""
        return np.where(x > 0, 1, alpha)
    
    @staticmethod
    def elu(x, alpha=1.0):
        """ELU函数"""
        return np.where(x > 0, x, alpha * (np.exp(x) - 1))
    
    @staticmethod
    def elu_derivative(x, alpha=1.0):
        """ELU导数"""
        return np.where(x > 0, 1, alpha * np.exp(x))
    
    @staticmethod
    def softmax(x):
        """Softmax函数"""
        exp_x = np.exp(x - np.max(x, axis=-1, keepdims=True))
        return exp_x / np.sum(exp_x, axis=-1, keepdims=True)
```

## 4. 前向传播

### 4.1 前向传播详解

```python
"""
前向传播详细实现
@author erik.zhou
"""
import numpy as np

class ForwardPropagation:
    """前向传播示例"""
    
    def __init__(self):
        # 简单的3层网络：2-3-1
        self.W1 = np.random.randn(2, 3) * 0.01
        self.b1 = np.zeros((1, 3))
        self.W2 = np.random.randn(3, 1) * 0.01
        self.b2 = np.zeros((1, 1))
    
    def relu(self, z):
        """ReLU激活"""
        return np.maximum(0, z)
    
    def sigmoid(self, z):
        """Sigmoid激活"""
        return 1 / (1 + np.exp(-np.clip(z, -500, 500)))
    
    def forward(self, X):
        """
        前向传播
        
        Args:
            X: 输入数据 (m, 2)
        
        Returns:
            A2: 输出 (m, 1)
        """
        # 第一层
        Z1 = np.dot(X, self.W1) + self.b1  # (m, 3)
        A1 = self.relu(Z1)  # (m, 3)
        
        # 第二层
        Z2 = np.dot(A1, self.W2) + self.b2  # (m, 1)
        A2 = self.sigmoid(Z2)  # (m, 1)
        
        # 保存中间值用于反向传播
        self.cache = {
            'X': X,
            'Z1': Z1,
            'A1': A1,
            'Z2': Z2,
            'A2': A2
        }
        
        return A2
    
    def compute_loss(self, y_true, y_pred):
        """计算二元交叉熵损失"""
        m = y_true.shape[0]
        loss = -np.mean(y_true * np.log(y_pred + 1e-8) + 
                       (1 - y_true) * np.log(1 - y_pred + 1e-8))
        return loss

# 使用示例
X = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
y = np.array([[0], [1], [1], [0]])

model = ForwardPropagation()
output = model.forward(X)
loss = model.compute_loss(y, output)

print("前向传播结果:")
print(f"输出: \n{output}")
print(f"损失: {loss:.4f}")
```

### 4.2 批量前向传播

```python
"""
批量前向传播
@author erik.zhou
"""
import numpy as np

def batch_forward_propagation(X, weights, biases, batch_size=32):
    """
    批量前向传播
    
    Args:
        X: 输入数据 (n_samples, n_features)
        weights: 权重列表
        biases: 偏置列表
        batch_size: 批量大小
    
    Returns:
        outputs: 所有批次的输出
    """
    n_samples = X.shape[0]
    n_batches = (n_samples + batch_size - 1) // batch_size
    
    outputs = []
    
    for i in range(n_batches):
        start_idx = i * batch_size
        end_idx = min((i + 1) * batch_size, n_samples)
        
        X_batch = X[start_idx:end_idx]
        
        # 前向传播
        A = X_batch
        for W, b in zip(weights, biases):
            Z = np.dot(A, W) + b
            A = np.maximum(0, Z)  # ReLU
        
        outputs.append(A)
    
    return np.vstack(outputs)

# 示例
X = np.random.randn(100, 10)
weights = [np.random.randn(10, 20), np.random.randn(20, 5)]
biases = [np.zeros((1, 20)), np.zeros((1, 5))]

outputs = batch_forward_propagation(X, weights, biases, batch_size=32)
print(f"批量前向传播输出形状: {outputs.shape}")
```

## 5. 反向传播

### 5.1 反向传播算法

```python
"""
反向传播算法详解
@author erik.zhou
"""
import numpy as np

class BackPropagation:
    """反向传播示例"""
    
    def __init__(self, layer_sizes):
        """初始化网络"""
        self.layer_sizes = layer_sizes
        self.weights = []
        self.biases = []
        
        for i in range(len(layer_sizes) - 1):
            w = np.random.randn(layer_sizes[i], layer_sizes[i+1]) * np.sqrt(2.0 / layer_sizes[i])
            b = np.zeros((1, layer_sizes[i+1]))
            self.weights.append(w)
            self.biases.append(b)
    
    def relu(self, z):
        """ReLU激活"""
        return np.maximum(0, z)
    
    def relu_derivative(self, z):
        """ReLU导数"""
        return (z > 0).astype(float)
    
    def sigmoid(self, z):
        """Sigmoid激活"""
        return 1 / (1 + np.exp(-np.clip(z, -500, 500)))
    
    def forward(self, X):
        """前向传播"""
        self.activations = [X]
        self.z_values = []
        
        A = X
        for i in range(len(self.weights) - 1):
            Z = np.dot(A, self.weights[i]) + self.biases[i]
            A = self.relu(Z)
            self.z_values.append(Z)
            self.activations.append(A)
        
        # 输出层使用sigmoid
        Z = np.dot(A, self.weights[-1]) + self.biases[-1]
        A = self.sigmoid(Z)
        self.z_values.append(Z)
        self.activations.append(A)
        
        return A
    
    def backward(self, y, learning_rate=0.01):
        """
        反向传播
        
        Args:
            y: 真实标签
            learning_rate: 学习率
        """
        m = y.shape[0]
        
        # 输出层梯度
        dA = self.activations[-1] - y
        
        # 反向传播
        for i in range(len(self.weights) - 1, -1, -1):
            # 计算梯度
            dZ = dA
            if i < len(self.weights) - 1:
                dZ = dA * self.relu_derivative(self.z_values[i])
            
            dW = np.dot(self.activations[i].T, dZ) / m
            db = np.sum(dZ, axis=0, keepdims=True) / m
            
            if i > 0:
                dA = np.dot(dZ, self.weights[i].T)
            
            # 更新参数
            self.weights[i] -= learning_rate * dW
            self.biases[i] -= learning_rate * db
    
    def train(self, X, y, epochs=1000, learning_rate=0.1, verbose=True):
        """训练网络"""
        losses = []
        
        for epoch in range(epochs):
            # 前向传播
            output = self.forward(X)
            
            # 计算损失
            loss = -np.mean(y * np.log(output + 1e-8) + 
                           (1 - y) * np.log(1 - output + 1e-8))
            losses.append(loss)
            
            # 反向传播
            self.backward(y, learning_rate)
            
            if verbose and epoch % 100 == 0:
                print(f"Epoch {epoch}, Loss: {loss:.4f}")
        
        return losses

# 使用示例
X = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
y = np.array([[0], [1], [1], [0]])

model = BackPropagation([2, 4, 1])
losses = model.train(X, y, epochs=5000, learning_rate=0.5)

print("\n测试结果:")
for xi, yi in zip(X, y):
    pred = model.forward(xi.reshape(1, -1))
    print(f"输入: {xi}, 预测: {pred[0][0]:.4f}, 真实: {yi[0]}")
```

### 5.2 梯度检查

```python
"""
梯度检查（验证反向传播实现）
@author erik.zhou
"""
import numpy as np

def numerical_gradient(f, x, epsilon=1e-5):
    """
    数值梯度计算
    
    Args:
        f: 损失函数
        x: 参数
        epsilon: 扰动值
    
    Returns:
        grad: 数值梯度
    """
    grad = np.zeros_like(x)
    it = np.nditer(x, flags=['multi_index'], op_flags=['readwrite'])
    
    while not it.finished:
        idx = it.multi_index
        old_value = x[idx]
        
        x[idx] = old_value + epsilon
        fxh_plus = f(x)
        
        x[idx] = old_value - epsilon
        fxh_minus = f(x)
        
        grad[idx] = (fxh_plus - fxh_minus) / (2 * epsilon)
        x[idx] = old_value
        it.iternext()
    
    return grad

def gradient_check(model, X, y, epsilon=1e-5):
    """
    梯度检查
    
    Args:
        model: 神经网络模型
        X: 输入数据
        y: 真实标签
        epsilon: 扰动值
    """
    # 前向传播
    output = model.forward(X)
```

```python
class WeightInitialization:
    """权重初始化方法集合"""
    
    @staticmethod
    def zero_init(shape):
        """全零初始化（不推荐）"""
        return np.zeros(shape)
    
    @staticmethod
    def random_init(shape, scale=0.01):
        """随机初始化"""
        return np.random.randn(*shape) * scale
    
    @staticmethod
    def xavier_init(shape):
        """Xavier初始化（适用于Sigmoid/Tanh）"""
        fan_in, fan_out = shape[0], shape[1]
        limit = np.sqrt(6 / (fan_in + fan_out))
        return np.random.uniform(-limit, limit, shape)
    
    @staticmethod
    def he_init(shape):
        """He初始化（适用于ReLU）"""
        fan_in = shape[0]
        std = np.sqrt(2.0 / fan_in)
        return np.random.randn(*shape) * std
    
    @staticmethod
    def lecun_init(shape):
        """LeCun初始化"""
        fan_in = shape[0]
        std = np.sqrt(1.0 / fan_in)
        return np.random.randn(*shape) * std
```

```python
    """使用不同初始化方法训练"""
    # 创建网络
    W1 = init_method((2, 10))
    b1 = np.zeros((1, 10))
    W2 = init_method((10, 1))
    b2 = np.zeros((1, 1))
    
    losses = []
    learning_rate = 0.1
    
    for epoch in range(epochs):
        # 前向传播
        Z1 = np.dot(X, W1) + b1
        A1 = np.maximum(0, Z1)
        Z2 = np.dot(A1, W2) + b2
        A2 = 1 / (1 + np.exp(-np.clip(Z2, -500, 500)))
        
        # 计算损失
        loss = -np.mean(y * np.log(A2 + 1e-8) + (1 - y) * np.log(1 - A2 + 1e-8))
        losses.append(loss)
        
        # 反向传播
        dZ2 = A2 - y
        dW2 = np.dot(A1.T, dZ2) / X.shape[0]
        db2 = np.sum(dZ2, axis=0, keepdims=True) / X.shape[0]
        
        dA1 = np.dot(dZ2, W2.T)
        dZ1 = dA1 * (Z1 > 0)
        dW1 = np.dot(X.T, dZ1) / X.shape[0]
        db1 = np.sum(dZ1, axis=0, keepdims=True) / X.shape[0]
        
        # 更新参数
        W1 -= learning_rate * dW1
        b1 -= learning_rate * db1
        W2 -= learning_rate * dW2
        b2 -= learning_rate * db2
    
    return losses
```

```python
import numpy as np

class RegularizedNN:
    """带正则化的神经网络"""
    
    def __init__(self, layer_sizes, reg_type='l2', reg_lambda=0.01):
        """
        初始化
        
        Args:
            layer_sizes: 各层大小
            reg_type: 正则化类型 ('l1', 'l2', 'none')
            reg_lambda: 正则化系数
        """
        self.layer_sizes = layer_sizes
        self.reg_type = reg_type
        self.reg_lambda = reg_lambda
        
        # 初始化权重
        self.weights = []
        self.biases = []
        for i in range(len(layer_sizes) - 1):
            w = np.random.randn(layer_sizes[i], layer_sizes[i+1]) * np.sqrt(2.0 / layer_sizes[i])
            b = np.zeros((1, layer_sizes[i+1]))
            self.weights.append(w)
            self.biases.append(b)
    
    def compute_regularization_loss(self):
        """计算正则化损失"""
        reg_loss = 0
        
        if self.reg_type == 'l1':
            for W in self.weights:
                reg_loss += np.sum(np.abs(W))
        elif self.reg_type == 'l2':
            for W in self.weights:
                reg_loss += np.sum(W ** 2)
        
        return self.reg_lambda * reg_loss
    
    def compute_regularization_gradient(self, W):
        """计算正则化梯度"""
        if self.reg_type == 'l1':
            return self.reg_lambda * np.sign(W)
        elif self.reg_type == 'l2':
            return self.reg_lambda * 2 * W
        else:
            return 0
    
    def forward(self, X):
        """前向传播"""
        A = X
        for i in range(len(self.weights) - 1):
            Z = np.dot(A, self.weights[i]) + self.biases[i]
            A = np.maximum(0, Z)
        
        Z = np.dot(A, self.weights[-1]) + self.biases[-1]
        A = 1 / (1 + np.exp(-np.clip(Z, -500, 500)))
        
        return A
    
    def train(self, X, y, epochs=1000, learning_rate=0.1):
        """训练"""
        losses = []
        
        for epoch in range(epochs):
            output = self.forward(X)
            
            # 计算损失（包括正则化）
            data_loss = -np.mean(y * np.log(output + 1e-8) + 
                                (1 - y) * np.log(1 - output + 1e-8))
            reg_loss = self.compute_regularization_loss()
            total_loss = data_loss + reg_loss
            losses.append(total_loss)
            
            # 反向传播（简化版）
            # 这里省略详细实现
            
            if epoch % 100 == 0:
                print(f"Epoch {epoch}, Loss: {total_loss:.4f} "
                      f"(Data: {data_loss:.4f}, Reg: {reg_loss:.4f})")
        
        return losses
```

```python
import numpy as np

class DropoutLayer:
    """Dropout层"""
    
    def __init__(self, dropout_rate=0.5):
        """
        初始化
        
        Args:
            dropout_rate: dropout比率
        """
        self.dropout_rate = dropout_rate
        self.mask = None
    
    def forward(self, X, training=True):
        """
        前向传播
        
        Args:
            X: 输入
            training: 是否训练模式
        
        Returns:
            输出
        """
        if training:
            # 训练模式：随机丢弃
            self.mask = np.random.rand(*X.shape) > self.dropout_rate
            return X * self.mask / (1 - self.dropout_rate)

            return X

        """
        反向传播
        
        Args:
            dout: 上游梯度
        
        Returns:
            下游梯度
        """
        return dout * self.mask / (1 - self.dropout_rate)
```

```python
"""
梯度下降优化算法
@author erik.zhou
"""
import numpy as np
import matplotlib.pyplot as plt

class Optimizers:
    """优化器集合"""
    
    @staticmethod
    def sgd(params, grads, learning_rate=0.01):
        """随机梯度下降"""
        for param, grad in zip(params, grads):
            param -= learning_rate * grad
    
    @staticmethod
    def momentum(params, grads, velocities, learning_rate=0.01, momentum=0.9):
        """动量法"""
        for i, (param, grad) in enumerate(zip(params, grads)):
            velocities[i] = momentum * velocities[i] - learning_rate * grad
            param += velocities[i]
    
    @staticmethod
    def rmsprop(params, grads, cache, learning_rate=0.01, decay_rate=0.99, epsilon=1e-8):
        """RMSprop"""
        for i, (param, grad) in enumerate(zip(params, grads)):
            cache[i] = decay_rate * cache[i] + (1 - decay_rate) * grad**2
            param -= learning_rate * grad / (np.sqrt(cache[i]) + epsilon)
    
    @staticmethod
    def adam(params, grads, m, v, t, learning_rate=0.001, beta1=0.9, beta2=0.999, epsilon=1e-8):
        """Adam优化器"""
        for i, (param, grad) in enumerate(zip(params, grads)):
            m[i] = beta1 * m[i] + (1 - beta1) * grad
            v[i] = beta2 * v[i] + (1 - beta2) * grad**2
            
            m_hat = m[i] / (1 - beta1**t)
            v_hat = v[i] / (1 - beta2**t)
            
            param -= learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)

    elif optimizer_name == 'adam':
```

```python
"""
学习率调度策略
@author erik.zhou
"""
import numpy as np
import matplotlib.pyplot as plt

class LearningRateSchedulers:
    """学习率调度器"""
    
    @staticmethod
    def step_decay(initial_lr, epoch, drop_rate=0.5, epochs_drop=10):
        """阶梯衰减"""
        return initial_lr * (drop_rate ** (epoch // epochs_drop))
    
    @staticmethod
    def exponential_decay(initial_lr, epoch, decay_rate=0.95):
        """指数衰减"""
        return initial_lr * (decay_rate ** epoch)
    
    @staticmethod
    def cosine_annealing(initial_lr, epoch, T_max=100, eta_min=0):
        """余弦退火"""
        return eta_min + (initial_lr - eta_min) * (1 + np.cos(np.pi * epoch / T_max)) / 2
    
    @staticmethod
    def warmup_cosine(initial_lr, epoch, warmup_epochs=10, T_max=100):
        """预热+余弦退火"""
        if epoch < warmup_epochs:
            return initial_lr * (epoch + 1) / warmup_epochs
        else:
            return LearningRateSchedulers.cosine_annealing(
                initial_lr, epoch - warmup_epochs, T_max - warmup_epochs
            )
```

```python
"""
批归一化（Batch Normalization）
@author erik.zhou
"""
import numpy as np

class BatchNormalization:
    """批归一化层"""
    
    def __init__(self, num_features, epsilon=1e-5, momentum=0.9):
        """
        初始化
        
        Args:
            num_features: 特征数量
            epsilon: 数值稳定性参数
            momentum: 移动平均动量
        """
        self.num_features = num_features
        self.epsilon = epsilon
        self.momentum = momentum
        
        # 可学习参数
        self.gamma = np.ones((1, num_features))
        self.beta = np.zeros((1, num_features))
        
        # 移动平均
        self.running_mean = np.zeros((1, num_features))
        self.running_var = np.ones((1, num_features))
    
    def forward(self, X, training=True):
        """
        前向传播
        
        Args:
            X: 输入 (batch_size, num_features)
            training: 是否训练模式
        
        Returns:
            输出
        """
        if training:
            # 计算批次统计量
            batch_mean = np.mean(X, axis=0, keepdims=True)
            batch_var = np.var(X, axis=0, keepdims=True)
            
            # 归一化
            X_norm = (X - batch_mean) / np.sqrt(batch_var + self.epsilon)
            
            # 更新移动平均
            self.running_mean = self.momentum * self.running_mean + (1 - self.momentum) * batch_mean
            self.running_var = self.momentum * self.running_var + (1 - self.momentum) * batch_var
            
            # 保存用于反向传播
            self.cache = (X, X_norm, batch_mean, batch_var)
        else:
            # 使用移动平均
            X_norm = (X - self.running_mean) / np.sqrt(self.running_var + self.epsilon)
        
        # 缩放和平移
        out = self.gamma * X_norm + self.beta
        
        return out
    
    def backward(self, dout):
        """
        反向传播
        
        Args:
            dout: 上游梯度
        
        Returns:
            下游梯度
        """
        X, X_norm, batch_mean, batch_var = self.cache
        m = X.shape[0]
        
        # 计算梯度
        dgamma = np.sum(dout * X_norm, axis=0, keepdims=True)
        dbeta = np.sum(dout, axis=0, keepdims=True)
        
        dX_norm = dout * self.gamma
        dvar = np.sum(dX_norm * (X - batch_mean) * -0.5 * (batch_var + self.epsilon)**(-1.5), 
                     axis=0, keepdims=True)
        dmean = np.sum(dX_norm * -1 / np.sqrt(batch_var + self.epsilon), axis=0, keepdims=True) + \
                dvar * np.sum(-2 * (X - batch_mean), axis=0, keepdims=True) / m
        
        dX = dX_norm / np.sqrt(batch_var + self.epsilon) + \
             dvar * 2 * (X - batch_mean) / m + \
             dmean / m
        
        return dX, dgamma, dbeta
```

```python
    """训练网络（有/无批归一化）"""
    # 简单的3层网络
    X = np.random.randn(100, 10)
    y = (np.sum(X, axis=1, keepdims=True) > 0).astype(float)
    
    W1 = np.random.randn(10, 20) * 0.01
    b1 = np.zeros((1, 20))
    W2 = np.random.randn(20, 1) * 0.01
    b2 = np.zeros((1, 1))
    
    if use_bn:
        bn = BatchNormalization(20)
    
    losses = []
    learning_rate = 0.1
    
    for epoch in range(epochs):
        # 前向传播
        Z1 = np.dot(X, W1) + b1
        
        if use_bn:
            Z1 = bn.forward(Z1, training=True)
        
        A1 = np.maximum(0, Z1)
        Z2 = np.dot(A1, W2) + b2
        A2 = 1 / (1 + np.exp(-np.clip(Z2, -500, 500)))
        
        # 计算损失
        loss = -np.mean(y * np.log(A2 + 1e-8) + (1 - y) * np.log(1 - A2 + 1e-8))
        losses.append(loss)
        
        # 反向传播（简化）
        dZ2 = A2 - y
        dW2 = np.dot(A1.T, dZ2) / X.shape[0]
        db2 = np.sum(dZ2, axis=0, keepdims=True) / X.shape[0]
        
        dA1 = np.dot(dZ2, W2.T)
        dZ1 = dA1 * (Z1 > 0)
        
        if use_bn:
            dZ1, _, _ = bn.backward(dZ1)
        
        dW1 = np.dot(X.T, dZ1) / X.shape[0]
        db1 = np.sum(dZ1, axis=0, keepdims=True) / X.shape[0]
        
        # 更新参数
        W1 -= learning_rate * dW1
        b1 -= learning_rate * db1
        W2 -= learning_rate * dW2
        b2 -= learning_rate * db2
    
    return losses

# 比较有无批归一化
```

```python
"""
完整的神经网络训练流程
@author erik.zhou
"""
import numpy as np
import matplotlib.pyplot as plt

class NeuralNetwork:
    """完整的神经网络实现"""
    
    def __init__(self, layer_sizes, activation='relu', use_bn=True, dropout_rate=0.5):
        """
        初始化神经网络
        
        Args:
            layer_sizes: 各层大小
            activation: 激活函数类型
            use_bn: 是否使用批归一化
            dropout_rate: dropout比率
        """
        self.layer_sizes = layer_sizes
        self.activation = activation
        self.use_bn = use_bn
        self.dropout_rate = dropout_rate
        
        # 初始化参数
        self.weights = []
        self.biases = []
        self.bn_layers = []
        
        for i in range(len(layer_sizes) - 1):
            w = np.random.randn(layer_sizes[i], layer_sizes[i+1]) * np.sqrt(2.0 / layer_sizes[i])
            b = np.zeros((1, layer_sizes[i+1]))
            self.weights.append(w)
            self.biases.append(b)
            
            if use_bn and i < len(layer_sizes) - 2:
                self.bn_layers.append(BatchNormalization(layer_sizes[i+1]))
    
    def forward(self, X, training=True):
        """前向传播"""
        A = X
        
        for i in range(len(self.weights) - 1):
            Z = np.dot(A, self.weights[i]) + self.biases[i]
            
            if self.use_bn:
                Z = self.bn_layers[i].forward(Z, training)
            
            A = np.maximum(0, Z)  # ReLU
            
            if training and self.dropout_rate > 0:
                mask = np.random.rand(*A.shape) > self.dropout_rate
                A = A * mask / (1 - self.dropout_rate)
        
        # 输出层
        Z = np.dot(A, self.weights[-1]) + self.biases[-1]
        A = 1 / (1 + np.exp(-np.clip(Z, -500, 500)))  # Sigmoid
        
        return A
    
    def train(self, X_train, y_train, X_val, y_val, 
              epochs=100, batch_size=32, learning_rate=0.01):
        """
        训练网络
        
        Args:
            X_train: 训练数据
            y_train: 训练标签
            X_val: 验证数据
            y_val: 验证标签
            epochs: 训练轮数
            batch_size: 批量大小
            learning_rate: 学习率
        
        Returns:
            history: 训练历史
        """
        history = {
            'train_loss': [],
            'val_loss': [],
            'train_acc': [],
            'val_acc': []
        }
        
        n_batches = (X_train.shape[0] + batch_size - 1) // batch_size
        
        for epoch in range(epochs):
            # 训练
            train_loss = 0
            train_correct = 0
            
            # 打乱数据
            indices = np.random.permutation(X_train.shape[0])
            X_train_shuffled = X_train[indices]
            y_train_shuffled = y_train[indices]
            
            for i in range(n_batches):
                start_idx = i * batch_size
                end_idx = min((i + 1) * batch_size, X_train.shape[0])
                
                X_batch = X_train_shuffled[start_idx:end_idx]
                y_batch = y_train_shuffled[start_idx:end_idx]
                
                # 前向传播
                output = self.forward(X_batch, training=True)
                
                # 计算损失
                loss = -np.mean(y_batch * np.log(output + 1e-8) + 
                               (1 - y_batch) * np.log(1 - output + 1e-8))
                train_loss += loss
                
                # 计算准确率
                predictions = (output > 0.5).astype(float)
                train_correct += np.sum(predictions == y_batch)
                
                # 反向传播（简化）
                # 这里省略详细实现
            
            # 验证
            val_output = self.forward(X_val, training=False)
            val_loss = -np.mean(y_val * np.log(val_output + 1e-8) + 
                               (1 - y_val) * np.log(1 - val_output + 1e-8))
            val_predictions = (val_output > 0.5).astype(float)
            val_correct = np.sum(val_predictions == y_val)
            
            # 记录历史
            history['train_loss'].append(train_loss / n_batches)
            history['val_loss'].append(val_loss)
            history['train_acc'].append(train_correct / X_train.shape[0])
            history['val_acc'].append(val_correct / X_val.shape[0])
            
            if epoch % 10 == 0:
                print(f"Epoch {epoch}/{epochs}")
                print(f"  Train Loss: {history['train_loss'][-1]:.4f}, "
                      f"Train Acc: {history['train_acc'][-1]:.4f}")
                print(f"  Val Loss: {history['val_loss'][-1]:.4f}, "
                      f"Val Acc: {history['val_acc'][-1]:.4f}")
        
        return history
```

```python
troubleshooting_guide = """
神经网络常见问题和解决方案：

1. 梯度消失
   - 问题：深层网络训练困难，梯度接近0
   - 解决：使用ReLU激活函数、批归一化、残差连接

2. 梯度爆炸
   - 问题：梯度过大导致参数更新不稳定
   - 解决：梯度裁剪、降低学习率、使用批归一化

3. 过拟合
   - 问题：训练集表现好，测试集表现差
   - 解决：增加数据、Dropout、L1/L2正则化、早停

4. 欠拟合
   - 问题：训练集和测试集表现都差
   - 解决：增加网络容量、训练更多轮次、降低正则化

5. 训练不收敛
   - 问题：损失不下降或震荡
   - 解决：降低学习率、检查数据预处理、使用批归一化

6. 神经元死亡（ReLU）
   - 问题：大量神经元输出恒为0
   - 解决：使用Leaky ReLU、降低学习率、检查初始化

7. 训练速度慢
   - 问题：训练时间过长
   - 解决：使用GPU、批归一化、更好的优化器（Adam）

8. 内存不足
   - 问题：显存溢出
   - 解决：减小批量大小、使用梯度累积、模型并行

调试技巧：
1. 先在小数据集上过拟合，确保模型有学习能力
2. 可视化激活值和梯度分布
3. 监控训练和验证指标
4. 使用学习率查找器
5. 检查数据预处理和标签
"""
```
