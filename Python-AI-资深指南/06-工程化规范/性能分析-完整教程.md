## 性能分析概述

### 为什么需要性能分析？

```python
# ❌ 盲目优化
def process_data(data):
    # 花了一周优化这个函数
    # 但它只占总运行时间的 1%
    return [x * 2 for x in data]

# ✅ 基于分析的优化
# 1. 先用 cProfile 找到瓶颈
# 2. 发现 database_query() 占 80% 时间
# 3. 优化数据库查询，性能提升 5 倍
```

### 性能分析工具对比

| 工具 | 分析对象 | 开销 | 适用场景 | 优先级 |
|------|---------|------|---------|--------|
| cProfile | CPU 时间 | 低 | 找出慢函数 | ⭐⭐⭐⭐⭐ |
| line_profiler | 行级 CPU | 中 | 优化具体代码 | ⭐⭐⭐⭐ |
| memory_profiler | 内存使用 | 高 | 内存泄漏 | ⭐⭐⭐⭐ |
| py-spy | CPU 采样 | 极低 | 生产环境 | ⭐⭐⭐⭐⭐ |
| tracemalloc | 内存追踪 | 中 | 内存分配 | ⭐⭐⭐ |

### 性能分析流程

```
1. 测量基准性能
   ↓
2. 使用 cProfile 找出热点函数
   ↓
3. 使用 line_profiler 分析具体代码行
   ↓
4. 使用 memory_profiler 检查内存问题
   ↓
5. 优化代码
   ↓
6. 重新测量，验证优化效果
```

## cProfile - CPU 性能分析

### 基本使用

```python
import cProfile
import pstats
from pstats import SortKey

def slow_function():
    """模拟慢函数"""
    total = 0
    for i in range(1000000):
        total += i
    return total

def fast_function():
    """快速函数"""
    return sum(range(1000000))

def main():
    slow_function()
    fast_function()

# 方法 1：命令行使用
# python -m cProfile -o output.prof my_script.py

# 方法 2：代码中使用
if __name__ == "__main__":
    profiler = cProfile.Profile()
    profiler.enable()
    
    main()
    
    profiler.disable()
    
    # 打印统计信息
    stats = pstats.Stats(profiler)
    stats.sort_stats(SortKey.CUMULATIVE)
    stats.print_stats(10)  # 打印前 10 个最慢的函数
```

### 输出解读

```
         5 function calls in 0.234 seconds

   Ordered by: cumulative time

   ncalls  tottime  percall  cumtime  percall filename:lineno(function)
        1    0.000    0.000    0.234    0.234 script.py:1(main)
        1    0.180    0.180    0.180    0.180 script.py:3(slow_function)
        1    0.054    0.054    0.054    0.054 script.py:9(fast_function)
        1    0.000    0.000    0.000    0.000 {built-in method builtins.sum}
        1    0.000    0.000    0.000    0.000 {method 'disable' of '_lsprof.Profiler'}
```

**字段说明**：
- `ncalls`: 函数调用次数
- `tottime`: 函数内部总时间（不含子函数）
- `percall`: tottime / ncalls
- `cumtime`: 累计时间（含子函数）
- `percall`: cumtime / ncalls

### 实用技巧

```python
import cProfile
import pstats
from io import StringIO

def profile_function(func):
    """装饰器：分析函数性能"""
    def wrapper(*args, **kwargs):
        profiler = cProfile.Profile()
        profiler.enable()
        
        result = func(*args, **kwargs)
        
        profiler.disable()
        
        # 输出到字符串
        s = StringIO()
        stats = pstats.Stats(profiler, stream=s)
        stats.sort_stats('cumulative')
        stats.print_stats(20)
        
        print(s.getvalue())
        return result
    
    return wrapper

@profile_function
def my_function():
    # 你的代码
    pass
```

### 内存优化技巧

```python
from memory_profiler import profile
import sys

# ❌ 内存浪费
@profile
def process_large_file_bad(filename):
    """一次性读取整个文件"""
    with open(filename, 'r') as f:
        data = f.read()  # 占用大量内存
        lines = data.split('\n')
        return len(lines)

# ✅ 内存优化
@profile
def process_large_file_good(filename):
    """逐行读取文件"""
    count = 0
    with open(filename, 'r') as f:
        for line in f:  # 逐行读取，内存占用小
            count += 1
    return count

# 使用生成器节省内存
@profile
def generate_numbers(n):
    """生成器版本"""
    for i in range(n):
        yield i ** 2

@profile
def list_numbers(n):
    """列表版本"""
    return [i ** 2 for i in range(n)]

# 测试
gen = generate_numbers(1000000)  # 几乎不占内存
lst = list_numbers(1000000)      # 占用大量内存

print(f"Generator size: {sys.getsizeof(gen)} bytes")
print(f"List size: {sys.getsizeof(lst)} bytes")
```

### tracemalloc 内存追踪

```python
import tracemalloc

def memory_leak_example():
    """内存泄漏示例"""
    # 启动内存追踪
    tracemalloc.start()
    
    # 记录初始快照
    snapshot1 = tracemalloc.take_snapshot()
    
    # 执行可能泄漏内存的代码
    big_list = []
    for i in range(100000):
        big_list.append([i] * 100)
    
    # 记录第二个快照
    snapshot2 = tracemalloc.take_snapshot()
    
    # 比较快照
    top_stats = snapshot2.compare_to(snapshot1, 'lineno')
    
    print("[ Top 10 differences ]")
    for stat in top_stats[:10]:
        print(stat)
    
    tracemalloc.stop()

if __name__ == "__main__":
    memory_leak_example()
```

## 性能优化技巧

### 1. 使用内置函数

```python
import time

# ❌ 慢：手动实现
def sum_slow(numbers):
    total = 0
    for num in numbers:
        total += num
    return total

# ✅ 快：使用内置函数
def sum_fast(numbers):
    return sum(numbers)

# 测试
numbers = list(range(1000000))

start = time.time()
sum_slow(numbers)
print(f"Slow: {time.time() - start:.4f}s")

start = time.time()
sum_fast(numbers)
print(f"Fast: {time.time() - start:.4f}s")

# 结果：内置函数快 10 倍以上
```

### 2. 列表推导式 vs 循环

```python
# ❌ 慢：循环 + append
def create_list_slow(n):
    result = []
    for i in range(n):
        result.append(i ** 2)
    return result

# ✅ 快：列表推导式
def create_list_fast(n):
    return [i ** 2 for i in range(n)]

# 测试
n = 100000
%timeit create_list_slow(n)  # 约 15 ms
%timeit create_list_fast(n)  # 约 10 ms
```

### 3. 使用生成器

```python
# ❌ 内存浪费：返回列表
def read_large_file_bad(filename):
    with open(filename) as f:
        return f.readlines()  # 一次性加载到内存

# ✅ 内存优化：使用生成器
def read_large_file_good(filename):
    with open(filename) as f:
        for line in f:
            yield line.strip()

# 使用
for line in read_large_file_good('large_file.txt'):
    process(line)  # 逐行处理，内存占用小
```

### 4. 字典查找 vs 列表查找

```python
# ❌ 慢：列表查找 O(n)
def check_in_list(item, items):
    return item in items  # 线性查找

# ✅ 快：字典/集合查找 O(1)
def check_in_set(item, items):
    return item in set(items)  # 哈希查找

# 测试
items = list(range(10000))
%timeit check_in_list(9999, items)  # 约 100 μs
%timeit check_in_set(9999, items)   # 约 1 μs
```

### 5. 字符串拼接

```python
# ❌ 慢：+ 拼接
def concat_slow(strings):
    result = ""
    for s in strings:
        result += s  # 每次创建新字符串
    return result

# ✅ 快：join
def concat_fast(strings):
    return "".join(strings)

# 测试
strings = ["hello"] * 10000
%timeit concat_slow(strings)  # 约 50 ms
%timeit concat_fast(strings)  # 约 0.5 ms
```

### 6. 局部变量 vs 全局变量

```python
# ❌ 慢：全局变量
CONSTANT = 100

def use_global():
    total = 0
    for i in range(1000000):
        total += CONSTANT  # 全局查找慢
    return total

# ✅ 快：局部变量
def use_local():
    constant = 100  # 局部变量
    total = 0
    for i in range(1000000):
        total += constant  # 局部查找快
    return total

%timeit use_global()  # 约 80 ms
%timeit use_local()   # 约 60 ms
```

### 自定义基准测试框架

```python
import time
from functools import wraps
from typing import Callable

class Benchmark:
    """基准测试工具"""
    
    def __init__(self, name: str, iterations: int = 1000):
        self.name = name
        self.iterations = iterations
        self.results = []
    
    def __call__(self, func: Callable) -> Callable:
        @wraps(func)
        def wrapper(*args, **kwargs):
            times = []
            for _ in range(self.iterations):
                start = time.perf_counter()
                result = func(*args, **kwargs)
                end = time.perf_counter()
                times.append(end - start)
            
            avg_time = sum(times) / len(times)
            min_time = min(times)
            max_time = max(times)
            
            print(f"\n{self.name} Benchmark Results:")
            print(f"  Iterations: {self.iterations}")
            print(f"  Average: {avg_time*1000:.4f} ms")
            print(f"  Min: {min_time*1000:.4f} ms")
            print(f"  Max: {max_time*1000:.4f} ms")
            
            return result
        
        return wrapper

# 使用
@Benchmark("List Comprehension", iterations=10000)
def test_list_comp():
    return [i ** 2 for i in range(1000)]

@Benchmark("Map Function", iterations=10000)
def test_map():
    return list(map(lambda x: x ** 2, range(1000)))

test_list_comp()
test_map()
```

### 案例 1：优化数据处理

```python
from line_profiler import profile
import pandas as pd

# ❌ 慢版本
@profile
def process_data_slow(df):
    """逐行处理 DataFrame"""
    result = []
    for idx, row in df.iterrows():  # 慢！
        if row['value'] > 0:
            result.append(row['value'] * 2)
    return result

# ✅ 快版本
@profile
def process_data_fast(df):
    """向量化操作"""
    mask = df['value'] > 0
    return (df.loc[mask, 'value'] * 2).tolist()

# 测试
df = pd.DataFrame({'value': range(-5000, 5000)})

# 慢版本：约 500 ms
result1 = process_data_slow(df)

# 快版本：约 5 ms（快 100 倍！）
result2 = process_data_fast(df)
```

### 案例 2：优化数据库查询

```python
from memory_profiler import profile
import sqlite3

# ❌ 慢版本：多次查询
@profile
def fetch_users_slow(user_ids):
    """逐个查询用户"""
    conn = sqlite3.connect('users.db')
    cursor = conn.cursor()
    
    users = []
    for user_id in user_ids:
        cursor.execute("SELECT * FROM users WHERE id = ?", (user_id,))
        users.append(cursor.fetchone())
    
    conn.close()
    return users

# ✅ 快版本：批量查询
@profile
def fetch_users_fast(user_ids):
    """批量查询用户"""
    conn = sqlite3.connect('users.db')
    cursor = conn.cursor()
    
    placeholders = ','.join('?' * len(user_ids))
    query = f"SELECT * FROM users WHERE id IN ({placeholders})"
    cursor.execute(query, user_ids)
    users = cursor.fetchall()
    
    conn.close()
    return users

# 测试
user_ids = list(range(1, 1001))

# 慢版本：约 5 秒（1000 次查询）
users1 = fetch_users_slow(user_ids)

# 快版本：约 0.05 秒（1 次查询，快 100 倍！）
users2 = fetch_users_fast(user_ids)
```

### 案例 3：优化 API 调用

```python
import asyncio
import aiohttp
import time

# ❌ 慢版本：同步调用
def fetch_urls_sync(urls):
    """同步获取多个 URL"""
    import requests
    results = []
    for url in urls:
        response = requests.get(url)
        results.append(response.text)
    return results

# ✅ 快版本：异步调用
async def fetch_url(session, url):
    """异步获取单个 URL"""
    async with session.get(url) as response:
        return await response.text()

async def fetch_urls_async(urls):
    """异步获取多个 URL"""
    async with aiohttp.ClientSession() as session:
        tasks = [fetch_url(session, url) for url in urls]
        return await asyncio.gather(*tasks)

# 测试
urls = ['http://example.com'] * 10

# 同步版本：约 10 秒（串行执行）
start = time.time()
results1 = fetch_urls_sync(urls)
print(f"Sync: {time.time() - start:.2f}s")

# 异步版本：约 1 秒（并行执行，快 10 倍！）
start = time.time()
results2 = asyncio.run(fetch_urls_async(urls))
print(f"Async: {time.time() - start:.2f}s")
```

## 总结

### 性能优化原则

1. **先测量，再优化**：不要盲目优化
2. **找到瓶颈**：优化占时间最多的部分
3. **权衡取舍**：性能 vs 可读性 vs 维护性
4. **持续监控**：定期检查性能指标

### 工具使用建议

| 场景 | 推荐工具 | 说明 |
|------|---------|------|
| 找出慢函数 | cProfile | 第一步，找到瓶颈 |
| 优化具体代码 | line_profiler | 第二步，行级分析 |
| 内存问题 | memory_profiler | 内存泄漏排查 |
| 生产环境 | py-spy | 无侵入式分析 |
| 基准测试 | timeit | 比较不同实现 |

### 常见优化技巧

1. 使用内置函数和标准库
2. 列表推导式代替循环
3. 生成器节省内存
4. 字典/集合代替列表查找
5. join 代替 + 拼接字符串
6. 局部变量代替全局变量
7. 向量化操作（NumPy/Pandas）
8. 批量操作代替逐个操作
9. 异步 IO 代替同步 IO
10. 缓存重复计算结果
