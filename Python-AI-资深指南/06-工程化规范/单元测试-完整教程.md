```python
# test_calculator.py

def add(a: int, b: int) -> int:
```

```bash
pytest -v

pytest -s
```

## 📖 pytest 基础

### 1. 测试函数命名

```python
# ✅ 正确示例 - 测试函数以 test_ 开头
def test_user_creation():
    """测试用户创建"""
    user = User(name="张三", age=30)
    assert user.name == "张三"
    assert user.age == 30

def test_user_validation():
    """测试用户验证"""
    with pytest.raises(ValueError):
        User(name="", age=-1)

# ❌ 错误示例 - 不以 test_ 开头的函数不会被执行
def check_user_creation():  # 不会被 pytest 发现
    user = User(name="张三", age=30)
    assert user.name == "张三"
```

### 2. 测试类组织

```python
import pytest

class TestUserService:
    """用户服务测试类"""
    
    def test_get_user(self):
        """测试获取用户"""
        service = UserService()
        user = service.get_user(1)
        assert user is not None
        assert user.id == 1
    
    def test_create_user(self):
        """测试创建用户"""
        service = UserService()
        user = service.create_user("张三", "zhangsan@example.com")
        assert user.name == "张三"
        assert user.email == "zhangsan@example.com"
    
    def test_delete_user(self):
        """测试删除用户"""
        service = UserService()
        result = service.delete_user(1)
        assert result is True
```

### 3. 断言方法

```python
import pytest

def test_assertions():
    """测试各种断言"""
    # 相等断言
    assert 1 + 1 == 2
    assert "hello" == "hello"
    
    # 不相等断言
    assert 1 != 2
    
    # 布尔断言
    assert True
    assert not False
    
    # 包含断言
    assert "hello" in "hello world"
    assert 1 in [1, 2, 3]
    
    # 类型断言
    assert isinstance("hello", str)
    assert isinstance(123, int)
    
    # None 断言
    value = None
    assert value is None
    
    # 异常断言
    with pytest.raises(ValueError):
        int("not a number")
    
    with pytest.raises(ValueError, match="invalid"):
        raise ValueError("invalid input")
```

## 🎯 Fixture 使用

### 1. 基础 Fixture

```python
import pytest

@pytest.fixture
def sample_user():
    """创建示例用户"""
    return User(name="张三", age=30, email="zhangsan@example.com")

def test_user_name(sample_user):
    """测试用户名"""
    assert sample_user.name == "张三"

def test_user_age(sample_user):
    """测试用户年龄"""
    assert sample_user.age == 30
```

### 2. Fixture 作用域

```python
import pytest

@pytest.fixture(scope="function")  # 默认，每个测试函数执行一次
def function_fixture():
    print("Setup function fixture")
    yield "function data"
    print("Teardown function fixture")

@pytest.fixture(scope="class")  # 每个测试类执行一次
def class_fixture():
    print("Setup class fixture")
    yield "class data"
    print("Teardown class fixture")

@pytest.fixture(scope="module")  # 每个模块执行一次
def module_fixture():
    print("Setup module fixture")
    yield "module data"
    print("Teardown module fixture")

@pytest.fixture(scope="session")  # 整个测试会话执行一次
def session_fixture():
    print("Setup session fixture")
    yield "session data"
    print("Teardown session fixture")
```

### 3. Fixture 依赖

```python
import pytest

@pytest.fixture
```

### 4. conftest.py 共享 Fixture

```python
from app.database import Database
from app.models import User

@pytest.fixture(scope="session")
def database():
    """数据库连接（会话级别）"""
    db = Database("sqlite:///:memory:")
    db.create_tables()
    yield db
    db.drop_tables()
    db.close()

@pytest.fixture
def sample_users():
    """示例用户列表"""
    return [
        User(name="张三", age=30),
        User(name="李四", age=25),
        User(name="王五", age=35),
    ]

@pytest.fixture(autouse=True)
def reset_database(database):
    """每个测试后重置数据库"""
```

## 🎭 Mock 使用

### 1. 基础 Mock

```python
from unittest.mock import Mock, patch
import pytest

def test_mock_basic():
    """基础 Mock 示例"""
    # 创建 Mock 对象
    mock_service = Mock()
    
    # 设置返回值
    mock_service.get_user.return_value = {"id": 1, "name": "张三"}
    
    # 调用并断言
    result = mock_service.get_user(1)
    assert result["name"] == "张三"
    
    # 验证调用
    mock_service.get_user.assert_called_once_with(1)

def test_mock_side_effect():
    """Mock 副作用"""
    mock_service = Mock()
    
    # 设置副作用（抛出异常）
    mock_service.get_user.side_effect = ValueError("用户不存在")
    
    with pytest.raises(ValueError, match="用户不存在"):
        mock_service.get_user(999)
```

### 2. patch 装饰器

```python
from unittest.mock import patch
import requests

def get_user_from_api(user_id: int) -> dict:
    """从 API 获取用户"""
    response = requests.get(f"https://api.example.com/users/{user_id}")
    return response.json()

@patch('requests.get')
def test_get_user_from_api(mock_get):
    """测试 API 调用"""
    # 设置 Mock 返回值
    mock_response = Mock()
    mock_response.json.return_value = {"id": 1, "name": "张三"}
    mock_get.return_value = mock_response
    
    # 调用函数
    result = get_user_from_api(1)
    
    # 断言
    assert result["name"] == "张三"
    mock_get.assert_called_once_with("https://api.example.com/users/1")
```

### 3. patch 上下文管理器

```python
from unittest.mock import patch
import os

def test_environment_variable():
    """测试环境变量"""
    with patch.dict(os.environ, {"API_KEY": "test_key"}):
        assert os.getenv("API_KEY") == "test_key"
    
    # 退出上下文后恢复原值
    assert os.getenv("API_KEY") != "test_key"

def test_file_operations():
    """测试文件操作"""
    with patch('builtins.open', create=True) as mock_open:
        mock_open.return_value.__enter__.return_value.read.return_value = "test content"
        
        with open("test.txt") as f:
            content = f.read()
        
        assert content == "test content"
```

### 4. pytest-mock 插件

```python
import pytest

def test_with_mocker(mocker):
    """使用 pytest-mock"""
    # 创建 Mock
    mock_service = mocker.Mock()
    mock_service.get_user.return_value = {"id": 1, "name": "张三"}
    
    # 使用 patch
    mocker.patch('app.services.UserService', return_value=mock_service)
    
    # 测试代码
    from app.services import UserService
    service = UserService()
    result = service.get_user(1)
    
    assert result["name"] == "张三"

def test_spy(mocker):
    """使用 spy 监控真实对象"""
    real_service = UserService()
    spy = mocker.spy(real_service, 'get_user')
    
    # 调用真实方法
    result = real_service.get_user(1)
    
    # 验证调用
    spy.assert_called_once_with(1)
    assert result is not None
```

## 📊 测试覆盖率

### 1. 生成覆盖率报告

```bash
# 运行测试并生成覆盖率报告
pytest --cov=app --cov-report=html --cov-report=term

# 只显示未覆盖的行
pytest --cov=app --cov-report=term-missing

# 设置最低覆盖率要求
pytest --cov=app --cov-fail-under=80
```

### 2. 配置文件

```ini
# pytest.ini 或 setup.cfg

[tool:pytest]
testpaths = tests
python_files = test_*.py
python_classes = Test*
python_functions = test_*
addopts = 
    --cov=app
    --cov-report=html
    --cov-report=term-missing
    --cov-fail-under=80
    -v
```

```ini
# .coveragerc

[report]

    def __repr__
    raise AssertionError
    raise NotImplementedError

    if TYPE_CHECKING:
```

### 2. AAA 模式（Arrange-Act-Assert）

```python
def test_calculate_discount():
    """测试计算折扣"""
    # Arrange - 准备测试数据
    price = 100.0
    discount_rate = 0.2
    expected_price = 80.0
    
    # Act - 执行被测试的代码
    result = calculate_discount(price, discount_rate)
    
    # Assert - 验证结果
    assert result == expected_price
```

### 3. 一个测试只测一个功能

```python
# ✅ 正确示例 - 每个测试只测一个功能
def test_user_name_validation():

def test_user_email_validation():

def test_user_validation():
```

### 5. 测试边界条件

```python
def test_calculate_discount_boundary_conditions():
    """测试计算折扣的边界条件"""
    # 正常情况
    assert calculate_discount(100, 0.2) == 80
    
    # 边界条件
    assert calculate_discount(100, 0) == 100  # 无折扣
    assert calculate_discount(100, 1) == 0    # 全折扣
    
    # 异常情况
    with pytest.raises(ValueError):
        calculate_discount(-100, 0.2)  # 负价格
    
    with pytest.raises(ValueError):
        calculate_discount(100, 1.5)  # 折扣率超过1
```

## ❓ 常见问题

### Q1: 测试覆盖率要达到多少？
A: 建议：
- 核心业务逻辑：90%+
- 整体项目：80%+
- 不要追求 100%，关注关键路径

### Q2: 什么时候使用 Mock？
A: 以下情况使用 Mock：
- 外部 API 调用
- 数据库操作
- 文件系统操作
- 时间相关操作
- 随机数生成

### Q3: 如何测试异步代码？
A: 使用 pytest-asyncio：

```python
import pytest

@pytest.mark.asyncio
async def test_async_function():
    result = await async_function()
    assert result == expected
```
