## 1. 异步编程基础

### 1.1 同步 vs 异步

```python
"""
同步与异步对比示例
@author erik.zhou
"""
import time
import asyncio

# 同步版本
def sync_task(name, delay):
    """同步任务"""
    print(f"{name} 开始")
    time.sleep(delay)
    print(f"{name} 完成")
    return f"{name} 结果"

def sync_main():
    """同步主函数"""
    start = time.time()
    sync_task("任务1", 2)
    sync_task("任务2", 2)
    sync_task("任务3", 2)
    print(f"总耗时: {time.time() - start:.2f}秒")  # 约6秒

# 异步版本
async def async_task(name, delay):
    """异步任务"""
    print(f"{name} 开始")
    await asyncio.sleep(delay)
    print(f"{name} 完成")
    return f"{name} 结果"

async def async_main():
    """异步主函数"""
    start = time.time()
    await asyncio.gather(
        async_task("任务1", 2),
        async_task("任务2", 2),
        async_task("任务3", 2)
    )
    print(f"总耗时: {time.time() - start:.2f}秒")  # 约2秒

# 运行
if __name__ == "__main__":
    # sync_main()
    asyncio.run(async_main())
```

```python
"""
异步 HTTP 请求示例
@author erik.zhou

import asyncio
import aiohttp

async def fetch_url(session, url):
    """获取单个 URL"""
    async with session.get(url) as response:
        return await response.text()

    """获取多个 URL"""
```

```python
"""
信号量示例
@author erik.zhou

import asyncio

async def worker(semaphore, name, delay):
    """工作任务"""
    async with semaphore:
        print(f"{name} 开始工作")
        await asyncio.sleep(delay)

async def main():
    """主函数"""
```

### 5.2 锁

```python
"""
锁示例
@author erik.zhou
"""
import asyncio

class Counter:
    """计数器类"""
    
    def __init__(self):
        self.value = 0
        self.lock = asyncio.Lock()
    
    async def increment(self):
        """增加计数"""
        async with self.lock:
            current = self.value
            await asyncio.sleep(0.1)  # 模拟耗时操作
            self.value = current + 1

async def main():
    """主函数"""
    counter = Counter()
    
    # 并发增加计数
    await asyncio.gather(*[counter.increment() for _ in range(10)])
    
    print(f"最终计数: {counter.value}")  # 10

asyncio.run(main())
```

## 7. 异步迭代器和生成器

### 7.1 异步迭代器

```python
"""
异步迭代器示例
@author erik.zhou
"""
import asyncio

class AsyncRange:
    """异步范围迭代器"""
    
    def __init__(self, start, end):
        self.current = start
        self.end = end
    
    def __aiter__(self):
        return self
    
    async def __anext__(self):
        if self.current >= self.end:
            raise StopAsyncIteration
        
        await asyncio.sleep(0.1)
        value = self.current
        self.current += 1
        return value

async def main():
    """主函数"""
    async for num in AsyncRange(0, 5):
        print(num)

asyncio.run(main())
```

```python
"""
异步生成器示例
@author erik.zhou

import asyncio

async def async_fibonacci(n):
    """异步斐波那契数列生成器"""
    a, b = 0, 1
    count = 0
    while count < n:
        await asyncio.sleep(0.1)
        yield a
        a, b = b, a + b
        count += 1

async def main():

    print()

asyncio.run(main())
```

## 8. 异常处理与调试

### 8.1 异常处理

```python
"""
异常处理示例
@author erik.zhou
"""
import asyncio

async def risky_operation(should_fail=False):
    """可能失败的操作"""
    await asyncio.sleep(1)
    if should_fail:
        raise ValueError("操作失败")
    return "成功"

async def main():
    """主函数"""
    # 单个任务异常处理
    try:
        result = await risky_operation(should_fail=True)
    except ValueError as e:
        print(f"捕获异常: {e}")
    
    # gather 异常处理
    results = await asyncio.gather(
        risky_operation(should_fail=False),
        risky_operation(should_fail=True),
        risky_operation(should_fail=False),
        return_exceptions=True  # 返回异常而不是抛出
    )
    
    for i, result in enumerate(results):
        if isinstance(result, Exception):
            print(f"任务{i} 失败: {result}")
        else:
            print(f"任务{i} 成功: {result}")

asyncio.run(main())
```

## 9. 性能优化

### 9.1 批量处理

```python
"""
批量处理示例
@author erik.zhou
"""
import asyncio

async def process_item(item):
    """处理单个项目"""
    await asyncio.sleep(0.1)
    return item * 2

async def process_batch(items, batch_size=10):
    """批量处理项目"""
    results = []
    for i in range(0, len(items), batch_size):
        batch = items[i:i + batch_size]
        batch_results = await asyncio.gather(*[process_item(item) for item in batch])
        results.extend(batch_results)
    return results

async def main():
    """主函数"""
    items = list(range(100))
    results = await process_batch(items, batch_size=20)
    print(f"处理了 {len(results)} 个项目")

asyncio.run(main())
```

### 9.2 连接池

```python
"""
连接池示例
@author erik.zhou
"""
import asyncio
from asyncio import Queue

class ConnectionPool:
    """连接池类"""
    
    def __init__(self, size=10):
        self.size = size
        self.pool = Queue(maxsize=size)
    
    async def initialize(self):
        """初始化连接池"""
        for i in range(self.size):
            await self.pool.put(f"连接{i}")
    
    async def acquire(self):
        """获取连接"""
        return await self.pool.get()
    
    async def release(self, conn):
        """释放连接"""
        await self.pool.put(conn)

async def worker(pool, name):
    """工作任务"""
    conn = await pool.acquire()
    try:
        print(f"{name} 使用 {conn}")
        await asyncio.sleep(1)
    finally:
        await pool.release(conn)

async def main():
    """主函数"""
    pool = ConnectionPool(size=5)
    await pool.initialize()
    
    # 10个任务共享5个连接
    await asyncio.gather(*[worker(pool, f"任务{i}") for i in range(10)])

asyncio.run(main())
```

## 10. 最佳实践

### 10.1 避免阻塞操作

```python
"""
避免阻塞操作示例
@author erik.zhou
"""
import asyncio
import time
from concurrent.futures import ThreadPoolExecutor

def blocking_operation():
    """阻塞操作"""
    time.sleep(2)
    return "完成"

async def main():
    """主函数"""
    loop = asyncio.get_event_loop()
    
    # 错误: 直接调用阻塞函数
    # result = blocking_operation()  # 会阻塞事件循环
    
    # 正确: 在线程池中运行阻塞函数
    with ThreadPoolExecutor() as executor:
        result = await loop.run_in_executor(executor, blocking_operation)
        print(result)

asyncio.run(main())
```

```python
优雅关闭示例
@author erik.zhou
"""
import asyncio
import signal

async def cleanup():
    """清理资源"""
    print("清理资源...")
    await asyncio.sleep(1)
    print("清理完成")

async def main():
    """主函数"""
    try:
        # 模拟长时间运行的任务
        await asyncio.sleep(100)
    except asyncio.CancelledError:
        await cleanup()
        raise

def shutdown(loop, signal=None):
    """关闭处理"""
    if signal:
        print(f"收到信号 {signal.name}")
    
    # 取消所有任务
    tasks = [task for task in asyncio.all_tasks(loop) if not task.done()]
    for task in tasks:
        task.cancel()
    
    # 等待所有任务完成
    loop.run_until_complete(asyncio.gather(*tasks, return_exceptions=True))
    loop.stop()
```

```python
@author erik.zhou

async def safe_task(coro, default=None):
    """安全执行任务"""
    try:
        return await coro
```
