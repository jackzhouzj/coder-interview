### 1.1 时间分析

```python
"""
时间性能分析
@author erik.zhou
"""
import time
import timeit
from functools import wraps

# 方式1：time模块
def measure_time_basic():
    """基本时间测量"""
    start = time.time()
    
    # 执行操作
    result = sum(range(1000000))
    
    end = time.time()
    print(f"执行时间: {end - start:.4f}秒")

measure_time_basic()

# 方式2：装饰器
def timer(func):
    """计时装饰器"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)
        end = time.perf_counter()
        print(f"{func.__name__} 执行时间: {end - start:.4f}秒")
        return result
    return wrapper

@timer
def slow_function():
    """慢函数"""
    time.sleep(1)
    return sum(range(1000000))

slow_function()

# 方式3：timeit模块（更精确）
def test_function():
    """测试函数"""
    return sum(range(1000))

# 测试执行时间
execution_time = timeit.timeit(test_function, number=10000)
print(f"timeit测试: {execution_time:.4f}秒")

# 比较不同实现
time1 = timeit.timeit('sum(range(1000))', number=10000)
time2 = timeit.timeit('[i for i in range(1000)]', number=10000)
print(f"sum: {time1:.4f}秒")
print(f"列表推导: {time2:.4f}秒")

print("""
时间分析工具：

1. time.time()：
   - 墙上时钟时间
   - 精度较低
   - 受系统影响

2. time.perf_counter()：
   - 高精度计时器
   - 推荐使用
   - 不受系统时间影响

3. timeit：
   - 多次执行取平均
   - 自动禁用垃圾回收
   - 适合微基准测试

最佳实践：
- 多次测量取平均
- 预热代码
- 隔离测试环境
- 使用统计分析
""")
```

### 2.1 避免不必要的计算

```python
"""
避免不必要的计算
@author erik.zhou
"""
import time

# 错误示例：重复计算
def bad_example(items):
    """低效代码"""
    result = []
    for i in range(len(items)):
        if i < len(items) - 1:  # 每次都计算len
            result.append(items[i] + items[i+1])
    return result

# 优化：缓存计算结果
def good_example(items):
    """高效代码"""
    n = len(items)  # 只计算一次
    result = []
    for i in range(n - 1):
        result.append(items[i] + items[i+1])
    return result

# 更好：使用zip
def better_example(items):
    """更高效的代码"""
    return [a + b for a, b in zip(items, items[1:])]

# 性能测试
items = list(range(10000))

start = time.perf_counter()
bad_example(items)
print(f"低效代码: {time.perf_counter() - start:.4f}秒")

start = time.perf_counter()
good_example(items)
print(f"优化代码: {time.perf_counter() - start:.4f}秒")

start = time.perf_counter()
better_example(items)
print(f"最优代码: {time.perf_counter() - start:.4f}秒")

print("""
避免不必要计算的技巧：

1. 缓存计算结果：
   - 提取循环不变量
   - 避免重复计算
   - 使用局部变量

2. 短路求值：
```

```python
def fibonacci_generator(n):
    """斐波那契生成器"""

        a, b = b, a + b
```

## 4. 算法优化

### 4.1 选择合适的算法

```python
"""
算法选择
@author erik.zhou
"""
import time

# 示例：查找重复元素
data = list(range(10000)) + list(range(5000))

# 方式1：暴力法 O(n²)
def find_duplicates_brute(lst):
    """暴力查找（慢）"""
    duplicates = []
    for i in range(len(lst)):
        for j in range(i + 1, len(lst)):
            if lst[i] == lst[j] and lst[i] not in duplicates:
                duplicates.append(lst[i])
    return duplicates

# 方式2：使用集合 O(n)
def find_duplicates_set(lst):
    """集合查找（快）"""
    seen = set()
    duplicates = set()
    for item in lst:
        if item in seen:
            duplicates.add(item)
        else:
            seen.add(item)
    return list(duplicates)

# 性能对比
start = time.perf_counter()
# result1 = find_duplicates_brute(data)  # 太慢，注释掉
# print(f"暴力法: {time.perf_counter() - start:.4f}秒")

start = time.perf_counter()
result2 = find_duplicates_set(data)
print(f"集合法: {time.perf_counter() - start:.4f}秒")

print("""
算法复杂度对比：

常见复杂度：
- O(1)：常数时间（哈希表查找）
- O(log n)：对数时间（二分查找）
- O(n)：线性时间（遍历）
- O(n log n)：线性对数（快速排序）
- O(n²)：平方时间（冒泡排序）
- O(2ⁿ)：指数时间（递归斐波那契）

优化策略：
1. 降低时间复杂度
2. 使用合适的数据结构
3. 避免重复计算
4. 空间换时间
""")
```

### 4.2 缓存和记忆化

```python
"""
缓存优化
@author erik.zhou
"""
import time
from functools import lru_cache

# 无缓存的递归
def fibonacci_no_cache(n):
    """无缓存斐波那契"""
    if n < 2:
        return n
    return fibonacci_no_cache(n-1) + fibonacci_no_cache(n-2)

# 使用lru_cache
@lru_cache(maxsize=None)
def fibonacci_cached(n):
    """缓存斐波那契"""
    if n < 2:
        return n
    return fibonacci_cached(n-1) + fibonacci_cached(n-2)

# 性能对比
start = time.perf_counter()
result1 = fibonacci_no_cache(30)
print(f"无缓存: {time.perf_counter() - start:.4f}秒, 结果: {result1}")

start = time.perf_counter()
result2 = fibonacci_cached(30)
print(f"有缓存: {time.perf_counter() - start:.4f}秒, 结果: {result2}")

# 查看缓存信息
print(f"缓存信息: {fibonacci_cached.cache_info()}")

# 手动实现缓存
def memoize(func):
    """记忆化装饰器"""
    cache = {}
    
    def wrapper(*args):
        if args not in cache:
            cache[args] = func(*args)
        return cache[args]
    
    return wrapper

@memoize
def fibonacci_manual(n):
    """手动缓存斐波那契"""
    if n < 2:
        return n
    return fibonacci_manual(n-1) + fibonacci_manual(n-2)

print("""
缓存策略：

1. lru_cache：
   - 最近最少使用缓存
   - 自动管理缓存大小
   - 线程安全
   - 适合纯函数

2. 手动缓存：
   - 更灵活
   - 可自定义策略
   - 需要手动管理

3. 使用场景：
   - 重复计算
   - 递归函数
   - 数据库查询
   - API调用

4. 注意事项：
   - 内存占用
   - 缓存失效
   - 线程安全
""")
```

## 6. Numba JIT

### 6.1 Numba基础

```python
"""
Numba JIT编译
@author erik.zhou
"""
import numpy as np
import time
from numba import jit

# 纯Python实现
def sum_array_python(arr):
    """纯Python求和"""
    total = 0.0
    for i in range(len(arr)):
        total += arr[i]
    return total

# Numba JIT编译
@jit(nopython=True)
def sum_array_numba(arr):
    """Numba加速求和"""
    total = 0.0
    for i in range(len(arr)):
        total += arr[i]
    return total

# 性能测试
arr = np.random.rand(10000000)

# Python版本
start = time.perf_counter()
result1 = sum_array_python(arr)
print(f"Python: {time.perf_counter() - start:.4f}秒")

# Numba版本（首次调用包含编译时间）
start = time.perf_counter()
result2 = sum_array_numba(arr)
print(f"Numba（首次）: {time.perf_counter() - start:.4f}秒")

# Numba版本（已编译）
```

```python
        y = np.random.random()
```

## 7. 缓存策略

### 7.1 应用级缓存

```python
"""
应用级缓存
@author erik.zhou
"""
import time
from functools import lru_cache
import hashlib

# 简单缓存
class SimpleCache:
    """简单缓存类"""
    
    def __init__(self, ttl=60):
        self.cache = {}
        self.ttl = ttl
    
    def get(self, key):
        """获取缓存"""
        if key in self.cache:
            value, timestamp = self.cache[key]
            if time.time() - timestamp < self.ttl:
                return value
            else:
                del self.cache[key]
        return None
    
    def set(self, key, value):
        """设置缓存"""
        self.cache[key] = (value, time.time())
    
    def clear(self):
        """清空缓存"""
        self.cache.clear()

# 使用示例
cache = SimpleCache(ttl=10)

def expensive_operation(x):
    """耗时操作"""
    time.sleep(1)
    return x * x

def cached_operation(x):
    """带缓存的操作"""
    key = f"op_{x}"
    result = cache.get(key)
    
    if result is None:
        print(f"计算 {x}")
        result = expensive_operation(x)
        cache.set(key, result)
    else:
        print(f"使用缓存 {x}")
    
    return result

# 测试
print(cached_operation(5))  # 计算
print(cached_operation(5))  # 使用缓存
print(cached_operation(10))  # 计算

print("""
缓存策略：

1. LRU（最近最少使用）：
   - 淘汰最久未使用
   - 适合热点数据
   - functools.lru_cache

2. LFU（最不经常使用）：
   - 淘汰使用次数最少
   - 适合访问频率差异大

3. TTL（过期时间）：
   - 设置过期时间
   - 自动失效
   - 适合时效性数据

4. 缓存穿透：
   - 缓存空值

   - 随机过期时间
```

```python
import sys

    """不使用__slots__"""
    def __init__(self, x, y):
        self.x = x
        self.y = y

class WithSlots:
    """使用__slots__"""
    __slots__ = ['x', 'y']
    
    def __init__(self, x, y):
        self.x = x
        self.y = y

# 内存对比

obj2 = WithSlots(1, 2)

    """大列表（占内存）"""
    return [i for i in range(1000000)]

    """生成器（省内存）"""
    return (i for i in range(1000000))

lst = large_list()
gen = large_generator()

print(f"列表: {sys.getsizeof(lst):,} 字节")
print(f"生成器: {sys.getsizeof(gen):,} 字节")
```
