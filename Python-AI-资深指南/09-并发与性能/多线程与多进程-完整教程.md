## 🔒 GIL 全局解释器锁

### 1. 什么是 GIL？

GIL（Global Interpreter Lock）是 CPython 解释器的一个互斥锁，它保证同一时刻只有一个线程在执行 Python 字节码。

```python
import threading
import time

# CPU 密集型任务
def cpu_bound_task(n):
    """CPU 密集型任务"""
    count = 0
    for i in range(n):
        count += i * i
    return count

# 单线程执行
start = time.time()
result1 = cpu_bound_task(10000000)
result2 = cpu_bound_task(10000000)
print(f"单线程耗时: {time.time() - start:.2f}秒")

# 多线程执行（受 GIL 影响）
start = time.time()
t1 = threading.Thread(target=cpu_bound_task, args=(10000000,))
t2 = threading.Thread(target=cpu_bound_task, args=(10000000,))
t1.start()
t2.start()
t1.join()
t2.join()
print(f"多线程耗时: {time.time() - start:.2f}秒")

# 结论：多线程可能比单线程还慢（因为 GIL 和线程切换开销）
```

### 2. GIL 的影响

```python
import threading
import time
import requests

# IO 密集型任务（多线程有效）
def io_bound_task(url):
    """IO 密集型任务"""
    response = requests.get(url)
    return len(response.content)

urls = ["https://www.python.org"] * 10

# 单线程执行
start = time.time()
for url in urls:
    io_bound_task(url)
print(f"IO密集单线程耗时: {time.time() - start:.2f}秒")

# 多线程执行（IO 密集型任务不受 GIL 影响）
start = time.time()
threads = []
for url in urls:
    t = threading.Thread(target=io_bound_task, args=(url,))
    t.start()
    threads.append(t)

for t in threads:
    t.join()
print(f"IO密集多线程耗时: {time.time() - start:.2f}秒")

# 结论：IO 密集型任务使用多线程可以显著提升性能
```

### 3. 如何绕过 GIL？

```python
# 方案1：使用多进程（推荐用于 CPU 密集型）
from multiprocessing import Process, Pool
import time

def cpu_task(n):
    count = 0
    for i in range(n):
        count += i * i
    return count

if __name__ == "__main__":
    start = time.time()
    with Pool(processes=2) as pool:
        results = pool.map(cpu_task, [10000000, 10000000])
    print(f"多进程耗时: {time.time() - start:.2f}秒")

# 方案2：使用 C 扩展（Cython、NumPy）
import numpy as np

# NumPy 操作会释放 GIL
arr = np.random.rand(10000000)
result = np.sum(arr ** 2)  # 这个操作会释放 GIL

# 方案3：使用其他 Python 实现（Jython、IronPython、PyPy）
```

## 🧵 多线程编程

### 1. 创建线程

```python
import threading
import time

# 方式1：使用 Thread 类
def worker(name, delay):
    """工作线程"""
    print(f"线程 {name} 开始")
    time.sleep(delay)
    print(f"线程 {name} 结束")

# 创建并启动线程
t1 = threading.Thread(target=worker, args=("A", 2))
t2 = threading.Thread(target=worker, args=("B", 1))

t1.start()
t2.start()

# 等待线程结束
t1.join()
t2.join()

print("所有线程执行完毕")

# 方式2：继承 Thread 类
class WorkerThread(threading.Thread):
    """工作线程类"""
    
    def __init__(self, name, delay):
        super().__init__()
        self.name = name
        self.delay = delay
    
    def run(self):
        """线程执行的方法"""
        print(f"线程 {self.name} 开始")
        time.sleep(self.delay)
        print(f"线程 {self.name} 结束")

# 创建并启动线程
t1 = WorkerThread("A", 2)
t2 = WorkerThread("B", 1)

t1.start()
t2.start()

t1.join()
t2.join()
```

### 2. 线程同步 - Lock

```python
import threading

# ❌ 错误示例 - 没有线程同步
counter = 0

def increment():
    global counter
    for _ in range(100000):
        counter += 1

threads = []
for _ in range(10):
    t = threading.Thread(target=increment)
    t.start()
    threads.append(t)

for t in threads:
    t.join()

print(f"最终计数（无锁）: {counter}")  # 结果不是 1000000

# ✅ 正确示例 - 使用 Lock
counter = 0
lock = threading.Lock()

def increment_with_lock():
    global counter
    for _ in range(100000):
        with lock:  # 使用上下文管理器
            counter += 1

threads = []
for _ in range(10):
    t = threading.Thread(target=increment_with_lock)
    t.start()
    threads.append(t)

for t in threads:
    t.join()

print(f"最终计数（有锁）: {counter}")  # 结果是 1000000
```

### 3. 线程同步 - RLock（可重入锁）

```python
import threading

# RLock 允许同一线程多次获取锁
rlock = threading.RLock()

def recursive_function(n):
    """递归函数需要多次获取锁"""
    with rlock:
        if n > 0:
            print(f"递归层级: {n}")
            recursive_function(n - 1)

t = threading.Thread(target=recursive_function, args=(5,))
t.start()
t.join()
```

### 4. 线程同步 - Semaphore（信号量）

```python
import threading
import time

# 限制同时访问资源的线程数量
semaphore = threading.Semaphore(3)  # 最多3个线程同时访问

threads = []
```

### 5. 线程通信 - Queue

```python
import threading
import queue
import time

# 生产者-消费者模式
task_queue = queue.Queue(maxsize=10)

def producer():
    """生产者"""
    for i in range(20):
        task_queue.put(f"任务-{i}")
        print(f"生产: 任务-{i}")
        time.sleep(0.1)
    
    # 发送结束信号
    task_queue.put(None)

def consumer(name):
    """消费者"""
    while True:
        task = task_queue.get()
        if task is None:
            task_queue.put(None)  # 传递结束信号给其他消费者
            break
        
        print(f"{name} 消费: {task}")
        time.sleep(0.2)
        task_queue.task_done()

# 启动生产者
producer_thread = threading.Thread(target=producer)
producer_thread.start()

# 启动多个消费者
consumer_threads = []
for i in range(3):
    t = threading.Thread(target=consumer, args=(f"消费者-{i}",))
    t.start()
    consumer_threads.append(t)

# 等待完成
producer_thread.join()
for t in consumer_threads:
    t.join()

print("所有任务处理完毕")
```

### 1. 创建进程

```python
from multiprocessing import Process
import os
import time

def worker(name):
    """工作进程"""
    print(f"进程 {name} 开始, PID: {os.getpid()}")
    time.sleep(2)
    print(f"进程 {name} 结束")

if __name__ == "__main__":
    print(f"主进程 PID: {os.getpid()}")
    
    # 创建进程
    p1 = Process(target=worker, args=("A",))

    # 启动进程
    p1.start()
    p2.start()
    
    # 等待进程结束
    p1.join()
    p2.join()
    
    print("所有进程执行完毕")
```

### 2. 进程通信 - Queue

```python
from multiprocessing import Process, Queue
import time

def producer(queue):
    """生产者进程"""
    for i in range(10):
        queue.put(f"数据-{i}")
        print(f"生产: 数据-{i}")
        time.sleep(0.1)

def consumer(queue):
    """消费者进程"""
    while True:
        if not queue.empty():
            data = queue.get()
            print(f"消费: {data}")
            time.sleep(0.2)
        else:
            break

if __name__ == "__main__":
    # 创建队列
    queue = Queue()
    
    # 创建进程
    p1 = Process(target=producer, args=(queue,))
    p2 = Process(target=consumer, args=(queue,))
    
    # 启动进程
    p1.start()
    p1.join()  # 等待生产者完成
    
    p2.start()
    p2.join()
    
    print("进程通信完成")
```

### 3. 进程通信 - Pipe

```python
from multiprocessing import Process, Pipe

def sender(conn):
    """发送数据"""
    conn.send(["消息1", "消息2", "消息3"])
    conn.close()

def receiver(conn):
    """接收数据"""
    messages = conn.recv()
    print(f"接收到: {messages}")
    conn.close()

if __name__ == "__main__":
    # 创建管道
    parent_conn, child_conn = Pipe()
    
    # 创建进程
    p1 = Process(target=sender, args=(parent_conn,))
    p2 = Process(target=receiver, args=(child_conn,))
    
    # 启动进程
    p1.start()
    p2.start()
    
    p1.join()
    p2.join()
```

### 4. 共享内存

```python
from multiprocessing import Process, Value, Array
import time

def increment_value(shared_value, shared_array):
    """修改共享内存"""
    for i in range(100):
        with shared_value.get_lock():
            shared_value.value += 1
        
        with shared_array.get_lock():
            shared_array[0] += 1

if __name__ == "__main__":
    # 创建共享变量
    shared_value = Value('i', 0)  # 'i' 表示整数
    shared_array = Array('i', [0, 0, 0])  # 整数数组
    
    # 创建进程
    processes = []
    for _ in range(10):
        p = Process(target=increment_value, args=(shared_value, shared_array))
        p.start()
        processes.append(p)
    
    # 等待所有进程完成
    for p in processes:
        p.join()
    
    print(f"共享变量值: {shared_value.value}")
    print(f"共享数组: {list(shared_array)}")
```

## 🏊 线程池和进程池

### 1. ThreadPoolExecutor

```python
from concurrent.futures import ThreadPoolExecutor, as_completed
import time
import requests

def fetch_url(url):
    """获取 URL 内容"""
    response = requests.get(url)
    return url, len(response.content)

urls = [
    "https://www.python.org",
    "https://www.github.com",
    "https://www.stackoverflow.com",
] * 3

# 使用线程池
with ThreadPoolExecutor(max_workers=5) as executor:
    # 方式1：submit + as_completed
    futures = [executor.submit(fetch_url, url) for url in urls]
    
    for future in as_completed(futures):
        url, size = future.result()
        print(f"{url}: {size} bytes")

# 方式2：map
with ThreadPoolExecutor(max_workers=5) as executor:
    results = executor.map(fetch_url, urls)
    for url, size in results:
        print(f"{url}: {size} bytes")
```

### 2. ProcessPoolExecutor

### 3. 线程池最佳实践

```python
from concurrent.futures import ThreadPoolExecutor, as_completed
import logging

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

def process_item(item):
    """处理单个项目"""
    try:
        # 处理逻辑
        result = expensive_operation(item)
        return {"success": True, "item": item, "result": result}
    except Exception as e:
        logger.error(f"处理 {item} 失败: {e}", exc_info=True)
        return {"success": False, "item": item, "error": str(e)}

def batch_process(items, max_workers=10):
    """批量处理"""
    results = []
    
    with ThreadPoolExecutor(max_workers=max_workers) as executor:
        # 提交所有任务
        future_to_item = {
            executor.submit(process_item, item): item 
            for item in items
        }
        
        # 处理完成的任务
        for future in as_completed(future_to_item):
            item = future_to_item[future]
            try:
                result = future.result(timeout=30)  # 设置超时
                results.append(result)
            except Exception as e:
                logger.error(f"获取 {item} 结果失败: {e}")
                results.append({
                    "success": False,
                    "item": item,
                    "error": str(e)
                })
    
    return results
```

### 2. 避免死锁

```python
import threading

# ❌ 可能死锁
lock1 = threading.Lock()
lock2 = threading.Lock()

def thread1():
    with lock1:
        time.sleep(0.1)
        with lock2:  # 等待 lock2
            pass

def thread2():
    with lock2:
        time.sleep(0.1)
        with lock1:  # 等待 lock1，死锁！
            pass

# ✅ 避免死锁 - 按固定顺序获取锁
def thread1_safe():
    with lock1:
        with lock2:
            pass

def thread2_safe():
    with lock1:  # 相同顺序
        with lock2:
            pass
```

## ❓ 常见问题

### Q1: 什么时候使用多线程，什么时候使用多进程？
A: 
- IO 密集型（网络请求、文件读写）：使用多线程或异步
- CPU 密集型（计算、数据处理）：使用多进程
- 需要共享大量数据：使用多线程（共享内存）
- 需要隔离性和稳定性：使用多进程

### Q2: 如何确定线程池/进程池的大小？
A:
- IO 密集型：`线程数 = CPU核心数 * (1 + IO等待时间/CPU计算时间)`
- CPU 密集型：`进程数 = CPU核心数`
- 实际应用中需要通过测试确定最优值

### Q3: 如何处理线程/进程中的异常？
A: 使用 try-except 捕获异常并记录日志：

```python
def safe_worker(item):
    try:
        return process(item)
    except Exception as e:
        logger.error(f"处理失败: {e}", exc_info=True)
        return None
```
