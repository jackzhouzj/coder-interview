### 2.2 循环引用问题

```python
循环引用问题
@author erik.zhou

import sys
import gc

# 创建循环引用
class Node:
    """节点类"""
    def __init__(self, value):
        self.value = value
        self.next = None

        print(f"删除节点: {self.value}")

# 示例1：简单循环引用

node2 = Node(2)
node1.next = node2
node2.next = node1  # 循环引用

# 删除引用
del node1

2. 弱引用：
   - 不增加引用计数

3. 手动打破循环：
```

```python
print("""
分代回收原理：

1. 三代机制：
   - 第0代：新创建的对象
   - 第1代：经历过一次回收
   - 第2代：经历过多次回收

2. 回收触发：
   - 第0代：分配次数 - 释放次数 > 阈值0
   - 第1代：第0代回收次数 > 阈值1
   - 第2代：第1代回收次数 > 阈值2

3. 默认阈值：
   - (700, 10, 10)
   - 700：第0代阈值
   - 10：第1代阈值
   - 10：第2代阈值

4. 弱代假说：
   - 大多数对象生命周期短
   - 存活时间越长，越可能继续存活
   - 分代回收提高效率

5. 回收算法：
   - 标记-清除
   - 检测循环引用
   - 清理不可达对象
""")
```

## 4. 内存泄漏诊断

### 4.1 常见内存泄漏场景

```python
"""
常见内存泄漏场景
@author erik.zhou
"""
import sys

print("""
常见内存泄漏场景：

1. 全局变量累积：
""")

# ❌ 错误：全局列表不断增长
global_cache = []

def add_to_cache(data):
    global_cache.append(data)  # 永不释放

# ✅ 正确：使用LRU缓存
from functools import lru_cache

@lru_cache(maxsize=100)
def cached_function(x):
    return x * x

print("""
2. 循环引用：
""")

# ❌ 错误：循环引用
class Parent:
    def __init__(self):
        self.child = None

class Child:
    def __init__(self, parent):
        self.parent = parent  # 强引用

parent = Parent()
child = Child(parent)
parent.child = child  # 循环引用

# ✅ 正确：使用弱引用
import weakref

class ChildFixed:
    def __init__(self, parent):
        self.parent = weakref.ref(parent)  # 弱引用

print("""
3. 未关闭的资源：
""")

# ❌ 错误：文件未关闭
def read_file_bad(filename):
    f = open(filename)
    data = f.read()
    # 忘记关闭
    return data

# ✅ 正确：使用上下文管理器
def read_file_good(filename):
    with open(filename) as f:
        return f.read()

print("""
4. 事件监听器未移除：
""")

# ❌ 错误：监听器未移除
class EventEmitter:
    def __init__(self):
        self.listeners = []
    
    def on(self, callback):
        self.listeners.append(callback)  # 永不移除

# ✅ 正确：提供移除方法
class EventEmitterFixed:
    def __init__(self):
        self.listeners = []
    
    def on(self, callback):
        self.listeners.append(callback)
    
    def off(self, callback):
        self.listeners.remove(callback)

print("""
5. 大对象缓存：
""")

# ❌ 错误：无限制缓存
cache = {}

def get_data(key):
    if key not in cache:
        cache[key] = load_large_data(key)  # 永不清理
    return cache[key]

# ✅ 正确：使用LRU缓存
from functools import lru_cache

@lru_cache(maxsize=100)
def get_data_fixed(key):
    return load_large_data(key)

def load_large_data(key):
    """模拟加载大数据"""
```

## 5. 内存优化技巧

### 5.1 使用__slots__

```python
"""
__slots__优化
@author erik.zhou
"""
import sys

# 不使用__slots__
class RegularClass:
    """常规类"""
    def __init__(self, x, y, z):
        self.x = x
        self.y = y
        self.z = z

# 使用__slots__
class SlottedClass:
    """使用__slots__的类"""
    __slots__ = ['x', 'y', 'z']
    
    def __init__(self, x, y, z):
        self.x = x
        self.y = y
        self.z = z

# 创建大量对象
n = 100000

# 常规类
regular_objects = [RegularClass(i, i+1, i+2) for i in range(n)]
regular_size = sum(sys.getsizeof(obj) for obj in regular_objects)

# __slots__类
slotted_objects = [SlottedClass(i, i+1, i+2) for i in range(n)]
slotted_size = sum(sys.getsizeof(obj) for obj in slotted_objects)

print(f"常规类总大小: {regular_size:,} 字节")
print(f"__slots__类总大小: {slotted_size:,} 字节")
print(f"节省: {(1 - slotted_size/regular_size)*100:.1f}%")

print("""
__slots__优化：

1. 原理：
   - 固定属性
   - 无__dict__
   - 紧凑存储

2. 优势：
   - 减少内存占用（40-50%）
   - 提高属性访问速度
   - 防止动态添加属性

3. 限制：
   - 不能动态添加属性
   - 不能使用__dict__
   - 继承需要注意

4. 使用场景：
   - 大量小对象
   - 固定属性
   - 性能敏感

5. 注意事项：
   - 子类需要重新定义__slots__
   - 不能使用弱引用（除非添加__weakref__）
   - 不能使用默认值
""")
```


### 5.2 生成器和迭代器

```python
"""
生成器优化内存
@author erik.zhou
"""
import sys

# 列表方式（占用内存）
def process_with_list(n):
    """使用列表"""
    data = [i * i for i in range(n)]
    return sum(data)

# 生成器方式（节省内存）
def process_with_generator(n):
    """使用生成器"""
    data = (i * i for i in range(n))
    return sum(data)

n = 1000000

# 内存对比
list_data = [i * i for i in range(n)]
gen_data = (i * i for i in range(n))

print(f"列表大小: {sys.getsizeof(list_data):,} 字节")
print(f"生成器大小: {sys.getsizeof(gen_data):,} 字节")

# 自定义生成器
def read_large_file(filename):
    """逐行读取大文件"""
    with open(filename) as f:
        for line in f:
            yield line.strip()

# 生成器管道
def process_pipeline(filename):
    """生成器管道"""
    lines = read_large_file(filename)
    filtered = (line for line in lines if line)
    processed = (line.upper() for line in filtered)
    return processed

print("""
生成器优化：

1. 优势：
   - 惰性求值
   - 按需生成
   - 内存效率高
   - 支持无限序列

2. 使用场景：
   - 大文件处理
   - 数据流处理
   - 管道操作
   - 无限序列

3. 生成器表达式：
   gen = (x*x for x in range(n))

4. 生成器函数：
   def generator():
       for i in range(n):
           yield i

5. 注意事项：
   - 只能迭代一次
   - 不支持索引
   - 不支持len()
   - 不能回退
""")
```

### 5.3 数据结构选择

```python
"""
数据结构内存优化
@author erik.zhou

import sys
from array import array
from collections import deque

# 列表 vs array
int_list = [i for i in range(1000)]
int_array = array('i', range(1000))

print(f"整数列表: {sys.getsizeof(int_list):,} 字节")
print(f"整数数组: {sys.getsizeof(int_array):,} 字节")

lst = list(range(1000))
dq = deque(range(1000))

print(f"列表: {sys.getsizeof(lst):,} 字节")
print(f"deque: {sys.getsizeof(dq):,} 字节")

# 字典 vs 命名元组
from collections import namedtuple

# 字典
dict_data = [{'x': i, 'y': i+1, 'z': i+2} for i in range(1000)]
dict_size = sum(sys.getsizeof(d) for d in dict_data)

# 命名元组
Point = namedtuple('Point', ['x', 'y', 'z'])
tuple_data = [Point(i, i+1, i+2) for i in range(1000)]
tuple_size = sum(sys.getsizeof(t) for t in tuple_data)

print(f"字典总大小: {dict_size:,} 字节")
print(f"命名元组总大小: {tuple_size:,} 字节")

print("""
数据结构选择：

1. array模块：
   - 紧凑存储
   - 类型统一
   - 适合数值数组
   - 比列表省内存

2. deque：
   - 双端队列
   - 两端操作O(1)
   - 适合队列/栈

3. 命名元组：
   - 不可变
   - 比字典省内存
   - 支持属性访问

4. set vs list：
   - 查找：set O(1), list O(n)
   - 去重：使用set
   - 顺序：使用list

5. 选择原则：
   - 根据操作特点

   - 权衡性能
""")
```

```python
对象池模式
@author erik.zhou
"""
from queue import Queue

class ObjectPool:
    """对象池"""
    
    def __init__(self, factory, max_size=10):
        """
        初始化对象池
        
        Args:
            factory: 对象工厂函数
            max_size: 最大对象数
        """
        self.factory = factory
        self.max_size = max_size
        self.pool = Queue(maxsize=max_size)
        self.size = 0
    
    def acquire(self):
        """获取对象"""
        if not self.pool.empty():
            return self.pool.get()
        elif self.size < self.max_size:
            self.size += 1
            return self.factory()
        else:
            # 等待可用对象
            return self.pool.get()
    
    def release(self, obj):
        """释放对象"""
        self.pool.put(obj)
    
    def __enter__(self):
        """上下文管理器"""
        return self.acquire()
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        """上下文管理器"""
        # 注意：这里需要保存对象引用
        pass

# 使用示例
class ExpensiveObject:
    """昂贵对象"""
    def __init__(self):
        print("创建昂贵对象")
        self.data = [0] * 1000000
    
    def reset(self):
        """重置对象"""
        self.data = [0] * 1000000

# 创建对象池
pool = ObjectPool(ExpensiveObject, max_size=5)

# 使用对象
obj1 = pool.acquire()
print("使用对象1")
pool.release(obj1)

obj2 = pool.acquire()  # 复用obj1
```

```python
"""
数据库连接池
@author erik.zhou
"""
from queue import Queue, Empty
import time

class Connection:
    """模拟数据库连接"""
    def __init__(self, conn_id):
        self.conn_id = conn_id
        print(f"创建连接 {conn_id}")
    
    def execute(self, query):
        """执行查询"""
        print(f"连接 {self.conn_id} 执行: {query}")
        time.sleep(0.1)
    
    def close(self):
        """关闭连接"""
        print(f"关闭连接 {self.conn_id}")

class ConnectionPool:
    """连接池"""
    
    def __init__(self, max_connections=5):
        self.max_connections = max_connections
        self.pool = Queue(maxsize=max_connections)
        self.current_connections = 0
    
    def get_connection(self, timeout=5):
        """获取连接"""
        try:
            # 尝试从池中获取
            return self.pool.get(timeout=timeout)
        except Empty:
            # 池为空，创建新连接
            if self.current_connections < self.max_connections:
                self.current_connections += 1
                return Connection(self.current_connections)
            else:
                raise Exception("连接池已满")
    
    def return_connection(self, conn):
        """归还连接"""
        self.pool.put(conn)
    
    def close_all(self):
        """关闭所有连接"""
        while not self.pool.empty():
            conn = self.pool.get()
            conn.close()

# 使用示例
pool = ConnectionPool(max_connections=3)

# 获取连接
conn1 = pool.get_connection()
conn1.execute("SELECT * FROM users")
pool.return_connection(conn1)

# 复用连接
conn2 = pool.get_connection()
conn2.execute("SELECT * FROM orders")
pool.return_connection(conn2)

# 清理
pool.close_all()
```

```python
"""
弱引用

import weakref
import sys

class MyObject:
    """示例对象"""
    def __init__(self, name):
        self.name = name
    
    def __del__(self):
        print(f"删除对象: {self.name}")

# 强引用
print("=== 强引用 ===")
obj = MyObject("强引用对象")
print(f"引用计数: {sys.getrefcount(obj) - 1}")

ref = obj  # 强引用
print(f"引用计数: {sys.getrefcount(obj) - 1}")

del ref
print(f"引用计数: {sys.getrefcount(obj) - 1}")

# 弱引用
print("\n=== 弱引用 ===")
obj2 = MyObject("弱引用对象")
print(f"引用计数: {sys.getrefcount(obj2) - 1}")

weak_ref = weakref.ref(obj2)  # 弱引用
print(f"引用计数: {sys.getrefcount(obj2) - 1}")  # 不增加

# 访问弱引用对象
if weak_ref() is not None:
    print(f"对象存在: {weak_ref().name}")

# 删除强引用
del obj2
print("删除强引用后...")

# 弱引用失效
if weak_ref() is None:
    print("弱引用已失效")

print("""
弱引用特点：

1. 不增加引用计数
2. 对象可能被回收
3. 访问前需要检查
4. 避免循环引用

使用场景：
- 缓存
- 观察者模式
- 父子关系
- 回调函数
""")
```

### 7.2 弱引用应用

```python
"""
弱引用应用
@author erik.zhou
"""
import weakref

# 1. 弱引用字典
class Cache:
    """使用弱引用的缓存"""
    def __init__(self):
        self.cache = weakref.WeakValueDictionary()
    
    def get(self, key):
        """获取缓存"""
        return self.cache.get(key)
    
    def set(self, key, value):
        """设置缓存"""
        self.cache[key] = value

# 2. 观察者模式
class Subject:
    """主题（使用弱引用）"""
    def __init__(self):
        self.observers = []
    
    def attach(self, observer):
        """添加观察者"""
        self.observers.append(weakref.ref(observer))
    
    def notify(self, message):
        """通知观察者"""
        # 清理失效的弱引用
        self.observers = [obs for obs in self.observers if obs() is not None]
        
        for obs_ref in self.observers:
            obs = obs_ref()
            if obs is not None:
                obs.update(message)

class Observer:
    """观察者"""
    def __init__(self, name):
        self.name = name
    
    def update(self, message):
        """更新"""
        print(f"{self.name} 收到: {message}")

# 使用示例
subject = Subject()

obs1 = Observer("观察者1")
obs2 = Observer("观察者2")

subject.attach(obs1)
subject.attach(obs2)

subject.notify("消息1")

# 删除观察者
del obs1
subject.notify("消息2")  # 只有obs2收到

print("""
弱引用类型：

1. weakref.ref：
   - 基本弱引用
   - 需要调用获取对象

2. WeakValueDictionary：
   - 值为弱引用的字典
   - 自动清理失效项

3. WeakKeyDictionary：
   - 键为弱引用的字典
   - 键被回收时自动删除

4. WeakSet：
   - 弱引用集合
   - 自动清理

5. WeakMethod：
   - 方法的弱引用
   - 避免循环引用

最佳实践：
- 缓存使用WeakValueDictionary
- 观察者使用弱引用
- 回调使用WeakMethod
- 定期清理失效引用
""")
```

```python
"""
生成器管道处理大数据
@author erik.zhou

def read_lines(filename):
    """读取文件行"""
    with open(filename) as f:
        for line in f:
            yield line.strip()

def filter_lines(lines, pattern):
    """过滤行"""
    for line in lines:
        if pattern in line:
            yield line

def transform_lines(lines):
    """转换行"""
    for line in lines:
        yield line.upper()

def process_pipeline(filename, pattern):
    """处理管道"""
    lines = read_lines(filename)
    filtered = filter_lines(lines, pattern)
    transformed = transform_lines(filtered)
    
    for line in transformed:
        print(line)
```
