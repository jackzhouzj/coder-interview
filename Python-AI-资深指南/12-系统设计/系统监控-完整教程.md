## APM 应用性能监控

### 使用 Prometheus

```python
from prometheus_client import Counter, Histogram, Gauge, start_http_server
import time
import random

# 定义指标
request_count = Counter('http_requests_total', '总请求数', ['method', 'endpoint'])
request_duration = Histogram('http_request_duration_seconds', '请求耗时')
active_users = Gauge('active_users', '活跃用户数')

def process_request(method: str, endpoint: str):
    """处理请求"""
    # 增加请求计数
    request_count.labels(method=method, endpoint=endpoint).inc()
    
    # 记录请求耗时
    with request_duration.time():
        time.sleep(random.uniform(0.1, 0.5))  # 模拟处理时间
    
    return "OK"

# 启动指标服务器
# start_http_server(8000)

# 模拟请求
for _ in range(10):
    process_request("GET", "/api/users")
    process_request("POST", "/api/orders")

# 更新活跃用户数
active_users.set(random.randint(100, 200))
```

## 链路追踪

### OpenTelemetry 示例

```python
from opentelemetry import trace
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import ConsoleSpanExporter, SimpleSpanProcessor

# 配置追踪
trace.set_tracer_provider(TracerProvider())
tracer = trace.get_tracer(__name__)

# 添加导出器
span_processor = SimpleSpanProcessor(ConsoleSpanExporter())
trace.get_tracer_provider().add_span_processor(span_processor)

def get_user(user_id: str):
    """获取用户"""
    with tracer.start_as_current_span("get_user") as span:
        span.set_attribute("user.id", user_id)
        time.sleep(0.1)  # 模拟数据库查询
        return {"id": user_id, "name": "张三"}

def get_orders(user_id: str):
    """获取订单"""
    with tracer.start_as_current_span("get_orders") as span:
        span.set_attribute("user.id", user_id)
        time.sleep(0.2)  # 模拟数据库查询
        return [{"id": "ORD001", "amount": 299.0}]

def get_user_profile(user_id: str):
    """获取用户资料（包含订单）"""
    with tracer.start_as_current_span("get_user_profile") as span:
        user = get_user(user_id)
        orders = get_orders(user_id)
        return {"user": user, "orders": orders}

# 使用示例
# profile = get_user_profile("USER001")
```

## 指标监控

### 自定义指标收集

```python
import time
from dataclasses import dataclass
from typing import Dict, List
from datetime import datetime

@dataclass
class Metric:
    """指标"""
    name: str
    value: float
    timestamp: datetime
    labels: Dict[str, str]

class MetricsCollector:
    """指标收集器"""
    
    def __init__(self):
        self.metrics: List[Metric] = []
    
    def record(self, name: str, value: float, labels: Dict[str, str] = None):
        """记录指标"""
        metric = Metric(
            name=name,
            value=value,
            timestamp=datetime.now(),
            labels=labels or {}
        )
        self.metrics.append(metric)
    
    def get_metrics(self, name: str = None) -> List[Metric]:
        """获取指标"""
        if name:
            return [m for m in self.metrics if m.name == name]
        return self.metrics

# 使用示例
collector = MetricsCollector()

# 记录业务指标
collector.record("order_amount", 299.0, {"status": "completed"})
collector.record("user_login", 1, {"platform": "web"})
collector.record("api_latency", 0.15, {"endpoint": "/api/users"})

# 查询指标
for metric in collector.get_metrics("order_amount"):
    print(f"{metric.name}: {metric.value} at {metric.timestamp}")
```

## 日志聚合

### 结构化日志

```python
import logging
import json
from datetime import datetime

class JSONFormatter(logging.Formatter):
    """JSON 格式化器"""
    
    def format(self, record):
        log_data = {
            "timestamp": datetime.now().isoformat(),
            "level": record.levelname,
            "logger": record.name,
            "message": record.getMessage(),
            "module": record.module,
            "function": record.funcName,
            "line": record.lineno
        }
        
        if record.exc_info:
            log_data["exception"] = self.formatException(record.exc_info)
        
        return json.dumps(log_data, ensure_ascii=False)

# 配置日志
logger = logging.getLogger("app")
logger.setLevel(logging.INFO)

handler = logging.StreamHandler()
handler.setFormatter(JSONFormatter())
logger.addHandler(handler)

# 使用示例
logger.info("用户登录", extra={"user_id": "USER001"})
logger.error("订单创建失败", extra={"order_id": "ORD001", "error": "库存不足"})
```

## 告警系统

### 简单告警实现

```python
from abc import ABC, abstractmethod
from typing import List

class Alert:
    """告警"""
    
    def __init__(self, level: str, message: str, labels: dict = None):
        self.level = level
        self.message = message
        self.labels = labels or {}
        self.timestamp = datetime.now()

class AlertChannel(ABC):
    """告警渠道接口"""
    
    @abstractmethod
    def send(self, alert: Alert):
        pass

class EmailChannel(AlertChannel):
    """邮件告警渠道"""
    
    def send(self, alert: Alert):
        print(f"[EMAIL] {alert.level}: {alert.message}")

class SMSChannel(AlertChannel):
    """短信告警渠道"""
    
    def send(self, alert: Alert):
        print(f"[SMS] {alert.level}: {alert.message}")

class AlertManager:
    """告警管理器"""
    
    def __init__(self):
        self.channels: List[AlertChannel] = []
        self.rules = []
    
    def add_channel(self, channel: AlertChannel):
        """添加告警渠道"""
        self.channels.append(channel)
    
    def send_alert(self, alert: Alert):
        """发送告警"""
        for channel in self.channels:
            channel.send(alert)

# 使用示例
manager = AlertManager()
manager.add_channel(EmailChannel())
manager.add_channel(SMSChannel())

# 发送告警
alert = Alert("CRITICAL", "CPU 使用率超过 90%", {"host": "server1"})
manager.send_alert(alert)
```
