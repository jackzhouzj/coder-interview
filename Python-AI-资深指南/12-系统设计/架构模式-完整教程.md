## 架构模式概述

### 什么是架构模式

架构模式是软件系统的高层结构设计方案，定义了系统的主要组件及其交互方式。

### 架构模式分类

| 类型 | 特点 | 适用场景 |
|------|------|---------|
| 分层架构 | 层次清晰，职责分离 | 传统 Web 应用 |
| MVC/MVT | 视图与逻辑分离 | Web 框架 |
| 微服务 | 服务独立，松耦合 | 大型分布式系统 |
| 事件驱动 | 异步通信，高吞吐 | 实时系统 |
| DDD | 业务驱动，领域模型 | 复杂业务系统 |

### 架构选择原则

```python
"""
架构选择考虑因素：

1. 业务复杂度
   - 简单业务：单体架构
   - 复杂业务：DDD、微服务

2. 团队规模
   - 小团队：单体架构
   - 大团队：微服务

3. 性能要求
   - 高并发：事件驱动、微服务
   - 低延迟：单体架构

4. 可维护性
   - 长期维护：分层架构、DDD
   - 快速迭代：敏捷架构

5. 技术栈
   - Python：Django(MVT)、FastAPI(微服务)
   - 混合：微服务架构
"""
```

## MVC/MVT 架构

### MVC 模式

Model-View-Controller，将应用分为三个核心组件。

```python
# Model：数据模型
class User:
    """用户模型"""
    
    def __init__(self, user_id: int, name: str, email: str):
        self.user_id = user_id
        self.name = name
        self.email = email
    
    @classmethod
    def find_by_id(cls, user_id: int):
        """从数据库查找用户"""
        # 模拟数据库查询
        return cls(user_id, "John Doe", "john@example.com")
    
    def save(self):
        """保存用户"""
        print(f"保存用户: {self.name}")

# View：视图
class UserView:
    """用户视图"""
    
    @staticmethod
    def render_user(user: User):
        """渲染用户信息"""
        return f"""
        <div class="user">
            <h2>{user.name}</h2>
            <p>Email: {user.email}</p>
        </div>
        """
    
    @staticmethod
    def render_error(message: str):
        """渲染错误信息"""
        return f'<div class="error">{message}</div>'

# Controller：控制器
class UserController:
    """用户控制器"""
    
    def __init__(self):
        self.view = UserView()
    
    def get_user(self, user_id: int):
        """获取用户"""
        try:
            user = User.find_by_id(user_id)
            return self.view.render_user(user)
        except Exception as e:
            return self.view.render_error(str(e))
    
    def update_user(self, user_id: int, name: str, email: str):
        """更新用户"""
        try:
            user = User.find_by_id(user_id)
            user.name = name
            user.email = email
            user.save()
            return self.view.render_user(user)
        except Exception as e:
            return self.view.render_error(str(e))

# 使用示例
print("=== MVC 模式 ===")
controller = UserController()
print(controller.get_user(1))
print(controller.update_user(1, "Jane Doe", "jane@example.com"))
```

```python
from django.db import models

class Article(models.Model):
    """文章模型"""
    title = models.CharField(max_length=200)
    content = models.TextField()
    author = models.ForeignKey('User', on_delete=models.CASCADE)
    created_at = models.DateTimeField(auto_now_add=True)
    
    class Meta:
        db_table = 'articles'
        ordering = ['-created_at']

        return self.title

# views.py
from django.shortcuts import render, get_object_or_404
from django.http import JsonResponse
from .models import Article

def article_list(request):
    """文章列表视图"""
    articles = Article.objects.all()
    return render(request, 'articles/list.html', {
        'articles': articles
    })

def article_detail(request, article_id):
    """文章详情视图"""
    article = get_object_or_404(Article, pk=article_id)
    return render(request, 'articles/detail.html', {
        'article': article
    })

def article_create(request):
    """创建文章视图"""
    if request.method == 'POST':
        title = request.POST.get('title')
        content = request.POST.get('content')
        
        article = Article.objects.create(
            title=title,
            content=content,
            author=request.user
        )
        
        return JsonResponse({
            'id': article.id,
            'title': article.title
        })
    
    return render(request, 'articles/create.html')

# urls.py
```

## 领域驱动设计（DDD）

### DDD 核心概念

```python
from abc import ABC, abstractmethod
from typing import List, Optional
from dataclasses import dataclass
from datetime import datetime

# 值对象（Value Object）
@dataclass(frozen=True)
class Money:
    """金额值对象"""
    amount: float
    currency: str = "CNY"
    
    def add(self, other: 'Money') -> 'Money':
        if self.currency != other.currency:
            raise ValueError("货币类型不匹配")
        return Money(self.amount + other.amount, self.currency)
    
    def __str__(self):
        return f"{self.amount} {self.currency}"

@dataclass(frozen=True)
class Address:
    """地址值对象"""
    province: str
    city: str
    district: str
    detail: str
    
    def full_address(self) -> str:
        return f"{self.province}{self.city}{self.district}{self.detail}"

# 实体（Entity）
class Order:
    """订单实体"""
    
    def __init__(self, order_id: str, customer_id: str):
        self.order_id = order_id
        self.customer_id = customer_id
        self.items: List['OrderItem'] = []
        self.status = "pending"
        self.created_at = datetime.now()
    
    def add_item(self, product_id: str, quantity: int, price: Money):
        """添加订单项"""
        item = OrderItem(product_id, quantity, price)
        self.items.append(item)
    
    def total_amount(self) -> Money:
        """计算总金额"""
        total = Money(0)
        for item in self.items:
            total = total.add(item.subtotal())
        return total
    
    def confirm(self):
        """确认订单"""
        if self.status != "pending":
            raise ValueError("订单状态不正确")
        self.status = "confirmed"
    
    def __eq__(self, other):
        if not isinstance(other, Order):
            return False
        return self.order_id == other.order_id

class OrderItem:
    """订单项"""
    
    def __init__(self, product_id: str, quantity: int, price: Money):
        self.product_id = product_id
        self.quantity = quantity
        self.price = price
    
    def subtotal(self) -> Money:
        """计算小计"""
        return Money(self.price.amount * self.quantity, self.price.currency)

# 聚合根（Aggregate Root）
class Customer:
    """客户聚合根"""
    
    def __init__(self, customer_id: str, name: str, email: str):
        self.customer_id = customer_id
        self.name = name
        self.email = email
        self.addresses: List[Address] = []
        self.orders: List[Order] = []
    
    def add_address(self, address: Address):
        """添加地址"""
        self.addresses.append(address)
    
    def create_order(self) -> Order:
        """创建订单"""
        order = Order(f"ORD{len(self.orders) + 1}", self.customer_id)
        self.orders.append(order)
        return order

# 仓储接口（Repository）
class OrderRepository(ABC):
    """订单仓储接口"""
    
    @abstractmethod
    def save(self, order: Order):
        pass
    
    @abstractmethod
    def find_by_id(self, order_id: str) -> Optional[Order]:
        pass
    
    @abstractmethod
    def find_by_customer(self, customer_id: str) -> List[Order]:
        pass

# 仓储实现
class InMemoryOrderRepository(OrderRepository):
    """内存订单仓储"""
    
    def __init__(self):
        self._orders: dict[str, Order] = {}
    
    def save(self, order: Order):
        self._orders[order.order_id] = order
    
    def find_by_id(self, order_id: str) -> Optional[Order]:
        return self._orders.get(order_id)
    
    def find_by_customer(self, customer_id: str) -> List[Order]:
        return [
            order for order in self._orders.values()
            if order.customer_id == customer_id
        ]

# 领域服务（Domain Service）
class OrderService:
    """订单领域服务"""
    
    def __init__(self, order_repo: OrderRepository):
        self.order_repo = order_repo
    
    def place_order(self, customer: Customer, items: List[tuple]) -> Order:
        """下单"""
        order = customer.create_order()
        
        for product_id, quantity, price in items:
            order.add_item(product_id, quantity, price)
        
        order.confirm()
        self.order_repo.save(order)
        
        return order

# 使用示例
print("=== DDD 示例 ===")

# 创建客户
customer = Customer("CUST001", "张三", "zhangsan@example.com")
customer.add_address(Address("北京市", "朝阳区", "望京", "SOHO 1号楼"))

# 创建订单
repo = InMemoryOrderRepository()
service = OrderService(repo)

items = [
    ("PROD001", 2, Money(99.9)),
    ("PROD002", 1, Money(199.0))
]

order = service.place_order(customer, items)
print(f"订单ID: {order.order_id}")
print(f"订单状态: {order.status}")
print(f"订单总额: {order.total_amount()}")
```

## CQRS 和事件溯源

### CQRS（命令查询职责分离）

```python
from abc import ABC, abstractmethod
from typing import List, Dict
from dataclasses import dataclass
from datetime import datetime

# 命令（Command）
@dataclass
class CreateUserCommand:
    """创建用户命令"""
    name: str
    email: str

@dataclass
class UpdateUserCommand:
    """更新用户命令"""
    user_id: str
    name: str
    email: str

# 查询（Query）
@dataclass
class GetUserQuery:
    """获取用户查询"""
    user_id: str

@dataclass
class ListUsersQuery:
    """列出用户查询"""
    page: int = 1
    size: int = 10

# 命令处理器
class CommandHandler(ABC):
    @abstractmethod
    def handle(self, command):
        pass

class CreateUserHandler(CommandHandler):
    """创建用户处理器"""
    
    def __init__(self, write_db):
        self.write_db = write_db
    
    def handle(self, command: CreateUserCommand):
        user_id = f"USER{len(self.write_db) + 1}"
        self.write_db[user_id] = {
            'id': user_id,
            'name': command.name,
            'email': command.email,
            'created_at': datetime.now()
        }
        print(f"用户已创建: {user_id}")
        return user_id

# 查询处理器
class QueryHandler(ABC):
    @abstractmethod
    def handle(self, query):
        pass

class GetUserHandler(QueryHandler):
    """获取用户处理器"""
    
    def __init__(self, read_db):
        self.read_db = read_db
    
    def handle(self, query: GetUserQuery):
        return self.read_db.get(query.user_id)

# CQRS 总线
class CQRSBus:
    """CQRS 总线"""
    
    def __init__(self):
        self.command_handlers: Dict = {}
        self.query_handlers: Dict = {}
    
    def register_command(self, command_type, handler: CommandHandler):
        self.command_handlers[command_type] = handler
    
    def register_query(self, query_type, handler: QueryHandler):
        self.query_handlers[query_type] = handler
    
    def execute_command(self, command):
        handler = self.command_handlers.get(type(command))
        if not handler:
            raise ValueError(f"未找到命令处理器: {type(command)}")
        return handler.handle(command)
    
    def execute_query(self, query):
        handler = self.query_handlers.get(type(query))
        if not handler:
            raise ValueError(f"未找到查询处理器: {type(query)}")
        return handler.handle(query)

# 使用示例
print("\n=== CQRS 模式 ===")

write_db = {}  # 写数据库
read_db = {}   # 读数据库

bus = CQRSBus()
bus.register_command(CreateUserCommand, CreateUserHandler(write_db))
bus.register_query(GetUserQuery, GetUserHandler(read_db))

# 执行命令
command = CreateUserCommand("张三", "zhangsan@example.com")
user_id = bus.execute_command(command)

# 同步到读库
read_db.update(write_db)

# 执行查询
query = GetUserQuery(user_id)
user = bus.execute_query(query)
print(f"查询结果: {user}")
```

## 事件驱动架构

### 事件驱动系统

```python
from abc import ABC, abstractmethod
from typing import List, Callable
from dataclasses import dataclass
from datetime import datetime
import asyncio

@dataclass
class DomainEvent:
    """领域事件"""
    event_id: str
    event_type: str
    aggregate_id: str
    data: dict
    occurred_at: datetime = None
    
    def __post_init__(self):
        if self.occurred_at is None:
            self.occurred_at = datetime.now()

class EventStore:
    """事件存储"""
    
    def __init__(self):
        self.events: List[DomainEvent] = []
    
    def append(self, event: DomainEvent):
        """追加事件"""
        self.events.append(event)
        print(f"事件已存储: {event.event_type}")
    
    def get_events(self, aggregate_id: str) -> List[DomainEvent]:
        """获取聚合的所有事件"""
        return [e for e in self.events if e.aggregate_id == aggregate_id]

class EventBus:
    """事件总线"""
    
    def __init__(self):
        self.handlers: dict[str, List[Callable]] = {}
    
    def subscribe(self, event_type: str, handler: Callable):
        """订阅事件"""
        if event_type not in self.handlers:
            self.handlers[event_type] = []
        self.handlers[event_type].append(handler)
    
    async def publish(self, event: DomainEvent):
        """发布事件"""
        handlers = self.handlers.get(event.event_type, [])
        tasks = [handler(event) for handler in handlers]
        await asyncio.gather(*tasks)

# 事件处理器
async def send_email_handler(event: DomainEvent):
    """发送邮件处理器"""
    print(f"[EMAIL] 发送邮件: {event.data.get('email')}")
    await asyncio.sleep(0.1)  # 模拟异步操作

async def update_cache_handler(event: DomainEvent):
    """更新缓存处理器"""
    print(f"[CACHE] 更新缓存: {event.aggregate_id}")
    await asyncio.sleep(0.1)

# 使用示例
async def main():
    print("\n=== 事件驱动架构 ===")
    
    event_store = EventStore()
    event_bus = EventBus()
    
    # 订阅事件
    event_bus.subscribe("UserCreated", send_email_handler)
    event_bus.subscribe("UserCreated", update_cache_handler)
    
    # 创建事件
    event = DomainEvent(
        event_id="EVT001",
        event_type="UserCreated",
        aggregate_id="USER001",
        data={'email': 'user@example.com', 'name': '张三'}
    )
    
    # 存储事件
    event_store.append(event)
    
    # 发布事件
    await event_bus.publish(event)
```

## 总结

### 架构模式对比

| 模式 | 优点 | 缺点 | 适用场景 |
|------|------|------|---------|
| MVC/MVT | 结构清晰，易于理解 | 不适合复杂业务 | Web 应用 |
| DDD | 业务驱动，模型清晰 | 学习曲线陡峭 | 复杂业务系统 |
| CQRS | 读写分离，性能优化 | 数据一致性复杂 | 高并发系统 |
| 微服务 | 独立部署，技术异构 | 运维复杂，分布式事务 | 大型系统 |
| 事件驱动 | 解耦，高吞吐 | 调试困难，最终一致性 | 实时系统 |

### 学习路径

1. 基础阶段：MVC/MVT 架构
2. 进阶阶段：DDD、CQRS
3. 高级阶段：微服务、事件驱动
