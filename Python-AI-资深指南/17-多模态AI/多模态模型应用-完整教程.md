### 文本生成图像

```python
"""
使用 Stable Diffusion 生成图像
"""
from diffusers import StableDiffusionPipeline
import torch

class TextToImageGenerator:
    """文本生成图像"""
    
    def __init__(self, model_id="runwayml/stable-diffusion-v1-5"):
        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        self.pipe = StableDiffusionPipeline.from_pretrained(
            model_id,
            torch_dtype=torch.float16 if self.device == "cuda" else torch.float32
        ).to(self.device)
    
    def generate(self, prompt, negative_prompt="", num_images=1):
        """
        生成图像
        
        Args:
            prompt: 正向提示词
            negative_prompt: 负向提示词
            num_images: 生成图像数量
        """
        images = self.pipe(
            prompt=prompt,
            negative_prompt=negative_prompt,
            num_images_per_prompt=num_images,
            num_inference_steps=50,
            guidance_scale=7.5
        ).images
        
        return images
    
    def generate_with_style(self, prompt, style="realistic"):
        """
        生成特定风格的图像
        
        Args:
            prompt: 提示词
            style: 风格（realistic, anime, oil_painting等）
        """
        style_prompts = {
            "realistic": "photorealistic, highly detailed, 8k",
            "anime": "anime style, manga, vibrant colors",
            "oil_painting": "oil painting, classical art style",
            "watercolor": "watercolor painting, soft colors"
        }
        
        full_prompt = f"{prompt}, {style_prompts.get(style, '')}"
        
        return self.generate(full_prompt)

# 使用示例
generator = TextToImageGenerator()

# 生成图像
images = generator.generate(
    prompt="a beautiful sunset over the ocean",
    negative_prompt="blurry, low quality",
    num_images=2
)

# 保存图像
for i, image in enumerate(images):
    image.save(f"generated_{i}.png")

# 生成特定风格
anime_image = generator.generate_with_style(
    prompt="a cute cat",
    style="anime"
)
anime_image[0].save("anime_cat.png")
```

```python
"""
图像描述生成系统
"""
from transformers import BlipProcessor, BlipForConditionalGeneration
from PIL import Image

class ImageCaptioningSystem:
    """图像描述生成系统"""
    
    def __init__(self):
        self.processor = BlipProcessor.from_pretrained(
            "Salesforce/blip-image-captioning-large"
        )
        self.model = BlipForConditionalGeneration.from_pretrained(
            "Salesforce/blip-image-captioning-large"
        )
    
    def generate_caption(self, image_path, style="descriptive"):
        """
        生成图像描述
        
        Args:
            image_path: 图像路径
            style: 描述风格
        """
        image = Image.open(image_path).convert('RGB')
        
        # 根据风格调整提示
        prompts = {
            "descriptive": "a photography of",
            "poetic": "a beautiful scene of",
            "technical": "an image showing"
        }
        
        prompt = prompts.get(style, "")
        
        inputs = self.processor(image, prompt, return_tensors="pt")
        outputs = self.model.generate(**inputs, max_length=50)
        caption = self.processor.decode(outputs[0], skip_special_tokens=True)
        
        return caption
    
    def generate_detailed_description(self, image_path):
        """生成详细描述"""
        # 生成多个角度的描述
        aspects = [
            "What is the main subject?",
            "What is the setting?",
            "What colors are prominent?",
            "What is the mood or atmosphere?"
        ]
        
        descriptions = []
        for aspect in aspects:
            caption = self.generate_caption(image_path)
            descriptions.append(caption)
        
        return "\n".join(descriptions)

# 使用示例
captioning = ImageCaptioningSystem()

# 生成描述
caption = captioning.generate_caption("image.jpg", style="descriptive")
print(f"图像描述: {caption}")

# 详细描述
detailed = captioning.generate_detailed_description("image.jpg")
print(f"详细描述:\n{detailed}")
```

### 视频内容分析

```python
"""
视频内容分析系统
"""
import cv2
from PIL import Image
import torch

class VideoAnalyzer:
    """视频分析器"""
    
    def __init__(self):
        from transformers import BlipProcessor, BlipForConditionalGeneration
        
        self.processor = BlipProcessor.from_pretrained(
            "Salesforce/blip-image-captioning-base"
        )
        self.model = BlipForConditionalGeneration.from_pretrained(
            "Salesforce/blip-image-captioning-base"
        )
    
    def extract_frames(self, video_path, fps=1):
        """
        提取视频帧
        
        Args:
            video_path: 视频路径
            fps: 每秒提取帧数
        """
        cap = cv2.VideoCapture(video_path)
        video_fps = cap.get(cv2.CAP_PROP_FPS)
        frame_interval = int(video_fps / fps)
        
        frames = []
        frame_count = 0
        
        while True:
            ret, frame = cap.read()
            if not ret:
                break
            
            if frame_count % frame_interval == 0:
                # 转换为 RGB
                frame_rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
                frames.append(Image.fromarray(frame_rgb))
            
            frame_count += 1
        
        cap.release()
        return frames
    
    def analyze_video(self, video_path, fps=1):
        """
        分析视频内容
        
        Args:
            video_path: 视频路径
            fps: 采样帧率
        """
        # 提取帧
        frames = self.extract_frames(video_path, fps)
        
        # 分析每一帧
        descriptions = []
        for i, frame in enumerate(frames):
            inputs = self.processor(frame, return_tensors="pt")
            outputs = self.model.generate(**inputs)
            caption = self.processor.decode(outputs[0], skip_special_tokens=True)
            
            descriptions.append({
                'frame_index': i,
                'timestamp': i / fps,
                'description': caption
            })
        
        return descriptions
    
    def generate_video_summary(self, video_path):
        """生成视频摘要"""
        descriptions = self.analyze_video(video_path, fps=0.5)
        
        # 合并描述
        summary = "Video Summary:\n"
        for desc in descriptions:
            summary += f"[{desc['timestamp']:.1f}s] {desc['description']}\n"
        
        return summary

# 使用示例
analyzer = VideoAnalyzer()

# 分析视频
descriptions = analyzer.analyze_video("video.mp4", fps=1)
for desc in descriptions:
    print(f"[{desc['timestamp']:.1f}s] {desc['description']}")

# 生成摘要
summary = analyzer.generate_video_summary("video.mp4")
print(summary)
```

### 视频字幕生成

```python
"""
自动生成视频字幕
"""
import whisper
from moviepy.editor import VideoFileClip

class VideoSubtitleGenerator:
    """视频字幕生成器"""
    
    def __init__(self):
        self.whisper_model = whisper.load_model("base")
    
    def extract_audio(self, video_path, audio_path="temp_audio.wav"):
        """提取视频音频"""
        video = VideoFileClip(video_path)
        video.audio.write_audiofile(audio_path)
        return audio_path
    
    def generate_subtitles(self, video_path, output_path):
        """
        生成字幕文件
        
        Args:
            video_path: 视频路径
            output_path: 字幕输出路径
        """
        # 提取音频
        audio_path = self.extract_audio(video_path)
        
        # 转录音频
        result = self.whisper_model.transcribe(audio_path)
        
        # 生成 SRT 字幕
        with open(output_path, 'w', encoding='utf-8') as f:
            for i, segment in enumerate(result['segments'], 1):
                start_time = self.format_timestamp(segment['start'])
                end_time = self.format_timestamp(segment['end'])
                
                f.write(f"{i}\n")
                f.write(f"{start_time} --> {end_time}\n")
                f.write(f"{segment['text'].strip()}\n\n")
        
        print(f"字幕已保存到: {output_path}")
    
    def format_timestamp(self, seconds):
        """格式化时间戳为 SRT 格式"""
        hours = int(seconds // 3600)
        minutes = int((seconds % 3600) // 60)
        secs = int(seconds % 60)
        millis = int((seconds % 1) * 1000)
        
        return f"{hours:02d}:{minutes:02d}:{secs:02d},{millis:03d}"

# 使用示例
subtitle_gen = VideoSubtitleGenerator()
subtitle_gen.generate_subtitles("video.mp4", "subtitles.srt")
```

### 音频分类

```python
"""
音频分类系统
"""
import torch
import torchaudio
from transformers import Wav2Vec2ForSequenceClassification, Wav2Vec2Processor

class AudioClassifier:
    """音频分类器"""
    
    def __init__(self, model_name="facebook/wav2vec2-base"):
        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        self.processor = Wav2Vec2Processor.from_pretrained(model_name)
        self.model = Wav2Vec2ForSequenceClassification.from_pretrained(
            model_name
        ).to(self.device)
    
    def classify(self, audio_path):
        """
        分类音频
        
        Args:
            audio_path: 音频文件路径
        """
        # 加载音频
        waveform, sample_rate = torchaudio.load(audio_path)
        
        # 重采样到 16kHz
        if sample_rate != 16000:
            resampler = torchaudio.transforms.Resample(sample_rate, 16000)
            waveform = resampler(waveform)
        
        # 处理输入
        inputs = self.processor(
            waveform.squeeze().numpy(),
            sampling_rate=16000,
            return_tensors="pt"
        ).to(self.device)
        
        # 预测
        with torch.no_grad():
            logits = self.model(**inputs).logits
            predicted_class = torch.argmax(logits, dim=-1).item()
        
        return predicted_class

# 使用示例
classifier = AudioClassifier()
result = classifier.classify("audio.wav")
print(f"分类结果: {result}")
```

```python
"""
多模态内容创作平台
"""
class MultimodalCreationPlatform:
    """多模态创作平台"""
    
    def __init__(self, api_key=None):
        # 初始化各个模型
        self.text_to_image = TextToImageGenerator()
        self.image_captioning = ImageCaptioningSystem()
        self.video_analyzer = VideoAnalyzer()
        self.subtitle_generator = VideoSubtitleGenerator()
    
    def create_story_with_images(self, story_text, num_images=4):
        """
        根据故事文本生成配图
        
        Args:
            story_text: 故事文本
            num_images: 生成图像数量
        """
        # 分割故事为段落
        paragraphs = story_text.split('\n\n')[:num_images]
        
        results = []
        for i, paragraph in enumerate(paragraphs):
            # 生成图像
            image = self.text_to_image.generate(paragraph)[0]
            
            # 保存图像
            image_path = f"story_image_{i}.png"
            image.save(image_path)
            
            results.append({
                'paragraph': paragraph,
                'image_path': image_path
            })
        
        return results
    
    def analyze_and_caption_video(self, video_path):
        """
        分析视频并生成描述和字幕
        
        Args:
            video_path: 视频路径
        """
        # 生成视频摘要
        summary = self.video_analyzer.generate_video_summary(video_path)
        
        # 生成字幕
        subtitle_path = video_path.replace('.mp4', '.srt')
        self.subtitle_generator.generate_subtitles(video_path, subtitle_path)
        
        return {
            'summary': summary,
            'subtitle_path': subtitle_path
        }
    
    def create_social_media_post(self, topic):
        """
        创建社交媒体帖子（图文）
        
        Args:
            topic: 主题
        """
        # 生成图像
        image = self.text_to_image.generate(topic)[0]
        image_path = "social_post.png"
        image.save(image_path)
        
        # 生成描述
        caption = self.image_captioning.generate_caption(image_path)
        
        return {
            'image_path': image_path,
            'caption': caption,
            'hashtags': self.generate_hashtags(topic)
        }
    
    def generate_hashtags(self, topic):
        """生成相关标签"""
        # 简单实现，实际可以使用 LLM
        words = topic.split()
        hashtags = [f"#{word}" for word in words if len(word) > 3]
        return hashtags

# 使用示例
platform = MultimodalCreationPlatform()

# 创建故事配图
story = """
A brave knight embarked on a quest.

He crossed dangerous mountains.

Finally, he found the treasure.
"""

results = platform.create_story_with_images(story)
for i, result in enumerate(results, 1):
    print(f"场景 {i}:")
    print(f"文本: {result['paragraph']}")
    print(f"图像: {result['image_path']}\n")

# 分析视频
video_result = platform.analyze_and_caption_video("video.mp4")
print(f"视频摘要:\n{video_result['summary']}")
print(f"字幕文件: {video_result['subtitle_path']}")

# 创建社交媒体帖子
post = platform.create_social_media_post("beautiful sunset at the beach")
print(f"图像: {post['image_path']}")
print(f"描述: {post['caption']}")
print(f"标签: {' '.join(post['hashtags'])}")
```
