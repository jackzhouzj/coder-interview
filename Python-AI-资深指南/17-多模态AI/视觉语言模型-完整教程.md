## GPT-4V

### 基础使用

```python
"""
使用 GPT-4V 进行图像理解
"""
from openai import OpenAI
import base64

class GPT4Vision:
    """GPT-4V 客户端"""
    
    def __init__(self, api_key):
        self.client = OpenAI(api_key=api_key)
    
    def encode_image(self, image_path):
        """将图像编码为 base64"""
        with open(image_path, "rb") as image_file:
            return base64.b64encode(image_file.read()).decode('utf-8')
    
    def analyze_image(self, image_path, prompt="What's in this image?"):
        """
        分析图像
        
        Args:
            image_path: 图像路径
            prompt: 提示文本
        """
        base64_image = self.encode_image(image_path)
        
        response = self.client.chat.completions.create(
            model="gpt-4-vision-preview",
            messages=[
                {
                    "role": "user",
                    "content": [
                        {"type": "text", "text": prompt},
                        {
                            "type": "image_url",
                            "image_url": {
                                "url": f"data:image/jpeg;base64,{base64_image}"
                            }
                        }
                    ]
                }
            ],
            max_tokens=300
        )
        
        return response.choices[0].message.content
    
    def multi_image_analysis(self, image_paths, prompt):
        """
        分析多张图像
        
        Args:
            image_paths: 图像路径列表
            prompt: 提示文本
        """
        content = [{"type": "text", "text": prompt}]
        
        for image_path in image_paths:
            base64_image = self.encode_image(image_path)
            content.append({
                "type": "image_url",
                "image_url": {
                    "url": f"data:image/jpeg;base64,{base64_image}"
                }
            })
        
        response = self.client.chat.completions.create(
            model="gpt-4-vision-preview",
            messages=[{"role": "user", "content": content}],
            max_tokens=500
        )
        
        return response.choices[0].message.content

# 使用示例
gpt4v = GPT4Vision(api_key="your-api-key")

# 单图分析
result = gpt4v.analyze_image("image.jpg", "Describe this image in detail")
print(result)

# 多图对比
images = ["image1.jpg", "image2.jpg"]
result = gpt4v.multi_image_analysis(images, "Compare these two images")
print(result)
```

### 基础使用

```python
"""
使用 LLaVA 进行视觉对话
"""
from llava.model.builder import load_pretrained_model
from llava.mm_utils import get_model_name_from_path
from llava.eval.run_llava import eval_model
from PIL import Image
import torch

class LLaVAModel:
    """LLaVA 模型封装"""
    
    def __init__(self, model_path="liuhaotian/llava-v1.5-7b"):
        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        
        # 加载模型
        self.tokenizer, self.model, self.image_processor, self.context_len = \
            load_pretrained_model(
                model_path=model_path,
                model_base=None,
                model_name=get_model_name_from_path(model_path)
            )
    
    def chat(self, image_path, question):
        """
        与模型对话
        
        Args:
            image_path: 图像路径
            question: 问题
        """
        # 加载图像
        image = Image.open(image_path).convert('RGB')
        
        # 准备输入
        prompt = f"USER: <image>\n{question}\nASSISTANT:"
        
        # 生成回答
        response = eval_model(
            self.model,
            self.tokenizer,
            self.image_processor,
            image,
            prompt
        )
        
        return response
    
    def multi_turn_chat(self, image_path, questions):
        """
        多轮对话
        
        Args:
            image_path: 图像路径
            questions: 问题列表
        """
        results = []
        
        for question in questions:
            response = self.chat(image_path, question)
            results.append({
                'question': question,
                'answer': response
            })
        
        return results

# 使用示例
llava = LLaVAModel()

# 单轮对话
answer = llava.chat("image.jpg", "What do you see in this image?")
print(f"回答: {answer}")

# 多轮对话
questions = [
    "What is the main object in the image?",
    "What color is it?",
    "Where is it located?"
]

results = llava.multi_turn_chat("image.jpg", questions)
for result in results:
    print(f"Q: {result['question']}")
    print(f"A: {result['answer']}\n")
```

## MiniGPT-4

### 基础使用

```python
"""
使用 MiniGPT-4 进行图像理解
"""
from minigpt4.common.config import Config
from minigpt4.common.registry import registry
from minigpt4.conversation.conversation import Chat
from PIL import Image

class MiniGPT4:
    """MiniGPT-4 模型封装"""
    
    def __init__(self, cfg_path, model_cfg):
        # 加载配置
        cfg = Config(cfg_path, model_cfg)
        
        # 初始化模型
        model_config = cfg.model_cfg
        model_cls = registry.get_model_class(model_config.arch)
        self.model = model_cls.from_config(model_config).to('cuda')
        
        # 初始化对话
        vis_processor_cfg = cfg.datasets_cfg.cc_sbu_align.vis_processor.train
        self.vis_processor = registry.get_processor_class(
            vis_processor_cfg.name
        ).from_config(vis_processor_cfg)
        
        self.chat = Chat(self.model, self.vis_processor)
    
    def generate(self, image_path, prompt):
        """
        生成回答
        
        Args:
            image_path: 图像路径
            prompt: 提示文本
        """
        # 加载图像
        image = Image.open(image_path).convert('RGB')
        
        # 上传图像
        self.chat.upload_img(image, self.vis_processor)
        
        # 生成回答
        response = self.chat.answer(
            prompt=prompt,
            num_beams=1,
            temperature=1.0,
            max_new_tokens=300
        )
        
        return response

# 使用示例
minigpt4 = MiniGPT4(
    cfg_path="eval_configs/minigpt4_eval.yaml",
    model_cfg="minigpt4_7b"
)

response = minigpt4.generate("image.jpg", "Describe this image")
print(response)
```

## 实战案例

### 智能图像助手

```python
"""
基于视觉语言模型的智能图像助手
"""
class ImageAssistant:
    """智能图像助手"""
    
    def __init__(self, model_type="gpt4v", api_key=None):
        if model_type == "gpt4v":
            self.model = GPT4Vision(api_key)
        elif model_type == "llava":
            self.model = LLaVAModel()
        else:
            raise ValueError(f"Unsupported model type: {model_type}")
    
    def describe_image(self, image_path):
        """详细描述图像"""
        prompt = """Please provide a detailed description of this image, including:
        1. Main objects and subjects
        2. Colors and lighting
        3. Setting and background
        4. Any notable details or actions
        """
        return self.model.analyze_image(image_path, prompt)
    
    def extract_text(self, image_path):
        """提取图像中的文本"""
        prompt = "Extract all text visible in this image"
        return self.model.analyze_image(image_path, prompt)
    
    def answer_questions(self, image_path, questions):
        """回答关于图像的问题"""
        results = []
        for question in questions:
            answer = self.model.analyze_image(image_path, question)
            results.append({
                'question': question,
                'answer': answer
            })
        return results
    
    def compare_images(self, image_paths):
        """对比多张图像"""
        prompt = "Compare these images and describe their similarities and differences"
        return self.model.multi_image_analysis(image_paths, prompt)

# 使用示例
assistant = ImageAssistant(model_type="gpt4v", api_key="your-api-key")

# 描述图像
description = assistant.describe_image("image.jpg")
print(f"图像描述:\n{description}\n")

# 提取文本
text = assistant.extract_text("document.jpg")
print(f"提取的文本:\n{text}\n")

# 回答问题
questions = [
    "What is the main subject?",
    "What time of day is it?",
    "Are there any people in the image?"
]
answers = assistant.answer_questions("image.jpg", questions)
for qa in answers:
    print(f"Q: {qa['question']}")
    print(f"A: {qa['answer']}\n")
```

### 图像内容审核

```python
"""
使用视觉语言模型进行内容审核
"""
class ContentModerator:
    """内容审核系统"""
    
    def __init__(self, api_key):
        self.model = GPT4Vision(api_key)
    
    def moderate_image(self, image_path):
        """
        审核图像内容
        
        Returns:
            dict: 审核结果
        """
        prompt = """Analyze this image for content moderation. Check for:
        1. Inappropriate or explicit content
        2. Violence or disturbing imagery
        3. Hate symbols or offensive material
        4. Spam or misleading content
        
        Provide a safety rating (safe/warning/unsafe) and explanation.
        """
        
        result = self.model.analyze_image(image_path, prompt)
        
        # 解析结果
        if "safe" in result.lower():
            rating = "safe"
        elif "warning" in result.lower():
            rating = "warning"
        else:
            rating = "unsafe"
        
        return {
            'rating': rating,
            'explanation': result
        }
    
    def batch_moderate(self, image_paths):
        """批量审核"""
        results = []
        
        for image_path in image_paths:
            result = self.moderate_image(image_path)
            results.append({
                'image_path': image_path,
                **result
            })
        
        return results

# 使用示例
moderator = ContentModerator(api_key="your-api-key")

# 单张图像审核
result = moderator.moderate_image("image.jpg")
print(f"审核结果: {result['rating']}")
print(f"说明: {result['explanation']}")

# 批量审核
images = ["image1.jpg", "image2.jpg", "image3.jpg"]
results = moderator.batch_moderate(images)

for result in results:
    print(f"\n图像: {result['image_path']}")
    print(f"评级: {result['rating']}")
```
