## Core Web Vitals

### LCP（Largest Contentful Paint）
最大内容绘制，衡量加载性能。

```javascript
// 监听LCP
new PerformanceObserver((entryList) => {
  const entries = entryList.getEntries();
  const lastEntry = entries[entries.length - 1];
  
  console.log('LCP:', lastEntry.renderTime || lastEntry.loadTime);
  
  // 上报数据
  reportMetric({
    name: 'LCP',
    value: lastEntry.renderTime || lastEntry.loadTime,
    rating: lastEntry.renderTime < 2500 ? 'good' : 
            lastEntry.renderTime < 4000 ? 'needs-improvement' : 'poor'
  });
}).observe({ type: 'largest-contentful-paint', buffered: true });

// 优化建议
// 1. 优化服务器响应时间
// 2. 使用CDN
// 3. 优化图片加载
// 4. 预加载关键资源
```

**LCP评分标准：**
- 优秀：< 2.5秒
- 需要改进：2.5-4秒
- 差：> 4秒

```javascript

new PerformanceObserver((entryList) => {
  const entries = entryList.getEntries();

    reportMetric({

      value: delay,
      rating: delay < 100 ? 'good' : 
              delay < 300 ? 'needs-improvement' : 'poor'

  });
}).observe({ type: 'first-input', buffered: true });

// 优化建议

```

**FID评分标准：**
- 优秀：< 100ms
- 需要改进：100-300ms
- 差：> 300ms

### CLS（Cumulative Layout Shift）
累积布局偏移，衡量视觉稳定性。

```javascript
// 监听CLS
let clsValue = 0;
let clsEntries = [];

new PerformanceObserver((entryList) => {
  for (const entry of entryList.getEntries()) {
    // 只统计没有用户输入的布局偏移
    if (!entry.hadRecentInput) {
      clsValue += entry.value;
      clsEntries.push(entry);
    }
  }
  
  console.log('CLS:', clsValue);
  
  reportMetric({
    name: 'CLS',
    value: clsValue,
    rating: clsValue < 0.1 ? 'good' : 
            clsValue < 0.25 ? 'needs-improvement' : 'poor'
  });
}).observe({ type: 'layout-shift', buffered: true });

// 优化建议
// 1. 为图片和视频设置尺寸
// 2. 避免在现有内容上方插入内容
// 3. 使用transform动画
// 4. 预留广告位空间
```

**CLS评分标准：**
- 优秀：< 0.1
- 需要改进：0.1-0.25
- 差：> 0.25

### INP（Interaction to Next Paint）
交互到下一次绘制，替代FID的新指标。

```javascript
// 监听INP
new PerformanceObserver((entryList) => {
  const entries = entryList.getEntries();
  
  entries.forEach((entry) => {
```

- 差：> 500ms

### TBT（Total Blocking Time）
总阻塞时间。

```javascript
function calculateTBT() {
  const entries = performance.getEntriesByType('longtask');
  let tbt = 0;
  
  entries.forEach(entry => {
    // 超过50ms的任务被认为是长任务
    if (entry.duration > 50) {
      tbt += entry.duration - 50;
    }
  });
  
  console.log('TBT:', tbt);
  
  reportMetric({
    name: 'TBT',
    value: tbt,
    rating: tbt < 200 ? 'good' : 
            tbt < 600 ? 'needs-improvement' : 'poor'
  });
}

// 监听长任务
new PerformanceObserver((entryList) => {
  calculateTBT();
}).observe({ type: 'longtask', buffered: true });
```

### Speed Index
速度指数。

```javascript
// 使用Lighthouse或WebPageTest测量
// 或使用Performance Observer API计算

function calculateSpeedIndex() {
  const paintEntries = performance.getEntriesByType('paint');
  const navigationEntry = performance.getEntriesByType('navigation')[0];
  
  // 简化计算
  const visualProgress = [];
  
  paintEntries.forEach(entry => {
    visualProgress.push({
      time: entry.startTime,
      progress: entry.name === 'first-contentful-paint' ? 0.5 : 1
    });
  });
  
  // 计算速度指数
  let speedIndex = 0;
  for (let i = 1; i < visualProgress.length; i++) {
    const timeDelta = visualProgress[i].time - visualProgress[i - 1].time;
    const progressDelta = visualProgress[i].progress - visualProgress[i - 1].progress;
    speedIndex += timeDelta * (1 - visualProgress[i - 1].progress);
  }
  
  return speedIndex;
}
```

### 用户自定义指标
```javascript
class PerformanceMetrics {
  constructor() {
    this.metrics = {};
  }
  
  // 记录指标
  recordMetric(name, value, tags = {}) {
    this.metrics[name] = {
      value,
      timestamp: Date.now(),
      tags
    };
  }
  
  // 记录时间
  startTimer(name) {
    this.metrics[name] = {
      startTime: performance.now()
    };
  }
  
  endTimer(name) {
    if (this.metrics[name] && this.metrics[name].startTime) {
      const duration = performance.now() - this.metrics[name].startTime;
      this.metrics[name] = {
        duration,
        timestamp: Date.now()
      };
      return duration;
    }
  }
  
  // 获取所有指标
  getAllMetrics() {
    return this.metrics;
  }
  
  // 上报指标
  report() {
    const data = this.getAllMetrics();
    
    // 发送到监控服务
    navigator.sendBeacon('/api/metrics', JSON.stringify(data));
  }
}

// 使用
const metrics = new PerformanceMetrics();

// 记录API请求时间
metrics.startTimer('api-request');
fetch('/api/data')
  .then(() => {
    const duration = metrics.endTimer('api-request');
    console.log('API请求耗时:', duration);
  });

// 记录组件渲染时间
metrics.startTimer('component-render');
ReactDOM.render(<App />, document.getElementById('root'));
metrics.endTimer('component-render');

// 页面卸载时上报
window.addEventListener('beforeunload', () => {
  metrics.report();
});
```

### 业务指标

```javascript
// 首屏时间
function measureFirstScreen() {
  const observer = new MutationObserver(() => {
    const images = document.querySelectorAll('img');
    const allImagesLoaded = Array.from(images).every(img => img.complete);
    
    if (allImagesLoaded) {
      const firstScreenTime = performance.now();
      console.log('首屏时间:', firstScreenTime);
      
      reportMetric({
        name: 'first-screen',
        value: firstScreenTime
      });
      
      observer.disconnect();
    }
  });
  
  observer.observe(document.body, {
    childList: true,
    subtree: true
  });
}

// 白屏时间
function measureWhiteScreen() {
  const whiteScreenTime = performance.timing.responseStart - 
                          performance.timing.navigationStart;
  
  console.log('白屏时间:', whiteScreenTime);
  
  reportMetric({
    name: 'white-screen',
    value: whiteScreenTime
  });
}

// 可交互时间（自定义）
function measureCustomTTI() {
  let tti = 0;
  
  // 监听DOMContentLoaded
  document.addEventListener('DOMContentLoaded', () => {
    tti = performance.now();
    
    // 等待所有异步资源加载完成
    Promise.all([
      ...Array.from(document.images).map(img => {
        if (img.complete) return Promise.resolve();
        return new Promise(resolve => {
          img.addEventListener('load', resolve);
          img.addEventListener('error', resolve);
        });
      })
    ]).then(() => {
      console.log('自定义TTI:', tti);
      
      reportMetric({
        name: 'custom-tti',
        value: tti
      });
    });
  });
}
```

### User Timing API
```javascript
// 标记和测量
class PerformanceTracker {
  // 开始计时
  static start(name) {
    performance.mark(`${name}-start`);
  }
  
  // 结束计时
  static end(name) {
    performance.mark(`${name}-end`);
    performance.measure(name, `${name}-start`, `${name}-end`);
    
    const measure = performance.getEntriesByName(name)[0];
    console.log(`${name}:`, measure.duration);
    
    return measure.duration;
  }
  
  // 清理
  static clear(name) {
    performance.clearMarks(`${name}-start`);
    performance.clearMarks(`${name}-end`);
    performance.clearMeasures(name);
  }
}

// 使用
PerformanceTracker.start('data-fetch');
await fetchData();
PerformanceTracker.end('data-fetch');

PerformanceTracker.start('render');
render();
PerformanceTracker.end('render');
```

## 性能监控实践

### 完整监控方案
```javascript
class PerformanceMonitor {
  constructor(options = {}) {
    this.options = {
      reportUrl: '/api/performance',
      sampleRate: 1, // 采样率
      ...options
    };
    
    this.metrics = {};
    this.init();
  }
  
  init() {
    // 监控Core Web Vitals
    this.observeLCP();
    this.observeFID();
    this.observeCLS();
    
    // 监控其他指标
    this.observeFCP();
    this.observeResources();
    this.observeLongTasks();
    
    // 页面卸载时上报
    this.setupReporting();
  }
  
  observeLCP() {
    new PerformanceObserver((entryList) => {
      const entries = entryList.getEntries();
      const lastEntry = entries[entries.length - 1];
      
      this.metrics.lcp = {
        value: lastEntry.renderTime || lastEntry.loadTime,
        element: lastEntry.element?.tagName
      };
    }).observe({ type: 'largest-contentful-paint', buffered: true });
  }
  
  observeFID() {
    new PerformanceObserver((entryList) => {
      const entries = entryList.getEntries();
      
      entries.forEach((entry) => {
        this.metrics.fid = {
          value: entry.processingStart - entry.startTime,
          eventType: entry.name
        };
      });
    }).observe({ type: 'first-input', buffered: true });
  }
  
  observeCLS() {
    let clsValue = 0;
    
    new PerformanceObserver((entryList) => {
      for (const entry of entryList.getEntries()) {
        if (!entry.hadRecentInput) {
          clsValue += entry.value;
        }
      }
      
      this.metrics.cls = { value: clsValue };
    }).observe({ type: 'layout-shift', buffered: true });
  }
  
  observeFCP() {
    new PerformanceObserver((entryList) => {
      const entries = entryList.getEntries();
      const fcpEntry = entries.find(e => e.name === 'first-contentful-paint');
      
      if (fcpEntry) {
        this.metrics.fcp = { value: fcpEntry.startTime };
      }
    }).observe({ type: 'paint', buffered: true });
  }
  
  observeResources() {
    new PerformanceObserver((entryList) => {
      const entries = entryList.getEntries();
      
      this.metrics.resources = entries.map(entry => ({
        name: entry.name,
        type: entry.initiatorType,
        duration: entry.duration,
        size: entry.transferSize
      }));
    }).observe({ type: 'resource', buffered: true });
  }
  
  observeLongTasks() {
    new PerformanceObserver((entryList) => {
      const entries = entryList.getEntries();
      
      this.metrics.longTasks = entries.map(entry => ({
        duration: entry.duration,
        startTime: entry.startTime
      }));
    }).observe({ type: 'longtask', buffered: true });
  }
  
  setupReporting() {
    // 页面可见性变化时上报
    document.addEventListener('visibilitychange', () => {
      if (document.visibilityState === 'hidden') {
        this.report();
      }
    });
    
    // 页面卸载时上报
    window.addEventListener('beforeunload', () => {
      this.report();
    });
  }
  
  report() {
    // 采样
    if (Math.random() > this.options.sampleRate) {
      return;
    }
    
    const data = {
      url: location.href,
      timestamp: Date.now(),
      userAgent: navigator.userAgent,
      metrics: this.metrics,
      // 添加用户信息
      userId: this.getUserId(),
      // 添加设备信息
      device: this.getDeviceInfo()
    };
    
    // 使用sendBeacon确保数据发送
    if (navigator.sendBeacon) {
      navigator.sendBeacon(
        this.options.reportUrl,
        JSON.stringify(data)
      );
    } else {
      // 降级方案
      fetch(this.options.reportUrl, {
        method: 'POST',
        body: JSON.stringify(data),
        keepalive: true
      });
    }
  }
  
  getUserId() {
    // 获取用户ID逻辑
    return localStorage.getItem('userId') || 'anonymous';
  }
  
  getDeviceInfo() {
    return {
      screen: `${screen.width}x${screen.height}`,
      viewport: `${window.innerWidth}x${window.innerHeight}`,
      devicePixelRatio: window.devicePixelRatio,
      connection: navigator.connection?.effectiveType
    };
  }
}

// 初始化监控
const monitor = new PerformanceMonitor({
  reportUrl: '/api/performance',
  sampleRate: 0.1 // 10%采样率
});
```

### 性能预算
```javascript
const performanceBudget = {
  lcp: 2500,
  fid: 100,
  cls: 0.1,
  fcp: 1800,
  tti: 3800,
  tbt: 200,
  resourceSize: 1024 * 1024, // 1MB
  resourceCount: 50
};

function checkPerformanceBudget(metrics) {
  const violations = [];
  
  Object.keys(performanceBudget).forEach(key => {
    if (metrics[key] && metrics[key].value > performanceBudget[key]) {
      violations.push({
        metric: key,
        actual: metrics[key].value,
        budget: performanceBudget[key],
        exceeded: metrics[key].value - performanceBudget[key]
      });
    }
  });
  
  if (violations.length > 0) {
    console.warn('性能预算超标:', violations);
    
    // 上报预算超标
    reportMetric({
      name: 'budget-violation',
      violations
    });
  }
  
  return violations;
}
```
