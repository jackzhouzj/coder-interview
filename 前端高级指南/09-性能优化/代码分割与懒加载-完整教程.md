## 1. 动态导入

### 1.1 基础动态导入

#### 示例1：动态导入模块
```javascript
/**
 * 动态导入示例
 * @author erik.zhou
 */

// 基础动态导入
async function loadModule() {
    const module = await import('./module.js');
    module.default();
}

// 条件导入
async function loadConditionally(condition) {
    if (condition) {
        const { feature } = await import('./feature.js');
        feature();
    }
}

// 动态导入工具类
class DynamicImporter {
    constructor() {
        this.cache = new Map();
    }

    async import(modulePath) {
        if (this.cache.has(modulePath)) {
            return this.cache.get(modulePath);
        }

        try {
            const module = await import(modulePath);
            this.cache.set(modulePath, module);
            return module;
        } catch (error) {
            console.error(`Failed to load module: ${modulePath}`, error);
            throw error;
        }
    }

    clearCache() {
        this.cache.clear();
    }
}

// 使用示例
const importer = new DynamicImporter();

importer.import('./utils.js').then(utils => {
    utils.doSomething();
});
```

## 2. 路由懒加载

### 2.1 React路由懒加载

#### 示例2：React Router懒加载
```javascript
/**
 * React Router懒加载
 * @author erik.zhou
 */
import React, { Suspense, lazy } from 'react';
import { BrowserRouter, Routes, Route } from 'react-router-dom';

// 懒加载组件
const Home = lazy(() => import('./pages/Home'));
const About = lazy(() => import('./pages/About'));
const Dashboard = lazy(() => import('./pages/Dashboard'));

// 加载中组件
function LoadingFallback() {
    return <div>Loading...</div>;
}

function App() {
    return (
        <BrowserRouter>
            <Suspense fallback={<LoadingFallback />}>
                <Routes>
                    <Route path="/" element={<Home />} />
                    <Route path="/about" element={<About />} />
                    <Route path="/dashboard" element={<Dashboard />} />
                </Routes>
            </Suspense>
        </BrowserRouter>
    );
}

export default App;
```

## 4. 按需加载

#### 示例5：按需加载库
```javascript
/**
 * 按需加载第三方库
 * @author erik.zhou
 */
class LibraryLoader {
    constructor() {
        this.loaded = new Set();
    }

    async loadChart() {
        if (this.loaded.has('chart')) {
            return;
        }

        const Chart = await import('chart.js');
        this.loaded.add('chart');
        return Chart;
    }

    async loadEditor() {
        if (this.loaded.has('editor')) {
            return;
        }

        const Editor = await import('monaco-editor');
        this.loaded.add('editor');
        return Editor;
    }
}

// 使用示例
const loader = new LibraryLoader();

document.getElementById('showChart').addEventListener('click', async () => {
    const Chart = await loader.loadChart();
    // 使用Chart.js
});
```

## 5. 预加载与预获取

#### 示例6：智能预加载
```javascript
/**
 * 智能预加载
 * @author erik.zhou
 */
class SmartPreloader {
    constructor() {
        this.preloaded = new Set();
    }

    preload(modulePath) {
        if (this.preloaded.has(modulePath)) {
            return;
        }

        const link = document.createElement('link');
        link.rel = 'prefetch';
        link.href = modulePath;
        document.head.appendChild(link);

        this.preloaded.add(modulePath);
    }

    preloadOnHover(element, modulePath) {
        let timer = null;

        element.addEventListener('mouseenter', () => {
            timer = setTimeout(() => {
                this.preload(modulePath);
            }, 300);
        });

        element.addEventListener('mouseleave', () => {
            if (timer) {
                clearTimeout(timer);
            }
        });
    }
}

// 使用示例
const preloader = new SmartPreloader();

const link = document.querySelector('a[href="/dashboard"]');
preloader.preloadOnHover(link, '/js/dashboard.js');
```

## 8. 性能监控

#### 示例9：懒加载性能监控
```javascript
/**
 * 懒加载性能监控
 * @author erik.zhou
 */
class LazyLoadMonitor {
    constructor() {
        this.metrics = [];
    }

    async measureLoad(modulePath, loader) {
        const startTime = performance.now();

        try {
            const module = await loader();
            const endTime = performance.now();
            const duration = endTime - startTime;

            this.metrics.push({
                modulePath,
                duration,
                success: true,
                timestamp: Date.now()
            });

            console.log(`Loaded ${modulePath} in ${duration.toFixed(2)}ms`);
            return module;
        } catch (error) {
            const endTime = performance.now();
            const duration = endTime - startTime;

            this.metrics.push({
                modulePath,
                duration,
                success: false,
                error: error.message,
                timestamp: Date.now()
            });

            throw error;
        }
    }

    getReport() {
        return {
            totalLoads: this.metrics.length,
            successRate: this.metrics.filter(m => m.success).length / this.metrics.length,
            avgDuration: this.metrics.reduce((sum, m) => sum + m.duration, 0) / this.metrics.length,
            metrics: this.metrics
        };
    }
}

// 使用示例
const monitor = new LazyLoadMonitor();

monitor.measureLoad('./module.js', () => import('./module.js'))
    .then(module => {
        console.log('Module loaded:', module);
    });
```

## 9. 错误处理

#### 示例10：懒加载错误处理
```javascript
/**
 * 懒加载错误处理
 * @author erik.zhou
 */
class LazyLoadErrorHandler {
    async loadWithRetry(loader, retries = 3) {
        for (let i = 0; i < retries; i++) {
            try {
                return await loader();
            } catch (error) {
                if (i === retries - 1) {
                    throw error;
                }
                await this.delay(1000 * (i + 1));
            }
        }
    }

    delay(ms) {
        return new Promise(resolve => setTimeout(resolve, ms));
    }
}

// 使用示例
const errorHandler = new LazyLoadErrorHandler();

errorHandler.loadWithRetry(() => import('./module.js'))
    .then(module => console.log('Loaded:', module))
    .catch(error => console.error('Failed:', error));
```
