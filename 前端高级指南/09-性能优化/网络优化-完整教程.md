## 1. HTTP协议优化

### 1.1 HTTP/2优化

#### 示例1：HTTP/2服务器推送
```javascript
/**
 * HTTP/2服务器推送配置
 * @author erik.zhou
 */
const express = require('express');
const spdy = require('spdy');
const fs = require('fs');

const app = express();

// HTTP/2服务器配置
const options = {
    key: fs.readFileSync('./server.key'),
    cert: fs.readFileSync('./server.crt')
};

// 服务器推送中间件
app.use((req, res, next) => {
    if (req.url === '/') {
        // 推送关键资源
        const stream = res.push('/css/critical.css', {
            status: 200,
            method: 'GET',
            request: {
                accept: 'text/css'
            },
            response: {
                'content-type': 'text/css'
            }
        });

        stream.on('error', (err) => {
            console.error('Push stream error:', err);
        });

        stream.end(fs.readFileSync('./public/css/critical.css'));

        // 推送JavaScript
        const jsStream = res.push('/js/main.js', {
            status: 200,
            method: 'GET',
            request: {
                accept: 'application/javascript'
            },
            response: {
                'content-type': 'application/javascript'
            }
        });

        jsStream.end(fs.readFileSync('./public/js/main.js'));
    }

    next();
});

app.use(express.static('public'));

// 启动HTTP/2服务器
spdy.createServer(options, app).listen(3000, () => {
    console.log('HTTP/2 server running on https://localhost:3000');
});
```

#### 示例2：HTTP/2多路复用
```javascript
/**
 * HTTP/2多路复用客户端
 * @author erik.zhou
 */
class HTTP2Client {
    constructor(baseUrl) {
        this.baseUrl = baseUrl;
        this.session = null;
    }

    /**
     * 建立连接
     */
    async connect() {
        if (this.session) {
            return;
        }

        // 使用fetch API（浏览器自动处理HTTP/2）
        this.session = true;
    }

    /**
     * 并行请求多个资源
     * @param {Array<string>} urls - 资源URL数组
     * @returns {Promise<Array>}
     */
    async fetchMultiple(urls) {
        await this.connect();

        // HTTP/2自动复用连接
        const promises = urls.map(url => 
            fetch(`${this.baseUrl}${url}`)
                .then(res => res.json())
        );

        return Promise.all(promises);
    }

    /**
     * 流式请求
     * @param {string} url - 资源URL
     * @param {Function} onChunk - 数据块回调
     */
    async streamFetch(url, onChunk) {
        const response = await fetch(`${this.baseUrl}${url}`);
        const reader = response.body.getReader();
        const decoder = new TextDecoder();

        while (true) {
            const { done, value } = await reader.read();
            
            if (done) {
                break;
            }

            const chunk = decoder.decode(value, { stream: true });
            onChunk(chunk);
        }
    }
}

// 使用示例
const client = new HTTP2Client('https://api.example.com');

// 并行请求
client.fetchMultiple([
    '/api/user',
    '/api/posts',
    '/api/comments'
]).then(([user, posts, comments]) => {
    console.log('User:', user);
    console.log('Posts:', posts);
    console.log('Comments:', comments);
});

// 流式请求
client.streamFetch('/api/stream', (chunk) => {
    console.log('Received chunk:', chunk);
});
```

## 2. 请求优化

### 2.1 请求合并

#### 示例3：请求批处理
```javascript
/**
 * 请求批处理器
 * @author erik.zhou
 */
class RequestBatcher {
    constructor(options = {}) {
        this.batchUrl = options.batchUrl || '/api/batch';
        this.maxBatchSize = options.maxBatchSize || 10;
        this.batchDelay = options.batchDelay || 50;
        
        this.queue = [];
        this.timer = null;
    }

    /**
     * 添加请求到批处理队列
     * @param {string} url - 请求URL
     * @param {Object} options - 请求选项
     * @returns {Promise<any>}
     */
    fetch(url, options = {}) {
        return new Promise((resolve, reject) => {
            this.queue.push({
                url,
                options,
                resolve,
                reject
            });

            // 达到批处理大小立即执行
            if (this.queue.length >= this.maxBatchSize) {
                this.flush();
            } else {
                // 延迟执行
                this.scheduleBatch();
            }
        });
    }

    /**
     * 调度批处理
     */
    scheduleBatch() {
        if (this.timer) {
            return;
        }

        this.timer = setTimeout(() => {
            this.flush();
        }, this.batchDelay);
    }

    /**
     * 执行批处理
     */
    async flush() {
        if (this.queue.length === 0) {
            return;
        }

        if (this.timer) {
            clearTimeout(this.timer);
            this.timer = null;
        }

        const batch = this.queue.splice(0, this.maxBatchSize);

        try {
            const response = await fetch(this.batchUrl, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify({
                    requests: batch.map(item => ({
                        url: item.url,
                        method: item.options.method || 'GET',
                        headers: item.options.headers,
                        body: item.options.body
                    }))
                })
            });

            const results = await response.json();

            // 分发结果
            batch.forEach((item, index) => {
                const result = results[index];
                if (result.error) {
                    item.reject(new Error(result.error));
                } else {
                    item.resolve(result.data);
                }
            });
        } catch (error) {
            // 批处理失败，拒绝所有请求
            batch.forEach(item => item.reject(error));
        }
    }

    /**
     * 清空队列
     */
    clear() {
        if (this.timer) {
            clearTimeout(this.timer);
            this.timer = null;
        }
        this.queue = [];
    }
}

// 使用示例
const batcher = new RequestBatcher({
    batchUrl: '/api/batch',
    maxBatchSize: 10,
    batchDelay: 50
});

// 发起多个请求
const requests = [
    batcher.fetch('/api/user/1'),
    batcher.fetch('/api/user/2'),
    batcher.fetch('/api/user/3'),
    batcher.fetch('/api/posts/1'),
    batcher.fetch('/api/posts/2')
];

Promise.all(requests).then(results => {
    console.log('Batch results:', results);
});
```

```javascript
/**
 * GraphQL批处理客户端
 * @author erik.zhou
 */
class GraphQLBatcher {
    constructor(endpoint, options = {}) {
        this.endpoint = endpoint;
        this.batchDelay = options.batchDelay || 10;
        this.maxBatchSize = options.maxBatchSize || 10;
        
        this.queue = [];
        this.timer = null;
    }

    /**
     * 执行查询
     * @param {string} query - GraphQL查询
     * @param {Object} variables - 变量
     * @returns {Promise<any>}
     */
    query(query, variables = {}) {
        return new Promise((resolve, reject) => {
            this.queue.push({
                query,
                variables,
                resolve,
                reject
            });

            if (this.queue.length >= this.maxBatchSize) {
                this.flush();
            } else {
                this.scheduleBatch();
            }
        });
    }

    /**
     * 调度批处理
     */
    scheduleBatch() {
        if (this.timer) {
            return;
        }

        this.timer = setTimeout(() => {
            this.flush();
        }, this.batchDelay);
    }

    /**
     * 执行批处理
     */
    async flush() {
        if (this.queue.length === 0) {
            return;
        }

        if (this.timer) {
            clearTimeout(this.timer);
            this.timer = null;
        }

        const batch = this.queue.splice(0, this.maxBatchSize);

        try {
            const response = await fetch(this.endpoint, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify(
                    batch.map(item => ({
                        query: item.query,
                        variables: item.variables
                    }))
                )
            });

            const results = await response.json();

            // 分发结果
            batch.forEach((item, index) => {
                const result = results[index];
                if (result.errors) {
                    item.reject(new Error(result.errors[0].message));
                } else {
                    item.resolve(result.data);
                }
            });
        } catch (error) {
            batch.forEach(item => item.reject(error));
        }
    }
}

// 使用示例
const gqlBatcher = new GraphQLBatcher('https://api.example.com/graphql');

// 批量查询
const queries = [
    gqlBatcher.query(`
        query GetUser($id: ID!) {
            user(id: $id) {
                id
                name
                email
            }
        }
    `, { id: '1' }),
    
    gqlBatcher.query(`
        query GetPosts($userId: ID!) {
            posts(userId: $userId) {
                id
                title
                content
            }
        }
    `, { userId: '1' })
];

Promise.all(queries).then(([user, posts]) => {
    console.log('User:', user);
    console.log('Posts:', posts);
});
```

### 2.2 请求去重

#### 示例5：请求去重器
```javascript
/**
 * 请求去重器
 * @author erik.zhou
 */
class RequestDeduplicator {
    constructor() {
        this.pendingRequests = new Map();
    }

    /**
     * 发起请求（自动去重）
     * @param {string} url - 请求URL
     * @param {Object} options - 请求选项
     * @returns {Promise<any>}
     */
    fetch(url, options = {}) {
        const key = this.generateKey(url, options);

        // 如果请求正在进行，返回现有Promise
        if (this.pendingRequests.has(key)) {
            return this.pendingRequests.get(key);
        }

        // 创建新请求
        const promise = fetch(url, options)
            .then(response => response.json())
            .then(data => {
                this.pendingRequests.delete(key);
                return data;
            })
            .catch(error => {
                this.pendingRequests.delete(key);
                throw error;
            });

        this.pendingRequests.set(key, promise);
        return promise;
    }

    /**
     * 生成请求键
     * @param {string} url - 请求URL
     * @param {Object} options - 请求选项
     * @returns {string}
     */
    generateKey(url, options) {
        const method = options.method || 'GET';
        const body = options.body ? JSON.stringify(options.body) : '';
        return `${method}:${url}:${body}`;
    }

    /**
     * 清空缓存
     */
    clear() {
        this.pendingRequests.clear();
    }

    /**
     * 获取统计信息
     */
    getStats() {
        return {
            pendingCount: this.pendingRequests.size,
            pendingUrls: Array.from(this.pendingRequests.keys())
        };
    }
}

// 使用示例
const deduplicator = new RequestDeduplicator();

// 同时发起多个相同请求
const requests = [
    deduplicator.fetch('/api/user/1'),
    deduplicator.fetch('/api/user/1'),
    deduplicator.fetch('/api/user/1')
];

// 实际只会发起一个请求
Promise.all(requests).then(results => {
    console.log('Results:', results);
    // 三个结果相同，但只发起了一个网络请求
});
```

## 4. CDN优化

#### 示例7：CDN故障转移
```javascript
/**
 * CDN故障转移
 * @author erik.zhou
 */
class CDNFailover {
    constructor(cdnUrls, fallbackUrl) {
        this.cdnUrls = cdnUrls;
        this.fallbackUrl = fallbackUrl;
        this.failedCDNs = new Set();
    }

    async loadResource(path) {
        for (const cdn of this.cdnUrls) {
            if (this.failedCDNs.has(cdn)) {
                continue;
            }

            try {
                const url = `${cdn}${path}`;
                const response = await fetch(url, { timeout: 3000 });
                if (response.ok) {
                    return response;
                }
            } catch (error) {
                this.failedCDNs.add(cdn);
            }
        }

        // 使用降级URL
        return fetch(`${this.fallbackUrl}${path}`);
    }
}
```

## 5. 缓存策略

#### 示例8：智能缓存管理
```javascript
/**
 * 智能缓存管理器
 * @author erik.zhou
 */
class CacheManager {
    constructor() {
        this.cache = new Map();
        this.maxSize = 100;
        this.ttl = 5 * 60 * 1000; // 5分钟
    }

    async get(key, fetcher) {
        const cached = this.cache.get(key);
        
        if (cached && Date.now() - cached.timestamp < this.ttl) {
            return cached.data;
        }

        const data = await fetcher();
        this.set(key, data);
        return data;
    }

    set(key, data) {
        if (this.cache.size >= this.maxSize) {
            const firstKey = this.cache.keys().next().value;
            this.cache.delete(firstKey);
        }

        this.cache.set(key, {
            data,
            timestamp: Date.now()
        });
    }

    clear() {
        this.cache.clear();
    }
}
```

## 7. 网络监控

#### 示例10：网络性能监控
```javascript
/**
 * 网络性能监控
 * @author erik.zhou
 */
class NetworkMonitor {
    constructor() {
        this.metrics = [];
    }

    observe() {
        if (!('PerformanceObserver' in window)) {
            return;
        }

        const observer = new PerformanceObserver((list) => {
            const entries = list.getEntries();
            entries.forEach(entry => {
                this.metrics.push({
                    name: entry.name,
                    duration: entry.duration,
                    size: entry.transferSize
                });
            });
        });

        observer.observe({ entryTypes: ['resource'] });
    }

    getReport() {
        return {
            totalRequests: this.metrics.length,
            totalSize: this.metrics.reduce((sum, m) => sum + (m.size || 0), 0),
            avgDuration: this.metrics.reduce((sum, m) => sum + m.duration, 0) / this.metrics.length
        };
    }
}
```

## 9. WebSocket优化

#### 示例12：WebSocket连接池
```javascript
/**
 * WebSocket连接池
 * @author erik.zhou
 */
class WebSocketPool {
    constructor(url, poolSize = 3) {
        this.url = url;
        this.poolSize = poolSize;
        this.connections = [];
        this.currentIndex = 0;
    }

    async getConnection() {
        if (this.connections.length < this.poolSize) {
            const ws = new WebSocket(this.url);
            await new Promise((resolve) => {
                ws.onopen = resolve;
            });
            this.connections.push(ws);
            return ws;
        }

        const ws = this.connections[this.currentIndex];
        this.currentIndex = (this.currentIndex + 1) % this.poolSize;
        return ws;
    }

    closeAll() {
        this.connections.forEach(ws => ws.close());
        this.connections = [];
    }
}
```
