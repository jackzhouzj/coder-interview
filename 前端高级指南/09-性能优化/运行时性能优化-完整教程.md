## 1. JavaScript执行优化

### 1.1 避免长任务

#### 示例1：任务分片
```javascript
/**
 * 任务分片执行
 * @author erik.zhou
 */
class TaskScheduler {
    constructor(options = {}) {
        this.chunkSize = options.chunkSize || 50;
        this.timeout = options.timeout || 16; // 一帧的时间
        this.tasks = [];
        this.isRunning = false;
    }

    /**
     * 添加任务
     * @param {Function} task - 任务函数
     */
    addTask(task) {
        this.tasks.push(task);
        if (!this.isRunning) {
            this.run();
        }
    }

    /**
     * 运行任务
     */
    async run() {
        this.isRunning = true;

        while (this.tasks.length > 0) {
            const startTime = performance.now();
            let count = 0;

            // 在一帧时间内执行尽可能多的任务
            while (
                this.tasks.length > 0 &&
                count < this.chunkSize &&
                performance.now() - startTime < this.timeout
            ) {
                const task = this.tasks.shift();
                try {
                    await task();
                } catch (error) {
                    console.error('Task execution failed:', error);
                }
                count++;
            }

            // 让出主线程，避免阻塞渲染
            if (this.tasks.length > 0) {
                await this.yieldToMain();
            }
        }

        this.isRunning = false;
    }

    /**
     * 让出主线程
     * @returns {Promise<void>}
     */
    yieldToMain() {
        return new Promise(resolve => {
            setTimeout(resolve, 0);
        });
    }

    /**
     * 批量处理数据
     * @param {Array} items - 数据数组
     * @param {Function} processor - 处理函数
     * @returns {Promise<void>}
     */
    async processBatch(items, processor) {
        for (let i = 0; i < items.length; i += this.chunkSize) {
            const chunk = items.slice(i, i + this.chunkSize);
            
            await new Promise(resolve => {
                requestAnimationFrame(() => {
                    chunk.forEach(item => processor(item));
                    resolve();
                });
            });
        }
    }
}

// 使用示例
const scheduler = new TaskScheduler({
    chunkSize: 100,
    timeout: 16
});

// 处理大量数据
const largeDataset = Array.from({ length: 10000 }, (_, i) => i);

scheduler.processBatch(largeDataset, (item) => {
    // 处理每个数据项
    console.log('Processing:', item);
}).then(() => {
    console.log('All items processed');
});

// 添加多个任务
for (let i = 0; i < 1000; i++) {
    scheduler.addTask(() => {
        // 执行任务
        return new Promise(resolve => {
            // 模拟异步操作
            setTimeout(resolve, 1);
        });
    });
}
```

#### 示例2：防抖与节流
```javascript
/**
 * 防抖与节流工具
 * @author erik.zhou
 */
class PerformanceUtils {
    /**
     * 防抖函数
     * @param {Function} func - 要防抖的函数
     * @param {number} wait - 等待时间
     * @param {boolean} immediate - 是否立即执行
     * @returns {Function}
     */
    static debounce(func, wait = 300, immediate = false) {
        let timeout = null;

        return function debounced(...args) {
            const context = this;

            const later = () => {
                timeout = null;
                if (!immediate) {
                    func.apply(context, args);
                }
            };

            const callNow = immediate && !timeout;

            clearTimeout(timeout);
            timeout = setTimeout(later, wait);

            if (callNow) {
                func.apply(context, args);
            }
        };
    }

    /**
     * 节流函数
     * @param {Function} func - 要节流的函数
     * @param {number} wait - 等待时间
     * @param {Object} options - 选项
     * @returns {Function}
     */
    static throttle(func, wait = 300, options = {}) {
        let timeout = null;
        let previous = 0;
        const { leading = true, trailing = true } = options;

        return function throttled(...args) {
            const context = this;
            const now = Date.now();

            if (!previous && !leading) {
                previous = now;
            }

            const remaining = wait - (now - previous);

            if (remaining <= 0 || remaining > wait) {
                if (timeout) {
                    clearTimeout(timeout);
                    timeout = null;
                }

                previous = now;
                func.apply(context, args);
            } else if (!timeout && trailing) {
                timeout = setTimeout(() => {
                    previous = leading ? Date.now() : 0;
                    timeout = null;
                    func.apply(context, args);
                }, remaining);
            }
        };
    }

    /**
     * requestAnimationFrame节流
     * @param {Function} func - 要节流的函数
     * @returns {Function}
     */
    static rafThrottle(func) {
        let rafId = null;
        let lastArgs = null;

        return function throttled(...args) {
            lastArgs = args;

            if (rafId === null) {
                rafId = requestAnimationFrame(() => {
                    func.apply(this, lastArgs);
                    rafId = null;
                    lastArgs = null;
                });
            }
        };
    }
}

// 使用示例

// 防抖：搜索输入
const searchInput = document.querySelector('#search');
const debouncedSearch = PerformanceUtils.debounce((value) => {
    console.log('Searching for:', value);
    // 执行搜索请求
    fetch(`/api/search?q=${value}`)
        .then(res => res.json())
        .then(data => console.log(data));
}, 500);

searchInput.addEventListener('input', (e) => {
    debouncedSearch(e.target.value);
});

// 节流：滚动事件
const throttledScroll = PerformanceUtils.throttle(() => {
    console.log('Scroll position:', window.scrollY);
    // 执行滚动相关操作
}, 200);

window.addEventListener('scroll', throttledScroll);

// RAF节流：鼠标移动
const rafThrottledMove = PerformanceUtils.rafThrottle((e) => {
    console.log('Mouse position:', e.clientX, e.clientY);
    // 更新UI
});

document.addEventListener('mousemove', rafThrottledMove);
```

/**
 * 循环优化示例
 * @author erik.zhou
 */
class LoopOptimizer {
    /**
     * 优化前：低效循环
     */
    static inefficientLoop(items) {
        const results = [];
        
        // 问题1：每次循环都访问length属性
        for (let i = 0; i < items.length; i++) {
            // 问题2：在循环内创建对象
            const obj = {
                id: items[i].id,
                name: items[i].name
            };
            
            // 问题3：频繁的DOM操作
            document.getElementById('list').innerHTML += `<li>${obj.name}</li>`;
            
            results.push(obj);
        }
        
        return results;
    }

    /**
     * 优化后：高效循环
     */
    static efficientLoop(items) {
        const results = [];
        const len = items.length; // 缓存长度
        const fragment = document.createDocumentFragment(); // 使用文档片段
        
        // 使用for循环（比forEach快）
        for (let i = 0; i < len; i++) {
            const item = items[i];
            
            // 对象复用
            results.push({
                id: item.id,
                name: item.name
            });
            
            // 批量DOM操作
            const li = document.createElement('li');
            li.textContent = item.name;
            fragment.appendChild(li);
        }
        
        // 一次性插入DOM
        document.getElementById('list').appendChild(fragment);
        
        return results;
    }

    /**
     * 数组操作优化
     */
    static arrayOptimization() {
        const data = Array.from({ length: 10000 }, (_, i) => i);

        // 优化前：链式调用多次遍历
        console.time('inefficient');
        const result1 = data
            .filter(n => n % 2 === 0)
            .map(n => n * 2)
            .filter(n => n > 100);
        console.timeEnd('inefficient');

        // 优化后：单次遍历
        console.time('efficient');
        const result2 = [];
        for (let i = 0; i < data.length; i++) {
            const n = data[i];
            if (n % 2 === 0) {
                const doubled = n * 2;
                if (doubled > 100) {
                    result2.push(doubled);
                }
            }
        }
        console.timeEnd('efficient');

        return { result1, result2 };
    }

    /**
     * 对象属性访问优化
     */
    static propertyAccessOptimization(obj) {
        // 优化前：重复访问深层属性
        console.time('inefficient-access');
        for (let i = 0; i < 10000; i++) {
            const value = obj.data.user.profile.settings.theme;
            // 使用value
        }
        console.timeEnd('inefficient-access');

        // 优化后：缓存属性访问
        console.time('efficient-access');
        const theme = obj.data.user.profile.settings.theme;
        for (let i = 0; i < 10000; i++) {
            const value = theme;
            // 使用value
        }
        console.timeEnd('efficient-access');
    }
}

// 性能测试
const testData = Array.from({ length: 1000 }, (_, i) => ({
    id: i,
    name: `Item ${i}`
}));

console.time('inefficient');
LoopOptimizer.inefficientLoop(testData);
console.timeEnd('inefficient');

console.time('efficient');
LoopOptimizer.efficientLoop(testData);
console.timeEnd('efficient');

### 2.1 避免重排重绘

#### 示例4：批量DOM操作
```javascript
/**
 * DOM操作优化
 * @author erik.zhou
 */
class DOMOptimizer {
    /**
     * 优化前：频繁的DOM操作
     */
    static inefficientDOMUpdate(items) {
        const container = document.getElementById('container');
        
        // 问题：每次操作都触发重排重绘
        items.forEach(item => {
            const div = document.createElement('div');
            div.textContent = item.text;
            div.style.width = '100px';
            div.style.height = '50px';
            div.style.backgroundColor = item.color;
            container.appendChild(div);
        });
    }

    /**
     * 优化后：批量DOM操作
     */
    static efficientDOMUpdate(items) {
        const container = document.getElementById('container');
        const fragment = document.createDocumentFragment();
        
        // 使用文档片段，减少重排重绘
        items.forEach(item => {
            const div = document.createElement('div');
            div.textContent = item.text;
            // 使用CSS类而不是内联样式
            div.className = 'item';
            div.dataset.color = item.color;
            fragment.appendChild(div);
        });
        
        // 一次性插入
        container.appendChild(fragment);
    }

    /**
     * 读写分离优化
     */
    static readWriteSeparation() {
        const elements = document.querySelectorAll('.item');
        
        // 优化前：读写交替导致强制同步布局
        console.time('inefficient-rw');
        elements.forEach(el => {
            const height = el.offsetHeight; // 读
            el.style.height = `${height + 10}px`; // 写
        });
        console.timeEnd('inefficient-rw');

        // 优化后：先读后写
        console.time('efficient-rw');
        const heights = Array.from(elements).map(el => el.offsetHeight); // 批量读
        elements.forEach((el, i) => {
            el.style.height = `${heights[i] + 10}px`; // 批量写
        });
        console.timeEnd('efficient-rw');
    }

    /**
     * 使用transform代替top/left
     */
    static useTransform(element, x, y) {
        // 优化前：触发重排
        // element.style.left = `${x}px`;
        // element.style.top = `${y}px`;

        // 优化后：只触发合成
        element.style.transform = `translate(${x}px, ${y}px)`;
    }

    /**
     * 虚拟滚动容器
     */
    static createVirtualScroll(container, items, itemHeight) {
        const visibleCount = Math.ceil(container.clientHeight / itemHeight);
        const totalHeight = items.length * itemHeight;
        
        let scrollTop = 0;
        let startIndex = 0;

        // 创建占位元素
        const spacer = document.createElement('div');
        spacer.style.height = `${totalHeight}px`;
        container.appendChild(spacer);

        // 渲染可见项
        function render() {
            startIndex = Math.floor(scrollTop / itemHeight);
            const endIndex = Math.min(startIndex + visibleCount + 1, items.length);
            
            // 清空容器
            container.innerHTML = '';
            container.appendChild(spacer);

            // 渲染可见项
            for (let i = startIndex; i < endIndex; i++) {
                const item = document.createElement('div');
                item.className = 'virtual-item';
                item.style.position = 'absolute';
                item.style.top = `${i * itemHeight}px`;
                item.style.height = `${itemHeight}px`;
                item.textContent = items[i];
                container.appendChild(item);
            }
        }

        // 监听滚动
        container.addEventListener('scroll', () => {
            scrollTop = container.scrollTop;
            requestAnimationFrame(render);
        });

        render();
    }
}

// 使用示例
const items = Array.from({ length: 100 }, (_, i) => ({
    text: `Item ${i}`,
    color: `hsl(${i * 3.6}, 70%, 50%)`
}));

// 批量DOM操作
DOMOptimizer.efficientDOMUpdate(items);

// 读写分离
DOMOptimizer.readWriteSeparation();

// 使用transform
const box = document.querySelector('.box');
DOMOptimizer.useTransform(box, 100, 200);
```

### 2.2 requestAnimationFrame优化

#### 示例6：动画优化
```javascript
/**
 * 动画性能优化
 * @author erik.zhou
 */
class AnimationOptimizer {
    constructor() {
        this.animations = new Map();
        this.rafId = null;
    }

    /**
     * 添加动画
     * @param {string} id - 动画ID
     * @param {Function} callback - 动画回调
     */
    addAnimation(id, callback) {
        this.animations.set(id, callback);
        
        if (!this.rafId) {
            this.start();
        }
    }

    /**
     * 移除动画
     * @param {string} id - 动画ID
     */
    removeAnimation(id) {
        this.animations.delete(id);
        
        if (this.animations.size === 0) {
            this.stop();
        }
    }

    /**
     * 开始动画循环
     */
    start() {
        const animate = (timestamp) => {
            this.animations.forEach((callback, id) => {
                try {
                    callback(timestamp);
                } catch (error) {
                    console.error(`Animation ${id} failed:`, error);
                    this.removeAnimation(id);
                }
            });

            if (this.animations.size > 0) {
                this.rafId = requestAnimationFrame(animate);
            }
        };

        this.rafId = requestAnimationFrame(animate);
    }

    /**
     * 停止动画循环
     */
    stop() {
        if (this.rafId) {
            cancelAnimationFrame(this.rafId);
            this.rafId = null;
        }
    }

    /**
     * 平滑滚动
     * @param {HTMLElement} element - 目标元素
     * @param {number} targetY - 目标位置
     * @param {number} duration - 持续时间
     */
    static smoothScroll(element, targetY, duration = 500) {
        const startY = element.scrollTop;
        const distance = targetY - startY;
        const startTime = performance.now();

        function easeInOutQuad(t) {
            return t < 0.5 ? 2 * t * t : -1 + (4 - 2 * t) * t;
        }

        function animate(currentTime) {
            const elapsed = currentTime - startTime;
            const progress = Math.min(elapsed / duration, 1);
            const eased = easeInOutQuad(progress);

            element.scrollTop = startY + distance * eased;

            if (progress < 1) {
                requestAnimationFrame(animate);
            }
        }

        requestAnimationFrame(animate);
    }

    /**
     * 数字动画
     * @param {HTMLElement} element - 目标元素
     * @param {number} start - 起始值
     * @param {number} end - 结束值
     * @param {number} duration - 持续时间
     */
    static animateNumber(element, start, end, duration = 1000) {
        const startTime = performance.now();
        const difference = end - start;

        function update(currentTime) {
            const elapsed = currentTime - startTime;
            const progress = Math.min(elapsed / duration, 1);
            
            const current = start + difference * progress;
            element.textContent = Math.round(current);

            if (progress < 1) {
                requestAnimationFrame(update);
            }
        }

        requestAnimationFrame(update);
    }

    /**
     * 视差滚动
     * @param {HTMLElement} element - 视差元素
     * @param {number} speed - 速度系数
     */
    static parallaxScroll(element, speed = 0.5) {
        let ticking = false;

        function update() {
            const scrollY = window.scrollY;
            const offset = scrollY * speed;
            element.style.transform = `translateY(${offset}px)`;
            ticking = false;
        }

        window.addEventListener('scroll', () => {
            if (!ticking) {
                requestAnimationFrame(update);
                ticking = true;
            }
        });
    }
}

// 使用示例
const optimizer = new AnimationOptimizer();

// 添加旋转动画
const rotateElement = document.querySelector('.rotate');
let rotation = 0;

optimizer.addAnimation('rotate', () => {
    rotation += 1;
    rotateElement.style.transform = `rotate(${rotation}deg)`;
});

// 平滑滚动
const scrollContainer = document.querySelector('.scroll-container');
AnimationOptimizer.smoothScroll(scrollContainer, 1000, 500);

// 数字动画
const counter = document.querySelector('.counter');
AnimationOptimizer.animateNumber(counter, 0, 1000, 2000);

// 视差滚动
const parallaxElement = document.querySelector('.parallax');
AnimationOptimizer.parallaxScroll(parallaxElement, 0.5);
```

/**
 * 内存泄漏检测与修复
 * @author erik.zhou
 */
class MemoryLeakPrevention {
    /**
     * 场景1：未清理的事件监听器
     */
    static eventListenerLeak() {
        // 问题代码
        class BadComponent {
            constructor() {
                this.handleClick = this.handleClick.bind(this);
                document.addEventListener('click', this.handleClick);
            }

            handleClick() {
                console.log('Clicked');
            }

            // 缺少清理方法
        }

        // 修复代码
        class GoodComponent {
            constructor() {
                this.handleClick = this.handleClick.bind(this);
                document.addEventListener('click', this.handleClick);
            }

            handleClick() {
                console.log('Clicked');
            }

            destroy() {
                // 清理事件监听器
                document.removeEventListener('click', this.handleClick);
            }
        }

        return { BadComponent, GoodComponent };
    }

    /**
     * 场景2：未清理的定时器
     */
    static timerLeak() {
        // 问题代码
        class BadTimer {
            constructor() {
                this.intervalId = setInterval(() => {
                    console.log('Tick');
                }, 1000);
            }

            // 缺少清理方法
        }

        // 修复代码
        class GoodTimer {
            constructor() {
                this.intervalId = setInterval(() => {
                    console.log('Tick');
                }, 1000);
            }

            destroy() {
                // 清理定时器
                if (this.intervalId) {
                    clearInterval(this.intervalId);
                    this.intervalId = null;
                }
            }
        }

        return { BadTimer, GoodTimer };
    }

    /**
     * 场景3：闭包引用
     */
    static closureLeak() {
        // 问题代码
        function createBadClosure() {
            const largeData = new Array(1000000).fill('data');
            
            return function() {
                // 闭包持有largeData引用
                console.log(largeData[0]);
            };
        }

        // 修复代码
        function createGoodClosure() {
            const largeData = new Array(1000000).fill('data');
            const firstItem = largeData[0]; // 只保存需要的数据
            
            return function() {
                console.log(firstItem);
            };
        }

        return { createBadClosure, createGoodClosure };
    }

    /**
     * 场景4：DOM引用
     */
    static domReferenceLeak() {
        // 问题代码
        class BadDOMManager {
            constructor() {
                this.elements = [];
            }

            addElement(id) {
                const el = document.getElementById(id);
                this.elements.push(el); // 持有DOM引用
            }

            // 即使DOM被移除，引用仍然存在
        }

        // 修复代码
        class GoodDOMManager {
            constructor() {
                this.elementIds = [];
            }

            addElement(id) {
                this.elementIds.push(id); // 只保存ID
            }

            getElement(id) {
                return document.getElementById(id); // 动态获取
            }

            destroy() {
                this.elementIds = [];
            }
        }

                    return null;

                    duration: last.timestamp - first.timestamp,
                    initialSize: first.usedJSHeapSize,
                    finalSize: last.usedJSHeapSize,
                    growth,
                    growthRate: growth / (last.timestamp - first.timestamp) * 1000,
                    measurements: this.measurements
                };
            },

                this.intervalId = setInterval(() => {

            },

            stopMonitoring() {

                    this.intervalId = null;

        this.reset = reset;

     * 获取对象
     * @returns {Object}
     */
    acquire() {
        let obj;

        if (this.pool.length > 0) {
            obj = this.pool.pop();
        } else {
            obj = this.factory();
        }

        this.active.add(obj);
        return obj;
    }

    /**
     * 释放对象
     * @param {Object} obj - 要释放的对象
     */
    release(obj) {
        if (!this.active.has(obj)) {
            console.warn('Object not from this pool');
            return;
        }

        this.active.delete(obj);
        this.reset(obj);
        this.pool.push(obj);
    }

    /**
     * 清空对象池
     */
    clear() {
        this.pool = [];
        this.active.clear();
    }

    /**
     * 获取统计信息
     */
    getStats() {
        return {
            poolSize: this.pool.length,
            activeCount: this.active.size,
            totalCount: this.pool.length + this.active.size
        };
    }
}

        easeOutCubic: t => (--t) * t * t + 1,

/**
 * 事件委托工具
 * @author erik.zhou
 */
class EventDelegation {
    constructor(container) {
        this.container = container;
        this.handlers = new Map();
    }

    /**
     * 添加委托事件
     * @param {string} eventType - 事件类型
     * @param {string} selector - 选择器
     * @param {Function} handler - 处理函数
     */
    on(eventType, selector, handler) {
        if (!this.handlers.has(eventType)) {
            this.handlers.set(eventType, []);
            
            // 添加容器级别的监听器
            this.container.addEventListener(eventType, (e) => {
                this.handleEvent(e, eventType);
            });
        }

        this.handlers.get(eventType).push({ selector, handler });
    }

    /**
     * 移除委托事件
     * @param {string} eventType - 事件类型
     * @param {string} selector - 选择器
     * @param {Function} handler - 处理函数
     */
    off(eventType, selector, handler) {
        const handlers = this.handlers.get(eventType);
        if (!handlers) {
            return;
        }

        const index = handlers.findIndex(
            h => h.selector === selector && h.handler === handler
        );

        if (index !== -1) {
            handlers.splice(index, 1);
        }
    }

    /**
     * 处理事件
     * @param {Event} event - 事件对象
     * @param {string} eventType - 事件类型
     */
    handleEvent(event, eventType) {
        const handlers = this.handlers.get(eventType);
        if (!handlers) {
            return;
        }

        let target = event.target;

        // 向上遍历DOM树
        while (target && target !== this.container) {
            handlers.forEach(({ selector, handler }) => {
                if (target.matches(selector)) {
                    handler.call(target, event);
                }
            });

            target = target.parentElement;
        }
    }

    /**
     * 销毁
     */
    destroy() {
        this.handlers.clear();
    }
}

/**
 * Worker线程池
 * @author erik.zhou
 */
class WorkerPool {
    constructor(scriptUrl, poolSize = 4) {
        this.scriptUrl = scriptUrl;
        this.poolSize = poolSize;
        this.workers = [];
        this.taskQueue = [];
        this.activeWorkers = new Set();

        this.init();
    }

    /**
     * 初始化线程池
     */
    init() {
        for (let i = 0; i < this.poolSize; i++) {
            const worker = new Worker(this.scriptUrl);
            worker.id = i;
            this.workers.push(worker);
        }
    }

    /**
     * 执行任务
     * @param {Object} data - 任务数据
     * @returns {Promise<any>}
     */
    execute(data) {
        return new Promise((resolve, reject) => {
            const task = { data, resolve, reject };

            // 查找空闲Worker
            const availableWorker = this.workers.find(
                w => !this.activeWorkers.has(w)
            );

            if (availableWorker) {
                this.runTask(availableWorker, task);
            } else {
                // 加入队列
                this.taskQueue.push(task);
            }
        });
    }

    /**
     * 运行任务
     * @param {Worker} worker - Worker实例
     * @param {Object} task - 任务对象
     */
    runTask(worker, task) {
        this.activeWorkers.add(worker);

        const handleMessage = (e) => {
            worker.removeEventListener('message', handleMessage);
            worker.removeEventListener('error', handleError);

            this.activeWorkers.delete(worker);

            if (e.data.error) {
                task.reject(new Error(e.data.error));
            } else {
                task.resolve(e.data.result);
            }

            // 处理队列中的下一个任务
            this.processQueue();
        };

        const handleError = (error) => {
            worker.removeEventListener('message', handleMessage);
            worker.removeEventListener('error', handleError);

            this.activeWorkers.delete(worker);
            task.reject(error);

            this.processQueue();
        };

        worker.addEventListener('message', handleMessage);
        worker.addEventListener('error', handleError);

        worker.postMessage(task.data);
    }

    /**
     * 处理队列
     */
    processQueue() {
        if (this.taskQueue.length === 0) {
            return;
        }

        const availableWorker = this.workers.find(
            w => !this.activeWorkers.has(w)
        );

        if (availableWorker) {
            const task = this.taskQueue.shift();
            this.runTask(availableWorker, task);
        }
    }

    /**
     * 销毁线程池
     */
    destroy() {
        this.workers.forEach(worker => worker.terminate());
        this.workers = [];
        this.taskQueue = [];
        this.activeWorkers.clear();
    }

    /**
     * 获取统计信息
     */
    getStats() {
        return {
            poolSize: this.poolSize,
            activeWorkers: this.activeWorkers.size,
            queuedTasks: this.taskQueue.length
        };
    }
}

// 使用示例
const pool = new WorkerPool('/workers/compute-worker.js', 4);

// 批量执行任务
const tasks = Array.from({ length: 100 }, (_, i) => ({
    id: i,
    iterations: 1000000
}));

Promise.all(
    tasks.map(task => pool.execute(task))
).then(results => {
    console.log('All tasks completed:', results);
    console.log('Pool stats:', pool.getStats());
}).catch(error => {
    console.error('Task execution failed:', error);
});

/**
 * 空闲任务调度器
 * @author erik.zhou
 */
class IdleTaskScheduler {
    constructor() {
        this.tasks = [];
        this.isRunning = false;
        this.deadline = null;
    }

    /**
     * 添加任务
     * @param {Function} task - 任务函数
     * @param {Object} options - 选项
     */
    addTask(task, options = {}) {
        this.tasks.push({
            task,
            priority: options.priority || 0,
            timeout: options.timeout
        });

        // 按优先级排序
        this.tasks.sort((a, b) => b.priority - a.priority);

        if (!this.isRunning) {
            this.schedule();
        }
    }

    /**
     * 调度任务
     */
    schedule() {
        if (this.tasks.length === 0) {
            this.isRunning = false;
            return;
        }

        this.isRunning = true;

        if ('requestIdleCallback' in window) {
            requestIdleCallback((deadline) => {
                this.processTasks(deadline);
            }, { timeout: 1000 });
        } else {
            // 降级方案
            setTimeout(() => {
                this.processTasks({
                    timeRemaining: () => 16,
                    didTimeout: false
                });
            }, 0);
        }
    }

    /**
     * 处理任务
     * @param {IdleDeadline} deadline - 空闲截止时间
     */
    processTasks(deadline) {
        while (
            this.tasks.length > 0 &&
            (deadline.timeRemaining() > 0 || deadline.didTimeout)
        ) {
            const { task } = this.tasks.shift();

            try {
                task();
            } catch (error) {
                console.error('Idle task failed:', error);
            }
        }

        // 继续调度
        this.schedule();
    }

    /**
     * 清空任务
     */
    clear() {
        this.tasks = [];
        this.isRunning = false;
    }
}

// 使用示例
const scheduler = new IdleTaskScheduler();

// 添加低优先级任务
scheduler.addTask(() => {
    console.log('Preloading images...');
    preloadImages(['/img1.jpg', '/img2.jpg']);
}, { priority: 1 });

scheduler.addTask(() => {
    console.log('Sending analytics...');
    sendAnalytics();
}, { priority: 2 });

scheduler.addTask(() => {
    console.log('Prefetching data...');
    prefetchData();
}, { priority: 3 });

        const img = new Image();

/**
 * 性能监控工具
 * @author erik.zhou
 */
class PerformanceMonitor {
    constructor() {
        this.metrics = {
            navigation: {},
            resources: [],
            marks: {},
            measures: {},
            longTasks: [],
            memory: []
        };

        this.init();
    }

    /**
     * 初始化监控
     */
    init() {
        this.observeNavigation();
        this.observeResources();
        this.observeLongTasks();
        this.observeMemory();
    }

    /**
     * 观察导航时间
     */
    observeNavigation() {
        if (!window.performance || !window.performance.timing) {
            return;
        }

        window.addEventListener('load', () => {
            const timing = performance.timing;
            const navigation = {
                // DNS查询时间
                dns: timing.domainLookupEnd - timing.domainLookupStart,
                // TCP连接时间
                tcp: timing.connectEnd - timing.connectStart,
                // 请求时间
                request: timing.responseStart - timing.requestStart,
                // 响应时间
                response: timing.responseEnd - timing.responseStart,
                // DOM解析时间
                domParse: timing.domInteractive - timing.domLoading,
                // DOM内容加载完成时间
                domContentLoaded: timing.domContentLoadedEventEnd - timing.navigationStart,
                // 页面完全加载时间
                load: timing.loadEventEnd - timing.navigationStart,
                // 首次渲染时间
                firstPaint: this.getFirstPaint(),
                // 首次内容渲染时间
                firstContentfulPaint: this.getFirstContentfulPaint()
            };

            this.metrics.navigation = navigation;
            console.log('Navigation metrics:', navigation);
        });
    }

    /**
     * 获取首次渲染时间
     */
    getFirstPaint() {
        const paint = performance.getEntriesByType('paint')
            .find(entry => entry.name === 'first-paint');
        return paint ? Math.round(paint.startTime) : null;
    }

    /**
     * 获取首次内容渲染时间
     */
    getFirstContentfulPaint() {
        const paint = performance.getEntriesByType('paint')
            .find(entry => entry.name === 'first-contentful-paint');
        return paint ? Math.round(paint.startTime) : null;
    }

    /**
     * 观察资源加载
     */
    observeResources() {
        if (!('PerformanceObserver' in window)) {
            return;
        }

        const observer = new PerformanceObserver((list) => {
            const entries = list.getEntries();
            entries.forEach(entry => {
                this.metrics.resources.push({
                    name: entry.name,
                    type: entry.initiatorType,
                    duration: Math.round(entry.duration),
                    size: entry.transferSize,
                    startTime: Math.round(entry.startTime)
                });
            });
        });

        observer.observe({ entryTypes: ['resource'] });
    }

    /**
     * 观察长任务
     */
    observeLongTasks() {
        if (!('PerformanceObserver' in window)) {
            return;
        }

        try {
            const observer = new PerformanceObserver((list) => {
                const entries = list.getEntries();
                entries.forEach(entry => {
                    this.metrics.longTasks.push({
                        duration: Math.round(entry.duration),
                        startTime: Math.round(entry.startTime)
                    });

                    console.warn('Long task detected:', {
                        duration: `${Math.round(entry.duration)}ms`,
                        startTime: `${Math.round(entry.startTime)}ms`
                    });
                });
            });

            observer.observe({ entryTypes: ['longtask'] });
        } catch (e) {
            // longtask可能不被支持
            console.log('Long task observation not supported');
        }
    }

    /**
     * 观察内存使用
     */
    observeMemory() {
        if (!performance.memory) {
            return;
        }

        setInterval(() => {
            this.metrics.memory.push({
                timestamp: Date.now(),
                usedJSHeapSize: performance.memory.usedJSHeapSize,
                totalJSHeapSize: performance.memory.totalJSHeapSize,
                jsHeapSizeLimit: performance.memory.jsHeapSizeLimit
            });
        }, 5000);
    }

    /**
     * 标记时间点
     * @param {string} name - 标记名称
     */
    mark(name) {
        performance.mark(name);
        this.metrics.marks[name] = performance.now();
    }

    /**
     * 测量时间段
     * @param {string} name - 测量名称
     * @param {string} startMark - 开始标记
     * @param {string} endMark - 结束标记
     */
    measure(name, startMark, endMark) {
        performance.measure(name, startMark, endMark);
        const measure = performance.getEntriesByName(name, 'measure')[0];
        this.metrics.measures[name] = Math.round(measure.duration);
    }

    /**
     * 获取报告
     */
    getReport() {
        return {
            navigation: this.metrics.navigation,
            resources: {
                count: this.metrics.resources.length,
                totalSize: this.metrics.resources.reduce((sum, r) => sum + (r.size || 0), 0),
                slowest: this.metrics.resources
                    .sort((a, b) => b.duration - a.duration)
                    .slice(0, 10)
            },
            longTasks: {
                count: this.metrics.longTasks.length,
                totalDuration: this.metrics.longTasks.reduce((sum, t) => sum + t.duration, 0)
            },
            memory: this.metrics.memory.length > 0 ? {
                current: this.metrics.memory[this.metrics.memory.length - 1],
                peak: this.metrics.memory.reduce((max, m) => 
                    m.usedJSHeapSize > max.usedJSHeapSize ? m : max
                )
            } : null,
            marks: this.metrics.marks,
            measures: this.metrics.measures
        };
    }

    /**
     * 导出数据
     */
    export() {
        const report = this.getReport();
        const blob = new Blob([JSON.stringify(report, null, 2)], {
            type: 'application/json'
        });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = `performance-report-${Date.now()}.json`;
        a.click();
        URL.revokeObjectURL(url);
    }
}

// 使用示例

// 标记关键时间点

// 应用初始化
initializeApp();

monitor.mark('app-init-end');
monitor.measure('app-init', 'app-init-start', 'app-init-end');

// 获取报告
setTimeout(() => {
    const report = monitor.getReport();
    console.log('Performance report:', report);
    
    // 导出报告
    monitor.export();
}, 5000);

    // 应用初始化逻辑
    console.log('App initialized');
}

/**
 * 性能优化检查器
 * @author erik.zhou
 */
class PerformanceChecker {
    constructor() {
        this.checks = [];
        this.results = [];
    }

    /**
     * 添加检查项
     * @param {string} name - 检查名称
     * @param {Function} check - 检查函数
     */
    addCheck(name, check) {
        this.checks.push({ name, check });
    }

    /**
     * 运行所有检查
     */
    async runAll() {
        this.results = [];

        for (const { name, check } of this.checks) {
            try {
                const result = await check();
                this.results.push({
                    name,
                    passed: result.passed,
                    message: result.message,
                    details: result.details
                });
            } catch (error) {
                this.results.push({
                    name,
                    passed: false,
                    message: error.message,
                    details: null
                });
            }
        }

        return this.results;
    }

    /**
     * 生成报告
     */
    generateReport() {
        const passed = this.results.filter(r => r.passed).length;
        const failed = this.results.length - passed;

        let report = `# Performance Check Report\n\n`;
        report += `Total: ${this.results.length} | Passed: ${passed} | Failed: ${failed}\n\n`;

        this.results.forEach(result => {
            const icon = result.passed ? '✅' : '❌';
            report += `${icon} ${result.name}\n`;
            report += `   ${result.message}\n`;
            if (result.details) {
                report += `   Details: ${JSON.stringify(result.details)}\n`;
            }
            report += '\n';
        });

        return report;
    }
}

// 创建检查器
const checker = new PerformanceChecker();

// 添加检查项

// 1. 检查资源大小
checker.addCheck('Resource Size', async () => {
    const resources = performance.getEntriesByType('resource');
    const totalSize = resources.reduce((sum, r) => sum + (r.transferSize || 0), 0);
    const sizeInMB = (totalSize / 1024 / 1024).toFixed(2);

    return {
        passed: totalSize < 2 * 1024 * 1024, // 2MB
        message: `Total resource size: ${sizeInMB}MB`,
        details: { totalSize, count: resources.length }
    };
});

// 2. 检查长任务
checker.addCheck('Long Tasks', async () => {
    return new Promise((resolve) => {
        const longTasks = [];
        
        if ('PerformanceObserver' in window) {
            try {
                const observer = new PerformanceObserver((list) => {
                    longTasks.push(...list.getEntries());
                });
                observer.observe({ entryTypes: ['longtask'] });

                setTimeout(() => {
                    observer.disconnect();
                    resolve({
                        passed: longTasks.length === 0,
                        message: `Found ${longTasks.length} long tasks`,
                        details: { count: longTasks.length }
                    });
                }, 5000);
            } catch (e) {
                resolve({
                    passed: true,
                    message: 'Long task observation not supported',
                    details: null
                });
            }
        } else {
            resolve({
                passed: true,
                message: 'PerformanceObserver not supported',
                details: null
            });
        }
    });
});

// 3. 检查内存使用
checker.addCheck('Memory Usage', async () => {
    if (!performance.memory) {
        return {
            passed: true,
            message: 'Memory API not available',
            details: null
        };
    }

    const used = performance.memory.usedJSHeapSize;
    const limit = performance.memory.jsHeapSizeLimit;
    const percentage = ((used / limit) * 100).toFixed(2);

    return {
        passed: used < limit * 0.9,
        message: `Memory usage: ${percentage}%`,
        details: {
            used: `${(used / 1024 / 1024).toFixed(2)}MB`,
            limit: `${(limit / 1024 / 1024).toFixed(2)}MB`
        }
    };
});

// 4. 检查FCP
checker.addCheck('First Contentful Paint', async () => {
    const fcp = performance.getEntriesByType('paint')
        .find(entry => entry.name === 'first-contentful-paint');

    if (!fcp) {
        return {
            passed: false,
            message: 'FCP not available',
            details: null
        };

    const fcpTime = Math.round(fcp.startTime);

    return {
        passed: fcpTime < 1800,
        message: `FCP: ${fcpTime}ms`,
        details: { fcp: fcpTime }
    };
});
