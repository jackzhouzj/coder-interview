#### 示例2：动态资源加载器
```javascript
/**
 * 资源加载管理器
 * @author erik.zhou
 */
class ResourceLoader {
    constructor() {
        this.loadedResources = new Set();
        this.loadingPromises = new Map();
    }

    /**
     * 加载脚本
     * @param {string} url - 脚本URL
     * @param {Object} options - 加载选项
     * @returns {Promise<void>}
     */
    loadScript(url, options = {}) {
        // 避免重复加载
        if (this.loadedResources.has(url)) {
            return Promise.resolve();
        }

        // 返回正在加载的Promise
        if (this.loadingPromises.has(url)) {
            return this.loadingPromises.get(url);
        }

        const promise = new Promise((resolve, reject) => {
            const script = document.createElement('script');
            script.src = url;
            script.async = options.async !== false;
            script.defer = options.defer || false;
            
            if (options.crossOrigin) {
                script.crossOrigin = options.crossOrigin;
            }

            script.onload = () => {
                this.loadedResources.add(url);
                this.loadingPromises.delete(url);
                resolve();
            };

            script.onerror = () => {
                this.loadingPromises.delete(url);
                reject(new Error(`Failed to load script: ${url}`));
            };

            document.head.appendChild(script);
        });

        this.loadingPromises.set(url, promise);
        return promise;
    }

    /**
     * 加载样式表
     * @param {string} url - 样式表URL
     * @returns {Promise<void>}
     */
    loadStylesheet(url) {
        if (this.loadedResources.has(url)) {
            return Promise.resolve();
        }

        if (this.loadingPromises.has(url)) {
            return this.loadingPromises.get(url);
        }

        const promise = new Promise((resolve, reject) => {
            const link = document.createElement('link');
            link.rel = 'stylesheet';
            link.href = url;

            link.onload = () => {
                this.loadedResources.add(url);
                this.loadingPromises.delete(url);
                resolve();
            };

            link.onerror = () => {
                this.loadingPromises.delete(url);
                reject(new Error(`Failed to load stylesheet: ${url}`));
            };

            document.head.appendChild(link);
        });

        this.loadingPromises.set(url, promise);
        return promise;
    }

    /**
     * 预加载资源
     * @param {string} url - 资源URL
     * @param {string} as - 资源类型
     * @returns {Promise<void>}
     */
    preload(url, as = 'fetch') {
        return new Promise((resolve, reject) => {
            const link = document.createElement('link');
            link.rel = 'preload';
            link.href = url;
            link.as = as;

            if (as === 'font') {
                link.crossOrigin = 'anonymous';
            }

            link.onload = resolve;
            link.onerror = reject;

            document.head.appendChild(link);
        });
    }

    /**
     * 批量加载资源
     * @param {Array} resources - 资源列表
     * @returns {Promise<void[]>}
     */
    async loadBatch(resources) {
        const promises = resources.map(resource => {
            if (resource.type === 'script') {
                return this.loadScript(resource.url, resource.options);
            } else if (resource.type === 'style') {
                return this.loadStylesheet(resource.url);
            }
            return Promise.resolve();
        });

        return Promise.all(promises);
    }
}

// 使用示例
const loader = new ResourceLoader();

// 加载单个脚本
loader.loadScript('/js/analytics.js', { async: true })
    .then(() => console.log('Analytics loaded'))
    .catch(err => console.error(err));

// 批量加载资源
loader.loadBatch([
    { type: 'script', url: '/js/vendor.js' },
    { type: 'script', url: '/js/app.js' },
    { type: 'style', url: '/css/theme.css' }
]).then(() => {
    console.log('All resources loaded');
});
```

```javascript
/**
 * 资源优先级管理器
 * @author erik.zhou
 */
class ResourcePriorityManager {
    constructor() {
        this.priorities = {
            critical: [],
            high: [],
            medium: [],
            low: []
        };
    }

    /**
     * 添加资源到优先级队列
     * @param {string} url - 资源URL
     * @param {string} priority - 优先级
     * @param {string} type - 资源类型
     */
    addResource(url, priority = 'medium', type = 'fetch') {
        if (!this.priorities[priority]) {
            console.warn(`Invalid priority: ${priority}`);
            priority = 'medium';
        }

        this.priorities[priority].push({ url, type });
    }

    /**
     * 按优先级加载资源
     * @returns {Promise<void>}
     */
    async loadByPriority() {
        const loader = new ResourceLoader();

        // 1. 加载关键资源
        await this.loadPriorityGroup('critical', loader);

        // 2. 加载高优先级资源
        await this.loadPriorityGroup('high', loader);

        // 3. 延迟加载中优先级资源
        requestIdleCallback(() => {
            this.loadPriorityGroup('medium', loader);
        });

        // 4. 延迟加载低优先级资源
        requestIdleCallback(() => {
            this.loadPriorityGroup('low', loader);
        }, { timeout: 2000 });
    }

    /**
     * 加载优先级组
     * @param {string} priority - 优先级
     * @param {ResourceLoader} loader - 加载器实例
     * @returns {Promise<void>}
     */
    async loadPriorityGroup(priority, loader) {
        const resources = this.priorities[priority];
        if (resources.length === 0) {
            return;
        }

        console.log(`Loading ${priority} priority resources:`, resources);

        const promises = resources.map(resource => {
            if (resource.type === 'script') {
                return loader.loadScript(resource.url);
            } else if (resource.type === 'style') {
                return loader.loadStylesheet(resource.url);
            } else {
                return loader.preload(resource.url, resource.type);
            }
        });

        await Promise.all(promises);
    }

    /**
     * 预加载下一页资源
     * @param {string} nextPageUrl - 下一页URL
     */
    prefetchNextPage(nextPageUrl) {
        const link = document.createElement('link');
        link.rel = 'prefetch';
        link.href = nextPageUrl;
        document.head.appendChild(link);
    }

    /**
     * 智能预加载（基于用户行为）
     * @param {HTMLElement} element - 触发元素
     * @param {string} url - 资源URL
     */
    smartPrefetch(element, url) {
        let prefetchTimer = null;

        element.addEventListener('mouseenter', () => {
            // 鼠标悬停300ms后开始预加载
            prefetchTimer = setTimeout(() => {
                this.prefetchNextPage(url);
            }, 300);
        });

        element.addEventListener('mouseleave', () => {
            // 鼠标离开时取消预加载
            if (prefetchTimer) {
                clearTimeout(prefetchTimer);
                prefetchTimer = null;
            }
        });
    }
}

// 使用示例
const priorityManager = new ResourcePriorityManager();

// 添加关键资源
priorityManager.addResource('/js/critical.js', 'critical', 'script');
priorityManager.addResource('/css/critical.css', 'critical', 'style');

// 添加高优先级资源
priorityManager.addResource('/js/main.js', 'high', 'script');
priorityManager.addResource('/fonts/main.woff2', 'high', 'font');

// 添加中优先级资源
priorityManager.addResource('/js/analytics.js', 'medium', 'script');

// 添加低优先级资源
priorityManager.addResource('/js/chat-widget.js', 'low', 'script');

// 按优先级加载
priorityManager.loadByPriority();

// 智能预加载
const nextPageLink = document.querySelector('a[href="/next-page"]');
if (nextPageLink) {
    priorityManager.smartPrefetch(nextPageLink, '/next-page.html');
}
```

```javascript
/**

import React, { Suspense, lazy } from 'react';
import { BrowserRouter, Routes, Route } from 'react-router-dom';

// 懒加载组件
const Home = lazy(() => import('./pages/Home'));
const About = lazy(() => import('./pages/About'));
const Products = lazy(() => import('./pages/Products'));
const ProductDetail = lazy(() => import('./pages/ProductDetail'));
const Dashboard = lazy(() => import('./pages/Dashboard'));

// 加载中组件
const LoadingFallback = () => (
    <div className="loading-container">
        <div className="spinner"></div>
        <p>加载中...</p>
    </div>
);

// 错误边界组件
class ErrorBoundary extends React.Component {
    constructor(props) {
        super(props);
        this.state = { hasError: false, error: null };
    }

    static getDerivedStateFromError(error) {
        return { hasError: true, error };
    }

    componentDidCatch(error, errorInfo) {
        console.error('Route loading error:', error, errorInfo);
    }

    render() {
        if (this.state.hasError) {
            return (
                <div className="error-container">
                    <h2>页面加载失败</h2>
                    <p>{this.state.error?.message}</p>
                    <button onClick={() => window.location.reload()}>
                        重新加载
                    </button>
                </div>
            );
        }

        return this.props.children;
    }
}

// 路由配置
function App() {
    return (
        <BrowserRouter>
            <ErrorBoundary>
                <Suspense fallback={<LoadingFallback />}>
                    <Routes>
                        <Route path="/" element={<Home />} />
                        <Route path="/about" element={<About />} />
                        <Route path="/products" element={<Products />} />
                        <Route path="/products/:id" element={<ProductDetail />} />
                        <Route path="/dashboard/*" element={<Dashboard />} />
                    </Routes>
                </Suspense>
            </ErrorBoundary>
        </BrowserRouter>
    );
}

export default App;
```

```javascript
// 带重试机制的懒加载
function lazyWithRetry(componentImport, retries = 3, interval = 1000) {
    return lazy(() => {
        return new Promise((resolve, reject) => {
            const attemptImport = (retriesLeft) => {
                componentImport()
                    .then(resolve)
                    .catch((error) => {
                        if (retriesLeft === 0) {
                            reject(error);
                            return;
                        }
                        
                        console.log(`Import failed, retrying... (${retriesLeft} attempts left)`);
                        setTimeout(() => {
                            attemptImport(retriesLeft - 1);
                        }, interval);
                    });
            };
            
            attemptImport(retries);
        });
    });
}

// 使用带重试的懒加载
const ReliableChart = lazyWithRetry(() => import('./components/Chart'));

    const [showChart, setShowChart] = useState(false);
```

```javascript
async function loadPolyfills() {
    const polyfills = [];

    // 检查IntersectionObserver支持

        polyfills.push(
            import('intersection-observer')
        );
    }

    // 检查ResizeObserver支持

        polyfills.push(
            import('@juggle/resize-observer')
        );
    }

    // 检查Intl.RelativeTimeFormat支持
    if (!('RelativeTimeFormat' in Intl)) {
        polyfills.push(
            import('@formatjs/intl-relativetimeformat/polyfill')
        );
    }
```

#### 示例10：智能预加载管理器
```javascript
/**
 * 智能预加载管理器
 * @author erik.zhou
 */
class PreloadManager {
    constructor() {
        this.preloadedResources = new Set();
        this.observer = null;
        this.initIntersectionObserver();
    }

    /**
     * 初始化Intersection Observer
     */
    initIntersectionObserver() {
        if (!('IntersectionObserver' in window)) {
            return;
        }

        this.observer = new IntersectionObserver(
            (entries) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        const url = entry.target.dataset.preload;
                        const type = entry.target.dataset.preloadType || 'fetch';
                        
                        if (url) {
                            this.preload(url, type);
                            this.observer.unobserve(entry.target);
                        }
                    }
                });
            },
            {
                rootMargin: '50px'
            }
        );
    }

    /**
     * 预加载资源
     * @param {string} url - 资源URL
     * @param {string} as - 资源类型
     * @param {Object} options - 选项
     */
    preload(url, as = 'fetch', options = {}) {
        if (this.preloadedResources.has(url)) {
            return;
        }

        const link = document.createElement('link');
        link.rel = 'preload';
        link.href = url;
        link.as = as;

        if (options.crossOrigin) {
            link.crossOrigin = options.crossOrigin;
        }

        if (options.type) {
            link.type = options.type;
        }

        if (options.media) {
            link.media = options.media;
        }

        document.head.appendChild(link);
        this.preloadedResources.add(url);

        console.log(`Preloaded: ${url} (${as})`);
    }

    /**
     * 预获取资源
     * @param {string} url - 资源URL
     */
    prefetch(url) {
        if (this.preloadedResources.has(url)) {
            return;
        }

        const link = document.createElement('link');
        link.rel = 'prefetch';
        link.href = url;

        document.head.appendChild(link);
        this.preloadedResources.add(url);

        console.log(`Prefetched: ${url}`);
    }

    /**
     * 预连接到域名
     * @param {string} url - 域名URL
     * @param {boolean} crossOrigin - 是否跨域
     */
    preconnect(url, crossOrigin = false) {
        const link = document.createElement('link');
        link.rel = 'preconnect';
        link.href = url;

        if (crossOrigin) {
            link.crossOrigin = 'anonymous';
        }

        document.head.appendChild(link);
        console.log(`Preconnected: ${url}`);
    }

    /**
     * DNS预解析
     * @param {string} url - 域名URL
     */
    dnsPrefetch(url) {
        const link = document.createElement('link');
        link.rel = 'dns-prefetch';
        link.href = url;

        document.head.appendChild(link);
        console.log(`DNS prefetched: ${url}`);
    }

    /**
     * 观察元素进行预加载
     * @param {HTMLElement} element - 要观察的元素
     */
    observeElement(element) {
        if (this.observer) {
            this.observer.observe(element);
        }
    }

    /**
     * 基于网络状态的智能预加载
     * @param {string} url - 资源URL
     * @param {string} as - 资源类型
     */
    smartPreload(url, as = 'fetch') {
        // 检查网络连接
        if (!navigator.onLine) {
            console.log('Offline, skipping preload');
            return;
        }

        // 检查网络类型
        const connection = navigator.connection || navigator.mozConnection || navigator.webkitConnection;
        
        if (connection) {
            const effectiveType = connection.effectiveType;
            const saveData = connection.saveData;

            // 如果用户启用了数据节省模式，跳过预加载
            if (saveData) {
                console.log('Save-Data enabled, skipping preload');
                return;
            }

            // 根据网络类型决定是否预加载
            if (effectiveType === 'slow-2g' || effectiveType === '2g') {
                console.log('Slow connection, skipping preload');
                return;
            }

            // 4G网络使用preload，3G使用prefetch
            if (effectiveType === '4g') {
                this.preload(url, as);
            } else {
                this.prefetch(url);
            }
        } else {
            // 无法检测网络类型，使用prefetch
            this.prefetch(url);
        }
    }

    /**
     * 批量预加载
     * @param {Array} resources - 资源列表
     */
    batchPreload(resources) {
        // 使用requestIdleCallback避免阻塞主线程
        if ('requestIdleCallback' in window) {
            requestIdleCallback(() => {
                resources.forEach(resource => {
                    this.smartPreload(resource.url, resource.as);
                });
            });
        } else {
            setTimeout(() => {
                resources.forEach(resource => {
                    this.smartPreload(resource.url, resource.as);
                });
            }, 1000);
        }
    }
}
```

```javascript
/**
 * 图片预加载器
 * @author erik.zhou
 */
class ImagePreloader {
    constructor() {
        this.loadedImages = new Map();
        this.loadingPromises = new Map();
    }

    /**
     * 预加载单张图片
     * @param {string} url - 图片URL
     * @returns {Promise<HTMLImageElement>}
     */
    preloadImage(url) {
        // 如果已加载，直接返回
        if (this.loadedImages.has(url)) {
            return Promise.resolve(this.loadedImages.get(url));
        }

        // 如果正在加载，返回现有Promise
        if (this.loadingPromises.has(url)) {
            return this.loadingPromises.get(url);
        }

        const promise = new Promise((resolve, reject) => {
            const img = new Image();
            
            img.onload = () => {
                this.loadedImages.set(url, img);
                this.loadingPromises.delete(url);
                resolve(img);
            };

            img.onerror = () => {
                this.loadingPromises.delete(url);
                reject(new Error(`Failed to load image: ${url}`));
            };

            img.src = url;
        });

        this.loadingPromises.set(url, promise);
        return promise;
    }

    /**
     * 批量预加载图片
     * @param {Array<string>} urls - 图片URL数组
     * @returns {Promise<Array<HTMLImageElement>>}
     */
    async preloadImages(urls) {
        const promises = urls.map(url => this.preloadImage(url));
        return Promise.all(promises);
    }

    /**
     * 带进度的批量预加载
     * @param {Array<string>} urls - 图片URL数组
     * @param {Function} onProgress - 进度回调
     * @returns {Promise<Array<HTMLImageElement>>}
     */
    async preloadImagesWithProgress(urls, onProgress) {
        let loaded = 0;
        const total = urls.length;

        const promises = urls.map(async (url) => {
            try {
                const img = await this.preloadImage(url);
                loaded++;
                onProgress && onProgress({
                    loaded,
                    total,
                    percentage: Math.round((loaded / total) * 100),
                    url
                });
                return img;
            } catch (error) {
                loaded++;
                onProgress && onProgress({
                    loaded,
                    total,
                    percentage: Math.round((loaded / total) * 100),
                    url,
                    error
                });
                throw error;
            }
        });

        return Promise.allSettled(promises);
    }

    /**
     * 响应式图片预加载
     * @param {Object} srcset - 响应式图片源
     * @returns {Promise<HTMLImageElement>}
     */
    preloadResponsiveImage(srcset) {
        const { src, srcset: srcsetStr, sizes } = srcset;
        
        return new Promise((resolve, reject) => {
            const img = new Image();
            
            if (srcsetStr) {
                img.srcset = srcsetStr;
            }
            
            if (sizes) {
                img.sizes = sizes;
            }
            
            img.onload = () => resolve(img);
            img.onerror = () => reject(new Error(`Failed to load responsive image`));
            
            img.src = src;
        });
    }

    /**
     * 懒加载图片预加载
     * @param {HTMLImageElement} img - 图片元素
     * @param {Object} options - 选项
     */
    lazyPreload(img, options = {}) {
        const {
            rootMargin = '50px',
            threshold = 0.01
        } = options;

        if (!('IntersectionObserver' in window)) {
            // 降级处理：直接加载
            this.loadImage(img);
            return;
        }

        const observer = new IntersectionObserver(
            (entries) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        this.loadImage(img);
                        observer.unobserve(img);
                    }
                });
            },
            { rootMargin, threshold }
        );

        observer.observe(img);
    }

    /**
     * 加载图片
     * @param {HTMLImageElement} img - 图片元素
     */
    loadImage(img) {
        const src = img.dataset.src;
        const srcset = img.dataset.srcset;
        const sizes = img.dataset.sizes;

        if (srcset) {
            img.srcset = srcset;
        }

        if (sizes) {
            img.sizes = sizes;
        }

        if (src) {
            img.src = src;
        }

        // 移除data属性
        delete img.dataset.src;
        delete img.dataset.srcset;
        delete img.dataset.sizes;
    }

    /**
     * 清除缓存
     */
    clearCache() {
        this.loadedImages.clear();
        this.loadingPromises.clear();
    }
}
```

```javascript
class ImageOptimizer {
    constructor(options = {}) {
        this.quality = options.quality || 0.8;
        this.maxWidth = options.maxWidth || 1920;
        this.maxHeight = options.maxHeight || 1080;
        this.formats = options.formats || ['webp', 'jpeg'];
    }

    /**
     * 压缩图片
     * @param {File} file - 图片文件
     * @param {Object} options - 压缩选项
     * @returns {Promise<Blob>}
     */
    async compressImage(file, options = {}) {
        const quality = options.quality || this.quality;
        const maxWidth = options.maxWidth || this.maxWidth;
        const maxHeight = options.maxHeight || this.maxHeight;

        return new Promise((resolve, reject) => {
            const reader = new FileReader();
            
            reader.onload = (e) => {
                const img = new Image();
                
                img.onload = () => {
                    const canvas = document.createElement('canvas');
                    const ctx = canvas.getContext('2d');

                    // 计算缩放比例
                    let width = img.width;
                    let height = img.height;

                    if (width > maxWidth) {
                        height = (height * maxWidth) / width;
                        width = maxWidth;
                    }

                    if (height > maxHeight) {
                        width = (width * maxHeight) / height;
                        height = maxHeight;
                    }

                    canvas.width = width;
                    canvas.height = height;

                    // 绘制图片
                    ctx.drawImage(img, 0, 0, width, height);

                    // 转换为Blob
                    canvas.toBlob(
                        (blob) => {
                            if (blob) {
                                resolve(blob);
                            } else {
                                reject(new Error('Failed to compress image'));
                            }
                        },
                        'image/jpeg',
                        quality
                    );
                };

                img.onerror = () => reject(new Error('Failed to load image'));
                img.src = e.target.result;
            };

            reader.onerror = () => reject(new Error('Failed to read file'));
            reader.readAsDataURL(file);
        });

                            resolve(newBlob);
                        } else {
                            reject(new Error('Failed to convert format'));
                        }
                    },
                    `image/${format}`,
                    this.quality
                );
            };

            img.onerror = () => {
                URL.revokeObjectURL(url);
                reject(new Error('Failed to load image'));
            };

            img.src = url;
        });
    }

     * 生成响应式图片

    async generateResponsiveImages(file, widths = [320, 640, 1024, 1920]) {

                });
            } catch (error) {

            }
        }

    }
```

```javascript
/**
 * 图片懒加载实现
 * @author erik.zhou
 */
class LazyImageLoader {
    constructor(options = {}) {
        this.rootMargin = options.rootMargin || '50px';
        this.threshold = options.threshold || 0.01;
        this.loadingClass = options.loadingClass || 'lazy-loading';
        this.loadedClass = options.loadedClass || 'lazy-loaded';
        this.errorClass = options.errorClass || 'lazy-error';
        
        this.observer = null;
        this.images = new Set();
        
        this.init();
    }

    /**
     * 初始化
     */
    init() {
        // 检查原生懒加载支持
        if ('loading' in HTMLImageElement.prototype) {
            this.useNativeLazyLoading();
        } else {
            this.useIntersectionObserver();
        }
    }

    /**
     * 使用原生懒加载
     */
    useNativeLazyLoading() {
        document.querySelectorAll('img[data-src]').forEach(img => {
            img.loading = 'lazy';
            img.src = img.dataset.src;
            
            if (img.dataset.srcset) {
                img.srcset = img.dataset.srcset;
            }
            
            delete img.dataset.src;
            delete img.dataset.srcset;
        });
    }

    /**
     * 使用Intersection Observer
     */
    useIntersectionObserver() {
        if (!('IntersectionObserver' in window)) {
            // 降级：直接加载所有图片
            this.loadAllImages();
            return;
        }

        this.observer = new IntersectionObserver(
            (entries) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        this.loadImage(entry.target);
                    }
                });
            },
            {
                rootMargin: this.rootMargin,
                threshold: this.threshold
            }
        );

        this.observeImages();
    }

    /**
     * 观察图片
     */
    observeImages() {
        document.querySelectorAll('img[data-src]').forEach(img => {
            this.images.add(img);
            this.observer.observe(img);
        });
    }

    /**
     * 加载图片
     * @param {HTMLImageElement} img - 图片元素
     */
    loadImage(img) {
        img.classList.add(this.loadingClass);

        const src = img.dataset.src;
        const srcset = img.dataset.srcset;
        const sizes = img.dataset.sizes;

        // 创建临时图片对象进行预加载
        const tempImg = new Image();

        tempImg.onload = () => {
            // 设置实际图片
            if (srcset) {
                img.srcset = srcset;
            }
            if (sizes) {
                img.sizes = sizes;
            }
            img.src = src;

            // 更新状态
            img.classList.remove(this.loadingClass);
            img.classList.add(this.loadedClass);

            // 移除data属性
            delete img.dataset.src;
```

```javascript
class FontLoader {
    constructor() {
        this.loadedFonts = new Set();
        this.loadingPromises = new Map();
    }

    /**
     * 加载字体
     * @param {string} fontFamily - 字体名称
     * @param {Object} options - 字体选项
     * @returns {Promise<FontFace>}
     */
    async loadFont(fontFamily, options = {}) {
        const {
            url,
            weight = '400',
            style = 'normal',
            display = 'swap',
            unicodeRange
        } = options;

        const fontKey = `${fontFamily}-${weight}-${style}`;

        // 避免重复加载
        if (this.loadedFonts.has(fontKey)) {
            return Promise.resolve();
        }

        if (this.loadingPromises.has(fontKey)) {
            return this.loadingPromises.get(fontKey);
        }

        // 使用Font Loading API
        if ('FontFace' in window) {
            const fontFace = new FontFace(fontFamily, `url(${url})`, {
                weight,
                style,
                display,
                unicodeRange
            });

            const promise = fontFace.load().then(loadedFace => {
                document.fonts.add(loadedFace);
                this.loadedFonts.add(fontKey);
                this.loadingPromises.delete(fontKey);
                console.log(`Font loaded: ${fontFamily}`);
                return loadedFace;
            }).catch(error => {
                this.loadingPromises.delete(fontKey);
                console.error(`Failed to load font: ${fontFamily}`, error);
                throw error;
            });

            this.loadingPromises.set(fontKey, promise);
            return promise;
        } else {
            // 降级方案：使用CSS
            return this.loadFontWithCSS(fontFamily, options);
        }
    }

    /**
     * 使用CSS加载字体（降级方案）
     * @param {string} fontFamily - 字体名称
     * @param {Object} options - 字体选项
     * @returns {Promise<void>}
     */
    loadFontWithCSS(fontFamily, options) {
        const { url, weight, style, display } = options;

        return new Promise((resolve, reject) => {
            const styleElement = document.createElement('style');
            styleElement.textContent = `
                @font-face {
                    font-family: '${fontFamily}';
                    src: url('${url}') format('woff2');
                    font-weight: ${weight};
                    font-style: ${style};
                    font-display: ${display};

                document.fonts.ready.then(() => {
                    resolve();
                });
            } else {
                // 简单延迟
                setTimeout(resolve, 1000);
            }
        });
    }

     * @param {Array} fonts - 字体配置数组
     * @returns {Promise<void>}

    async loadFonts(fonts) {
        const promises = fonts.map(font => 
            this.loadFont(font.family, font.options)
        );
        return Promise.all(promises);
    }

    /**
     * 检测字体是否已加载
     * @param {string} fontFamily - 字体名称
     * @param {string} weight - 字重
     * @returns {boolean}
     */
```

```javascript
const CACHE_VERSION = 'v1.0.0';
const CACHE_NAME = `app-cache-${CACHE_VERSION}`;

// 需要预缓存的资源
const PRECACHE_URLS = [
    '/',

    '/css/main.css',
    '/js/main.js',

    '/fonts/main.woff2'
];

self.addEventListener('install', (event) => {
    event.waitUntil(
        caches.open(CACHE_NAME).then((cache) => {
            console.log('Precaching resources');
            return cache.addAll(PRECACHE_URLS);
        }).then(() => {
            return self.skipWaiting();
        })
    );
});

// 激活事件：清理旧缓存
self.addEventListener('activate', (event) => {
    event.waitUntil(
        caches.keys().then((cacheNames) => {
            return Promise.all(
                cacheNames.map((cacheName) => {
                    if (cacheName !== CACHE_NAME) {
                        console.log('Deleting old cache:', cacheName);
                        return caches.delete(cacheName);
                    }
                })
            );
        }).then(() => {
            return self.clients.claim();
        })
    );
});

// Fetch事件：缓存策略
self.addEventListener('fetch', (event) => {
    const { request } = event;
    const url = new URL(request.url);

    // 根据资源类型选择缓存策略
    if (request.method !== 'GET') {
        // 非GET请求不缓存
        return;
    }

    if (url.pathname.startsWith('/api/')) {
        // API请求：网络优先
        event.respondWith(networkFirst(request));
    } else if (url.pathname.match(/\.(js|css|woff2?)$/)) {
        // 静态资源：缓存优先
        event.respondWith(cacheFirst(request));
    } else if (url.pathname.match(/\.(jpg|jpeg|png|gif|webp|svg)$/)) {
        // 图片：缓存优先，过期重新验证
        event.respondWith(staleWhileRevalidate(request));
    } else {
        // 其他请求：网络优先
        event.respondWith(networkFirst(request));
    }
});

/**
 * 缓存优先策略
 * @param {Request} request - 请求对象
 * @returns {Promise<Response>}
 */
async function cacheFirst(request) {
    const cache = await caches.open(CACHE_NAME);
    const cached = await cache.match(request);

    if (cached) {
        return cached;
    }

    try {
        const response = await fetch(request);
        if (response.ok) {
            cache.put(request, response.clone());
        }
        return response;
    } catch (error) {
        console.error('Fetch failed:', error);
        return new Response('Network error', { status: 408 });
    }
}

/**
 * 网络优先策略
 * @param {Request} request - 请求对象
 * @returns {Promise<Response>}
 */
async function networkFirst(request) {
    const cache = await caches.open(CACHE_NAME);

    try {
        const response = await fetch(request);
        if (response.ok) {
            cache.put(request, response.clone());
        }
        return response;
    } catch (error) {
        console.error('Network failed, trying cache:', error);
        const cached = await cache.match(request);
        if (cached) {
            return cached;
        }
        return new Response('Network error and no cache', { status: 408 });
    }
}

/**
 * 过期重新验证策略
 * @param {Request} request - 请求对象
 * @returns {Promise<Response>}
 */
async function staleWhileRevalidate(request) {
    const cache = await caches.open(CACHE_NAME);
    const cached = await cache.match(request);

    // 后台更新缓存
    const fetchPromise = fetch(request).then((response) => {
        if (response.ok) {
            cache.put(request, response.clone());
        }
        return response;
    });

    // 返回缓存或等待网络响应
    return cached || fetchPromise;
}
```

```javascript
class CDNManager {

            const url = `${this.cdnUrls[i]}${path}`;
            
            try {
                await this.tryLoadResource(url, type);
                console.log(`Loaded from CDN ${i + 1}:`, url);
                return;
            } catch (error) {
                console.warn(`CDN ${i + 1} failed:`, error.message);
                
                if (i === this.cdnUrls.length - 1) {
                    // 所有CDN都失败，使用降级方案
                    const fallbackUrl = `${this.fallbackUrl}${path}`;
                    await this.tryLoadResource(fallbackUrl, type);
                    console.log('Loaded from fallback:', fallbackUrl);
                }
            }
        }

     * 尝试加载资源
     * @param {string} url - 资源URL
     * @param {string} type - 资源类型
     * @returns {Promise<void>}
     */
    tryLoadResource(url, type) {
        return new Promise((resolve, reject) => {
            const timer = setTimeout(() => {
                reject(new Error('Timeout'));
            }, this.timeout);

            if (type === 'script') {
                const script = document.createElement('script');
                script.src = url;
                script.async = true;

                script.onload = () => {
                    clearTimeout(timer);
                    resolve();
                };

                script.onerror = () => {
                    clearTimeout(timer);
                    reject(new Error(`Failed to load script: ${url}`));
                };

                document.head.appendChild(script);
            } else if (type === 'style') {
                const link = document.createElement('link');
                link.rel = 'stylesheet';
                link.href = url;

                link.onload = () => {
                    clearTimeout(timer);
                    resolve();
                };

                link.onerror = () => {
                    clearTimeout(timer);
                    reject(new Error(`Failed to load stylesheet: ${url}`));
                };

                document.head.appendChild(link);
            }
        });
    }

    /**
     * 测试CDN速度
     * @returns {Promise<Array>}

        for (const cdnUrl of this.cdnUrls) {
            const url = `${cdnUrl}${testPath}`;
            const startTime = performance.now();

            try {
                await fetch(url, { method: 'HEAD' });
                const endTime = performance.now();
                const latency = endTime - startTime;

                    url: cdnUrl,
                    latency,
                    available: true
                });
            } catch (error) {
                results.push({
                    url: cdnUrl,
                    latency: Infinity,
                    available: false
                });
            }
        }

        // 按延迟排序
        results.sort((a, b) => a.latency - b.latency);
        return results;

            console.log('Fastest CDN:', fastest.url, `(${fastest.latency.toFixed(2)}ms)`);
            return fastest.url;
        }

        return this.fallbackUrl;
    }
}
```

```javascript
/**
 * 客户端解压缩工具
 * @author erik.zhou
 */
class CompressionUtils {
    /**
     * 检测Brotli支持
     * @returns {boolean}
     */
    static supportsBrotli() {
        return 'DecompressionStream' in window && 
               'CompressionStream' in window;
    }

    /**
     * 解压缩响应
     * @param {Response} response - Fetch响应
     * @returns {Promise<Response>}
     */
    static async decompressResponse(response) {
        const encoding = response.headers.get('Content-Encoding');

        if (!encoding || encoding === 'identity') {
            return response;
        }

        if (encoding === 'br' && this.supportsBrotli()) {
            const stream = response.body
                .pipeThrough(new DecompressionStream('deflate'));
            
            return new Response(stream, {
                headers: response.headers
            });
        }

        return response;
    }

    /**
     * 压缩数据
     * @param {string} data - 要压缩的数据
     * @param {string} format - 压缩格式
     * @returns {Promise<Blob>}
     */
    static async compressData(data, format = 'gzip') {
        if (!this.supportsBrotli()) {
            throw new Error('Compression not supported');
        }

        const encoder = new TextEncoder();
        const stream = new ReadableStream({
            start(controller) {
                controller.enqueue(encoder.encode(data));
                controller.close();
            }
        });

        const compressedStream = stream.pipeThrough(
            new CompressionStream(format)
        );

        const reader = compressedStream.getReader();
        const chunks = [];

        while (true) {
            const { done, value } = await reader.read();
            if (done) break;
            chunks.push(value);
        }

        return new Blob(chunks);
    }

    /**
     * 解压缩数据
     * @param {Blob} blob - 压缩的数据
     * @param {string} format - 压缩格式
     * @returns {Promise<string>}
     */
    static async decompressData(blob, format = 'gzip') {
        if (!this.supportsBrotli()) {
            throw new Error('Decompression not supported');
        }

        const stream = blob.stream().pipeThrough(
            new DecompressionStream(format)
        );

        const reader = stream.getReader();
        const decoder = new TextDecoder();
        let result = '';

        while (true) {
            const { done, value } = await reader.read();
            if (done) break;
            result += decoder.decode(value, { stream: true });
        }

        return result;
    }
}
```

```javascript
/**
 * 资源优先级管理
 * @author erik.zhou
 */
class ResourcePriorityOptimizer {
    constructor() {
        this.criticalResources = [];
        this.deferredResources = [];
    }

    /**
     * 标记关键资源
     * @param {string} url - 资源URL
     * @param {string} type - 资源类型
     */
    markAsCritical(url, type) {
        this.criticalResources.push({ url, type });
    }

    /**
     * 标记延迟资源
     * @param {string} url - 资源URL
     * @param {string} type - 资源类型
     */
    markAsDeferred(url, type) {
        this.deferredResources.push({ url, type });
    }

    /**
     * 优化资源加载顺序
     */
    async optimizeLoading() {
        // 1. 预加载关键资源
        await this.preloadCriticalResources();

        // 2. 加载关键资源
        await this.loadCriticalResources();

        // 3. 延迟加载非关键资源
        this.deferNonCriticalResources();
    }

    /**
     * 预加载关键资源
     */
    async preloadCriticalResources() {
        this.criticalResources.forEach(resource => {
            const link = document.createElement('link');
            link.rel = 'preload';
            link.href = resource.url;
            link.as = resource.type;

            if (resource.type === 'font') {
                link.crossOrigin = 'anonymous';
            }

            document.head.appendChild(link);
        });
    }

    /**
     * 加载关键资源
     */
    async loadCriticalResources() {
        const promises = this.criticalResources.map(resource => {
            if (resource.type === 'script') {
                return this.loadScript(resource.url);
            } else if (resource.type === 'style') {
                return this.loadStylesheet(resource.url);
            }
            return Promise.resolve();
        });

        await Promise.all(promises);
    }

    /**
     * 延迟加载非关键资源
     */
    deferNonCriticalResources() {
        if ('requestIdleCallback' in window) {
            requestIdleCallback(() => {
                this.loadDeferredResources();
            });
        } else {
            setTimeout(() => {
                this.loadDeferredResources();
            }, 1000);
        }
    }

    /**
     * 加载延迟资源
     */
    loadDeferredResources() {
        this.deferredResources.forEach(resource => {
            if (resource.type === 'script') {
                this.loadScript(resource.url);
            } else if (resource.type === 'style') {
                this.loadStylesheet(resource.url);
            }
        });
    }

    /**
     * 加载脚本
     * @param {string} url - 脚本URL
     * @returns {Promise<void>}
     */
    loadScript(url) {
        return new Promise((resolve, reject) => {
            const script = document.createElement('script');
            script.src = url;
            script.async = true;
            script.onload = resolve;
            script.onerror = reject;
            document.head.appendChild(script);
        });
    }

    /**
     * 加载样式表
     * @param {string} url - 样式表URL
     * @returns {Promise<void>}
     */
    loadStylesheet(url) {
        return new Promise((resolve, reject) => {
            const link = document.createElement('link');
            link.rel = 'stylesheet';
            link.href = url;
            link.onload = resolve;
            link.onerror = reject;
            document.head.appendChild(link);
        });
    }
}
```

```javascript
     * 检查资源大小

    async checkResourceSizes() {
        if (!window.performance || !window.performance.getEntriesByType) {

        }

        const resources = performance.getEntriesByType('resource');
        const sizes = {
            total: 0,
            js: 0,
            css: 0,
            image: 0,
            font: 0

            total: resources.length,
            js: 0,
            css: 0,
            image: 0,
            font: 0
        };

        resources.forEach(resource => {
            const size = resource.transferSize || 0;
            sizes.total += size;

            if (resource.name.match(/\.js$/)) {
                sizes.js += size;
                counts.js++;
            } else if (resource.name.match(/\.css$/)) {
                sizes.css += size;
                counts.css++;
            } else if (resource.name.match(/\.(jpg|jpeg|png|gif|webp|svg)$/)) {
                sizes.image += size;
                counts.image++;
            } else if (resource.name.match(/\.(woff|woff2|ttf|eot)$/)) {
                sizes.font += size;
                counts.font++;
            }
        });

        // 转换为KB
        Object.keys(sizes).forEach(key => {
            sizes[key] = Math.round(sizes[key] / 1024);
        });

        return { sizes, counts };
    }

    /**
     * 检查性能指标
     * @returns {Promise<Object>}
     */
    async checkPerformanceMetrics() {
        return new Promise((resolve) => {
            if ('PerformanceObserver' in window) {
                const metrics = {};

                // 观察FCP
                const fcpObserver = new PerformanceObserver((list) => {
                    const entries = list.getEntries();
                    entries.forEach(entry => {
                        if (entry.name === 'first-contentful-paint') {
                            metrics.fcp = Math.round(entry.startTime);
                        }
                    });

                    metrics.lcp = Math.round(lastEntry.startTime);

                // 观察FID
                const fidObserver = new PerformanceObserver((list) => {
                    const entries = list.getEntries();
                    entries.forEach(entry => {
                        metrics.fid = Math.round(entry.processingStart - entry.startTime);
                    });
                });
                fidObserver.observe({ entryTypes: ['first-input'] });

        // 检查资源大小

                type: 'size',
                metric: 'CSS Size',
                actual: sizes.css,
                budget: this.budgets.cssSize,
                unit: 'KB'
            });
        }

        // 检查性能指标

                type: 'metric',
                metric: 'FCP',
                actual: metrics.fcp,
                budget: this.budgets.fcp,
                unit: 'ms'
            });
        }

        if (metrics.lcp && metrics.lcp > this.budgets.lcp) {
            this.violations.push({
                type: 'metric',
                metric: 'LCP',
                actual: metrics.lcp,
                budget: this.budgets.lcp,
                unit: 'ms'
            });
        }

        if (metrics.cls && metrics.cls > this.budgets.cls) {
            this.violations.push({
                type: 'metric',
                metric: 'CLS',
                actual: metrics.cls,
                budget: this.budgets.cls,
                unit: ''
            });

                type: 'count',
                metric: 'Total Requests',
                actual: counts.total,
                budget: this.budgets.totalRequests,

            passed: this.violations.length === 0,

            sizes,
            counts,
            metrics
```

```javascript
/**
 * 性能监控系统
 * @author erik.zhou
 */
class PerformanceMonitor {
    constructor(options = {}) {
        this.apiEndpoint = options.apiEndpoint || '/api/performance';
        this.sampleRate = options.sampleRate || 1.0;
        this.batchSize = options.batchSize || 10;
        this.batchTimeout = options.batchTimeout || 5000;
        
        this.metrics = [];
        this.batchTimer = null;
        
        this.init();
    }

    /**
     * 初始化监控
     */
    init() {
        // 采样控制
        if (Math.random() > this.sampleRate) {
            return;
        }

        this.observePerformance();
        this.observeErrors();
        this.observeResources();
    }

    /**
     * 观察性能指标
     */
    observePerformance() {
        if (!('PerformanceObserver' in window)) {
            return;
        }

        // 观察导航时间
        const navigationObserver = new PerformanceObserver((list) => {
            const entries = list.getEntries();
            entries.forEach(entry => {
                this.recordMetric({
                    type: 'navigation',
                    metrics: {
                        dns: entry.domainLookupEnd - entry.domainLookupStart,
                        tcp: entry.connectEnd - entry.connectStart,
                        request: entry.responseStart - entry.requestStart,
                        response: entry.responseEnd - entry.responseStart,
                        domParse: entry.domContentLoadedEventEnd - entry.domContentLoadedEventStart,
                        domReady: entry.domContentLoadedEventEnd - entry.fetchStart,
                        load: entry.loadEventEnd - entry.fetchStart
                    }
                });
            });
        });
        navigationObserver.observe({ entryTypes: ['navigation'] });

        // 观察绘制时间
        const paintObserver = new PerformanceObserver((list) => {
            const entries = list.getEntries();
            entries.forEach(entry => {
                this.recordMetric({
                    type: 'paint',
                    name: entry.name,
                    value: Math.round(entry.startTime)
                });
            });
        });
        paintObserver.observe({ entryTypes: ['paint'] });

        // 观察LCP
        const lcpObserver = new PerformanceObserver((list) => {
            const entries = list.getEntries();
            const lastEntry = entries[entries.length - 1];
            this.recordMetric({
                type: 'lcp',
                value: Math.round(lastEntry.startTime),
                element: lastEntry.element?.tagName
            });
        });
        lcpObserver.observe({ entryTypes: ['largest-contentful-paint'] });

        // 观察FID
        const fidObserver = new PerformanceObserver((list) => {
            const entries = list.getEntries();
            entries.forEach(entry => {
                this.recordMetric({
                    type: 'fid',
                    value: Math.round(entry.processingStart - entry.startTime)
                });
            });
        });
        fidObserver.observe({ entryTypes: ['first-input'] });

        // 观察CLS
        let clsValue = 0;
        const clsObserver = new PerformanceObserver((list) => {
            const entries = list.getEntries();
            entries.forEach(entry => {
                if (!entry.hadRecentInput) {
                    clsValue += entry.value;
                }
            });
            this.recordMetric({
                type: 'cls',
                value: Math.round(clsValue * 1000) / 1000
            });
        });
        clsObserver.observe({ entryTypes: ['layout-shift'] });
    }

    /**
     * 观察错误
     */
    observeErrors() {
        // JavaScript错误
        window.addEventListener('error', (event) => {
            this.recordMetric({
                type: 'error',
                category: 'javascript',
                message: event.message,
                filename: event.filename,
                lineno: event.lineno,
                colno: event.colno
            });
        });

        // Promise错误
        window.addEventListener('unhandledrejection', (event) => {
            this.recordMetric({
                type: 'error',
                category: 'promise',
                message: event.reason?.message || String(event.reason)
            });
        });

        // 资源加载错误
        window.addEventListener('error', (event) => {
            if (event.target !== window) {
                this.recordMetric({
                    type: 'error',
                    category: 'resource',
                    url: event.target.src || event.target.href
                });
            }
        }, true);
    }

    /**
     * 观察资源加载
     */
    observeResources() {
        if (!('PerformanceObserver' in window)) {
            return;
        }

        const resourceObserver = new PerformanceObserver((list) => {
            const entries = list.getEntries();
            entries.forEach(entry => {
                // 只记录慢资源
                if (entry.duration > 1000) {
                    this.recordMetric({
                        type: 'slow-resource',
                        url: entry.name,
                        duration: Math.round(entry.duration),
                        size: entry.transferSize,
                        initiatorType: entry.initiatorType
                    });
                }
            });
        });
        resourceObserver.observe({ entryTypes: ['resource'] });
    }

    /**
     * 记录指标
     * @param {Object} metric - 指标数据
     */
    recordMetric(metric) {
        this.metrics.push({
            ...metric,
            timestamp: Date.now(),
            url: window.location.href,
            userAgent: navigator.userAgent
        });

        // 批量上报
        if (this.metrics.length >= this.batchSize) {
            this.flush();
        } else {
            this.scheduleBatch();
        }
    }

    /**
     * 调度批量上报
     */
    scheduleBatch() {
        if (this.batchTimer) {
            return;
        }

        this.batchTimer = setTimeout(() => {
            this.flush();
        }, this.batchTimeout);
    }

    /**
     * 上报数据
     */
    async flush() {
        if (this.metrics.length === 0) {
            return;
        }

        const data = [...this.metrics];
        this.metrics = [];

        if (this.batchTimer) {
            clearTimeout(this.batchTimer);
            this.batchTimer = null;
        }

        try {
            // 使用sendBeacon确保数据发送
            if (navigator.sendBeacon) {
                const blob = new Blob([JSON.stringify(data)], {
                    type: 'application/json'
                });
                navigator.sendBeacon(this.apiEndpoint, blob);
            } else {
                // 降级使用fetch
                await fetch(this.apiEndpoint, {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json'
                    },
                    body: JSON.stringify(data),
                    keepalive: true
                });
            }

            console.log(`Reported ${data.length} metrics`);
        } catch (error) {
            console.error('Failed to report metrics:', error);
            // 失败的数据放回队列
            this.metrics.unshift(...data);
        }
    }

    /**
     * 页面卸载时上报
     */
    setupBeforeUnload() {
        window.addEventListener('beforeunload', () => {
            this.flush();
        });

        // 使用visibilitychange作为备份
        document.addEventListener('visibilitychange', () => {
            if (document.visibilityState === 'hidden') {
                this.flush();
            }
        });
    }
}
```
