## 类型体操基础

### 1.1 什么是类型体操

类型体操是指通过TypeScript的类型系统进行复杂的类型操作和转换。

```typescript
/**
 * 类型体操示例：提取函数返回值类型
 * @author erik.zhou
 */

// 基础示例
type GetReturnType<T> = T extends (...args: any[]) => infer R ? R : never;

function getUserInfo() {
    return {
        name: 'John',
        age: 30
    };
}

type UserInfo = GetReturnType<typeof getUserInfo>;
// 结果：{ name: string; age: number; }
```

### 1.2 类型体操的核心概念

```typescript
/**
 * 类型体操核心概念
 * @author erik.zhou
 */

// 1. 泛型约束
type OnlyString<T extends string> = T;
type Result1 = OnlyString<'hello'>; // ✅
// type Result2 = OnlyString<123>; // ❌ 类型错误

// 2. 条件类型
type IsString<T> = T extends string ? true : false;
type Test1 = IsString<'hello'>; // true
type Test2 = IsString<123>; // false

// 3. 类型推断
type GetFirstElement<T> = T extends [infer First, ...any[]] ? First : never;
type First = GetFirstElement<[string, number, boolean]>; // string

// 4. 映射类型
type ReadonlyType<T> = {
    readonly [K in keyof T]: T[K];
};

interface User {
    name: string;
    age: number;
}

type ReadonlyUser = ReadonlyType<User>;
// 结果：{ readonly name: string; readonly age: number; }

// 5. 递归类型
type DeepReadonly<T> = {
    readonly [K in keyof T]: T[K] extends object
        ? DeepReadonly<T[K]>
        : T[K];
};

interface NestedUser {
    name: string;
    profile: {
        email: string;
        address: {
            city: string;
        };
    };
}

type DeepReadonlyUser = DeepReadonly<NestedUser>;
```

## 内置工具类型

### 2.1 Partial 和 Required

```typescript
/**
 * Partial 和 Required 工具类型
 * @author erik.zhou
 */

interface User {
    id: number;
    name: string;
    email: string;
    age: number;
}

// Partial - 所有属性变为可选
type PartialUser = Partial<User>;
// 等价于：{ id?: number; name?: string; email?: string; age?: number; }

// Required - 所有属性变为必填
type RequiredUser = Required<PartialUser>;
// 等价于：{ id: number; name: string; email: string; age: number; }

// 自己实现 Partial
type MyPartial<T> = {
    [K in keyof T]?: T[K];
};

// 自己实现 Required
type MyRequired<T> = {
    [K in keyof T]-?: T[K];
};

// 实战示例
function updateUser(id: number, updates: Partial<User>) {
    // 只更新传入的字段
    console.log(`Updating user ${id}`, updates);
}

updateUser(1, { name: 'John' }); // ✅
updateUser(2, { email: 'john@example.com', age: 30 }); // ✅
```

### 2.2 Pick 和 Omit

```typescript
/**
 * Pick 和 Omit 工具类型
 * @author erik.zhou
 */

interface User {
    id: number;
    name: string;
    email: string;
    password: string;
    createdAt: Date;
}

// Pick - 选择指定属性
type UserBasicInfo = Pick<User, 'id' | 'name' | 'email'>;
// 结果：{ id: number; name: string; email: string; }

// Omit - 排除指定属性
type UserWithoutPassword = Omit<User, 'password'>;
// 结果：{ id: number; name: string; email: string; createdAt: Date; }

// 自己实现 Pick
type MyPick<T, K extends keyof T> = {
    [P in K]: T[P];
};

// 自己实现 Omit
type MyOmit<T, K extends keyof T> = {
    [P in Exclude<keyof T, K>]: T[P];
};

// 实战示例：API 响应类型
type UserResponse = Omit<User, 'password'>;

function getUserInfo(id: number): UserResponse {
    return {
        id,
        name: 'John',
        email: 'john@example.com',
        createdAt: new Date()
    };
}
```

// Record - 创建对象类型
type UserRoles = 'admin' | 'user' | 'guest';

class ConfigManager {
    private config: Readonly<Config>;

### 2.4 Exclude 和 Extract

```typescript
/**
 * Exclude 和 Extract 工具类型
 * @author erik.zhou
 */

type AllTypes = 'a' | 'b' | 'c' | 'd';

// Exclude - 排除类型
type ExcludedTypes = Exclude<AllTypes, 'a' | 'b'>;
// 结果：'c' | 'd'

// Extract - 提取类型
type ExtractedTypes = Extract<AllTypes, 'a' | 'b'>;
// 结果：'a' | 'b'

// 自己实现 Exclude
type MyExclude<T, U> = T extends U ? never : T;

// 自己实现 Extract
type MyExtract<T, U> = T extends U ? T : never;

// 实战示例：事件类型过滤
type MouseEvent = 'click' | 'dblclick' | 'mousedown' | 'mouseup';
type KeyboardEvent = 'keydown' | 'keyup' | 'keypress';
type AllEvents = MouseEvent | KeyboardEvent;

type OnlyMouseEvents = Extract<AllEvents, MouseEvent>;
type NonMouseEvents = Exclude<AllEvents, MouseEvent>;

// 高级用法：过滤函数类型
type MixedTypes = string | number | (() => void) | boolean;
type OnlyFunctions = Extract<MixedTypes, Function>;
// 结果：() => void
type NonFunctions = Exclude<MixedTypes, Function>;
// 结果：string | number | boolean
```

## 类型推断与条件类型

### 3.1 infer 关键字

```typescript
/**
 * infer 关键字详解
 * @author erik.zhou
 */

// 1. 推断函数参数类型
type GetParameters<T> = T extends (...args: infer P) => any ? P : never;

function add(a: number, b: number): number {
    return a + b;
}

type AddParams = GetParameters<typeof add>;
// 结果：[a: number, b: number]

// 2. 推断数组元素类型
type GetArrayElement<T> = T extends (infer U)[] ? U : never;

type NumberArray = number[];
type Element = GetArrayElement<NumberArray>;
// 结果：number

// 3. 推断 Promise 类型
type UnwrapPromise<T> = T extends Promise<infer U> ? U : T;

type PromiseNumber = Promise<number>;
type Unwrapped = UnwrapPromise<PromiseNumber>;
// 结果：number

// 4. 推断对象属性类型
type GetValueType<T> = T extends { value: infer V } ? V : never;

interface Container {
    value: string;
}

type ValueType = GetValueType<Container>;
// 结果：string

// 5. 多重推断
type GetFirstAndLast<T> = T extends [infer First, ...any[], infer Last]
    ? [First, Last]
    : never;

type Result = GetFirstAndLast<[1, 2, 3, 4, 5]>;
// 结果：[1, 5]
```

### 3.2 条件类型分发

```typescript
/**
 * 条件类型分发机制
 * @author erik.zhou
 */

// 1. 联合类型分发
type ToArray<T> = T extends any ? T[] : never;

type StringOrNumber = string | number;
type Arrays = ToArray<StringOrNumber>;
// 结果：string[] | number[]（分发）

// 2. 阻止分发
type ToArrayNonDistributive<T> = [T] extends [any] ? T[] : never;

type NonDistributive = ToArrayNonDistributive<StringOrNumber>;
// 结果：(string | number)[]（不分发）

// 3. 实战示例：类型过滤
type Filter<T, U> = T extends U ? T : never;

type Mixed = string | number | boolean | null;
type OnlyPrimitive = Filter<Mixed, string | number | boolean>;
// 结果：string | number | boolean

// 4. 复杂分发示例
type FunctionPropertyNames<T> = {
    [K in keyof T]: T[K] extends Function ? K : never;
}[keyof T];

interface Example {
    name: string;
    age: number;
    getName: () => string;
    getAge: () => number;
}

type FunctionKeys = FunctionPropertyNames<Example>;
// 结果：'getName' | 'getAge'
```

### 3.3 递归条件类型

```typescript
/**
 * 递归条件类型
 * @author erik.zhou
 */

// 1. 深度 Readonly
type DeepReadonly<T> = {
    readonly [K in keyof T]: T[K] extends object
        ? T[K] extends Function
            ? T[K]
            : DeepReadonly<T[K]>
        : T[K];
};

interface NestedObject {
    a: string;
    b: {
        c: number;
        d: {
            e: boolean;
        };
    };
}

type ReadonlyNested = DeepReadonly<NestedObject>;

// 2. 深度 Partial
type DeepPartial<T> = {
    [K in keyof T]?: T[K] extends object
        ? DeepPartial<T[K]>
        : T[K];
};

type PartialNested = DeepPartial<NestedObject>;

// 3. 路径类型
type PathsToStringProps<T> = T extends string
    ? []
    : {
        [K in Extract<keyof T, string>]: [K, ...PathsToStringProps<T[K]>];
    }[Extract<keyof T, string>];

interface Config {
    server: {
        host: string;
        port: number;
    };
    database: {
        url: string;
    };
}

type ConfigPaths = PathsToStringProps<Config>;

// 4. 扁平化类型
type Flatten<T> = T extends any[]
    ? T[number]
    : T extends object
    ? { [K in keyof T]: Flatten<T[K]> }[keyof T]
    : T;

type NestedArray = [1, [2, [3, [4]]]];
type Flattened = Flatten<NestedArray>;
// 结果：1 | 2 | 3 | 4
```

## 映射类型

### 4.1 基础映射类型

```typescript
/**
 * 基础映射类型
 * @author erik.zhou
 */

// 1. 基本映射
type Mapped<T> = {
    [K in keyof T]: T[K];
};

// 2. 添加修饰符
type AddReadonly<T> = {
    readonly [K in keyof T]: T[K];
};

type AddOptional<T> = {
    [K in keyof T]?: T[K];
};

// 3. 移除修饰符
type RemoveReadonly<T> = {
    -readonly [K in keyof T]: T[K];
};

type RemoveOptional<T> = {
    [K in keyof T]-?: T[K];
};

// 4. 类型转换
type Stringify<T> = {
    [K in keyof T]: string;
};

interface Numbers {
    a: number;
    b: number;
}

type StringNumbers = Stringify<Numbers>;
// 结果：{ a: string; b: string; }

// 5. 条件映射
type Nullable<T> = {
    [K in keyof T]: T[K] | null;
};

interface User {
    name: string;
    age: number;
}

type NullableUser = Nullable<User>;
// 结果：{ name: string | null; age: number | null; }
```

### 4.2 键名重映射

```typescript
/**
 * 键名重映射（TypeScript 4.1+）
 * @author erik.zhou
 */

// 1. 基础重映射
type Getters<T> = {
    [K in keyof T as `get${Capitalize<string & K>}`]: () => T[K];
};

interface Person {
    name: string;
    age: number;
}

type PersonGetters = Getters<Person>;
// 结果：{ getName: () => string; getAge: () => number; }

// 2. 过滤属性
type RemoveKindField<T> = {
    [K in keyof T as Exclude<K, 'kind'>]: T[K];
};

interface Circle {
    kind: 'circle';
    radius: number;
}

type CircleWithoutKind = RemoveKindField<Circle>;
// 结果：{ radius: number; }

// 3. 条件重映射
type ExtractPII<T> = {
    [K in keyof T as T[K] extends string | number ? K : never]: T[K];
};

interface Data {
    name: string;
    age: number;
    active: boolean;
    tags: string[];
}

type PIIData = ExtractPII<Data>;
// 结果：{ name: string; age: number; }

// 4. 复杂重映射
type EventHandlers<T> = {
    [K in keyof T as `on${Capitalize<string & K>}Change`]: (
        value: T[K]
    ) => void;
};

interface FormData {
    username: string;
    password: string;
    email: string;
}

type FormHandlers = EventHandlers<FormData>;
// 结果：{
//   onUsernameChange: (value: string) => void;
//   onPasswordChange: (value: string) => void;
//   onEmailChange: (value: string) => void;
// }
```

### 4.3 高级映射模式

```typescript
/**
 * 高级映射模式
 * @author erik.zhou
 */

// 1. 双向映射
type CreateMutable<T> = {
    -readonly [K in keyof T]: T[K];
};

type CreateImmutable<T> = {
    readonly [K in keyof T]: T[K];
};

// 2. 深度映射
type DeepMutable<T> = {
    -readonly [K in keyof T]: T[K] extends object
        ? DeepMutable<T[K]>
        : T[K];
};

// 3. 选择性映射
type PickByType<T, U> = {
    [K in keyof T as T[K] extends U ? K : never]: T[K];
};

interface Mixed {
    id: number;
    name: string;
    age: number;
    active: boolean;
}

type OnlyNumbers = PickByType<Mixed, number>;
// 结果：{ id: number; age: number; }

// 4. 映射联合类型
type UnionToIntersection<U> = (
    U extends any ? (k: U) => void : never
) extends (k: infer I) => void
    ? I
    : never;

type Union = { a: string } | { b: number };
type Intersection = UnionToIntersection<Union>;
// 结果：{ a: string } & { b: number }

// 5. 实战示例：表单验证器
type Validator<T> = {
    [K in keyof T]: (value: T[K]) => boolean;
};

interface LoginForm {
    username: string;
    password: string;
    rememberMe: boolean;
}

const validators: Validator<LoginForm> = {
    username: (value) => value.length >= 3,
    password: (value) => value.length >= 8,
    rememberMe: (value) => typeof value === 'boolean'
};
```

## 递归类型

### 6.1 基础递归类型

```typescript
/**
 * 基础递归类型
 * @author erik.zhou
 */

// 1. 数组扁平化
type Flatten<T> = T extends any[]
    ? T[number] extends infer U
        ? U extends any[]
            ? Flatten<U>
            : U
        : never
    : T;

type NestedArray = [1, [2, [3, [4, 5]]]];
type FlatArray = Flatten<NestedArray>;
// 结果：1 | 2 | 3 | 4 | 5

// 2. 对象深度只读
type DeepReadonly<T> = {
    readonly [K in keyof T]: T[K] extends object
        ? T[K] extends Function
            ? T[K]
            : DeepReadonly<T[K]>
        : T[K];
};

interface Config {
    server: {
        host: string;
        port: number;
        ssl: {
            enabled: boolean;
            cert: string;
        };
    };
}

type ReadonlyConfig = DeepReadonly<Config>;

// 3. 对象深度可选
type DeepPartial<T> = {
    [K in keyof T]?: T[K] extends object
        ? DeepPartial<T[K]>
        : T[K];
};

type PartialConfig = DeepPartial<Config>;

// 4. 对象深度必填
type DeepRequired<T> = {
    [K in keyof T]-?: T[K] extends object
        ? DeepRequired<T[K]>
        : T[K];
};

type RequiredConfig = DeepRequired<PartialConfig>;
```

### 6.2 高级递归类型

```typescript
/**
 * 高级递归类型
 * @author erik.zhou
 */

// 1. JSON 类型
type JSONValue =
    | string
    | number
    | boolean
    | null
    | JSONValue[]
    | { [key: string]: JSONValue };

interface JSONObject {
    [key: string]: JSONValue;
}

interface JSONArray extends Array<JSONValue> {}

// 2. 树形结构
interface TreeNode<T> {
    value: T;
    children?: TreeNode<T>[];
}

type FlattenTree<T> = T extends TreeNode<infer U>
    ? U | (T['children'] extends TreeNode<U>[]
        ? FlattenTree<T['children'][number]>
        : never)
    : never;

// 3. 路径访问类型
type PathValue<T, P extends string> = 
    P extends `${infer Key}.${infer Rest}`
        ? Key extends keyof T
            ? PathValue<T[Key], Rest>
            : never
        : P extends keyof T
        ? T[P]
        : never;

interface Data {
    user: {
        profile: {
            name: string;
            age: number;
        };
    };
}

type UserName = PathValue<Data, 'user.profile.name'>;
// 结果：string

// 4. 类型合并
type DeepMerge<T, U> = {
    [K in keyof T | keyof U]: K extends keyof U
        ? K extends keyof T
            ? T[K] extends object
                ? U[K] extends object
                    ? DeepMerge<T[K], U[K]>
                    : U[K]
                : U[K]
            : U[K]
        : K extends keyof T
        ? T[K]
        : never;
};

interface Base {
    a: string;
    b: {
        c: number;
    };
}

interface Override {
    b: {
        d: boolean;
    };
    e: string;
}

type Merged = DeepMerge<Base, Override>;
// 结果：{ a: string; b: { c: number; d: boolean; }; e: string; }
```

 * 状态管理类型系统实战
 * @author erik.zhou

// 1. Action 类型
interface Action<T = any> {
    type: string;
    payload?: T;
}

// 2. Reducer 类型
type Reducer<S, A extends Action> = (state: S, action: A) => S;

// 3. Action Creator 类型
type ActionCreator<T = any> = (...args: any[]) => Action<T>;

// 4. 提取 Action 类型
type ExtractActionType<T> = T extends ActionCreator<infer P>
    ? Action<P>
    : never;

// 5. Store 类型
interface Store<S, A extends Action> {
    getState(): S;
    dispatch(action: A): void;
    subscribe(listener: () => void): () => void;
}

// 6. 实战示例：Todo Store
interface Todo {
    id: number;
    text: string;
    completed: boolean;
}

interface TodoState {
    todos: Todo[];
    filter: 'all' | 'active' | 'completed';
}

// Action Creators
const addTodo = (text: string): Action<{ text: string }> => ({
    type: 'ADD_TODO',
    payload: { text }
});

const toggleTodo = (id: number): Action<{ id: number }> => ({
    type: 'TOGGLE_TODO',
    payload: { id }
});

const setFilter = (
    filter: 'all' | 'active' | 'completed'
): Action<{ filter: 'all' | 'active' | 'completed' }> => ({
    type: 'SET_FILTER',
    payload: { filter }
});

// Action 联合类型
type TodoAction =
    | ReturnType<typeof addTodo>
    | ReturnType<typeof toggleTodo>
    | ReturnType<typeof setFilter>;

// Reducer
const todoReducer: Reducer<TodoState, TodoAction> = (state, action) => {
    switch (action.type) {
        case 'ADD_TODO':
            return {
                ...state,
                todos: [
                    ...state.todos,
                    {
                        id: Date.now(),
                        text: action.payload!.text,
                        completed: false
                    }
                ]
            };
        case 'TOGGLE_TODO':
            return {
                ...state,
                todos: state.todos.map((todo) =>
                    todo.id === action.payload!.id
                        ? { ...todo, completed: !todo.completed }
                        : todo
                )
            };
        case 'SET_FILTER':
            return {
                ...state,
                filter: action.payload!.filter
            };
        default:
            return state;
    }
};

        ? (arg: First) => Rest extends []

        : R;

// 7. 实战示例：类型安全的事件系统
interface EventMap {
    click: { x: number; y: number };
    keypress: { key: string };
    focus: void;
}

type EventHandler<T> = T extends void
    ? () => void
    : (event: T) => void;

class EventEmitter<T extends Record<string, any>> {
    private listeners: {
        [K in keyof T]?: EventHandler<T[K]>[];
    } = {};

    on<K extends keyof T>(
        event: K,
        handler: EventHandler<T[K]>
    ): void {
        if (!this.listeners[event]) {
            this.listeners[event] = [];
        }
        this.listeners[event]!.push(handler);
    }

    emit<K extends keyof T>(
        event: K,
        ...args: T[K] extends void ? [] : [T[K]]
    ): void {
        const handlers = this.listeners[event];
        if (handlers) {
            handlers.forEach((handler) => {
                if (args.length > 0) {
                    (handler as any)(args[0]);
                } else {
                    (handler as any)();
                }
            });
        }
    }
}

// 使用示例
const emitter = new EventEmitter<EventMap>();

emitter.on('click', (event) => {
    console.log(event.x, event.y); // ✅ 类型安全
});

emitter.on('keypress', (event) => {
    console.log(event.key); // ✅ 类型安全
});

emitter.on('focus', () => {
    console.log('Focused'); // ✅ 类型安全
});

emitter.emit('click', { x: 10, y: 20 }); // ✅
emitter.emit('keypress', { key: 'Enter' }); // ✅
emitter.emit('focus'); // ✅
