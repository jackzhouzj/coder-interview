### 1.3 基础CRUD操作

```javascript
/**
 * MySQL CRUD操作
 * @author erik.zhou
 */

const mysql = require('mysql2/promise');

class UserRepository {
    constructor(pool) {
        this.pool = pool;
    }

    // 查询所有用户
    async findAll() {
        const [rows] = await this.pool.query(
            'SELECT id, name, email, created_at FROM users'
        );
        return rows;
    }

    // 根据ID查询用户
    async findById(id) {
        const [rows] = await this.pool.query(
            'SELECT id, name, email, created_at FROM users WHERE id = ?',
            [id]
        );
        return rows[0];
    }

    // 创建用户
    async create(user) {
        const [result] = await this.pool.query(
            'INSERT INTO users (name, email, password) VALUES (?, ?, ?)',
            [user.name, user.email, user.password]
        );
        return {
            id: result.insertId,
            ...user
        };
    }

    // 更新用户
    async update(id, user) {
        const [result] = await this.pool.query(
            'UPDATE users SET name = ?, email = ? WHERE id = ?',
            [user.name, user.email, id]
        );
        return result.affectedRows > 0;
    }

    // 删除用户
    async delete(id) {
        const [result] = await this.pool.query(
            'DELETE FROM users WHERE id = ?',
            [id]
        );
        return result.affectedRows > 0;
    }

    // 分页查询
    async findWithPagination(page = 1, limit = 10) {
        const offset = (page - 1) * limit;
        const [rows] = await this.pool.query(
            'SELECT id, name, email FROM users ORDER BY id LIMIT ? OFFSET ?',
            [limit, offset]
        );
        const [countResult] = await this.pool.query(
            'SELECT COUNT(*) as total FROM users'
        );
        return {
            data: rows,
            total: countResult[0].total,
            page: page,
            limit: limit
        };
    }
}
```

### 2.3 Repository操作

```typescript
/**
 * TypeORM Repository操作
 * @author erik.zhou
 */

import { AppDataSource } from './ormconfig';
import { User } from './entities/User';

class UserService {
    private userRepository = AppDataSource.getRepository(User);

    // 查询所有用户
    async findAll(): Promise<User[]> {
        return await this.userRepository.find();
    }

    // 根据ID查询
    async findById(id: number): Promise<User | null> {
        return await this.userRepository.findOne({
            where: { id: id }
        });
    }

    // 创建用户
    async create(userData: Partial<User>): Promise<User> {
        const user = this.userRepository.create(userData);
        return await this.userRepository.save(user);
    }

    // 更新用户
    async update(id: number, userData: Partial<User>): Promise<User | null> {
        await this.userRepository.update(id, userData);
        return await this.findById(id);
    }

    // 删除用户
    async delete(id: number): Promise<boolean> {
        const result = await this.userRepository.delete(id);
        return result.affected !== null && result.affected > 0;
    }

    // 条件查询
    async findByEmail(email: string): Promise<User | null> {
        return await this.userRepository.findOne({
            where: { email: email }
        });
    }

    // 复杂查询
    async findActiveUsers(): Promise<User[]> {
        return await this.userRepository.find({
            where: { status: 'active' },
            order: { createdAt: 'DESC' },
            take: 10
        });
    }

    // 分页查询
    async findWithPagination(page: number = 1, limit: number = 10) {
        const [users, total] = await this.userRepository.findAndCount({
            skip: (page - 1) * limit,
            take: limit,
            order: { id: 'DESC' }
        });

        return {
            data: users,
            total: total,
            page: page,
            limit: limit,
            pages: Math.ceil(total / limit)
        };
    }
}
```

### 2.4 QueryBuilder

```typescript
/**
 * TypeORM QueryBuilder
 * @author erik.zhou
 */

import { AppDataSource } from './ormconfig';
import { User } from './entities/User';

class UserQueryService {
    private userRepository = AppDataSource.getRepository(User);

    // 基础查询
    async findByName(name: string): Promise<User[]> {
        return await this.userRepository
            .createQueryBuilder('user')
            .where('user.name LIKE :name', { name: `%${name}%` })
            .getMany();
    }

    // 关联查询
    async findWithPosts(userId: number): Promise<User | null> {
        return await this.userRepository
            .createQueryBuilder('user')
            .leftJoinAndSelect('user.posts', 'post')
            .where('user.id = :id', { id: userId })
            .getOne();
    }

    // 聚合查询
    async getUserStats() {
        return await this.userRepository
            .createQueryBuilder('user')
            .select('COUNT(user.id)', 'total')
            .addSelect('AVG(user.age)', 'avgAge')
            .addSelect('MAX(user.age)', 'maxAge')
            .addSelect('MIN(user.age)', 'minAge')
            .getRawOne();
    }

    // 分组查询
    async getUsersByStatus() {
        return await this.userRepository
            .createQueryBuilder('user')
            .select('user.status', 'status')
            .addSelect('COUNT(user.id)', 'count')
            .groupBy('user.status')
            .getRawMany();
    }

    // 子查询
    async findUsersWithManyPosts() {
        const subQuery = AppDataSource
            .createQueryBuilder()
            .select('post.userId')
            .from('posts', 'post')
            .groupBy('post.userId')
            .having('COUNT(post.id) > :count', { count: 5 });

        return await this.userRepository
            .createQueryBuilder('user')
            .where(`user.id IN (${subQuery.getQuery()})`)
            .setParameters(subQuery.getParameters())
            .getMany();
    }

    // 原始SQL
    async executeRawQuery() {
        return await this.userRepository.query(
            'SELECT id, name, email FROM users WHERE age > ?',
            [18]
        );
    }
}
```

```typescript
import { AppDataSource } from './ormconfig';
import { User } from './entities/User';
import { Post } from './entities/Post';

class RelationService {
    private userRepository = AppDataSource.getRepository(User);
    private postRepository = AppDataSource.getRepository(Post);

    // 创建用户和文章
    async createUserWithPosts() {
        const user = this.userRepository.create({
            name: 'John',
            email: 'john@example.com',
            password: 'password123',
            posts: [
                {
                    title: '第一篇文章',
                    content: '文章内容...'
                },
                {
                    title: '第二篇文章',
                    content: '文章内容...'
                }
            ]
        });

        return await this.userRepository.save(user);
    }

    // 查询用户及其文章
    async findUserWithPosts(userId: number) {
        return await this.userRepository.findOne({
            where: { id: userId },
            relations: ['posts']
        });
    }

    // 延迟加载
    async lazyLoadPosts(userId: number) {
        const user = await this.userRepository.findOne({
            where: { id: userId }
        });

        if (user) {
            const posts = await this.postRepository.find({
                where: { userId: user.id }
            });
            return { user, posts };
        }

        return null;
    }

    // 更新关系
    async addPostToUser(userId: number, postData: Partial<Post>) {
        const user = await this.userRepository.findOne({
            where: { id: userId }
        });

        if (!user) {
            throw new Error('用户不存在');
        }

        const post = this.postRepository.create({
            ...postData,
            userId: user.id
        });

        return await this.postRepository.save(post);
    }
}
```

```typescript
/**
 * Sequelize模型定义
 * @author erik.zhou
 */

import { DataTypes, Model, Optional } from 'sequelize';
import { sequelize } from './config';

interface UserAttributes {
    id: number;
    name: string;
    email: string;
    password: string;
    age: number;
    status: 'active' | 'inactive';
    createdAt?: Date;
    updatedAt?: Date;
}

interface UserCreationAttributes extends Optional<UserAttributes, 'id'> {}

class User extends Model<UserAttributes, UserCreationAttributes>
    implements UserAttributes {
    public id!: number;
    public name!: string;
    public email!: string;
    public password!: string;
    public age!: number;
    public status!: 'active' | 'inactive';
    public readonly createdAt!: Date;
    public readonly updatedAt!: Date;
}

User.init(
    {
        id: {
            type: DataTypes.INTEGER,
            autoIncrement: true,
            primaryKey: true
        },
        name: {
            type: DataTypes.STRING(50),
            allowNull: false
        },
        email: {
            type: DataTypes.STRING(100),
            allowNull: false,
            unique: true,
            validate: {
                isEmail: true
            }
        },
        password: {
            type: DataTypes.STRING(255),
            allowNull: false
        },
        age: {
            type: DataTypes.INTEGER,
            defaultValue: 0,
            validate: {
                min: 0,
                max: 150
            }
        },
        status: {
            type: DataTypes.ENUM('active', 'inactive'),
            defaultValue: 'active'
        }
    },
    {
        sequelize: sequelize,
        tableName: 'users',
        timestamps: true,
        underscored: true
    }
);

export { User };
```

```typescript
/**
 * Sequelize CRUD操作
 * @author erik.zhou
 */

import { User } from './models/User';
import { Op } from 'sequelize';

class UserService {
    // 创建用户
    async create(userData: any) {
        return await User.create(userData);
    }

    // 批量创建
    async bulkCreate(users: any[]) {
        return await User.bulkCreate(users);
    }

    // 查询所有
    async findAll() {
        return await User.findAll();
    }

    // 根据ID查询
    async findById(id: number) {
        return await User.findByPk(id);
    }

    // 条件查询
    async findByEmail(email: string) {
        return await User.findOne({
            where: { email: email }
        });
    }

    // 复杂查询
    async findActiveUsers() {
        return await User.findAll({
            where: {
                status: 'active',
                age: {
                    [Op.gte]: 18
                }
            },
            order: [['createdAt', 'DESC']],
            limit: 10
        });
    }

    // 更新
    async update(id: number, userData: any) {
        const [affectedCount] = await User.update(userData, {
            where: { id: id }
        });
        return affectedCount > 0;
    }

    // 删除
    async delete(id: number) {
        const affectedCount = await User.destroy({
            where: { id: id }
        });
        return affectedCount > 0;
    }

    // 分页查询
    async findWithPagination(page: number = 1, limit: number = 10) {
        const offset = (page - 1) * limit;
        const { count, rows } = await User.findAndCountAll({
            offset: offset,
            limit: limit,
            order: [['id', 'DESC']]
        });

        return {
            data: rows,
            total: count,
            page: page,
            limit: limit,
            pages: Math.ceil(count / limit)
        };
    }

    // 聚合查询
    async getStats() {
        const result = await User.findAll({
            attributes: [
                [sequelize.fn('COUNT', sequelize.col('id')), 'total'],
                [sequelize.fn('AVG', sequelize.col('age')), 'avgAge'],
                [sequelize.fn('MAX', sequelize.col('age')), 'maxAge'],
                [sequelize.fn('MIN', sequelize.col('age')), 'minAge']
            ],
            raw: true
        });
        return result[0];
    }
}
```

```typescript
/**
 * Sequelize关联关系
 * @author erik.zhou
 */

import { DataTypes, Model } from 'sequelize';
import { sequelize } from './config';
import { User } from './models/User';

// 文章模型
class Post extends Model {
    public id!: number;
    public title!: string;
    public content!: string;
    public userId!: number;
}

Post.init(
    {
        id: {
            type: DataTypes.INTEGER,
            autoIncrement: true,
            primaryKey: true
        },
        title: {
            type: DataTypes.STRING(200),
            allowNull: false
        },
        content: {
            type: DataTypes.TEXT,
            allowNull: false
        },
        userId: {
            type: DataTypes.INTEGER,
            allowNull: false,
            field: 'user_id'
        }
    },
    {
        sequelize: sequelize,
        tableName: 'posts',
        timestamps: true,
        underscored: true
    }
);

// 定义关联
User.hasMany(Post, {
    foreignKey: 'userId',
    as: 'posts'
});

Post.belongsTo(User, {
    foreignKey: 'userId',
    as: 'user'
});

// 使用关联
class RelationService {
    // 查询用户及其文章
    async findUserWithPosts(userId: number) {
        return await User.findByPk(userId, {
            include: [
                {
                    model: Post,
                    as: 'posts'
                }
            ]
        });
    }

    // 查询文章及其作者
    async findPostWithUser(postId: number) {
        return await Post.findByPk(postId, {
            include: [
                {
                    model: User,
                    as: 'user',
                    attributes: ['id', 'name', 'email']
                }
            ]
        });
    }

    // 创建用户和文章
    async createUserWithPosts(userData: any, posts: any[]) {
        const user = await User.create(userData);
        const createdPosts = await Promise.all(
            posts.map(post =>
                Post.create({
                    ...post,
                    userId: user.id
                })
            )
        );
        return { user, posts: createdPosts };
    }
}

export { Post, RelationService };
```

### 6.3 缓存策略

```typescript
/**
 * Redis缓存策略
 * @author erik.zhou
 */

import { redisClient } from './client';
import { User } from './models/User';

class CacheService {
    private readonly CACHE_PREFIX = 'cache:';
    private readonly DEFAULT_TTL = 3600; // 1小时

    // 生成缓存键
    private getCacheKey(key: string): string {
        return `${this.CACHE_PREFIX}${key}`;
    }

    // 缓存穿透防护
    async getWithCache<T>(
        key: string,
        fetchFn: () => Promise<T>,
        ttl: number = this.DEFAULT_TTL
    ): Promise<T | null> {
        const cacheKey = this.getCacheKey(key);

        // 尝试从缓存获取
        const cached = await redisClient.get(cacheKey);
        if (cached) {
            return JSON.parse(cached);
        }

        // 从数据库获取
        const data = await fetchFn();

        // 缓存空值防止缓存穿透
        if (data === null) {
            await redisClient.setEx(cacheKey, 60, 'null');
            return null;
        }

        // 缓存数据
        await redisClient.setEx(cacheKey, ttl, JSON.stringify(data));
        return data;
    }

    // 缓存雪崩防护（随机过期时间）
    async setWithRandomTTL(key: string, value: any, baseTTL: number = 3600) {
        const randomTTL = baseTTL + Math.floor(Math.random() * 300);
        const cacheKey = this.getCacheKey(key);
        await redisClient.setEx(cacheKey, randomTTL, JSON.stringify(value));
    }

    // 缓存击穿防护（互斥锁）
    async getWithLock<T>(
        key: string,
        fetchFn: () => Promise<T>,
        ttl: number = this.DEFAULT_TTL
    ): Promise<T | null> {
        const cacheKey = this.getCacheKey(key);
        const lockKey = `lock:${key}`;

        // 尝试从缓存获取
        const cached = await redisClient.get(cacheKey);
        if (cached) {
            return JSON.parse(cached);
        }

        // 尝试获取锁
        const lockAcquired = await redisClient.setNX(lockKey, '1');
        if (!lockAcquired) {
            // 等待并重试
            await new Promise(resolve => setTimeout(resolve, 100));
            return this.getWithLock(key, fetchFn, ttl);
        }

        try {
            // 设置锁过期时间
            await redisClient.expire(lockKey, 10);

            // 再次检查缓存
            const cachedAgain = await redisClient.get(cacheKey);
            if (cachedAgain) {
                return JSON.parse(cachedAgain);
            }

            // 从数据库获取
            const data = await fetchFn();

            // 缓存数据
            if (data !== null) {
                await redisClient.setEx(cacheKey, ttl, JSON.stringify(data));
            }

            return data;
        } finally {
            // 释放锁
            await redisClient.del(lockKey);
        }
    }

    // 删除缓存
    async invalidate(key: string) {
        const cacheKey = this.getCacheKey(key);
        await redisClient.del(cacheKey);
    }

    // 批量删除缓存
    async invalidatePattern(pattern: string) {
        const keys = await redisClient.keys(`${this.CACHE_PREFIX}${pattern}`);
        if (keys.length > 0) {
            await redisClient.del(keys);
        }
    }
}

export { CacheService };
```

```typescript
/**
 * Redis分布式锁
 * @author erik.zhou
 */

import { redisClient } from './client';

class DistributedLock {
    private readonly lockPrefix = 'lock:';
    private readonly defaultTTL = 30; // 30秒

    // 获取锁
    async acquire(
        key: string,
        ttl: number = this.defaultTTL
    ): Promise<string | null> {
        const lockKey = `${this.lockPrefix}${key}`;
        const lockValue = `${Date.now()}-${Math.random()}`;

        const acquired = await redisClient.set(lockKey, lockValue, {
            NX: true,
            EX: ttl
        });

        return acquired ? lockValue : null;
    }

    // 释放锁
    async release(key: string, lockValue: string): Promise<boolean> {
        const lockKey = `${this.lockPrefix}${key}`;

        // Lua脚本确保原子性
        const script = `
            if redis.call("get", KEYS[1]) == ARGV[1] then
                return redis.call("del", KEYS[1])
            else
                return 0
            end
        `;

        const result = await redisClient.eval(script, {
            keys: [lockKey],
            arguments: [lockValue]
        });

        return result === 1;
    }

    // 使用锁执行操作
    async withLock<T>(
        key: string,
        fn: () => Promise<T>,
        ttl: number = this.defaultTTL
    ): Promise<T> {
        const lockValue = await this.acquire(key, ttl);

        if (!lockValue) {
            throw new Error('获取锁失败');
        }

        try {
            return await fn();
        } finally {
            await this.release(key, lockValue);
        }
    }
}

export { DistributedLock };
```

```sql
/**
```

```typescript
import { AppDataSource } from './ormconfig';
import { User } from './entities/User';
import { Post } from './entities/Post';

class TransactionService {
    // TypeORM事务
    async createUserWithPost(userData: any, postData: any) {
        return await AppDataSource.transaction(async (manager) => {
            // 创建用户
            const user = manager.create(User, userData);
            await manager.save(user);

            // 创建文章
            const post = manager.create(Post, {
                ...postData,
                userId: user.id
            });
            await manager.save(post);

            return { user, post };
        });
    }

    // 手动事务控制
    async manualTransaction() {
        const queryRunner = AppDataSource.createQueryRunner();
        await queryRunner.connect();
        await queryRunner.startTransaction();

        try {
            const user = queryRunner.manager.create(User, {
                name: 'John',
                email: 'john@example.com',
                password: 'password123'
            });
            await queryRunner.manager.save(user);

            const post = queryRunner.manager.create(Post, {
                title: '标题',
                content: '内容',
                userId: user.id
            });
            await queryRunner.manager.save(post);

            await queryRunner.commitTransaction();
            return { user, post };
        } catch (error) {
            await queryRunner.rollbackTransaction();
            throw error;
        } finally {
            await queryRunner.release();
        }
    }

    // 嵌套事务
    async nestedTransaction() {
        return await AppDataSource.transaction(async (manager) => {
            const user = await manager.save(User, {
                name: 'John',
                email: 'john@example.com',
                password: 'password123'
            });

            // 嵌套事务
            await manager.transaction(async (nestedManager) => {
                await nestedManager.save(Post, {
                    title: '标题',
                    content: '内容',
                    userId: user.id
                });
            });

            return user;
        });
    }
}
```

```typescript
                    password: 'password123'

                    title: '标题',
                    content: '内容',
                    userId: user.id
                },
                { transaction: t }
            );

            return { user, post };
        });
    }

    // 非托管事务
    async unmanagedTransaction() {
        const t = await sequelize.transaction();

        try {
            const user = await User.create(
                {
                    name: 'John',
                    email: 'john@example.com',
                    password: 'password123'
                },
                { transaction: t }
            );

            const post = await Post.create(
                {
                    title: '标题',
                    content: '内容',
                    userId: user.id
                },
                { transaction: t }
            );

            await t.commit();
            return { user, post };
        } catch (error) {
            await t.rollback();
            throw error;
        }
    }

    // 事务隔离级别
    async isolationLevel() {
        return await sequelize.transaction(
            {
                isolationLevel: Transaction.ISOLATION_LEVELS.READ_COMMITTED
            },
            async (t) => {
```

### 9.1 查询优化

```typescript
/**
 * 查询性能优化
 * @author erik.zhou
 */

import { AppDataSource } from './ormconfig';
import { User } from './entities/User';

class QueryOptimization {
    private userRepository = AppDataSource.getRepository(User);

    // 使用索引
    async optimizedQuery() {
        return await this.userRepository.find({
            where: { email: 'test@example.com' }, // 使用索引字段
            select: ['id', 'name', 'email'] // 只查询需要的字段
        });
    }

    // 避免N+1查询
    async avoidNPlusOne() {
        return await this.userRepository.find({
            relations: ['posts'], // 一次性加载关联数据
            where: { status: 'active' }
        });
    }

    // 分页优化
    async optimizedPagination(page: number, limit: number) {
        // 使用游标分页代替偏移分页
        const lastId = (page - 1) * limit;
        return await this.userRepository
            .createQueryBuilder('user')
            .where('user.id > :lastId', { lastId: lastId })
            .orderBy('user.id', 'ASC')
            .take(limit)
            .getMany();
    }

    // 批量操作
    async batchInsert(users: any[]) {
        // 使用批量插入代替循环插入
        return await this.userRepository
            .createQueryBuilder()
            .insert()
            .into(User)
            .values(users)
            .execute();
    }

    // 使用缓存
    async cachedQuery(userId: number) {
        const cacheKey = `user:${userId}`;
        
        // 先从缓存获取
        const cached = await redisClient.get(cacheKey);
        if (cached) {
            return JSON.parse(cached);
        }

        // 从数据库查询
        const user = await this.userRepository.findOne({
            where: { id: userId }
        });

        // 缓存结果
        if (user) {
            await redisClient.setEx(cacheKey, 3600, JSON.stringify(user));
        }

        return user;
    }
}
```
