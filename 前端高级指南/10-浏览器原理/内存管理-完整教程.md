```javascript
        if (obj.hasOwnProperty(key)) {
            clonedObj[key] = deepClone(obj[key]);
        }
    }
    return clonedObj;
```

#### 标记阶段
```javascript
/**
 * 标记阶段模拟
 * @author erik.zhou
 */

class MarkSweepSimulator {
    constructor() {
        this.objects = new Map();
        this.roots = new Set();
    }
    
    // 分配对象
    allocate(id, data) {
        this.objects.set(id, {
            data: data,
            marked: false,
            references: []
        });
        return id;
    }
    
    // 添加引用
    addReference(fromId, toId) {
        const obj = this.objects.get(fromId);
        if (obj) {
            obj.references.push(toId);
        }
    }
    
    // 添加根对象
    addRoot(id) {
        this.roots.add(id);
    }
    
    // 标记阶段
    mark() {
        const visited = new Set();
        
        const markRecursive = (id) => {
            if (visited.has(id)) return;
            
            visited.add(id);
            const obj = this.objects.get(id);
            
            if (obj) {
                obj.marked = true;
                obj.references.forEach(refId => markRecursive(refId));
            }
        };
        
        // 从根对象开始标记
        this.roots.forEach(rootId => markRecursive(rootId));
    }
    
    // 清除阶段
    sweep() {
        const toDelete = [];
        
        this.objects.forEach((obj, id) => {
            if (!obj.marked) {
                toDelete.push(id);
            } else {
                obj.marked = false; // 重置标记
            }
        });
        
        toDelete.forEach(id => this.objects.delete(id));
        
        return toDelete.length;
    }
    
    // 执行垃圾回收
    collectGarbage() {
        this.mark();
        const collected = this.sweep();
        console.log(`回收了 ${collected} 个对象`);
        return collected;
    }
}

// 使用示例
const gc = new MarkSweepSimulator();
const obj1 = gc.allocate('obj1', { name: 'Object 1' });
const obj2 = gc.allocate('obj2', { name: 'Object 2' });
const obj3 = gc.allocate('obj3', { name: 'Object 3' });

gc.addReference('obj1', 'obj2');
gc.addRoot('obj1'); // obj1是根对象

gc.collectGarbage(); // obj3将被回收
```

### 2.2 引用计数算法

#### 基本实现
```javascript
/**
 * 引用计数算法示例
 * @author erik.zhou
 */

class ReferenceCountingGC {
    constructor() {
        this.objects = new Map();
    }
    
    // 创建对象
    createObject(id, data) {
        this.objects.set(id, {
            data: data,
            refCount: 0,
            references: []
        });
        return id;
    }
    
    // 增加引用
    addReference(fromId, toId) {
        const fromObj = this.objects.get(fromId);
        const toObj = this.objects.get(toId);
        
        if (fromObj && toObj) {
            fromObj.references.push(toId);
            toObj.refCount++;
        }
    }
    
    // 移除引用
    removeReference(fromId, toId) {
        const fromObj = this.objects.get(fromId);
        const toObj = this.objects.get(toId);
        
        if (fromObj && toObj) {
            const index = fromObj.references.indexOf(toId);
            if (index > -1) {
                fromObj.references.splice(index, 1);
                toObj.refCount--;
                
                // 引用计数为0时回收
                if (toObj.refCount === 0) {
                    this.collectObject(toId);
                }
            }
        }
    }
    
    // 回收对象
    collectObject(id) {
        const obj = this.objects.get(id);
        if (obj) {
            // 递归减少引用的对象的计数
            obj.references.forEach(refId => {
                this.removeReference(id, refId);
            });
            
            this.objects.delete(id);
            console.log(`回收对象: ${id}`);
        }
    }
    
    // 获取对象信息
    getObjectInfo(id) {
        const obj = this.objects.get(id);
        if (obj) {
            return {
                id: id,
                refCount: obj.refCount,
                references: obj.references
            };
        }
        return null;
    }
}

// 使用示例
const refGC = new ReferenceCountingGC();
const a = refGC.createObject('a', { name: 'A' });
const b = refGC.createObject('b', { name: 'B' });
const c = refGC.createObject('c', { name: 'C' });

refGC.addReference('a', 'b');
refGC.addReference('a', 'c');
refGC.addReference('b', 'c');

console.log(refGC.getObjectInfo('c')); // refCount: 2

refGC.removeReference('a', 'c');
console.log(refGC.getObjectInfo('c')); // refCount: 1
```

### 2.3 分代回收

#### 新生代回收
```javascript
/**
 * 新生代回收示例（Scavenge算法）
 * @author erik.zhou
 */

class ScavengeGC {
    constructor() {
        this.fromSpace = [];
        this.toSpace = [];
        this.maxSize = 100;
    }
    
    // 分配对象
    allocate(obj) {
        if (this.fromSpace.length >= this.maxSize) {
            this.scavenge();
        }
        
        this.fromSpace.push({
            data: obj,
            age: 0,
            alive: true
        });
    }
    
    // Scavenge回收
    scavenge() {
        console.log('执行Scavenge回收...');
        
        // 复制存活对象到toSpace
        this.fromSpace.forEach(obj => {
            if (obj.alive) {
                obj.age++;
                this.toSpace.push(obj);
            }
        });
        
        // 交换空间
        [this.fromSpace, this.toSpace] = [this.toSpace, this.fromSpace];
        this.toSpace = [];
        
        // 晋升老年代
        this.promoteToOldGeneration();
    }
    
    // 晋升到老年代
    promoteToOldGeneration() {
        const promoted = [];
        this.fromSpace = this.fromSpace.filter(obj => {
            if (obj.age > 2) { // 经历2次回收后晋升
                promoted.push(obj);
                return false;
            }
            return true;
        });
        
        if (promoted.length > 0) {
            console.log(`晋升 ${promoted.length} 个对象到老年代`);
        }
    }
    
    // 标记对象为死亡
    markDead(index) {
        if (this.fromSpace[index]) {
            this.fromSpace[index].alive = false;
        }
    }
}

// 使用示例
const scavengeGC = new ScavengeGC();

for (let i = 0; i < 150; i++) {
    scavengeGC.allocate({ id: i, data: new Array(100) });
    
    // 模拟部分对象死亡
    if (i % 3 === 0) {
        scavengeGC.markDead(i);
    }
}
```

#### 老年代回收
```javascript
/**
 * 老年代回收示例（Mark-Compact算法）
 * @author erik.zhou
 */

class MarkCompactGC {
    constructor() {
        this.oldGeneration = [];
        this.roots = new Set();
    }
    
    // 添加对象到老年代
    addToOldGeneration(obj) {
        this.oldGeneration.push({
            data: obj,
            marked: false,
            forwarding: null
        });
    }
    
    // 标记阶段
    mark() {
        const markRecursive = (obj) => {
            if (obj.marked) return;
            
            obj.marked = true;
            
            // 标记引用的对象
            if (obj.data && obj.data.references) {
                obj.data.references.forEach(ref => {
                    const refObj = this.oldGeneration.find(o => o.data === ref);
                    if (refObj) {
                        markRecursive(refObj);
                    }
                });
            }
        };
        
        // 从根对象开始标记
        this.roots.forEach(rootIndex => {
            if (this.oldGeneration[rootIndex]) {
                markRecursive(this.oldGeneration[rootIndex]);
            }
        });
    }
    
    // 整理阶段
    compact() {
        let writeIndex = 0;
        
        // 计算新位置
        for (let i = 0; i < this.oldGeneration.length; i++) {
            const obj = this.oldGeneration[i];
            if (obj.marked) {
                obj.forwarding = writeIndex++;
            }
        }
        
        // 更新引用
        this.oldGeneration.forEach(obj => {
            if (obj.marked && obj.data && obj.data.references) {
                obj.data.references = obj.data.references.map(ref => {
                    const refObj = this.oldGeneration.find(o => o.data === ref);
                    return refObj && refObj.forwarding !== null 
                        ? this.oldGeneration[refObj.forwarding].data 
                        : ref;
                });
            }
        });
        
        // 压缩内存
        const compacted = [];
        this.oldGeneration.forEach(obj => {
            if (obj.marked) {
                obj.marked = false;
                obj.forwarding = null;
                compacted.push(obj);
            }
        });
        
        const collected = this.oldGeneration.length - compacted.length;
        this.oldGeneration = compacted;
        
        console.log(`Mark-Compact回收了 ${collected} 个对象`);
        return collected;
    }
    
    // 执行垃圾回收
    collectGarbage() {
        this.mark();
        return this.compact();
    }
}

// 使用示例
const markCompactGC = new MarkCompactGC();

for (let i = 0; i < 100; i++) {
    markCompactGC.addToOldGeneration({ id: i, data: new Array(1000) });
}

markCompactGC.roots.add(0);
markCompactGC.roots.add(10);
markCompactGC.roots.add(20);

markCompactGC.collectGarbage();
```

### 2.4 增量标记

#### 增量标记实现
```javascript
/**
 * 增量标记示例
 * @author erik.zhou
 */

class IncrementalMarkingGC {
    constructor() {
        this.objects = new Map();
        this.markingQueue = [];
        this.isMarking = false;
        this.markingStep = 10; // 每次标记10个对象
    }
    
    // 分配对象
    allocate(id, data) {
        this.objects.set(id, {
            data: data,
            marked: false,
            references: []
        });
    }
    
    // 添加引用
    addReference(fromId, toId) {
        const obj = this.objects.get(fromId);
        if (obj) {
            obj.references.push(toId);
        }
    }
    
    // 开始增量标记
    startIncrementalMarking(roots) {
        this.isMarking = true;
        this.markingQueue = [...roots];
        console.log('开始增量标记...');
    }
    
    // 执行一步增量标记
    performIncrementalMarkingStep() {
        if (!this.isMarking || this.markingQueue.length === 0) {
            if (this.isMarking) {
                this.finishMarking();
            }
            return false;
        }
        
        const step = Math.min(this.markingStep, this.markingQueue.length);
        
        for (let i = 0; i < step; i++) {
            const id = this.markingQueue.shift();
            const obj = this.objects.get(id);
            
            if (obj && !obj.marked) {
                obj.marked = true;
                this.markingQueue.push(...obj.references);
            }
        }
        
        console.log(`标记进度: ${this.markingQueue.length} 个对象待标记`);
        return true;
    }
    
    // 完成标记
    finishMarking() {
        this.isMarking = false;
        console.log('标记完成，开始清除...');
        this.sweep();
    }
    
    // 清除未标记对象
    sweep() {
        const toDelete = [];
        
        this.objects.forEach((obj, id) => {
            if (!obj.marked) {
                toDelete.push(id);
            } else {
                obj.marked = false;
            }
        });
        
        toDelete.forEach(id => this.objects.delete(id));
        console.log(`清除了 ${toDelete.length} 个对象`);
    }
    
    // 模拟与应用程序交替执行
    runWithApplication(roots, appSteps) {
        this.startIncrementalMarking(roots);
        
        let step = 0;
        const interval = setInterval(() => {
            // 执行一步GC
            const hasMore = this.performIncrementalMarkingStep();
            
            // 模拟应用程序执行
            console.log(`应用程序执行步骤 ${step++}`);
            
            if (!hasMore || step >= appSteps) {
                clearInterval(interval);
                console.log('增量标记完成');
            }
        }, 100);
    }
}

// 使用示例
const incrementalGC = new IncrementalMarkingGC();

// 创建对象图
for (let i = 0; i < 100; i++) {
    incrementalGC.allocate(`obj${i}`, { id: i });
}

// 添加引用关系
for (let i = 0; i < 99; i++) {
    incrementalGC.addReference(`obj${i}`, `obj${i + 1}`);
}

// 执行增量标记
incrementalGC.runWithApplication(['obj0'], 20);
```

```javascript
        return largeData ? largeData[0] : null;

            }
```
```javascript
        },
        get(key) {

class LRUCache {
    constructor(maxSize = 100) {
        this.maxSize = maxSize;
        this.cache = new Map();
    }
    
    get(key) {
        if (!this.cache.has(key)) {
            return undefined;
        }
        
        // 移到最后（最近使用）
        const value = this.cache.get(key);
        this.cache.delete(key);
        this.cache.set(key, value);
        
        return value;
    }
    
    set(key, value) {
        // 如果已存在，先删除
        if (this.cache.has(key)) {
            this.cache.delete(key);
        }
        
        // 如果超过最大容量，删除最旧的
        if (this.cache.size >= this.maxSize) {
            const firstKey = this.cache.keys().next().value;
            this.cache.delete(firstKey);
        }
        
        this.cache.set(key, value);
    }
    
    clear() {
        this.cache.clear();
    }
```

```javascript
        this.listeners = new Map();
    }
    
    on(target, event, handler, options) {
        if (!this.listeners.has(target)) {
            this.listeners.set(target, []);
        }
        
        target.addEventListener(event, handler, options);
        this.listeners.get(target).push({ event, handler, options });

                targetListeners.splice(index, 1);

            });
            this.listeners.delete(target);

            listeners.forEach(({ event, handler }) => {
                target.removeEventListener(event, handler);
```

```javascript
class ObjectPool {
    constructor(createFn, resetFn, initialSize = 10) {
        this.createFn = createFn;
        this.resetFn = resetFn;
        this.pool = [];
        this.inUse = new Set();
        
        // 预创建对象
        for (let i = 0; i < initialSize; i++) {
            this.pool.push(this.createFn());
        }
    }

    acquire() {
        let obj;

            obj = this.pool.pop();
        } else {
            obj = this.createFn();
        }
        
        this.inUse.add(obj);
        return obj;
    }
    
    // 释放对象

            this.resetFn(obj);
            this.pool.push(obj);
        }

    // 获取统计信息
    getStats() {
        return {
            poolSize: this.pool.length,
            inUse: this.inUse.size,
            total: this.pool.length + this.inUse.size
```

```javascript
/**
 * 通用对象池实现
 * @author erik.zhou
 */

class GenericPool {
    constructor(factory, options = {}) {
        this.factory = factory;
        this.minSize = options.minSize || 0;
        this.maxSize = options.maxSize || 100;
        this.idleTimeout = options.idleTimeout || 30000;
        
        this.available = [];
        this.inUse = new Map();
        this.creating = 0;
        
        // 初始化最小数量
        this.initialize();
    }
    
    async initialize() {
        const promises = [];
        for (let i = 0; i < this.minSize; i++) {
            promises.push(this.createObject());
        }
        await Promise.all(promises);
    }
    
    async createObject() {
        this.creating++;
        try {
            const obj = await this.factory.create();
            this.available.push({
                obj: obj,
                createdAt: Date.now(),
                lastUsed: Date.now()
            });
            return obj;
        } finally {
            this.creating--;
        }
    }
    
    async acquire() {
        // 清理过期对象
        this.cleanupIdle();
        
        // 如果有可用对象，直接返回
        if (this.available.length > 0) {
            const wrapper = this.available.pop();
            this.inUse.set(wrapper.obj, wrapper);
            return wrapper.obj;
        }
        
        // 如果未达到最大数量，创建新对象
        if (this.size() < this.maxSize) {
            const obj = await this.createObject();
            const wrapper = this.available.pop();
            this.inUse.set(wrapper.obj, wrapper);
            return wrapper.obj;
        }
        
        // 等待对象可用
        return new Promise((resolve) => {
            const checkAvailable = setInterval(() => {
                if (this.available.length > 0) {
                    clearInterval(checkAvailable);
                    const wrapper = this.available.pop();
                    this.inUse.set(wrapper.obj, wrapper);
                    resolve(wrapper.obj);
                }
            }, 100);
        });
    }
    
    async release(obj) {
        const wrapper = this.inUse.get(obj);
        if (wrapper) {
            this.inUse.delete(obj);
            
            // 验证对象是否仍然有效
            if (this.factory.validate && !await this.factory.validate(obj)) {
                await this.factory.destroy(obj);
                return;
            }
            
            // 重置对象
            if (this.factory.reset) {
                await this.factory.reset(obj);
            }
            
            wrapper.lastUsed = Date.now();
            this.available.push(wrapper);
        }
    }
    
    cleanupIdle() {
        const now = Date.now();
        this.available = this.available.filter(wrapper => {
            if (now - wrapper.lastUsed > this.idleTimeout) {
                this.factory.destroy(wrapper.obj);
                return false;
            }
            return true;
        });
    }
    
    size() {
        return this.available.length + this.inUse.size + this.creating;
    }
    
    async drain() {
        // 等待所有使用中的对象释放
        while (this.inUse.size > 0) {
            await new Promise(resolve => setTimeout(resolve, 100));
        }
        
        // 销毁所有对象
        for (const wrapper of this.available) {
            await this.factory.destroy(wrapper.obj);
        }
        
        this.available = [];
    }
}

// 使用示例：数据库连接池
```

```javascript
class BufferPool {
    constructor(bufferSize = 1024, poolSize = 10) {
        this.bufferSize = bufferSize;
        this.pool = [];
        
        // 预分配缓冲区
        for (let i = 0; i < poolSize; i++) {
            this.pool.push(new ArrayBuffer(bufferSize));
        }
    }
    
    acquire() {
        if (this.pool.length > 0) {
            return this.pool.pop();
        }
        return new ArrayBuffer(this.bufferSize);
    }
    
    release(buffer) {
        if (buffer.byteLength === this.bufferSize) {
            this.pool.push(buffer);
        }
    }
    
    clear() {
        this.pool = [];
    }
}

    const view = new Uint8Array(buffer);
    
    // 处理数据
    for (let i = 0; i < data.length && i < view.length; i++) {
        view[i] = data[i];
    }
    
    // 使用完毕后释放
    bufferPool.release(buffer);
}
```

```javascript
 * 字符串池实现
 * @author erik.zhou

class StringPool {
    constructor() {
        this.pool = new Map();
        this.stats = {
            hits: 0,
            misses: 0,
            size: 0
        };
    }
    
    intern(str) {
        if (this.pool.has(str)) {
            this.stats.hits++;
            return this.pool.get(str);
        }
        
        this.stats.misses++;
        this.pool.set(str, str);
        this.stats.size = this.pool.size;
        return str;
    }
    
    clear() {
        this.pool.clear();
        this.stats.size = 0;
    }
    
    getStats() {
        return {
            ...this.stats,
            hitRate: this.stats.hits / (this.stats.hits + this.stats.misses)
        };
    }
}

// 使用示例

// 重复字符串会复用同一个实例

const str2 = stringPool.intern('hello');
console.log(str1 === str2); // true

console.log(stringPool.getStats());
```

```javascript
// 使用WeakMap存储对象元数据
const metadata = new WeakMap();

class DOMElement {
    constructor(tag) {
        this.element = document.createElement(tag);
        
        // 使用WeakMap存储私有数据
        metadata.set(this.element, {
            createdAt: Date.now(),
            listeners: []
        });
    }
    
    addEventListener(event, handler) {
        const data = metadata.get(this.element);
        data.listeners.push({ event, handler });
        this.element.addEventListener(event, handler);
    }
    
    destroy() {
        const data = metadata.get(this.element);
        if (data) {
            data.listeners.forEach(({ event, handler }) => {
                this.element.removeEventListener(event, handler);
            });
        }
        
        if (this.element.parentNode) {
            this.element.parentNode.removeChild(this.element);
        }
        
        // WeakMap会自动清理
    }
}

// 使用WeakSet跟踪对象
const processedObjects = new WeakSet();

function processObject(obj) {

        console.log('对象已处理');

    }

    console.log('处理对象');
    processedObjects.add(obj);
}
```

```javascript
/**
 * 懒加载实现
 * @author erik.zhou

class LazyLoader {
    constructor() {
        this.cache = new Map();
    }
    
    // 懒加载属性
    lazy(target, key, loader) {
        Object.defineProperty(target, key, {
            get() {
                if (!this.cache.has(key)) {
                    const value = loader();
                    this.cache.set(key, value);
                }
                return this.cache.get(key);
            },
            configurable: true
        });
    }
    
    // 懒加载方法
    lazyMethod(target, key, loader) {
        const original = target[key];
        
        target[key] = function(...args) {
            if (!this.cache.has(key)) {
                const value = loader.call(this);
                this.cache.set(key, value);
            }
            
            const cached = this.cache.get(key);
            return cached.call(this, ...args);
        };
    }
}

// 使用示例
class DataService {
    constructor() {
        this.loader = new LazyLoader();
        
        // 懒加载大数据
        this.loader.lazy(this, 'largeData', () => {
            console.log('加载大数据...');
            return new Array(1000000).fill(0).map((_, i) => i);
        });
    }
    
    processData() {
        // 只有在访问时才加载
        return this.largeData.length;
    }
}

const service = new DataService();
console.log('服务已创建');
// 此时largeData还未加载

console.log(service.processData());
// 现在才加载largeData
```

```javascript
/**
 * 虚拟滚动实现
 * @author erik.zhou

class VirtualScroller {
    constructor(container, options = {}) {
        this.container = container;
        this.itemHeight = options.itemHeight || 50;
        this.bufferSize = options.bufferSize || 5;
        this.items = [];
        this.visibleItems = [];
        
        this.setupContainer();
        this.setupScrollListener();
    }
    
    setupContainer() {
        this.container.style.overflow = 'auto';
        this.container.style.position = 'relative';
        
        this.content = document.createElement('div');
        this.content.style.position = 'relative';
        this.container.appendChild(this.content);
    }
    
    setupScrollListener() {
        this.container.addEventListener('scroll', () => {
            this.render();
        });
    }
    
    setItems(items) {
        this.items = items;
        this.content.style.height = `${items.length * this.itemHeight}px`;
        this.render();
    }
    
    render() {
        const scrollTop = this.container.scrollTop;
        const containerHeight = this.container.clientHeight;
        
        // 计算可见范围
        const startIndex = Math.max(0, 
            Math.floor(scrollTop / this.itemHeight) - this.bufferSize
        );
        const endIndex = Math.min(this.items.length,
            Math.ceil((scrollTop + containerHeight) / this.itemHeight) + this.bufferSize
        );
        
        // 清空现有元素
        this.content.innerHTML = '';
        
        // 只渲染可见元素
        for (let i = startIndex; i < endIndex; i++) {
            const item = this.createItem(this.items[i], i);
            this.content.appendChild(item);
        }
    }
    
    createItem(data, index) {
        const item = document.createElement('div');
        item.style.position = 'absolute';
        item.style.top = `${index * this.itemHeight}px`;
        item.style.height = `${this.itemHeight}px`;
        item.style.width = '100%';
        item.textContent = data;
        return item;
    }
}

// 使用示例
const container = document.createElement('div');
container.style.height = '400px';
document.body.appendChild(container);

const scroller = new VirtualScroller(container, {
    itemHeight: 50,
    bufferSize: 5
});

// 设置大量数据
const items = Array.from({ length: 10000 }, (_, i) => `Item ${i}`);
scroller.setItems(items);
```

```javascript
class MemoryProfiler {
    constructor() {
        this.snapshots = [];

    // 记录堆快照
    takeHeapSnapshot(label) {
        if (performance.memory) {
            const snapshot = {
                label: label,
                timestamp: Date.now(),
                usedJSHeapSize: performance.memory.usedJSHeapSize,
                totalJSHeapSize: performance.memory.totalJSHeapSize,
                jsHeapSizeLimit: performance.memory.jsHeapSizeLimit
            };
            
            this.snapshots.push(snapshot);
            console.log(`[${label}] 堆快照:`, snapshot);
            return snapshot;
        }
        return null;
    }
    
    // 比较快照
    compareSnapshots(label1, label2) {
        const snap1 = this.snapshots.find(s => s.label === label1);
        const snap2 = this.snapshots.find(s => s.label === label2);
        
        if (snap1 && snap2) {
            const diff = {
                usedDiff: snap2.usedJSHeapSize - snap1.usedJSHeapSize,
                totalDiff: snap2.totalJSHeapSize - snap1.totalJSHeapSize,
                timeDiff: snap2.timestamp - snap1.timestamp
            };
            
            console.log('快照比较:', {
                from: label1,
                to: label2,
                usedDiff: `${(diff.usedDiff / 1024 / 1024).toFixed(2)} MB`,
                totalDiff: `${(diff.totalDiff / 1024 / 1024).toFixed(2)} MB`,
                timeDiff: `${diff.timeDiff} ms`
            });
            
            return diff;
        }
        return null;
    }
    
    // 生成报告
    generateReport() {
        if (this.snapshots.length === 0) {
            return '没有快照数据';
        }
        
        const report = {
            totalSnapshots: this.snapshots.length,
            firstSnapshot: this.snapshots[0],
            lastSnapshot: this.snapshots[this.snapshots.length - 1],
            maxUsed: Math.max(...this.snapshots.map(s => s.usedJSHeapSize)),
            minUsed: Math.min(...this.snapshots.map(s => s.usedJSHeapSize)),
            avgUsed: this.snapshots.reduce((sum, s) => sum + s.usedJSHeapSize, 0) / this.snapshots.length
        };
        
        console.log('内存分析报告:', {
            ...report,
            maxUsed: `${(report.maxUsed / 1024 / 1024).toFixed(2)} MB`,
            minUsed: `${(report.minUsed / 1024 / 1024).toFixed(2)} MB`,
            avgUsed: `${(report.avgUsed / 1024 / 1024).toFixed(2)} MB`
        });
        
        return report;
    }
```

#### 内存监控系统
```javascript
/**
 * 内存监控系统
 * @author erik.zhou
 */

class MemoryMonitoringSystem {
    constructor(options = {}) {
        this.interval = options.interval || 1000;
        this.threshold = options.threshold || 0.8; // 80%
        this.callbacks = {
            onWarning: options.onWarning || (() => {}),
            onCritical: options.onCritical || (() => {}),
            onNormal: options.onNormal || (() => {})
        };
        
        this.history = [];
        this.isMonitoring = false;
        this.monitoringTimer = null;
    }
    
    start() {
        if (this.isMonitoring) return;
        
        this.isMonitoring = true;
        this.monitoringTimer = setInterval(() => {
            this.check();
        }, this.interval);
        
        console.log('内存监控已启动');
    }
    
    stop() {
        if (!this.isMonitoring) return;
        
        this.isMonitoring = false;
        if (this.monitoringTimer) {
            clearInterval(this.monitoringTimer);
            this.monitoringTimer = null;
        }
        
        console.log('内存监控已停止');
    }
    
    check() {
        if (!performance.memory) {
            console.warn('performance.memory不可用');
            return;
        }
        
        const memory = performance.memory;
        const usage = memory.usedJSHeapSize / memory.jsHeapSizeLimit;
        
        const status = {
            timestamp: Date.now(),
            usedJSHeapSize: memory.usedJSHeapSize,
            totalJSHeapSize: memory.totalJSHeapSize,
            jsHeapSizeLimit: memory.jsHeapSizeLimit,
            usage: usage,
            level: this.getLevel(usage)
        };
        
        this.history.push(status);
        
        // 保持历史记录在合理范围内
        if (this.history.length > 100) {
            this.history.shift();
        }
        
        // 触发回调
        this.triggerCallbacks(status);
        
        return status;
    }
    
    getLevel(usage) {
        if (usage >= 0.9) return 'critical';
        if (usage >= this.threshold) return 'warning';
        return 'normal';
    }
    
    triggerCallbacks(status) {
        switch (status.level) {
            case 'critical':
                this.callbacks.onCritical(status);
                break;
            case 'warning':
                this.callbacks.onWarning(status);
                break;
            case 'normal':
                this.callbacks.onNormal(status);
                break;
        }
    }
    
    getReport() {
        if (this.history.length === 0) {
            return null;
        }
        
        const usages = this.history.map(h => h.usage);
        const maxUsage = Math.max(...usages);
        const minUsage = Math.min(...usages);
        const avgUsage = usages.reduce((sum, u) => sum + u, 0) / usages.length;
        
        const levels = this.history.reduce((acc, h) => {
            acc[h.level] = (acc[h.level] || 0) + 1;
            return acc;
        }, {});
        
        return {
            duration: this.history[this.history.length - 1].timestamp - this.history[0].timestamp,
            samples: this.history.length,
            maxUsage: (maxUsage * 100).toFixed(2) + '%',
            minUsage: (minUsage * 100).toFixed(2) + '%',
            avgUsage: (avgUsage * 100).toFixed(2) + '%',
            levels: levels
        };
    }
}

// 使用示例
const monitor = new MemoryMonitoringSystem({
    interval: 1000,
    threshold: 0.7,
    onWarning: (status) => {
        console.warn('内存使用警告:', (status.usage * 100).toFixed(2) + '%');
    },
    onCritical: (status) => {
        console.error('内存使用严重:', (status.usage * 100).toFixed(2) + '%');
        // 可以触发清理操作
    },
    onNormal: (status) => {
        console.log('内存使用正常:', (status.usage * 100).toFixed(2) + '%');
    }
});

monitor.start();

// 运行一段时间后查看报告
setTimeout(() => {
    const report = monitor.getReport();
    console.log('监控报告:', report);
    monitor.stop();
}, 10000);
```
