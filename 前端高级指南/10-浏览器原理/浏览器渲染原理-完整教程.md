## 渲染流程概述

### 完整渲染流程
```
1. 解析HTML → 构建DOM树
2. 解析CSS → 构建CSSOM树
3. DOM + CSSOM → 构建渲染树（Render Tree）
4. 布局（Layout）→ 计算元素位置和大小
5. 绘制（Paint）→ 绘制像素
6. 合成（Composite）→ 合成图层
```

### 渲染树构建
```javascript
// 合并DOM和CSSOM生成渲染树
function buildRenderTree(domNode, cssomTree) {
  // 跳过不可见元素
  if (domNode.style?.display === 'none') {
    return null;
  }
  
  // 创建渲染节点
  const renderNode = {
    type: domNode.type,
    styles: getComputedStyles(domNode, cssomTree),
    children: []
  };
  
  // 递归处理子节点
  domNode.children?.forEach(child => {
    const childRenderNode = buildRenderTree(child, cssomTree);
    if (childRenderNode) {
      renderNode.children.push(childRenderNode);
    }
  });
  
  return renderNode;
}

function getComputedStyles(domNode, cssomTree) {
  // 计算最终样式（包括继承、层叠等）
  const styles = {};
  
  // 1. 浏览器默认样式
  Object.assign(styles, getDefaultStyles(domNode.type));
  
  // 2. 用户样式表
  Object.assign(styles, getUserStyles(domNode));
  
  // 3. 作者样式表
  Object.assign(styles, getAuthorStyles(domNode, cssomTree));
  
  // 4. 内联样式
  Object.assign(styles, domNode.style || {});
  
  return styles;
}
```

## 关键渲染路径

### 阻塞渲染的资源
```html
<!DOCTYPE html>
<html>
<head>
  <!-- CSS会阻塞渲染 -->
  <link rel="stylesheet" href="style.css">
  
  <!-- JavaScript会阻塞DOM解析 -->
  <script src="script.js"></script>
  
  <!-- async不阻塞DOM解析 -->
  <script async src="analytics.js"></script>
  
  <!-- defer延迟执行 -->
  <script defer src="app.js"></script>
</head>
<body>
  <div id="app"></div>
</body>
</html>
```

### 优化关键渲染路径
```html
<!DOCTYPE html>
<html>
<head>
  <!-- 1. 内联关键CSS -->
  <style>
    /* 首屏关键样式 */
    .hero { display: flex; }
  </style>
  
  <!-- 2. 异步加载非关键CSS -->
  <link rel="preload" href="style.css" as="style" onload="this.onload=null;this.rel='stylesheet'">
  
  <!-- 3. 预连接第三方域名 -->
  <link rel="preconnect" href="https://fonts.googleapis.com">
  
  <!-- 4. 预加载关键资源 -->
  <link rel="preload" href="hero.jpg" as="image">
</head>
<body>
  <!-- 5. 延迟加载非关键JavaScript -->
  <script defer src="app.js"></script>
</body>
</html>
```

## 重排与重绘

### 重排（Reflow）
触发重排的操作：
```javascript
// 1. 修改几何属性
element.style.width = '100px';
element.style.height = '100px';
element.style.padding = '10px';
element.style.margin = '10px';

// 2. 添加/删除元素
document.body.appendChild(newElement);
element.remove();

// 3. 改变字体
element.style.fontSize = '20px';

// 4. 激活伪类
element.classList.add('hover');

// 5. 获取布局信息（强制同步布局）
const width = element.offsetWidth;
const height = element.offsetHeight;
const top = element.offsetTop;
const rect = element.getBoundingClientRect();
```

### 重绘（Repaint）
触发重绘的操作：
```javascript
// 修改视觉属性（不影响布局）
element.style.color = 'red';
element.style.backgroundColor = 'blue';
element.style.visibility = 'hidden';
element.style.outline = '1px solid red';
```

### 避免强制同步布局
```javascript
// ❌ 错误：强制同步布局
function badLayout() {
  const elements = document.querySelectorAll('.item');
  
  elements.forEach(element => {
    // 读取布局信息
    const width = element.offsetWidth;
    
    // 修改样式（触发重排）
    element.style.width = width + 10 + 'px';
    
    // 再次读取（强制浏览器立即重排）
    const newWidth = element.offsetWidth;
  });
}

// ✅ 正确：批量读取，批量写入
function goodLayout() {
  const elements = document.querySelectorAll('.item');
  
  // 批量读取
  const widths = Array.from(elements).map(el => el.offsetWidth);
  
  // 批量写入
  elements.forEach((element, index) => {
    element.style.width = widths[index] + 10 + 'px';
  });
}
```

### 使用DocumentFragment减少重排
```javascript
// ❌ 错误：多次触发重排
function badAppend() {
  const container = document.getElementById('container');
  
  for (let i = 0; i < 1000; i++) {
    const div = document.createElement('div');
    div.textContent = `Item ${i}`;
    container.appendChild(div); // 每次都触发重排
  }
}

// ✅ 正确：使用DocumentFragment
function goodAppend() {
  const container = document.getElementById('container');
  const fragment = document.createDocumentFragment();
  
  for (let i = 0; i < 1000; i++) {
    const div = document.createElement('div');
    div.textContent = `Item ${i}`;
    fragment.appendChild(div);
  }
  
  container.appendChild(fragment); // 只触发一次重排
}
```

### 使用CSS类名代替直接修改样式

## 合成层优化

### 创建合成层
```css
/* 1. 使用transform */
.element {
  transform: translateZ(0);
  /* 或 */
  will-change: transform;
}

/* 2. 使用opacity */
.element {
  opacity: 0.99;
}

/* 3. 使用filter */
.element {
  filter: blur(5px);
}

/* 4. 使用will-change */
.element {
  will-change: transform, opacity;
}

/* 5. 使用video、canvas、iframe */
video, canvas, iframe {
  /* 自动创建合成层 */
}
```

### 合成层的优势
```javascript
// 使用transform和opacity不会触发重排和重绘
function animateWithComposite(element) {
  element.style.transform = 'translateX(100px)';
  element.style.opacity = '0.5';
  // 只触发合成，性能最优
}

// 使用left和width会触发重排
function animateWithReflow(element) {
  element.style.left = '100px';
  element.style.width = '200px';
  // 触发重排和重绘，性能较差
}
```

### 合成层最佳实践
```css
/* ✅ 推荐：使用transform实现动画 */
.slide-in {
  animation: slideIn 0.3s ease-out;
}

@keyframes slideIn {
  from {
    transform: translateX(-100%);
  }
  to {
    transform: translateX(0);
  }
}

/* ❌ 不推荐：使用left实现动画 */
.slide-in-bad {
  animation: slideInBad 0.3s ease-out;
}

@keyframes slideInBad {
  from {
    left: -100%;
  }
  to {
    left: 0;
  }
}
```

### 避免层爆炸
```css
/* ❌ 错误：过度使用will-change */
.item {
  will-change: transform, opacity, width, height;
}

/* ✅ 正确：只在需要时使用 */
.item:hover {
  will-change: transform;
}

.item {
  transition: transform 0.3s;
}
```

## 渲染性能优化

### 使用requestAnimationFrame
```javascript
// ❌ 错误：使用setTimeout
function badAnimation() {
  let position = 0;
  
  setInterval(() => {
    position += 1;
    element.style.left = position + 'px';
  }, 16); // 可能与浏览器刷新率不同步
}

// ✅ 正确：使用requestAnimationFrame
function goodAnimation() {
  let position = 0;
  
  function animate() {
    position += 1;
    element.style.transform = `translateX(${position}px)`;
    
    if (position < 100) {
      requestAnimationFrame(animate);
    }
  }
  
  requestAnimationFrame(animate);
}
```

### 虚拟滚动优化长列表
```javascript
class VirtualScroll {
  constructor(container, itemHeight, totalItems) {
    this.container = container;
    this.itemHeight = itemHeight;
    this.totalItems = totalItems;
    this.visibleItems = Math.ceil(container.clientHeight / itemHeight);
    this.scrollTop = 0;
    
    this.init();
  }
  
  init() {
    // 设置容器高度
    this.container.style.height = this.totalItems * this.itemHeight + 'px';
    
    // 监听滚动
    this.container.addEventListener('scroll', () => {
      this.scrollTop = this.container.scrollTop;
      this.render();
    });
    
    this.render();
  }
  
  render() {
    // 计算可见范围
    const startIndex = Math.floor(this.scrollTop / this.itemHeight);
    const endIndex = Math.min(
      startIndex + this.visibleItems + 1,
      this.totalItems
    );
    
    // 只渲染可见项
    const fragment = document.createDocumentFragment();
    
    for (let i = startIndex; i < endIndex; i++) {
      const item = document.createElement('div');
      item.className = 'item';
      item.style.position = 'absolute';
      item.style.top = i * this.itemHeight + 'px';
      item.style.height = this.itemHeight + 'px';
      item.textContent = `Item ${i}`;
      
      fragment.appendChild(item);
    }
    
    // 清空并重新渲染
    this.container.innerHTML = '';
    this.container.appendChild(fragment);
  }
}

// 使用
const virtualScroll = new VirtualScroll(
  document.getElementById('container'),
  50,  // 每项高度
  10000 // 总项数
);
```

### 防抖和节流
```javascript
// 防抖：延迟执行
function debounce(func, wait) {
  let timeout;
  
  return function executedFunction(...args) {
    const later = () => {
      clearTimeout(timeout);
      func(...args);
    };
    
    clearTimeout(timeout);
    timeout = setTimeout(later, wait);
  };
}

// 节流：限制执行频率
function throttle(func, limit) {
  let inThrottle;
  
  return function(...args) {
    if (!inThrottle) {
      func.apply(this, args);
      inThrottle = true;
      setTimeout(() => inThrottle = false, limit);
    }
  };
}

// 使用
window.addEventListener('scroll', throttle(() => {
  console.log('滚动事件');
}, 100));

window.addEventListener('resize', debounce(() => {
  console.log('窗口大小改变');
}, 300));
```

### 性能监控
```javascript
class RenderPerformanceMonitor {
  constructor() {
    this.metrics = {
      fps: [],
      reflows: 0,
      repaints: 0
    };
    
    this.init();
  }
  
  init() {
    // 监控FPS
    this.monitorFPS();
    
    // 监控重排重绘
    this.monitorReflowRepaint();
  }
  
  monitorFPS() {
    let lastTime = performance.now();
    let frames = 0;
    
    const measureFPS = () => {
      const currentTime = performance.now();
      frames++;
      
      if (currentTime >= lastTime + 1000) {
        const fps = Math.round((frames * 1000) / (currentTime - lastTime));
        this.metrics.fps.push(fps);
        
        console.log('FPS:', fps);
        
        frames = 0;
        lastTime = currentTime;
      }
      
      requestAnimationFrame(measureFPS);
    };
    
    requestAnimationFrame(measureFPS);
  }
  
  monitorReflowRepaint() {
    // 使用Performance Observer监控长任务
    const observer = new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        if (entry.duration > 50) {
          console.warn('长任务检测:', entry.duration + 'ms');
        }
      }
    });
    
    observer.observe({ entryTypes: ['longtask'] });
  }
  
  getAverageFPS() {
    const sum = this.metrics.fps.reduce((a, b) => a + b, 0);
    return sum / this.metrics.fps.length;
  }
}

// 使用
const monitor = new RenderPerformanceMonitor();
```
