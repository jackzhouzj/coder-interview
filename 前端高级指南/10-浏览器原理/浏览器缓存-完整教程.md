### 1.2 协商缓存

#### Last-Modified / If-Modified-Since
```javascript
/**
 * Last-Modified协商缓存
 * @author erik.zhou
 */

const fs = require('fs');
const path = require('path');

app.get('/file/:name', (req, res) => {
    const filePath = path.join(__dirname, 'files', req.params.name);
    
    // 获取文件修改时间
    fs.stat(filePath, (err, stats) => {
        if (err) {
            return res.status(404).send('File not found');
        }
        
        const lastModified = stats.mtime.toUTCString();
        const ifModifiedSince = req.headers['if-modified-since'];
        
        // 检查是否需要返回304
        if (ifModifiedSince && ifModifiedSince === lastModified) {
            res.status(304).end();
            return;
        }
        
        // 设置Last-Modified头部
        res.set({
            'Last-Modified': lastModified,
            'Cache-Control': 'public, max-age=0, must-revalidate'
        });
        
        res.sendFile(filePath);
    });
});

// 客户端请求示例
async function fetchWithLastModified(url) {
    // 首次请求
    const response1 = await fetch(url);
    const lastModified = response1.headers.get('Last-Modified');
    
    console.log('首次请求:', {
        status: response1.status,
        lastModified: lastModified
    });
    
    // 再次请求，带上If-Modified-Since
    const response2 = await fetch(url, {
        headers: {
            'If-Modified-Since': lastModified
        }
    });
    
    console.log('再次请求:', {
        status: response2.status, // 304 Not Modified
        fromCache: response2.status === 304
    });
}
```

```javascript
// 1. 文件名哈希
class AssetVersioning {
    constructor() {
        this.manifest = new Map();
    }
    
    // 生成带哈希的文件名
    generateHashedFilename(filename, content) {
        const hash = crypto
            .createHash('md5')
            .update(content)
            .digest('hex')
            .substring(0, 8);
        
        const ext = path.extname(filename);
        const base = path.basename(filename, ext);
        
        return `${base}.${hash}${ext}`;
    }
    
    // 构建资源清单
    buildManifest(files) {
        files.forEach(file => {
            const content = fs.readFileSync(file);
            const hashedName = this.generateHashedFilename(file, content);
            this.manifest.set(file, hashedName);
        });
        
        return this.manifest;
    }
    
    // 获取资源URL
    getAssetUrl(filename) {
        return this.manifest.get(filename) || filename;
    }
}

// 2. 查询参数版本
class QueryVersioning {
    constructor() {
        this.version = Date.now();
    }
    
    getAssetUrl(url) {
        return `${url}?v=${this.version}`;
    }
    
    updateVersion() {
        this.version = Date.now();
    }
}
```

#### 缓存分层策略

```javascript
/**
 * 缓存分层策略
 * @author erik.zhou
 */

class CacheStrategy {
    constructor() {
        this.strategies = {
            // 1. 永久缓存 - 带哈希的静态资源
            immutable: {
                pattern: /\.(js|css|png|jpg|gif|woff2)$/,
                headers: {
                    'Cache-Control': 'public, max-age=31536000, immutable'
                }
            },
            
            // 2. 协商缓存 - HTML文件
            revalidate: {
                pattern: /\.html$/,
                headers: {
                    'Cache-Control': 'public, max-age=0, must-revalidate'
                }
            },
            
            // 3. 短期缓存 - API数据
            shortTerm: {
                pattern: /^\/api\//,
                headers: {
                    'Cache-Control': 'private, max-age=300'
                }
            },
            
            // 4. 不缓存 - 敏感数据
            noCache: {
                pattern: /^\/auth\//,
                headers: {
                    'Cache-Control': 'no-store'
                }
            }
        };
    }
    
    getStrategy(url) {
        for (const [name, strategy] of Object.entries(this.strategies)) {
            if (strategy.pattern.test(url)) {
                return strategy;
            }
        }
        return null;
    }
    
    applyStrategy(req, res, next) {
        const strategy = this.getStrategy(req.url);
        
        if (strategy) {
            Object.entries(strategy.headers).forEach(([key, value]) => {
                res.set(key, value);
            });
        }
        
        next();
    }
}

// 使用中间件
const cacheStrategy = new CacheStrategy();
app.use((req, res, next) => {
    cacheStrategy.applyStrategy(req, res, next);
});
```

```javascript
        caches.open(CACHE_NAME)

                    '/css/style.css',

            })

self.addEventListener('activate', event => {
    console.log('Service Worker激活中...');

        caches.keys()
            .then(cacheNames => {
                // 删除旧缓存
                return Promise.all(
                    cacheNames
                        .filter(name => name !== CACHE_NAME)
                        .map(name => {
                            console.log('删除旧缓存:', name);

                );

            .then(() => {
                console.log('旧缓存已清理');
                // 立即控制所有页面
                return self.clients.claim();
            })
    );
});

    event.respondWith(
```

```javascript
/**
 * 路由匹配策略
 * @author erik.zhou

class CacheRouter {
    constructor() {
        this.routes = [];
    }
    
    // 注册路由
    register(pattern, strategy) {
        this.routes.push({ pattern, strategy });
    }
    
    // 匹配路由
    match(request) {
        for (const route of this.routes) {
            if (route.pattern.test(request.url)) {
                return route.strategy;
            }
        }
        return null;
    }
}

const router = new CacheRouter();

// 注册不同的缓存策略
router.register(/\.(?:png|jpg|jpeg|svg|gif)$/, 'cacheFirst');
router.register(/\.(?:js|css)$/, 'staleWhileRevalidate');
router.register(/^\/api\//, 'networkFirst');
router.register(/\.html$/, 'networkFirst');

self.addEventListener('fetch', event => {
    const strategy = router.match(event.request);
    
    if (!strategy) {
        event.respondWith(fetch(event.request));
        return;
    }
    
    event.respondWith(handleRequest(event.request, strategy));
});

async function handleRequest(request, strategy) {
    switch (strategy) {
        case 'cacheFirst':
            return cacheFirst(request);
        case 'networkFirst':
            return networkFirst(request);
        case 'staleWhileRevalidate':
            return staleWhileRevalidate(request);
        default:
            return fetch(request);
    }
}
```

#### 缓存过期策略
```javascript
/**
 * 缓存过期策略
 * @author erik.zhou
 */

class CacheExpiration {
    constructor(cacheName) {
        this.cacheName = cacheName;
        this.expirationMap = new Map();
    }
    
    // 设置缓存及过期时间
    async put(request, response, maxAge) {
        const cache = await caches.open(this.cacheName);
        await cache.put(request, response);
        
        const expirationTime = Date.now() + maxAge * 1000;
        this.expirationMap.set(request.url, expirationTime);
        
        // 保存过期时间到IndexedDB
        await this.saveExpiration(request.url, expirationTime);
    }
    
    // 获取缓存（检查过期）
    async match(request) {
        const cache = await caches.open(this.cacheName);
        const response = await cache.match(request);
        
        if (!response) {
            return null;
        }
        
        // 检查是否过期
        const expirationTime = await this.getExpiration(request.url);
        if (expirationTime && Date.now() > expirationTime) {
            // 已过期，删除缓存
            await cache.delete(request);
            await this.deleteExpiration(request.url);
            return null;
        }
        
        return response;
    }
    
    // 清理过期缓存
    async cleanup() {
        const cache = await caches.open(this.cacheName);
        const requests = await cache.keys();
        
        for (const request of requests) {
            const expirationTime = await this.getExpiration(request.url);
            if (expirationTime && Date.now() > expirationTime) {
                await cache.delete(request);
                await this.deleteExpiration(request.url);
            }
        }
    }
    
    // 保存过期时间到IndexedDB
    async saveExpiration(url, time) {
        const db = await this.openDB();
        const tx = db.transaction('expirations', 'readwrite');
        await tx.objectStore('expirations').put({ url, time });
    }
    
    // 获取过期时间
    async getExpiration(url) {
        const db = await this.openDB();
        const tx = db.transaction('expirations', 'readonly');
        const record = await tx.objectStore('expirations').get(url);
        return record ? record.time : null;
    }
    
    // 删除过期时间
    async deleteExpiration(url) {
        const db = await this.openDB();
        const tx = db.transaction('expirations', 'readwrite');
        await tx.objectStore('expirations').delete(url);
    }
    
    // 打开IndexedDB
    async openDB() {
        return new Promise((resolve, reject) => {
            const request = indexedDB.open('CacheExpiration', 1);
            
            request.onerror = () => reject(request.error);
            request.onsuccess = () => resolve(request.result);
            
            request.onupgradeneeded = (event) => {
                const db = event.target.result;
                if (!db.objectStoreNames.contains('expirations')) {
                    db.createObjectStore('expirations', { keyPath: 'url' });
                }
            };
        });
    }
}

// 在Service Worker中使用
const cacheExpiration = new CacheExpiration('app-cache-v1');

self.addEventListener('fetch', event => {
    event.respondWith(
        cacheExpiration.match(event.request)
            .then(cachedResponse => {
                if (cachedResponse) {
                    return cachedResponse;
                }
                
                return fetch(event.request)
                    .then(response => {
                        if (response.status === 200) {
                            // 缓存1小时
                            cacheExpiration.put(
                                event.request,
                                response.clone(),
                                3600
                            );
                        }
                        return response;
                    });
            })
    );
});

// 定期清理过期缓存
self.addEventListener('message', event => {
    if (event.data.type === 'CLEANUP_CACHE') {
```

```javascript
class CacheSizeLimit {
    constructor(cacheName, maxSize = 50) {
        this.cacheName = cacheName;
        this.maxSize = maxSize;
    }
    
    // 添加缓存并检查大小
    async put(request, response) {
        const cache = await caches.open(this.cacheName);
        await cache.put(request, response);
        
        // 检查并清理超出限制的缓存
        await this.trimCache();
    }
    
    // 清理超出限制的缓存
    async trimCache() {
        const cache = await caches.open(this.cacheName);
        const keys = await cache.keys();
        
        if (keys.length > this.maxSize) {
            // 删除最旧的缓存
            const deleteCount = keys.length - this.maxSize;
            for (let i = 0; i < deleteCount; i++) {
                await cache.delete(keys[i]);
            }
        }
    }
    
    // 获取缓存大小
    async getSize() {
        const cache = await caches.open(this.cacheName);
        const keys = await cache.keys();
        return keys.length;
    }
    
    // 获取缓存使用的存储空间
    async getStorageSize() {
        if ('storage' in navigator && 'estimate' in navigator.storage) {
            const estimate = await navigator.storage.estimate();
            return {
                usage: estimate.usage,
                quota: estimate.quota,
                percentage: (estimate.usage / estimate.quota * 100).toFixed(2) + '%'
            };
        }
        return null;
    }
}

const cacheSizeLimit = new CacheSizeLimit('app-cache-v1', 100);

        caches.match(event.request)
            .then(response => {
                if (response) {
                    return response;
                }
                
                return fetch(event.request)
                    .then(response => {
                        if (response.status === 200) {
                            cacheSizeLimit.put(event.request, response.clone());
                        }
                        return response;
                    });
            })
    );
});
```

```javascript
                reject(request.error);
            };
        });
    }
    
    // 游标查询
    async queryWithCursor(storeName, indexName, callback) {
        return new Promise((resolve, reject) => {
            const transaction = this.db.transaction([storeName], 'readonly');
            const objectStore = transaction.objectStore(storeName);
            const index = objectStore.index(indexName);
            const request = index.openCursor();
            
            const results = [];
            
            request.onsuccess = (event) => {
                const cursor = event.target.result;
                
                if (cursor) {
                    const shouldContinue = callback(cursor.value);
                    results.push(cursor.value);
                    
                    if (shouldContinue !== false) {
                        cursor.continue();
                    } else {
                        resolve(results);
                    }
                } else {
                    resolve(results);
                }
            };
            
            request.onerror = () => {
                reject(request.error);
            };
        });
    }
    
    // 分页查询
    async paginate(storeName, indexName, page = 1, pageSize = 10) {
        return new Promise((resolve, reject) => {
            const transaction = this.db.transaction([storeName], 'readonly');
            const objectStore = transaction.objectStore(storeName);
            const index = objectStore.index(indexName);
            const request = index.openCursor();
            
            const results = [];
            let skipped = 0;
            let collected = 0;
            const skipCount = (page - 1) * pageSize;

                const cursor = event.target.result;
                
                if (cursor) {
                    if (skipped < skipCount) {
                        skipped++;
                        cursor.continue();
                    } else if (collected < pageSize) {
                        results.push(cursor.value);
                        collected++;
                        cursor.continue();
                    } else {
                        resolve(results);
                    }
                } else {
                    resolve(results);
                }
            };
            
            request.onerror = () => {
                reject(request.error);
            };
        });
    }
}
```

```javascript
                this.db = request.result;
                resolve();

                        keyPath: 'key'
                    });
                    
                    objectStore.createIndex('expireAt', 'expireAt', { unique: false });
                }
            };
        });
    }
    
    // 设置缓存
    async set(key, value, ttl = 3600) {
        const expireAt = Date.now() + ttl * 1000;
        
        return new Promise((resolve, reject) => {
            const transaction = this.db.transaction(['cache'], 'readwrite');
            const objectStore = transaction.objectStore('cache');
            
            const request = objectStore.put({
                key: key,
                value: value,
                expireAt: expireAt,
                createdAt: Date.now()
            });
            
            request.onsuccess = () => resolve();
            request.onerror = () => reject(request.error);
        });
    }
    
    // 获取缓存
    async get(key) {
        return new Promise((resolve, reject) => {
            const transaction = this.db.transaction(['cache'], 'readonly');
            const objectStore = transaction.objectStore('cache');
            const request = objectStore.get(key);
            
            request.onsuccess = () => {
                const record = request.result;
                
                if (!record) {
                    resolve(null);
                    return;
                }
                
                // 检查是否过期
                if (Date.now() > record.expireAt) {
                    this.delete(key);
                    resolve(null);
                    return;
                }
                
                resolve(record.value);
            };
            
            request.onerror = () => reject(request.error);
        });
    }
    
    // 删除缓存
    async delete(key) {
        return new Promise((resolve, reject) => {
            const transaction = this.db.transaction(['cache'], 'readwrite');
            const objectStore = transaction.objectStore('cache');
            const request = objectStore.delete(key);
            
            request.onsuccess = () => resolve();
            request.onerror = () => reject(request.error);
        });
    }
    
    // 清理过期缓存
    async cleanup() {
        return new Promise((resolve, reject) => {
            const transaction = this.db.transaction(['cache'], 'readwrite');
            const objectStore = transaction.objectStore('cache');
            const index = objectStore.index('expireAt');
            
            const range = IDBKeyRange.upperBound(Date.now());
            const request = index.openCursor(range);
            
            let deletedCount = 0;
            
            request.onsuccess = (event) => {
                const cursor = event.target.result;
                
                if (cursor) {
                    cursor.delete();
                    deletedCount++;
                    cursor.continue();
                } else {
                    console.log(`清理了 ${deletedCount} 个过期缓存`);
                    resolve(deletedCount);
                }
            };
            
            request.onerror = () => reject(request.error);
        });
    }
    
    // 获取缓存统计
    async getStats() {
        return new Promise((resolve, reject) => {
            const transaction = this.db.transaction(['cache'], 'readonly');
            const objectStore = transaction.objectStore('cache');
            const request = objectStore.getAll();
            
            request.onsuccess = () => {
                const records = request.result;
                const now = Date.now();
                
                const stats = {
                    total: records.length,
                    valid: 0,
                    expired: 0,
                    size: 0
                };
                
                records.forEach(record => {
                    if (now > record.expireAt) {
                        stats.expired++;
                    } else {
                        stats.valid++;
                    }
                    
                    stats.size += JSON.stringify(record.value).length;
                });
                
                resolve(stats);
            };
            
            request.onerror = () => reject(request.error);
        });
    }
}
```

```javascript
        try {
            const item = localStorage.getItem(this.prefix + key);
            
            if (!item) {
                return null;
            }
            
            const data = JSON.parse(item);
            
            // 检查是否过期
            if (data.expireAt && Date.now() > data.expireAt) {
                this.remove(key);
                return null;
            }
            
            return data.value;

            console.error('LocalStorage获取失败:', error);

            if (key.startsWith(this.prefix)) {

            }
        });

            const testKey = this.prefix + '__test__';
            localStorage.setItem(testKey, '1');
            localStorage.removeItem(testKey);

            return true;
        }
    }
```

```javascript
class StorageQuotaManager {
    constructor() {
        this.storageTypes = ['localStorage', 'sessionStorage'];
    }
    
    // 获取存储配额信息
    async getQuotaInfo() {
        if ('storage' in navigator && 'estimate' in navigator.storage) {
            const estimate = await navigator.storage.estimate();
            
            return {
                usage: estimate.usage,
                quota: estimate.quota,
                usagePercentage: ((estimate.usage / estimate.quota) * 100).toFixed(2),
                available: estimate.quota - estimate.usage,
                usageDetails: estimate.usageDetails || {}
            };
        }
        
        return null;
    }
    
    // 检查存储是否可用
    isStorageAvailable(type) {
        try {
            const storage = window[type];
            const testKey = '__storage_test__';
            storage.setItem(testKey, 'test');
            storage.removeItem(testKey);
            return true;
        } catch (error) {
            return false;
        }
    }
    
    // 获取存储使用情况
    getStorageUsage(type) {
        const storage = window[type];
        let totalSize = 0;
        
        for (let key in storage) {
            if (storage.hasOwnProperty(key)) {
                totalSize += storage[key].length + key.length;
            }
        }
        
        return {
            type: type,
            size: totalSize,
            itemCount: storage.length,
            sizeInKB: (totalSize / 1024).toFixed(2),
            sizeInMB: (totalSize / 1024 / 1024).toFixed(2)
        };
    }
    
    // 清理最旧的数据
    cleanOldest(type, count = 10) {
        const storage = window[type];
        const items = [];
        
        // 收集所有项及其时间戳
        for (let i = 0; i < storage.length; i++) {
            const key = storage.key(i);
            try {
                const value = JSON.parse(storage.getItem(key));
                if (value && value.createdAt) {
                    items.push({
                        key: key,
                        createdAt: value.createdAt
                    });
                }
            } catch (error) {
                // 忽略解析错误
            }
        }
        
        // 按时间排序
        items.sort((a, b) => a.createdAt - b.createdAt);
        
        // 删除最旧的项
        const toDelete = items.slice(0, count);
        toDelete.forEach(item => {
            storage.removeItem(item.key);
        });
        
        return toDelete.length;
    }
    
    // 请求持久化存储
    async requestPersistentStorage() {
        if ('storage' in navigator && 'persist' in navigator.storage) {
            const isPersisted = await navigator.storage.persist();
            console.log(`持久化存储: ${isPersisted ? '已启用' : '未启用'}`);
            return isPersisted;
        }
        return false;
    }
    
    // 检查是否已持久化
    async isPersisted() {
        if ('storage' in navigator && 'persisted' in navigator.storage) {
            return await navigator.storage.persisted();
        }
        return false;
    }
}
```

### 5.1 缓存策略选择

```javascript
/**
 * 资源类型缓存策略
 * @author erik.zhou
 */

const cacheStrategies = {
    // HTML文件 - 网络优先
    html: {
        strategy: 'networkFirst',
        cacheName: 'html-cache',
        maxAge: 0,
        maxEntries: 50
    },
    
    // CSS/JS文件 - 缓存优先（带版本号）
    static: {
        strategy: 'cacheFirst',
        cacheName: 'static-cache',
        maxAge: 31536000, // 1年
        maxEntries: 100
    },
    
    // 图片 - 缓存优先
    images: {
        strategy: 'cacheFirst',
        cacheName: 'image-cache',
        maxAge: 2592000, // 30天
        maxEntries: 200
    },
    
    // API数据 - 网络优先
    api: {
        strategy: 'networkFirst',
        cacheName: 'api-cache',
        maxAge: 300, // 5分钟
        maxEntries: 50
    },
    
    // 字体文件 - 缓存优先
    fonts: {
        strategy: 'cacheFirst',
        cacheName: 'font-cache',
        maxAge: 31536000, // 1年
        maxEntries: 30
    }
};

// 根据URL选择策略
function selectStrategy(url) {
    if (url.endsWith('.html')) {
        return cacheStrategies.html;
    }
    
    if (/\.(css|js)$/.test(url)) {
        return cacheStrategies.static;
    }
    
    if (/\.(png|jpg|jpeg|gif|svg|webp)$/.test(url)) {
        return cacheStrategies.images;
    }
    
    if (url.includes('/api/')) {
        return cacheStrategies.api;
    }
    
    if (/\.(woff|woff2|ttf|eot)$/.test(url)) {
        return cacheStrategies.fonts;
    }
    
    return null;
}
```

### 5.2 缓存更新策略

#### 版本化缓存更新
```javascript
/**
 * 版本化缓存更新
 * @author erik.zhou
 */

class CacheVersionManager {
    constructor() {
        this.currentVersion = '1.0.0';
        this.versionKey = 'cache_version';
    }
    
    // 检查版本
    async checkVersion() {
        const storedVersion = localStorage.getItem(this.versionKey);
        
        if (storedVersion !== this.currentVersion) {
            console.log(`版本更新: ${storedVersion} -> ${this.currentVersion}`);
            await this.updateCache();
            localStorage.setItem(this.versionKey, this.currentVersion);
        }
    }
    
    // 更新缓存
    async updateCache() {
        // 清除所有旧缓存
        const cacheNames = await caches.keys();
        await Promise.all(
            cacheNames.map(name => caches.delete(name))
        );
        
        // 预缓存新资源
        const cache = await caches.open(`app-cache-${this.currentVersion}`);
        await cache.addAll([
            '/',
            '/index.html',
            '/css/style.css',
            '/js/app.js'
        ]);
        
        console.log('缓存已更新');
    }
    
    // 通知用户更新
    notifyUpdate() {
        if ('serviceWorker' in navigator && navigator.serviceWorker.controller) {
            navigator.serviceWorker.controller.postMessage({
                type: 'CHECK_UPDATE'
            });
        }
    }
}

// 使用示例
const versionManager = new CacheVersionManager();
versionManager.checkVersion();
```

```javascript
/**
 * 缓存性能监控
 * @author erik.zhou
 */

class CachePerformanceMonitor {
    constructor() {
        this.metrics = {
            hits: 0,
            misses: 0,
            totalRequests: 0,
            cacheTime: [],
            networkTime: []
        };
    }
    
    // 记录缓存命中
    recordHit(duration) {
        this.metrics.hits++;
        this.metrics.totalRequests++;
        this.metrics.cacheTime.push(duration);
    }
    
    // 记录缓存未命中
    recordMiss(duration) {
        this.metrics.misses++;
        this.metrics.totalRequests++;
        this.metrics.networkTime.push(duration);
    }
    
    // 获取统计信息
    getStats() {
        const hitRate = this.metrics.totalRequests > 0
            ? (this.metrics.hits / this.metrics.totalRequests * 100).toFixed(2)
            : 0;
        
        const avgCacheTime = this.metrics.cacheTime.length > 0
            ? (this.metrics.cacheTime.reduce((a, b) => a + b, 0) / this.metrics.cacheTime.length).toFixed(2)
            : 0;
        
        const avgNetworkTime = this.metrics.networkTime.length > 0
            ? (this.metrics.networkTime.reduce((a, b) => a + b, 0) / this.metrics.networkTime.length).toFixed(2)
            : 0;
        
        return {
            totalRequests: this.metrics.totalRequests,
            hits: this.metrics.hits,
            misses: this.metrics.misses,
            hitRate: hitRate + '%',
            avgCacheTime: avgCacheTime + 'ms',
            avgNetworkTime: avgNetworkTime + 'ms',
            timeSaved: ((avgNetworkTime - avgCacheTime) * this.metrics.hits).toFixed(2) + 'ms'
        };
    }
    
    // 重置统计
    reset() {
        this.metrics = {
            hits: 0,
            misses: 0,
            totalRequests: 0,
            cacheTime: [],
            networkTime: []
        };
    }
    
    // 生成报告
    generateReport() {
        const stats = this.getStats();
        
        console.log('=== 缓存性能报告 ===');
        console.log(`总请求数: ${stats.totalRequests}`);
        console.log(`缓存命中: ${stats.hits}`);
        console.log(`缓存未命中: ${stats.misses}`);
        console.log(`命中率: ${stats.hitRate}`);
        console.log(`平均缓存响应时间: ${stats.avgCacheTime}`);
        console.log(`平均网络响应时间: ${stats.avgNetworkTime}`);
        console.log(`节省时间: ${stats.timeSaved}`);
        
        return stats;
    }
}

// 在Service Worker中使用
const perfMonitor = new CachePerformanceMonitor();

self.addEventListener('fetch', event => {
    const startTime = Date.now();
    
    event.respondWith(
        caches.match(event.request)
            .then(response => {
                if (response) {
                    const duration = Date.now() - startTime;
                    perfMonitor.recordHit(duration);
                    return response;
                }
                
                return fetch(event.request)
                    .then(response => {
                        const duration = Date.now() - startTime;
                        perfMonitor.recordMiss(duration);
                        return response;
                    });
            })
    );
});

// 定期生成报告
setInterval(() => {
    perfMonitor.generateReport();
}, 60000); // 每分钟
```
