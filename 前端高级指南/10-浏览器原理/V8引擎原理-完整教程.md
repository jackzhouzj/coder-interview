// - 复制存活对象到To空间

## 6. 隐藏类

### 6.1 隐藏类优化

```javascript
/**
 * 隐藏类优化
 * @author erik.zhou
 */

// 优化前：动态添加属性
function Point(x, y) {
    this.x = x;
    this.y = y;
}

const p1 = new Point(1, 2);
p1.z = 3; // 改变隐藏类

// 优化后：固定属性
function Point(x, y, z) {
    this.x = x;
    this.y = y;
    this.z = z || 0;
}

const p2 = new Point(1, 2);
```

## 7. 内联缓存

### 7.1 IC优化

```javascript
/**
 * 内联缓存示例
 * @author erik.zhou
 */

// 单态（Monomorphic）- 最快
function getX(obj) {
    return obj.x; // 总是访问相同类型的对象
}

// 多态（Polymorphic）- 较慢
function getProperty(obj, prop) {
    return obj[prop]; // 访问不同类型的对象
}
```

## 8. 性能优化

### 8.1 优化技巧

```javascript
/**
 * V8性能优化技巧
 * @author erik.zhou
 */

// 1. 避免删除属性
const obj = { x: 1, y: 2 };
// delete obj.x; // 不推荐
obj.x = undefined; // 推荐

// 2. 使用数组而不是类数组对象
const arr = [1, 2, 3]; // 推荐
// const arrLike = { 0: 1, 1: 2, 2: 3, length: 3 }; // 不推荐

// 3. 避免稀疏数组
const dense = [1, 2, 3]; // 推荐
// const sparse = [1, , 3]; // 不推荐

// 4. 使用单态函数
function add(a, b) {
    return a + b;
}
add(1, 2); // 数字
add(1, 2); // 数字（单态）
// add('1', '2'); // 字符串（变为多态）
```
