### 1.1 同源策略基础

#### 什么是同源
```javascript
/**
 * 同源策略示例
 * @author erik.zhou
 */

// 同源判断
function isSameOrigin(url1, url2) {
    const parseURL = (url) => {
        const a = document.createElement('a');
        a.href = url;
        return {
            protocol: a.protocol,
            hostname: a.hostname,
            port: a.port || (a.protocol === 'https:' ? '443' : '80')
        };
    };
    
    const origin1 = parseURL(url1);
    const origin2 = parseURL(url2);
    
    return origin1.protocol === origin2.protocol &&
           origin1.hostname === origin2.hostname &&
           origin1.port === origin2.port;
}

// 测试示例
const currentURL = 'https://example.com:443/page';

console.log(isSameOrigin(currentURL, 'https://example.com/api')); // true
console.log(isSameOrigin(currentURL, 'http://example.com/api'));  // false (协议不同)
console.log(isSameOrigin(currentURL, 'https://api.example.com/data')); // false (域名不同)
console.log(isSameOrigin(currentURL, 'https://example.com:8080/api')); // false (端口不同)
```

```javascript
    res.json({ message: '受限API' });
});

// 4. 动态CORS配置
app.use((req, res, next) => {
    const origin = req.headers.origin;
    
    // 根据请求动态设置CORS
    if (origin && isAllowedOrigin(origin)) {
        res.set('Access-Control-Allow-Origin', origin);
        res.set('Access-Control-Allow-Credentials', 'true');
    }
    
    next();
});

function isAllowedOrigin(origin) {
    const allowedPatterns = [
        /^https:\/\/.*\.example\.com$/,
        /^http:\/\/localhost:\d+$/
    ];
    
    return allowedPatterns.some(pattern => pattern.test(origin));
}
```

#### 自定义CORS中间件
```javascript
/**
 * 自定义CORS中间件
 * @author erik.zhou
 */

class CORSMiddleware {
    constructor(options = {}) {
        this.options = {
            origin: options.origin || '*',
            methods: options.methods || ['GET', 'POST', 'PUT', 'DELETE'],
            allowedHeaders: options.allowedHeaders || ['Content-Type'],
            exposedHeaders: options.exposedHeaders || [],
            credentials: options.credentials || false,
            maxAge: options.maxAge || 86400
        };
    }
    
    // 中间件函数
    middleware() {
        return (req, res, next) => {
            const origin = req.headers.origin;
            
            // 设置Access-Control-Allow-Origin
            if (this.isOriginAllowed(origin)) {
                res.set('Access-Control-Allow-Origin', origin);
            } else if (this.options.origin === '*') {
                res.set('Access-Control-Allow-Origin', '*');
            }
            
            // 设置其他CORS头部
            if (this.options.credentials) {
                res.set('Access-Control-Allow-Credentials', 'true');
            }
            
            if (this.options.exposedHeaders.length > 0) {
                res.set('Access-Control-Expose-Headers', this.options.exposedHeaders.join(', '));
            }
            
            // 处理预检请求
            if (req.method === 'OPTIONS') {
                res.set('Access-Control-Allow-Methods', this.options.methods.join(', '));
                res.set('Access-Control-Allow-Headers', this.options.allowedHeaders.join(', '));
                res.set('Access-Control-Max-Age', this.options.maxAge.toString());
                return res.sendStatus(204);
            }
            
            next();
        };
    }
    
    // 检查源是否允许
    isOriginAllowed(origin) {
        if (!origin) return false;
        
        if (typeof this.options.origin === 'string') {
            return this.options.origin === origin;
        }
        
        if (Array.isArray(this.options.origin)) {
            return this.options.origin.includes(origin);
        }
        
        if (this.options.origin instanceof RegExp) {
            return this.options.origin.test(origin);
        }
        
        if (typeof this.options.origin === 'function') {
            return this.options.origin(origin);
        }
        
        return false;
    }
}

// 使用示例
const corsMiddleware = new CORSMiddleware({
    origin: ['https://example.com', 'https://app.example.com'],
    methods: ['GET', 'POST', 'PUT', 'DELETE'],
    allowedHeaders: ['Content-Type', 'Authorization'],
    exposedHeaders: ['X-Total-Count'],
    credentials: true,
    maxAge: 86400
});

app.use(corsMiddleware.middleware());
```

### 3.1 JSONP原理

#### JSONP实现
```javascript
/**
 * JSONP实现
 * @author erik.zhou
 */

// 客户端实现
class JSONP {
    constructor() {
        this.callbackCounter = 0;
    }
    
    // 发送JSONP请求
    request(url, options = {}) {
        return new Promise((resolve, reject) => {
            const callbackName = options.callbackName || 
                `jsonp_callback_${Date.now()}_${this.callbackCounter++}`;
            
            const timeout = options.timeout || 10000;
            const params = options.params || {};
            
            // 添加回调参数
            params[options.callbackParam || 'callback'] = callbackName;
            
            // 构建URL
            const queryString = Object.keys(params)
                .map(key => `${encodeURIComponent(key)}=${encodeURIComponent(params[key])}`)
                .join('&');
            
            const fullURL = `${url}?${queryString}`;
            
            // 创建script标签
            const script = document.createElement('script');
            script.src = fullURL;
            
            // 设置超时
            const timer = setTimeout(() => {
                this.cleanup(script, callbackName);
                reject(new Error('JSONP请求超时'));
            }, timeout);
            
            // 定义全局回调函数
            window[callbackName] = (data) => {
                clearTimeout(timer);
                this.cleanup(script, callbackName);
                resolve(data);
            };
            
            // 错误处理
            script.onerror = () => {
                clearTimeout(timer);
                this.cleanup(script, callbackName);
                reject(new Error('JSONP请求失败'));
            };
            
            // 添加到DOM
            document.head.appendChild(script);
        });
    }
    
    // 清理资源
    cleanup(script, callbackName) {
        if (script.parentNode) {
            script.parentNode.removeChild(script);
        }
        delete window[callbackName];
    }
}

// 使用示例
const jsonp = new JSONP();

jsonp.request('https://api.example.com/data', {
    params: { id: 123 },
    timeout: 5000
})
.then(data => {
    console.log('JSONP数据:', data);
})
.catch(error => {
    console.error('JSONP错误:', error);
});
```

### 3.2 JSONP限制与安全

#### JSONP安全问题
```javascript
/**
 * JSONP安全问题与防护
 * @author erik.zhou
 */

// 1. 回调函数名验证
function validateCallback(callback) {
    // 只允许字母、数字、下划线和$
    const pattern = /^[a-zA-Z_$][a-zA-Z0-9_$]*$/;
    return pattern.test(callback);
}

// 2. 内容安全策略
app.get('/api/jsonp', (req, res) => {
    const callback = req.query.callback;
    
    if (!validateCallback(callback)) {
        return res.status(400).json({ error: '无效的回调函数' });
    }
    
    // 设置CSP头部
    res.set('Content-Security-Policy', "script-src 'self'");
    
    const data = { message: '安全的JSONP响应' };
    res.set('Content-Type', 'application/javascript');
    res.send(`${callback}(${JSON.stringify(data)})`);
});

// 3. Referer检查
function checkReferer(req, res, next) {
    const referer = req.headers.referer;
    const allowedDomains = ['example.com', 'app.example.com'];
    
    if (!referer) {
        return res.status(403).json({ error: '缺少Referer' });
    }
    
    const refererDomain = new URL(referer).hostname;
    const isAllowed = allowedDomains.some(domain => 
        refererDomain === domain || refererDomain.endsWith(`.${domain}`)
    );
    
    if (!isAllowed) {
        return res.status(403).json({ error: '不允许的来源' });
    }
    
    next();
}

app.get('/api/secure-jsonp', checkReferer, (req, res) => {
    // 处理JSONP请求
});
```

### 5.1 窗口间通信

#### 基本使用
```javascript
/**
 * PostMessage基本使用
 * @author erik.zhou
 */

// 父窗口代码
class ParentWindow {
    constructor() {
        this.childWindow = null;
        this.setupMessageListener();
    }
    
    // 打开子窗口
    openChild(url) {
        this.childWindow = window.open(url, '_blank');
    }
    
    // 发送消息到子窗口
    sendToChild(message) {
        if (this.childWindow) {
            this.childWindow.postMessage(message, 'https://child.example.com');
        }
    }
    
    // 监听子窗口消息
    setupMessageListener() {
        window.addEventListener('message', (event) => {
            // 验证来源
            if (event.origin !== 'https://child.example.com') {
                console.warn('收到未知来源的消息:', event.origin);
                return;
            }
            
            console.log('收到子窗口消息:', event.data);
            this.handleChildMessage(event.data);
        });
    }
    
    // 处理子窗口消息
    handleChildMessage(data) {
        switch (data.type) {
            case 'READY':
                console.log('子窗口已就绪');
                this.sendToChild({ type: 'INIT', config: {} });
                break;
            case 'DATA':
                console.log('收到数据:', data.payload);
                break;
            default:
                console.log('未知消息类型:', data.type);
        }
    }
}

// 子窗口代码
class ChildWindow {
    constructor() {
        this.parentWindow = window.opener;
        this.setupMessageListener();
        this.notifyReady();
    }
    
    // 通知父窗口已就绪
    notifyReady() {
        if (this.parentWindow) {
            this.parentWindow.postMessage(
                { type: 'READY' },
                'https://parent.example.com'
            );
        }
    }
    
    // 发送消息到父窗口
```

```javascript
/**
 * 安全的PostMessage通信
 * @author erik.zhou
 */

class SecureMessenger {
    constructor(targetOrigin, secretKey) {
        this.targetOrigin = targetOrigin;
        this.secretKey = secretKey;
        this.messageQueue = new Map();
        this.setupMessageListener();
    }
    
    // 发送加密消息
    async sendSecure(targetWindow, message) {
        const messageId = this.generateMessageId();
        const timestamp = Date.now();
        
        // 构建消息
        const payload = {
            id: messageId,
            timestamp: timestamp,
            data: message
        };
        
        // 生成签名
        const signature = await this.sign(payload);
        
        // 发送消息
        targetWindow.postMessage({
            payload: payload,
            signature: signature
        }, this.targetOrigin);
        
        // 返回Promise用于等待响应
        return new Promise((resolve, reject) => {
            this.messageQueue.set(messageId, { resolve, reject });
            
            // 超时处理
            setTimeout(() => {
                if (this.messageQueue.has(messageId)) {
                    this.messageQueue.delete(messageId);
                    reject(new Error('消息超时'));
                }
            }, 10000);
        });
    }
    
    // 生成消息ID
    generateMessageId() {
        return `${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
    }
    
    // 签名消息
    async sign(payload) {
        const message = JSON.stringify(payload);
        const encoder = new TextEncoder();
        const data = encoder.encode(message + this.secretKey);
        
        const hashBuffer = await crypto.subtle.digest('SHA-256', data);
        const hashArray = Array.from(new Uint8Array(hashBuffer));
        const hashHex = hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
        
        return hashHex;
    }
    
    // 验证签名
    async verify(payload, signature) {
        const expectedSignature = await this.sign(payload);
        return expectedSignature === signature;
    }
    
    // 监听消息
    setupMessageListener() {
        window.addEventListener('message', async (event) => {
            // 验证来源
            if (event.origin !== this.targetOrigin) {
                console.warn('收到未知来源的消息:', event.origin);
                return;
            }
            
            const { payload, signature } = event.data;
            
            // 验证签名
            const isValid = await this.verify(payload, signature);
            if (!isValid) {
                console.error('消息签名验证失败');
                return;
            }
            
            // 验证时间戳（防止重放攻击）
            const age = Date.now() - payload.timestamp;
            if (age > 60000) { // 1分钟
                console.error('消息已过期');
                return;
            }
            
            // 处理消息
            this.handleMessage(payload);
        });
    }
    
    // 处理消息
    handleMessage(payload) {
        const { id, data } = payload;
        
        // 如果是响应消息
        if (this.messageQueue.has(id)) {
            const { resolve } = this.messageQueue.get(id);
            this.messageQueue.delete(id);
            resolve(data);
            return;
        }
        
        // 处理新消息
        console.log('收到消息:', data);
    }
}

// 使用示例
const messenger = new SecureMessenger('https://target.example.com', 'secret-key-123');

// 发送安全消息
const childWindow = window.open('https://target.example.com');
messenger.sendSecure(childWindow, { type: 'HELLO', message: 'Hello World' })
    .then(response => {
        console.log('收到响应:', response);
    })
    .catch(error => {
        console.error('消息发送失败:', error);
    });
```

```javascript
// 客户端代码
class WebSocketClient {
    constructor(url) {
        this.url = url;
        this.ws = null;
        this.reconnectAttempts = 0;
        this.maxReconnectAttempts = 5;
        this.reconnectDelay = 1000;
    }
    
    // 连接WebSocket
    connect() {
        return new Promise((resolve, reject) => {
            this.ws = new WebSocket(this.url);
            
            this.ws.onopen = () => {
                console.log('WebSocket已连接');
                this.reconnectAttempts = 0;
                resolve();
            };
            
            this.ws.onmessage = (event) => {
                this.handleMessage(event.data);
            };
            
            this.ws.onerror = (error) => {
                console.error('WebSocket错误:', error);
                reject(error);
            };
            
            this.ws.onclose = () => {
                console.log('WebSocket已关闭');
                this.reconnect();
            };
        });
    }
    
    // 发送消息
    send(data) {
        if (this.ws && this.ws.readyState === WebSocket.OPEN) {
            this.ws.send(JSON.stringify(data));
        } else {
            console.error('WebSocket未连接');
        }
    }
    
    // 处理消息
    handleMessage(data) {
        try {
            const message = JSON.parse(data);
            console.log('收到消息:', message);
            
            // 触发自定义事件
            this.onMessage(message);
        } catch (error) {
            console.error('消息解析失败:', error);
        }
    }
    
    // 消息回调（可被覆盖）
    onMessage(message) {
        console.log('默认消息处理:', message);
    }
    
    // 重连
    reconnect() {
        if (this.reconnectAttempts >= this.maxReconnectAttempts) {
            console.error('达到最大重连次数');
            return;
        }
        
        this.reconnectAttempts++;
        const delay = this.reconnectDelay * this.reconnectAttempts;
        
        console.log(`${delay}ms后尝试重连...`);
        setTimeout(() => {
            this.connect();
        }, delay);
    }
    
    // 关闭连接
    close() {
        if (this.ws) {
            this.ws.close();
            this.ws = null;
        }
    }
}
```

## 总结

### 核心要点

1. **同源策略**
   - 协议、域名、端口必须完全相同
   - 限制Cookie、DOM、AJAX访问
   - 保护用户数据安全

2. **CORS**
   - 最标准的跨域解决方案
   - 支持简单请求和预检请求
   - 可携带凭证
   - 服务器端配置灵活

3. **JSONP**
   - 利用script标签不受同源限制
   - 只支持GET请求
   - 存在安全风险
   - 逐渐被CORS取代

4. **代理服务器**
   - 开发环境常用方案
   - 生产环境使用Nginx
   - 完全绕过浏览器限制
   - 可实现负载均衡

5. **PostMessage**
   - 窗口间安全通信
   - 支持跨域iframe
   - 需要验证消息来源
   - 适合复杂交互场景

6. **其他方案**
   - WebSocket天然支持跨域
   - document.domain适用子域名
   - window.name传输数据
   - 各有适用场景

### 方案选择建议

| 场景 | 推荐方案 | 原因 |
|------|---------|------|
| 现代Web应用 | CORS | 标准、安全、功能完整 |
| 开发环境 | 代理服务器 | 配置简单、无需后端改动 |
| 生产环境 | Nginx代理 | 性能好、功能强大 |
| 窗口通信 | PostMessage | 专为跨窗口设计 |
| 实时通信 | WebSocket | 双向通信、低延迟 |
| 老旧浏览器 | JSONP | 兼容性好（不推荐） |

### 安全注意事项

1. 始终验证消息来源（origin）
2. 不要使用`Access-Control-Allow-Origin: *`携带凭证
3. JSONP需要验证回调函数名
4. PostMessage需要验证和加密
5. 避免在响应中包含敏感信息
6. 实施CSRF防护
7. 使用HTTPS传输敏感数据

### 学习建议

1. 深入理解同源策略原理
2. 优先掌握CORS标准方案
3. 熟练配置开发环境代理
4. 实践PostMessage安全通信
5. 了解各方案的适用场景
6. 关注安全最佳实践
