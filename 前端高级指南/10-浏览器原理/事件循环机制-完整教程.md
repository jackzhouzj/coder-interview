## 1. 事件循环基础

### 1.1 事件循环模型

```javascript
/**
 * 事件循环基础
 * @author erik.zhou
 */

// 调用栈（Call Stack）
// 任务队列（Task Queue）
// 微任务队列（Microtask Queue）

console.log('1'); // 同步任务

setTimeout(() => {
    console.log('2'); // 宏任务
}, 0);

Promise.resolve().then(() => {
    console.log('3'); // 微任务
});

console.log('4'); // 同步任务

// 输出顺序：1, 4, 3, 2
```

## 2. 宏任务与微任务

### 2.1 任务类型

```javascript
/**
 * 宏任务与微任务
 * @author erik.zhou
 */

// 宏任务（Macrotask）
// - setTimeout
// - setInterval
// - setImmediate (Node.js)
// - I/O
// - UI渲染

// 微任务（Microtask）
// - Promise.then/catch/finally
// - MutationObserver
// - queueMicrotask
// - process.nextTick (Node.js)

// 示例
console.log('start');

setTimeout(() => {
    console.log('setTimeout'); // 宏任务
}, 0);

Promise.resolve().then(() => {
    console.log('promise1'); // 微任务
}).then(() => {
    console.log('promise2'); // 微任务
});

console.log('end');

// 输出：start, end, promise1, promise2, setTimeout
```

## 3. 执行顺序

### 3.1 执行流程

```javascript
/**
 * 事件循环执行顺序
 * @author erik.zhou
 */

// 执行流程：
// 1. 执行同步代码
// 2. 执行所有微任务
// 3. 执行一个宏任务
// 4. 重复步骤2-3

console.log('1');

setTimeout(() => {
    console.log('2');
    Promise.resolve().then(() => {
        console.log('3');
    });
}, 0);

Promise.resolve().then(() => {
    console.log('4');
    setTimeout(() => {
        console.log('5');
    }, 0);
});

console.log('6');

// 输出：1, 6, 4, 2, 3, 5
```

## 4. Promise与async/await

### 4.1 Promise执行顺序

```javascript
/**
 * Promise执行顺序
 * @author erik.zhou
 */

console.log('1');

new Promise((resolve) => {
    console.log('2'); // Promise构造函数同步执行
    resolve();
}).then(() => {
    console.log('3'); // 微任务
});

console.log('4');

// 输出：1, 2, 4, 3
```

### 4.2 async/await执行顺序

```javascript
/**
 * async/await执行顺序
 * @author erik.zhou
 */

async function async1() {
    console.log('async1 start');
    await async2();
    console.log('async1 end'); // 微任务
}

async function async2() {
    console.log('async2');
}

console.log('script start');

setTimeout(() => {
    console.log('setTimeout');
}, 0);

async1();

new Promise((resolve) => {
    console.log('promise1');
    resolve();
}).then(() => {
    console.log('promise2');
});

console.log('script end');

// 输出：
// script start
// async1 start
// async2
// promise1
// script end
// async1 end
// promise2
// setTimeout
```

## 5. 浏览器事件循环

### 5.1 渲染时机

```javascript
/**
 * 浏览器渲染时机
 * @author erik.zhou
 */

// 浏览器在每次事件循环后可能进行渲染
// 微任务在渲染前执行

const button = document.querySelector('button');

button.addEventListener('click', () => {
    // 宏任务
    console.log('click');
    
    Promise.resolve().then(() => {
        // 微任务（渲染前执行）
        console.log('microtask');
    });
    
    setTimeout(() => {
        // 下一个宏任务（渲染后执行）
        console.log('timeout');
    }, 0);
});
```

## 6. Node.js事件循环

### 6.1 Node.js事件循环阶段

```javascript
/**
 * Node.js事件循环
 * @author erik.zhou
 */

// Node.js事件循环阶段：
// 1. timers - 执行setTimeout/setInterval回调
// 2. pending callbacks - 执行延迟到下一个循环的I/O回调
// 3. idle, prepare - 内部使用
// 4. poll - 检索新的I/O事件
// 5. check - 执行setImmediate回调
// 6. close callbacks - 执行关闭回调

setTimeout(() => {
    console.log('setTimeout');
}, 0);

setImmediate(() => {
    console.log('setImmediate');
});

process.nextTick(() => {
    console.log('nextTick');
});

// 输出顺序可能：nextTick, setTimeout, setImmediate
```

## 7. 常见陷阱

### 7.1 执行顺序陷阱

```javascript
/**
 * 常见陷阱
 * @author erik.zhou
 */

// 陷阱1：Promise构造函数是同步的
new Promise((resolve) => {
    console.log('1'); // 同步执行
    resolve();
}).then(() => {
    console.log('2'); // 微任务
});

// 陷阱2：await后的代码是微任务
async function test() {
    console.log('3');
    await Promise.resolve();
    console.log('4'); // 微任务
}

// 陷阱3：setTimeout(fn, 0)不是立即执行
setTimeout(() => {
    console.log('5'); // 宏任务
}, 0);
```

## 8. 性能优化

### 8.1 避免阻塞事件循环

```javascript
/**
 * 避免阻塞事件循环
 * @author erik.zhou
 */

// 不推荐：长时间同步操作
function heavyTask() {
    for (let i = 0; i < 1000000000; i++) {
        // 阻塞事件循环
    }
}

// 推荐：分片处理
async function heavyTaskOptimized() {
    const chunkSize = 1000000;
    const total = 1000000000;
    
    for (let i = 0; i < total; i += chunkSize) {
        // 处理一块数据
        for (let j = i; j < Math.min(i + chunkSize, total); j++) {
            // 处理
        }
        
        // 让出控制权
        await new Promise(resolve => setTimeout(resolve, 0));
    }
}
```
